+ * 应用后用同套契约测试复验；本类契约测试用 H2 内存库（不碰宿主 MySQL）。

+ * @author dowei
+ * @date 2026-11-24

+    static final String TABLE = "iot_calc_state";

+    private final JdbcTemplate jdbc;
+
+    public JdbcCalcStateStore(JdbcTemplate jdbc) {
+        this.jdbc = jdbc;
+    }
+
+    @Override
+    public CalcState get(long tenantId, String fullKey) {
+        try {
+            List<CalcState> rows = jdbc.query(
+                    "SELECT tenant_id, full_key, last_calc_time, watermark, calc_status, "
+                            + "error_count, last_error, update_time FROM " + TABLE
+                            + " WHERE tenant_id=? AND full_key=?",
+                    (rs, i) -> map(rs.getLong("tenant_id"), rs.getString("full_key"),
+                            rs.getTimestamp("last_calc_time"), rs.getTimestamp("watermark"),
+                            rs.getInt("calc_status"), rs.getInt("error_count"),
+                            rs.getString("last_error"), rs.getTimestamp("update_time")),

+        } catch (RuntimeException e) {
+            warn("get", tenantId, fullKey, e);
+            return null;
+        }
+    }
+
+    @Override
+    public void upsert(CalcState state) {
+        if (state == null) {
+            return;
+        }
+        try {
+            long updateTime = state.updateTime() > 0 ? state.updateTime() : System.currentTimeMillis();
+            java.sql.Timestamp lastCalc = ts(state.lastCalcTime());
+            java.sql.Timestamp wm = ts(state.watermark());
+            java.sql.Timestamp ut = new java.sql.Timestamp(updateTime);
+            // 先 UPDATE 再按影响行数 INSERT——跨 MySQL/H2 通用（不依赖 ON DUPLICATE KEY 的方言差异）
+            int updated = jdbc.update(
+                    "UPDATE " + TABLE + " SET last_calc_time=?, watermark=?, calc_status=?, "
+                            + "error_count=?, last_error=?, update_time=? WHERE tenant_id=? AND full_key=?",
+                    lastCalc, wm, state.calcStatus(), state.errorCount(), state.lastError(), ut,
+                    state.tenantId(), state.fullKey());
+            if (updated == 0) {
+                jdbc.update(
+                        "INSERT INTO " + TABLE + " (tenant_id, full_key, last_calc_time, watermark, "
+                                + "calc_status, error_count, last_error, update_time) VALUES (?,?,?,?,?,?,?,?)",
+                        state.tenantId(), state.fullKey(), lastCalc, wm, state.calcStatus(),
+                        state.errorCount(), state.lastError(), ut);
+            }
+        } catch (RuntimeException e) {
+            warn("upsert", state.tenantId(), state.fullKey(), e);
+        }
+    }
+
+    @Override
+    public List<CalcState> byTenant(long tenantId) {
+        try {
+            return jdbc.query(
+                    "SELECT tenant_id, full_key, last_calc_time, watermark, calc_status, "
+                            + "error_count, last_error, update_time FROM " + TABLE
+                            + " WHERE tenant_id=? ORDER BY full_key",

+                            rs.getInt("calc_status"), rs.getInt("error_count"),
+                            rs.getString("last_error"), rs.getTimestamp("update_time")),
+                    tenantId);
+        } catch (RuntimeException e) {
+            warn("byTenant", tenantId, null, e);
+            return List.of();
+        }
+    }

+        try {
+            return jdbc.query(
+                    "SELECT tenant_id, full_key, last_calc_time, watermark, calc_status, "
+                            + "error_count, last_error, update_time FROM " + TABLE
+                            + " ORDER BY tenant_id, full_key",

+                            rs.getTimestamp("last_calc_time"), rs.getTimestamp("watermark"),
+                            rs.getInt("calc_status"), rs.getInt("error_count"),
+                            rs.getString("last_error"), rs.getTimestamp("update_time")));
+        } catch (RuntimeException e) {
+            warn("all", null, null, e);
+            return List.of();
+        }
+    }
+

+                                 java.sql.Timestamp wm, int status, int errorCount,
+                                 String lastError, java.sql.Timestamp updateTime) {
+        return CalcState.of(tenantId, fullKey)
+                .lastCalcTime(lastCalc == null ? 0L : lastCalc.getTime())
+                .watermark(wm == null ? 0L : wm.getTime())
+                .calcStatus(status).errorCount(errorCount).lastError(lastError)
+                .updateTime(updateTime == null ? 0L : updateTime.getTime())
+                .build();
+    }
+

+        return millis > 0 ? new java.sql.Timestamp(millis) : null;
+    }

+        log.warn("[CALC-STATE-JDBC] {} 降级（tenant={}, fullKey={}）——在线计算不阻塞: {}",
+                op, tenantId, fullKey, e.getMessage());
+    }

+--   4) 幂等：schema/表/账号均 IF NOT EXISTS 或 information_schema 判存，可重复执行无副作用。
+--   5) 执行纪律：backend/scripts/sql-exec.sh 每条语句单独一次调用。
+-- ============================================================
+
+-- ① 独立 schema（Q1 定版 A：同实例分 schema，iot_calc 与业务库同 MySQL 实例）
+CREATE SCHEMA IF NOT EXISTS `iot_calc` DEFAULT CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci;
+
+-- ② iot_calc_state（§1.9A；不套审计四件套）
+CREATE TABLE IF NOT EXISTS `iot_calc`.`iot_calc_state` (
+  `id`             bigint       NOT NULL AUTO_INCREMENT COMMENT '主键 ID',
+  `tenant_id`      bigint       NOT NULL                COMMENT '租户 ID',
+  `full_key`       varchar(180) NOT NULL                COMMENT '数据项/指标唯一键',
+  `last_calc_time` datetime     DEFAULT NULL            COMMENT '最近一次计算窗口起',
+  `watermark`      datetime     DEFAULT NULL            COMMENT '水位（该时间前已处理完，补算从此后开始）',
+  `calc_status`    tinyint      DEFAULT '1'             COMMENT '状态：1 正常 2 滞后 3 异常停算',
+  `error_count`    int          DEFAULT '0'             COMMENT '连续错误次数',
+  `last_error`     varchar(1000) DEFAULT NULL           COMMENT '最近错误原因',
+  `update_time`    datetime     DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
+  PRIMARY KEY (`id`),
+  UNIQUE KEY `uk_tenant_full_key` (`tenant_id`, `full_key`)
+) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='计算状态（iot_calc_state；无审计四件套，TASK-0063）';
+
+-- ③ iot_calc_log（§1.9A 全字段 + 三索引）
+CREATE TABLE IF NOT EXISTS `iot_calc`.`iot_calc_log` (
+  `id`            bigint        NOT NULL AUTO_INCREMENT COMMENT '主键 ID',
+  `tenant_id`     bigint        NOT NULL                COMMENT '租户 ID',
+  `task_id`       varchar(64)   NOT NULL                COMMENT '任务 ID（数据项/指标重算链）',
+  `full_key`      varchar(180)  NOT NULL                COMMENT '数据项/指标唯一键',
+  `period_start`  datetime      NOT NULL                COMMENT '计算窗口起',
+  `period_end`    datetime      NOT NULL                COMMENT '计算窗口止',
+  `trigger_type`  varchar(16)   NOT NULL                COMMENT '触发方式：AUTO/BACKFILL/MANUAL',
+  `input_summary` json          DEFAULT NULL            COMMENT '输入摘要（上游全键/窗口条数/关键输入）',
+  `result_status` tinyint       NOT NULL                COMMENT '结果：1 成功 2 失败',
+  `error_msg`     varchar(1000) DEFAULT NULL            COMMENT '失败原因',
+  `is_recalc`     tinyint       DEFAULT '0'             COMMENT '是否重算：0 否 1 是',
+  `started_at`    datetime      NOT NULL                COMMENT '开始时间',
+  `finished_at`   datetime      DEFAULT NULL            COMMENT '结束时间',
+  `duration_ms`   bigint        DEFAULT NULL            COMMENT '耗时（毫秒）',
+  PRIMARY KEY (`id`),
+  KEY `idx_task` (`task_id`),
+  KEY `idx_key_period` (`tenant_id`, `full_key`, `period_start`),
+  KEY `idx_trigger_time` (`trigger_type`, `started_at`)
+) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='计算日志（iot_calc_log；追加写，TASK-0063）';
+
+-- ④ 计算引擎账号（读写 iot_calc schema）——口令由部署注入，占位符不硬编码
+-- 幂等：information_schema 判存，存在则跳过 CREATE（MySQL 8.0 无 CREATE USER IF NOT EXISTS 的
+--       跨版本差异口径，统一用预处理判存，可重复执行）。
+SET @rw_user_exists = (SELECT COUNT(*) FROM information_schema.USER_PRIVILEGES WHERE GRANTEE = "'iot_calc_rw'@'%'");
+SET @rw_create = IF(@rw_user_exists = 0,
+  'CREATE USER ''iot_calc_rw''@''%'' IDENTIFIED BY ''${IOT_CALC_RW_PASSWORD}''',
+  'SELECT 1');
+PREPARE rw_stmt FROM @rw_create; EXECUTE rw_stmt; DEALLOCATE PREPARE rw_stmt;
+GRANT SELECT, INSERT, UPDATE ON `iot_calc`.* TO 'iot_calc_rw'@'%';
+FLUSH PRIVILEGES;
+
+-- ⑤ 管理系统只读账号（仅 SELECT iot_calc schema；与计算引擎写账号分离）
+SET @ro_user_exists = (SELECT COUNT(*) FROM information_schema.USER_PRIVILEGES WHERE GRANTEE = "'iot_calc_ro'@'%'");
+SET @ro_create = IF(@ro_user_exists = 0,
+  'CREATE USER ''iot_calc_ro''@''%'' IDENTIFIED BY ''${IOT_CALC_RO_PASSWORD}''',
+  'SELECT 1');
+PREPARE ro_stmt FROM @ro_create; EXECUTE ro_stmt; DEALLOCATE PREPARE ro_stmt;
+GRANT SELECT ON `iot_calc`.* TO 'iot_calc_ro'@'%';
+FLUSH PRIVILEGES;
