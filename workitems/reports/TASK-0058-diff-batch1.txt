+  ancestors?: string;

+}
+
+/** 节点类型（与 iot-consts 对齐：1 工厂 2 车间 3 产线 4 设备 5 组件） */
+export const DISPLAY_NODE_TYPE = {
+  FACTORY: 1,
+  WORKSHOP: 2,
+  LINE: 3,
+  DEVICE: 4,
+  COMPONENT: 5,
+} as const;
+
+/** 后端响应体 { code, message, data }；code!==200 已由 axios 拦截器全局处理并 reject */
+interface TreeResponse {
+  code: number;
+  message?: string;
+  data: DisplayTreeNode[];
+}
+
+export const displayApi = {
+  /**
+   * 查询完整实例树（已按数据权限裁剪——无权节点后端不返回，前端不二次过滤）。
+   * 返回 data 数组；空数组=无权/空厂，由视图渲染空态。
+   */
+  async queryInstanceTree(): Promise<DisplayTreeNode[]> {
+    const res: TreeResponse = await getRequest('/iot/instance/tree', undefined);
+    return Array.isArray(res?.data) ? res.data : [];
+  },
+};
+
+/** 按父节点 id 取子节点（模型树 lazy load 的纯函数；parentId=null 取根；无权节点不在树中自然不出现）。 */
+export function childrenOf(nodes: DisplayTreeNode[], parentId: number | null): DisplayTreeNode[] {
+  if (parentId == null) return [...nodes].sort((a, b) => (a.sort ?? 0) - (b.sort ?? 0));
+  const parent = findNode(nodes, parentId);
+  return parent ? [...childrenOfRaw(parent)].sort((a, b) => (a.sort ?? 0) - (b.sort ?? 0)) : [];
+}
+
+/** 在树中按 id 深度查找节点（下钻定位用；找不到返回 null）。 */
+export function findNode(nodes: DisplayTreeNode[], id: number): DisplayTreeNode | null {
+  for (const node of nodes) {
+    if (node.id === id) return node;
+    const hit = findNode(childrenOfRaw(node), id);
+    if (hit) return hit;
+  }
+  return null;
+}
+
+function childrenOfRaw(node: DisplayTreeNode): DisplayTreeNode[] {
+  return Array.isArray(node.children) ? node.children : [];
+}
+
+/**
+ * 取指定类型的直接子节点（跳过中间层级）：工厂页取车间列表（2），车间/产线页取设备列表（4）。
+ * 不做跨权限补节点——只见已裁剪树中存在的节点。
+ */
+export function descendantsOfType(nodes: DisplayTreeNode[], nodeId: number, nodeType: number): DisplayTreeNode[] {
+  const root = findNode(nodes, nodeId);
+  if (!root) return [];
+  const out: DisplayTreeNode[] = [];
+  const walk = (n: DisplayTreeNode) => {
+    for (const child of childrenOfRaw(n)) {
+      if (child.nodeType === nodeType) out.push(child);
+      walk(child);
+    }
+  };
+  walk(root);
+  return out;
+}

+/*
+ * 展示端模型树加载（TASK-0058 批次一）
+ *
+ * 三页共用：一次性拉取 GET /iot/instance/tree（已按数据权限裁剪，无权节点不返回），
+ * 前端只做树结构派生（子节点/类型筛选），不再二次过滤权限。加载失败走 error 态，
+ * 空树走 empty 态——两种都不造假数据。
+ */
+import { ref } from 'vue';
+import { displayApi, type DisplayTreeNode } from '/@/api/display/display-api';
+import { OtLoading } from '/@/components/framework/ot-loading';
+
+export function useDisplayTree() {
+  const tree = ref<DisplayTreeNode[]>([]);
+  const loading = ref(false);
+  const error = ref('');
+  const loaded = ref(false);
+
+  async function loadTree(force = false) {
+    if (loaded.value && !force) return;
+    loading.value = true;
+    error.value = '';
+    OtLoading.show();
+    try {
+      tree.value = await displayApi.queryInstanceTree();
+      loaded.value = true;
+    } catch (e) {
+      tree.value = [];
+      error.value = '模型树加载失败，请重试';
+      console.error('[display] load instance tree failed:', e);
+    } finally {
+      loading.value = false;
+      OtLoading.hide();
+    }
+  }
+
+  return { tree, loading, error, loaded, loadTree };
+}
