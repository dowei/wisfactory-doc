+    }

+    }

+        pendingParamChangeTs.put(key, ts);
+    }

+    public Result evaluate(Input in) {
+        return evaluate(in, in.expectedPeriodMs(), in.maxRange(), in.paramChangeEvent());
+    }

+     * 差分求值（每 key 独立基线）。
+     *
+     * @param expectedPeriodMs 期望采样周期（毫秒）；非空且实际间隔超容差 → 跨周期打码 9
+     * @param maxRange        满量程上限；非空且差值超过 → 溢出归零 + 码 1
+     * @param paramChangeEvent 本次样本携带的换表事件标记（与 markParamChange 等价）
+     */

+        String key = in.key();
+        Baseline prev = baselines.get(key);
+        long pendingTs = pendingParamChangeTs.getOrDefault(key, Long.MIN_VALUE);
+
+        // 换表分段：PARAM 变更发生在上次基线之后 → 本样本只建新表底数，跨段不差分。
+        boolean paramBoundary = paramChangeEvent
+                || (prev != null && pendingTs != Long.MIN_VALUE && pendingTs >= prev.ts() && pendingTs < in.ts());
+        if (paramBoundary) {
+            baselines.put(key, new Baseline(in.value(), in.ts()));
+            pendingParamChangeTs.remove(key);
+            return new Result(0.0, 0, Kind.NEW_BASELINE, "换表PARAM分段边界，新表底数建基线", in.ts());
+        }
+        if (prev == null) {
+            baselines.put(key, new Baseline(in.value(), in.ts()));
+            return new Result(0.0, 0, Kind.NEW_BASELINE, "首样本建基线", in.ts());
+        }
+        if (in.quality() != 0) {
+            return new Result(0.0, in.quality(), Kind.BAD_INPUT, "上游质量码非0，不差分", in.ts());
+        }
+        long gap = in.ts() - prev.ts();
+        if (expectedPeriodMs != null && expectedPeriodMs > 0
+                && gap > expectedPeriodMs * GAP_TOLERANCE) {
+            // 跨周期：以当前样本重建基线（恢复后不跨缺口），打码 9 不冒充单周期流量
+            baselines.put(key, new Baseline(in.value(), in.ts()));
+            return new Result(0.0, 9, Kind.CROSS_PERIOD, "采样窗不连续（跨周期）不差分", in.ts());
+        }
+        double diff = in.value() - prev.value();
+        if (maxRange != null && maxRange > 0 && Math.abs(diff) > maxRange) {
+            baselines.put(key, new Baseline(in.value(), in.ts()));
+            return new Result(0.0, 1, Kind.OVERFLOW, "差值超量程，溢出归零", in.ts());
+        }
+        if (diff < 0) {
+            // 回退：不写负值（值 0），打码 8 提示人工确认（CONV-02 / REQ-0019 验收 2）
+            baselines.put(key, new Baseline(in.value(), in.ts()));
+            return new Result(0.0, 8, Kind.BACKFLOW, "读数回退（差值负）不写负值，打码待确认", in.ts());
+        }
+        baselines.put(key, new Baseline(in.value(), in.ts()));
+        return new Result(diff, 0, Kind.NORMAL, "", in.ts());
+    }

+/**
+ * INTEGRAL 原生梯形积分算子（TASK-0042 brief 决策 2；REQ-0019；ADR-0010）：
+ * 瞬时流量 → 累计量；按相邻样本**真实 ts 间隔**梯形积分逐段累加（非固定周期）；
+ * 缺口不积分（质量码非 0 / 样本缺失的间隔跳过，禁止用最后流量填满缺口，计入完整度）；
+ * 停机零增量（正常质量、流量=0 → quality=0 的零增量）与断线（GapSkipped + 质量码）严格区分。
+ *
+ * <p>ADR-0010：INTEGRAL 不开放用户手写——仅随能源计量预置公式提供（isHandwritable=false，
+ * 供 FormulaExpressionValidator 白名单判定）。原生实现、不走表达式引擎。
+ *
+ * <p>线程安全：状态按 key 存于 ConcurrentHashMap。
+ *
+ * @author dowei
+ * @date 2026-11-24
+ */
+public final class IntegralOperator {
+
+    /** 样本（value=瞬时流量 / quality=上游质量 / ts=真实采样时刻） */
+    public record Sample(String key, double value, long ts, int quality) {
+        public Sample {
+            java.util.Objects.requireNonNull(key, "key");
+        }
+    }
+
+    public enum Kind {
+        INITIAL, INTEGRATED, GAP_SKIPPED, STOPPED_ZERO
+    }
+
+    /** increment=本段增量；value=累计量；quality=本段质量 */
+    public record Result(double increment, double value, int quality, Kind kind, long ts) {
+    }
+
+    private record LastSample(double value, long ts, int quality) {
+    }
+
+    private final Map<String, Accum> accums = new ConcurrentHashMap<>();
+
+    private static final class Accum {
+        LastSample last;
+        double total;
+    }
+
+    /** ADR-0010：INTEGRAL 仅随预置公式提供，不开放手写 */
+    public boolean isHandwritable() {
+        return false;
+    }
+
+    /** 累计量查询（观测/同构写用），无样本返回 0 */
+    public double total(String key) {
+        Accum a = accums.get(key);
+        return a == null ? 0 : a.total;
+    }
+
+    public void reset(String key) {
+        accums.remove(key);
+    }
+
+    /**
+     * 接纳一个样本并返回本段积分结果。
+     *
+     * <p>规则：首样本仅建基线（INITIAL 增量 0）；当前样本质量非 0 或距上个样本质量非 0 →
+     * 该段不积分（GAP_SKIPPED，累计不变），并以当前样本重建基线（防跨缺口累计）；
+     * 当前值=0 且上值=0（稳态停机）→ STOPPED_ZERO（quality=0，正常零增量）；
+     * 其余按真实 ts 间隔梯形积分 ((v0+v1)/2)*dt（dt 秒）。
+     */
+    public Result accept(Sample sample) {
+        Accum a = accums.computeIfAbsent(sample.key(), k -> new Accum());
+        LastSample prev = a.last;
+        if (prev == null) {
+            a.last = new LastSample(sample.value(), sample.ts(), sample.quality());
+            return new Result(0.0, a.total, sample.quality(), Kind.INITIAL, sample.ts());
+        }
+        if (sample.quality() != 0 || prev.quality() != 0) {
+            a.last = new LastSample(sample.value(), sample.ts(), sample.quality());
+            return new Result(0.0, a.total, sample.quality() != 0 ? sample.quality() : prev.quality(),
+                    Kind.GAP_SKIPPED, sample.ts());
+        }
+        long deltaMs = sample.ts() - prev.ts();
+        if (deltaMs <= 0) {
+            a.last = new LastSample(sample.value(), sample.ts(), sample.quality());
+            return new Result(0.0, a.total, 0, Kind.INITIAL, sample.ts());
+        }
+        double dtSec = deltaMs / 1000.0;
+        double increment = (prev.value() + sample.value()) / 2.0 * dtSec;
+        a.last = new LastSample(sample.value(), sample.ts(), sample.quality());
+        Kind kind = (sample.value() == 0.0 && prev.value() == 0.0) ? Kind.STOPPED_ZERO : Kind.INTEGRATED;
+        a.total += increment;
+        return new Result(increment, a.total, 0, kind, sample.ts());
+    }

+ *
+ * <p>失败语义（brief 决策 5）：连接/命令失败只 WARN、不上抛——不阻塞计算；
+ * 点位缓存是查询结果，写失败不影响 st_item_value 主存储（真源）。
+ *
+ * @author dowei
+ * @date 2026-11-24
+ */
+public final class RedisItemValueCache implements HomogeneousResultSink.ItemValueCache {
+
+    private static final Logger log = LoggerFactory.getLogger(RedisItemValueCache.class);
+
+    private final String host;
+    private final int port;
+    private final String password;
+    private final int database;
+    private final String tenantCode;
+
+    public RedisItemValueCache(String addr, String password, int database, String tenantCode) {
+        String[] parts = (addr == null || addr.isBlank() ? "127.0.0.1:6379" : addr).split(":");
+        this.host = parts[0];
+        this.port = parts.length > 1 ? Integer.parseInt(parts[1]) : 6379;
+        this.password = password == null ? "" : password;
+        this.database = database;
+        this.tenantCode = tenantCode == null ? "" : tenantCode;
+    }
+
+    @Override
+    public void put(ItemValue v) {
+        String key = keyOf(tenantCode, v.fullKey());
+        String json = jsonOf(v);
+        try (Socket s = new Socket()) {
+            s.connect(new InetSocketAddress(host, port), 2000);
+            s.setSoTimeout(2000);
+            OutputStream out = s.getOutputStream();
+            Resp2Reply in = new Resp2Reply(s.getInputStream());
+            if (!password.isBlank()) {
+                send(out, "AUTH", password);
+                in.read();
+            }
+            if (database > 0) {
+                send(out, "SELECT", String.valueOf(database));
+                in.read();
+            }
+            send(out, "SET", key, json);
+            String reply = in.read();
+            if (!"OK".equalsIgnoreCase(reply)) {
+                log.warn("[CALC-STORAGE] Redis 点位缓存 SET 非 OK key={} reply={}", key, reply);
+            }
+        } catch (IOException | RuntimeException e) {
+            log.warn("[CALC-STORAGE] Redis 点位缓存写失败（不阻塞）key={}: {}", key, e.getMessage());
+        }
+    }
+
+    static String keyOf(String tenant, String fullKey) {
+        return "iot:item:" + tenant + ":" + fullKey;
+    }
+
+    static String jsonOf(ItemValue v) {
+        return "{\"fullKey\":\"" + esc(v.fullKey()) + "\",\"value\":" + v.value()
+                + ",\"quality\":" + v.quality() + ",\"ts\":" + v.ts() + "}";
+    }
+
+    private static String esc(String s) {
+        return s == null ? "" : s.replace("\\", "\\\\").replace("\"", "\\\"");
+    }
+
+    private static void send(OutputStream out, String... args) throws IOException {
+        StringBuilder sb = new StringBuilder();
+        sb.append('*').append(args.length).append("\r\n");
+        for (String arg : args) {
+            byte[] b = arg.getBytes(StandardCharsets.UTF_8);
+            sb.append('$').append(b.length).append("\r\n").append(arg).append("\r\n");
+        }
+        out.write(sb.toString().getBytes(StandardCharsets.UTF_8));
+        out.flush();
+    }
+
+    /** 字节级 RESP2 回复（0041/0049 同款帧解析） */
+    private static final class Resp2Reply {
+        private final InputStream in;
+
+        Resp2Reply(InputStream in) {
+            this.in = in;
+        }
+
+        String read() throws IOException {
+            int b = in.read();
+            if (b < 0) {
+                throw new IOException("连接关闭");
+            }
+            return switch (b) {
+                case '+' -> line();
+                case '-' -> throw new IOException("Redis 错误: " + line());
+                case ':' -> line();
+                case '$' -> {
+                    int len = Integer.parseInt(line());
+                    if (len < 0) {
+                        yield "";
+                    }
+                    byte[] buf = new byte[len];
+                    int off = 0;
+                    while (off < buf.length) {
+                        int n = in.read(buf, off, buf.length - off);
+                        if (n < 0) {
+                            throw new IOException("RESP 批量串截断");
+                        }
+                        off += n;
+                    }
+                    in.read();
+                    in.read();
+                    yield new String(buf, StandardCharsets.UTF_8);
+                }
+                default -> throw new IOException("RESP 类型非法: " + (char) b);
+            };
+        }
+
+        private String line() throws IOException {
+            ByteArrayOutputStream bout = new ByteArrayOutputStream();
+            int c;
+            while ((c = in.read()) >= 0 && c != '\r') {
+                bout.write(c);
+            }
+            in.read();
+            return bout.toString(StandardCharsets.UTF_8);
+        }
+    }

+    public static TdengineItemValueWriter rest(String baseUrl, String tenantCode, String user, String password) {
+        HttpClient client = HttpClient.newBuilder().connectTimeout(Duration.ofSeconds(5)).build();
+        String auth = "Basic " + Base64.getEncoder()
+                .encodeToString((user + ":" + password).getBytes(StandardCharsets.UTF_8));
+        SqlRunner runner = sql -> {
+            HttpRequest req = HttpRequest.newBuilder()
+                    .uri(URI.create(baseUrl + "/rest/sql"))
+                    .timeout(Duration.ofSeconds(60))
+                    .header("Authorization", auth)
+                    .header("Content-Type", "text/plain")
+                    .POST(HttpRequest.BodyPublishers.ofString(sql, StandardCharsets.UTF_8))
+                    .build();
+            HttpResponse<String> resp = client.send(req, HttpResponse.BodyHandlers.ofString(StandardCharsets.UTF_8));
+            if (resp.statusCode() >= 400) {
+                throw new ItemWriteException("TDengine REST HTTP " + resp.statusCode(), null);

+        }
+    }

+                + v.instanceCode() + "', '" + v.dataCode() + "', '"

+    private static String[] tagsOf(String sql) {
+        int a = sql.indexOf("TAGS (");
+        int b = sql.indexOf(')', a);
+        String inside = sql.substring(a + "TAGS (".length(), b);
+        String[] raw = inside.split(",", -1);
+        String[] out = new String[raw.length];
+        for (int i = 0; i < raw.length; i++) {
+            out[i] = raw[i].trim().replaceAll("^'|'$", "");
+        }
+        return out;
+    }
+
+    private static int countSemicolons(String s) {
+        int n = 0;
+        for (char c : s.toCharArray()) {
+            if (c == ';') n++;
+        }
+        return n;
+    }
