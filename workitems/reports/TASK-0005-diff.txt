+import java.util.Locale;
+import java.util.Set;

+/**
+ * 越权拒绝策略常量与接口形态判定辅助（TASK-0005，ADR-0016 裁决 2）。
+ *
+ * <p>裁决 2 定版：空结果 vs 403 按接口形态判定而非逐个枚举——GET 集合类（/list、/tree、
+ * /history、/curve、/realtime 等集合语义）越权返回空集（SCOPE_EMPTY）；GET 单对象类
+ * （/{id}、/{code} 详情）越权返回 403（SCOPE_FORBIDDEN）。接口实现按形态选用常量，
+ * 未来新增接口自动归类，无需维护接口清单。
+ *
+ * <p>本任务只提供常量与判定辅助，不接线任何端点；最终选择权在接口实现
+ * （TASK-0007 红线用例按本裁决构造）。
+ *
+ * @author dowei
+ * @date 2026-11-22
+ */
+public final class DataPermAccessStrategy {
+
+    /**
+     * 越权拒绝策略（二选一，按接口形态）
+     */
+    public enum DenyStrategy {
+        /** 集合语义接口：越权返回空集（数据不存在口径） */
+        SCOPE_EMPTY,
+        /** 单对象语义接口：越权返回 403（4xx 错误码口径，含数据权限不命中） */
+        SCOPE_FORBIDDEN
+    }
+
+    /**
+     * 集合语义尾段词表（裁决 2 五例 + 同族动词）。这是<b>形态词表</b>而非接口清单——
+     * 新增接口只要尾段落在集合语义即自动归 SCOPE_EMPTY，无需登记。
+     */
+    private static final Set<String> COLLECTION_TAIL_KEYWORDS = Set.of(
+            "list", "tree", "history", "curve", "realtime",
+            "page", "export", "count", "stat", "stats", "options");
+
+    private DataPermAccessStrategy() {
+    }
+
+    /**
+     * 按接口形态判定是否单对象（详情类）访问——组件提供的判定辅助，最终选择权在接口实现。
+     *
+     * <p>判定规则：取路径最后一个非空段（剥离查询串、大小写不敏感）——
+     * 命中集合语义词表 → false（SCOPE_EMPTY）；其余（数字 id、路径式编码、占位符、
+     * 无法识别形态）→ true（SCOPE_FORBIDDEN，fail-closed：403 是更明确的拒绝方向）。
+     *
+     * @param resource 接口路径（如 /api/v1/iot/instance/{code}），允许 null/空白（按详情类处理）
+     */
+    public static boolean isDetailAccess(String resource) {
+        String tail = tailSegment(resource);
+        if (tail == null) {
+            return true;
+        }
+        return !COLLECTION_TAIL_KEYWORDS.contains(tail);
+    }
+
+    /**
+     * 剥离查询串后取最后一个非空路径段，小写归一；无可用段返回 null
+     */
+    private static String tailSegment(String resource) {
+        if (resource == null) {
+            return null;
+        }
+        int query = resource.indexOf('?');
+        String path = query >= 0 ? resource.substring(0, query) : resource;
+        String[] segments = path.split("/");
+        for (int i = segments.length - 1; i >= 0; i--) {
+            String segment = segments[i].trim();
+            if (!segment.isEmpty()) {
+                return segment.toLowerCase(Locale.ROOT);
+            }
+        }
+        return null;
+    }
+}

+
+import org.springframework.stereotype.Component;
+
+import java.util.ArrayList;
+import java.util.List;
+import java.util.Set;
+import java.util.stream.Collectors;

+/**
+ * 步骤 5：查询条件拼装（TASK-0005，ADR-0016 裁决 2/3/4/6）。
+ *
+ * <p>按表类别三态，渲染结果为<b>条件片段</b>（不含 WHERE）供消费方拼 SQL / 传入
+ * MyBatis-Plus wrapper apply：
+ * <ul>
+ * <li><b>配置类</b>（iot_ 配置表管理面列表：字典/模板/实例/点位/网关/通道/指标等）——
+ *     {@code instance_id IN (可见集)}，存在 scope=4 角色时追加 {@code OR create_by = 当前用户}
+ *     分支（裁决 3/4：最终条件 = (instance_id ∈ 并集) OR (create_by=me，仅配置类)）；
+ *     无实例锚点的表（如字典）传 null 列名，只承载 create_by 语义；</li>
+ * <li><b>点位类</b>——{@code dept_id IN (部门集) OR bound_instance_id IN (可见集)}；
+ *     点位列表同时属配置类对象，scope=4 追加 create_by 分支（裁决 3）；</li>
+ * <li><b>时序类</b>（数据项/指标/曲线查询）——仅 {@code instance_id IN (可见集)}；
+ *     scope=4 无贡献（时序值没有 create_by，TD-0044）。</li>
+ * </ul>
+ *
+ * <p>返回值约定：{@code null} = allTenant 无需追加条件；{@code "1 = 0"} = 无任何可见性
+ * （fail-closed，返回空列表而非全量）；其余为条件片段。
+ *
+ * <p>NULL 归类（裁决 6）为结构性满足：只渲染 IN 比较、绝不渲染 IS NULL/IS NOT NULL
+ * 分支——SQL NULL 不命中任何 IN，未绑部门行对非 scope=1 天然不可见，不做特判。
+ *
+ * @author dowei
+ * @date 2026-11-22
+ */
+@Component
+public class DataPermConditionAssembler {
+
+    /** 零可见性 fail-closed 片段：恒假条件，消费方直接 AND 拼接 */
+    public static final String FAIL_CLOSED = "1 = 0";
+
+    private static final String OR = " OR ";
+
+    /**
+     * 配置类条件：{@code instance_id IN (并集)} [OR {@code create_by = userId}]。
+     *
+     * @param scope            resolve 产物
+     * @param userId           当前用户 id（create_by 分支取值；中台场景无本人语义传 null）
+     * @param instanceIdColumn 实例 id 列名（如 "instance_id"；实例表自身传 "id"；
+     *                         无实例锚点的表传 null——只承载 create_by 语义）
+     * @param createByColumn   创建人列名（如 "create_by"）
+     */
+    public String configCondition(VisibleScope scope, Long userId, String instanceIdColumn, String createByColumn) {
+        if (scope == null) {
+            return FAIL_CLOSED;
+        }
+        if (scope.allTenant()) {
+            return null;
+        }
+        List<String> disjuncts = new ArrayList<>();
+        if (instanceIdColumn != null && !scope.instanceIds().isEmpty()) {
+            disjuncts.add(instanceIdColumn + " IN (" + joinSorted(scope.instanceIds()) + ")");
+        }
+        if (createByColumn != null && scope.createByOnly() && userId != null) {
+            disjuncts.add(createByColumn + " = " + userId);
+        }
+        // 有实例锚点的表：零可见性 fail-closed；无锚点表（列名 null）不产出任何数据权限条件
+        return joinDisjuncts(disjuncts, instanceIdColumn != null);
+    }
+
+    /**
+     * 点位类条件：{@code dept_id IN (部门集)} OR {@code bound_instance_id IN (可见集)}
+     * [OR {@code create_by = userId}]。
+     *
+     * <p>两个 IN 列均无 IS NULL 特判（裁决 6）：scope=1 全租户不做过滤（未绑部门行对
+     * 租户管理员可见）；其余 scope 下 NULL 行两条件皆不命中=不可见（fail-closed）。
+     *
+     * @param scope                resolve 产物
+     * @param userId               当前用户 id（scope=4 点位列表按 create_by 过滤，裁决 3）
+     * @param deptIdColumn         部门列名（如 "dept_id"）
+     * @param boundInstanceIdColumn 绑定实例列名（如 "bound_instance_id"）
+     * @param createByColumn       创建人列名（如 "create_by"）
+     */
+    public String pointCondition(VisibleScope scope, Long userId, String deptIdColumn,
+                                 String boundInstanceIdColumn, String createByColumn) {
+        if (scope == null) {
+            return FAIL_CLOSED;
+        }
+        if (scope.allTenant()) {
+            return null;
+        }
+        List<String> disjuncts = new ArrayList<>();
+        if (deptIdColumn != null && !scope.deptIds().isEmpty()) {
+            disjuncts.add(deptIdColumn + " IN (" + joinSorted(scope.deptIds()) + ")");
+        }
+        if (boundInstanceIdColumn != null && !scope.instanceIds().isEmpty()) {
+            disjuncts.add(boundInstanceIdColumn + " IN (" + joinSorted(scope.instanceIds()) + ")");
+        }
+        if (createByColumn != null && scope.createByOnly() && userId != null) {
+            disjuncts.add(createByColumn + " = " + userId);
+        }
+        return joinDisjuncts(disjuncts, true);
+    }
+
+    /**
+     * 时序类条件：仅 {@code instance_id IN (可见集)}——scope=4 无贡献（不追加 create_by，
+     * TD-0044：时序值没有 create_by，硬套只会出错）。
+     *
+     * @param scope            resolve 产物
+     * @param instanceIdColumn 实例 id 列名（如 "instance_id"）
+     */
+    public String timeseriesCondition(VisibleScope scope, String instanceIdColumn) {
+        if (scope == null) {
+            return FAIL_CLOSED;
+        }
+        if (scope.allTenant()) {
+            return null;
+        }
+        if (instanceIdColumn == null || scope.instanceIds().isEmpty()) {
+            return FAIL_CLOSED;
+        }
+        return instanceIdColumn + " IN (" + joinSorted(scope.instanceIds()) + ")";
+    }
+
+    /**
+     * 单分支裸片段（供 AND 拼接）；多分支加括号按 OR 合并；零分支按 failClosed 参数
+     * 返回 {@code 1 = 0}（有锚点表）或 null（无锚点表不承载权限过滤）
+     */
+    private String joinDisjuncts(List<String> disjuncts, boolean failClosedWhenEmpty) {
+        if (disjuncts.isEmpty()) {
+            return failClosedWhenEmpty ? FAIL_CLOSED : null;
+        }
+        if (disjuncts.size() == 1) {
+            return disjuncts.get(0);
+        }
+        return disjuncts.stream().collect(Collectors.joining(OR, "(", ")"));
+    }
+
+    /**
+     * id 升序渲染（确定性输出，利于 SQL 复用与断言）
+     */
+    private String joinSorted(Set<Long> ids) {
+        return ids.stream().sorted().map(String::valueOf).collect(Collectors.joining(","));
+    }
+}

+    @Override
+    public VisibleScope resolveClientScope(Long clientId, String scopeJson) {
+        ClientScopeSpec spec = parseClientScope(scopeJson);
+        if (spec == null || spec.instanceIds() == null || spec.instanceIds().isEmpty()) {
+            // 裁决 5：scope 为 null 或空数组 = 无任何数据权限（默认拒绝，与「为空表示无权限」一致）
+            return VisibleScope.denied();
+        }
+        List<Long> scopeIds = spec.instanceIds().stream().distinct().toList();
+        if (!spec.subtree()) {

+            List<DataPermInstanceRow> rows = instanceMapper.selectByIds(scopeIds);
+            return exactScope(rows);
+        }
+        // subtree=true：与登录用户完全相同的步骤 3-5——scope 实例为锚点，ancestors 展开全部后代（含上限截断）
+        List<DataPermInstanceRow> rows = instanceMapper.selectAllForPerm();
+        Set<Long> anchorIds = rows.stream()
+                .filter(r -> scopeIds.contains(r.getId()))
+                .map(DataPermInstanceRow::getId)
+                .collect(Collectors.toCollection(LinkedHashSet::new));
+        if (anchorIds.isEmpty()) {

+        }

+        return exactScope(rows.stream().filter(r -> visibleIds.contains(r.getId())).toList());
+    }

+                .collect(Collectors.toCollection(LinkedHashSet::new));
+        Set<Long> deptIds = rows.stream()
+                .map(DataPermInstanceRow::getDeptId)
+                .filter(d -> d != null)
+                .collect(Collectors.toCollection(LinkedHashSet::new));
+        return new VisibleScope(false, deptIds, ids, false);
+    }

+    private ClientScopeSpec parseClientScope(String scopeJson) {
+        if (scopeJson == null || scopeJson.isBlank()) {
+            return null;
+        }
+        try {
+            return SCOPE_JSON.readValue(scopeJson, ClientScopeSpec.class);

+            log.warn("[DATAPERM] client scope invalid, deny by default: {}", e.getMessage());
+            return null;
+        }
+    }

+     * 步骤 3：部门集合 → 实例锚点（dept_id 命中，通常是工厂/车间节点）
+     */
+    private Set<Long> resolveInstancesByDept(Set<Long> deptIds, Long userId) {
+        List<DataPermInstanceRow> rows = instanceMapper.selectAllForPerm();
+        Set<Long> anchorIds = new LinkedHashSet<>();
+        for (DataPermInstanceRow row : rows) {
+            // dept_id=NULL 不命中锚点（结构性 fail-closed，裁决 6：不做特判）
+            if (row.getDeptId() != null && deptIds.contains(row.getDeptId())) {
+                anchorIds.add(row.getId());
+            }
+        }
+        if (anchorIds.isEmpty()) {
+            return Set.of();
+        }
+        return expandSubtree(rows, anchorIds, userId);
+    }

+    /**
+     * 步骤 4：锚点 ancestors 前缀展开全部后代实例，并按上限截断（ADR-0016 裁决 1）。
+     *
+     * <p>展开语义 = 元素级路径前缀匹配：后代实例的 ancestors 逗号路径中<b>整 token</b>
+     * 包含锚点 id（「0,1,11」命中锚点 1，「0,11」不命中——防字符串前缀误判），
+     * 与 sys_dept.ancestors 同范式，不递归查库。锚点本身计入可见集合。
+     *
+     * <p>截断按 ancestors 路径序（深度升序，同深按 id）保近端实例：路径序下祖先必先于
+     * 后代，截断保留的前缀对祖先闭包——不会出现「后代可见而其唯一锚点被截掉」的裂缝。
+     */
+    private Set<Long> expandSubtree(List<DataPermInstanceRow> rows, Set<Long> anchorIds, Long subjectId) {
+        List<DataPermInstanceRow> visibleRows = new ArrayList<>();
+        Set<String> anchorTokens = anchorIds.stream().map(String::valueOf).collect(Collectors.toSet());
+        for (DataPermInstanceRow row : rows) {
+            if (anchorIds.contains(row.getId()) || ancestorsHitsAny(row.getAncestors(), anchorTokens)) {
+                visibleRows.add(row);
+            }
+        }
+        int cap = properties.getSubtreeMax();
+        if (cap > 0 && visibleRows.size() > cap) {
+            visibleRows.sort(Comparator
+                    .comparingInt((DataPermInstanceRow r) -> ancestorsDepth(r.getAncestors()))
+                    .thenComparing(DataPermInstanceRow::getId));
+            visibleRows = visibleRows.subList(0, cap);
+            log.warn("[DATAPERM] subtree truncated user={} kept={} cap={}", subjectId, visibleRows.size(), cap);
+        }
+        return visibleRows.stream().map(DataPermInstanceRow::getId)
+                .collect(Collectors.toCollection(LinkedHashSet::new));
+    }
+
+    /**
+     * ancestors 逗号路径是否整 token 包含指定部门 id（元素级匹配，防前缀误判）
+     */
+    private static boolean ancestorsContains(String ancestors, Long deptId) {
+        if (ancestors == null || ancestors.isBlank()) {
+            return false;
+        }
+        String token = String.valueOf(deptId);
+        for (String part : ancestors.split(",")) {
+            if (part.trim().equals(token)) {
+                return true;
+            }
+        }
+        return false;
+    }
+
+    /**
+     * ancestors 逗号路径是否整 token 命中任一锚点（展开用；token 预转字符串集合）
+     */
+    private static boolean ancestorsHitsAny(String ancestors, Set<String> anchorTokens) {
+        if (ancestors == null || ancestors.isBlank()) {
+            return false;
+        }
+        for (String part : ancestors.split(",")) {
+            if (anchorTokens.contains(part.trim())) {
+                return true;
+            }
+        }
+        return false;
+    }
+
+    /**
+     * ancestors 路径深度（元素数）：排序键，路径越浅实例越靠近根（近端）
+     */
+    private static int ancestorsDepth(String ancestors) {
+        if (ancestors == null || ancestors.isBlank()) {
+            return 0;
+        }
+        return ancestors.split(",").length;
+    }
+
+    private static boolean isScope(DataPermRoleRow role, DataScopeEnum expected) {
+        return role.getDataScope() != null && role.getDataScope().equals(expected.getValue());
+    }

+ * 渲染结果为条件片段供消费方拼 SQL；编码值做单引号转义防注入。

+ * @author dowei

+class DataPermTagConditionRendererTest {
+
+    @Mock
+    private DataPermInstanceMapper instanceMapper;
+
+    private TdengineTagConditionRenderer renderer;
+
+    private AutoCloseable mocks;
+
+    @BeforeEach
+    void setUp() {
+        mocks = MockitoAnnotations.openMocks(this);
+        renderer = new TdengineTagConditionRenderer(instanceMapper);
+    }
+
+    private DataPermInstanceRow inst(Long id, String code, Long deptId, String ancestors) {
+        DataPermInstanceRow row = new DataPermInstanceRow();
+        row.setId(id);
+        row.setInstanceCode(code);
+        row.setDeptId(deptId);
+        row.setAncestors(ancestors);
+        return row;

+    // ---------- 时序语义下推（指标/曲线/历史：instance_key IN） ----------
+
+    @Test
+    @DisplayName("时序下推 - 标签名 instance_key，编码升序渲染、单引号包裹")
+    void testTimeseries_InstanceKeyTag_SortedCodes() {
+        when(instanceMapper.selectByIds(anyCollection())).thenReturn(List.of(
+                inst(1L, "F01", 10L, "0"),
+                inst(2L, "F01.WS01", 20L, "0,1")));
+
+        String cond = renderer.renderTimeseriesCondition(
+                new VisibleScope(false, Set.of(10L), Set.of(2L, 1L), false));

+    @Test
+    @DisplayName("时序下推 - 全租户返回 null（库级租户隔离已由库名保证，无需标签过滤）")

+    }
+
+    @Test
+    @DisplayName("时序下推 - 仅本人角色：不追加 create_by（TDengine 时序无 create_by 标签，scope=4 无贡献）")

+                inst(2L, "F01.WS01", 20L, "0,1")));

+        String cond = renderer.renderTimeseriesCondition(

+    @Test

+                inst(1L, "F01", 10L, "0"),

+        when(instanceMapper.selectByIds(anyCollection())).thenReturn(List.of(
