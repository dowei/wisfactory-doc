+            bytes = snapshotStorage.readBytes(row.getSnapshotUri());

+                    row.getTenantId(), row.getScopeType(), row.getScopeId(), row.getVersion(), row.getSnapshotUri(), e);

+        }

+            log.error("[SNAPSHOT] snapshot file missing tenant={} scope={}-{} version={} uri={}",
+                    row.getTenantId(), row.getScopeType(), row.getScopeId(), row.getVersion(), row.getSnapshotUri());
+            throw new BusinessException(ResultCode.ERROR.getCode(),
+                    "快照文件缺失（存储损坏探测，见服务端告警日志）");
+        }
+        String actualHash = SnapshotStorage.sha256Hex(bytes);

+                    row.getVersion(), row.getSnapshotHash(), actualHash);

+        }
+        pointMapper.update(null, new LambdaUpdateWrapper<IotPoint>()
+                .set(IotPoint::getConfigVersion, version)
+                .in(IotPoint::getId, pointIds));
+    }

+    /** 从快照体 JSON 提取对象数组 id 列（points/dataItems；数组缺省/空返回空集） */
+    private List<Long> idArrayOf(byte[] snapshotBytes, String arrayKey) {
+        try {
+            com.fasterxml.jackson.databind.JsonNode body =
+                    BODY_MAPPER.readTree(new String(snapshotBytes, StandardCharsets.UTF_8));
+            com.fasterxml.jackson.databind.JsonNode array = body.get(arrayKey);
+            if (array == null || !array.isArray()) {
+                return List.of();
+            }
+            List<Long> ids = new java.util.ArrayList<>();
+            for (com.fasterxml.jackson.databind.JsonNode element : array) {
+                if (element.isObject() && element.has("id")) {
+                    ids.add(element.get("id").asLong());
+                }
+            }
+            return ids;
+        } catch (IOException e) {
+            throw new BusinessException(ResultCode.ERROR.getCode(),
+                    "回滚目标快照解析失败: " + e.getMessage());
+        }
+    }

+    private void assembleGatewayScope(ConfigSnapshot snapshot, Long gatewayId, Long version) {
+        // 通道：该网关全部通道（deleted 由 @TableLogic 过滤、tenant 由插件注入）
+        List<IotChannel> channels = channelMapper.selectList(new LambdaQueryWrapper<IotChannel>()
+                .eq(IotChannel::getGatewayId, gatewayId)
+                .orderByAsc(IotChannel::getId));

+        List<Long> channelIds = channels.stream().map(IotChannel::getId).toList();

+        Map<Long, List<IotChannelCredential>> credentialsByChannel = loadEnabledCredentials(channelIds);
+        for (IotChannel channel : sortById(channels, IotChannel::getId)) {
+            channelObjects.add(toChannelObject(channel, credentialsByChannel.get(channel.getId())));
+        }
+        snapshot.setChannels(channelObjects);

+        // 点位：该网关全部通道下的点位（经通道挂网关）；无通道（手工/虚拟点位）不属网关范围

+                .in(IotPoint::getChannelId, channelIds)

+    /** 启用凭据行（status=1）按通道分组——§1.3A 派生标记来源；各列本体不进快照 */
+    private Map<Long, List<IotChannelCredential>> loadEnabledCredentials(List<Long> channelIds) {
+        if (channelIds.isEmpty()) {
+            return Map.of();
+        }
+        List<IotChannelCredential> credentials = credentialMapper.selectList(
+                new LambdaQueryWrapper<IotChannelCredential>()
+                        .in(IotChannelCredential::getChannelId, channelIds)
+                        .eq(IotChannelCredential::getStatus, 1));
+        Map<Long, List<IotChannelCredential>> grouped = new LinkedHashMap<>();
+        for (IotChannelCredential cred : credentials) {
+            grouped.computeIfAbsent(cred.getChannelId(), k -> new ArrayList<>()).add(cred);
+        }
+        return grouped;
+    }

+                .orderByAsc(IotDataItem::getId));
+        List<IotDataItemRef> refs = dataItemRefMapper.selectList(new LambdaQueryWrapper<IotDataItemRef>()
+                .orderByAsc(IotDataItemRef::getId));
+        List<IotMetric> metrics = metricMapper.selectList(new LambdaQueryWrapper<IotMetric>()
+                .orderByAsc(IotMetric::getId));

+        List<SnapshotDataItem> itemObjects = new ArrayList<>();

+            itemObjects.add(toDataItemObject(item, version));

+        List<SnapshotFormula> formulaObjects = new ArrayList<>();

+        }

+        }
+        snapshot.setDataItems(itemObjects);
+        snapshot.setFormulas(formulaObjects);
+        snapshot.setMetrics(metricObjects);
+    }

+    /** 行序按 id 升序兜底（DB 侧 ORDER BY id；确定性序列化的行序来源） */
+    private <T> List<T> sortById(List<T> rows, java.util.function.Function<T, Long> idGetter) {
+        if (rows == null) {
+            return List.of();
+        }
+        return rows.stream()
+                .sorted(java.util.Comparator.comparing(idGetter,
+                        java.util.Comparator.nullsLast(java.util.Comparator.naturalOrder())))
+                .toList();
+    }

+     * @throws BusinessException PARAM_ERROR——顶层键越集 / 对象键越集 / 凭据关键词键 /

+        try {
+            root = MAPPER.readTree(snapshotJson);
+        } catch (JsonProcessingException e) {
+            throw new BusinessException(ResultCode.PARAM_ERROR.getCode(),

+            }

+            throw new BusinessException(ResultCode.PARAM_ERROR.getCode(),
+                    "快照终检失败：快照体出现 Schema v1.0.1 之外顶层键: " + String.join("、", topLevelViolations));
+        }

+        // ② 五对象数组元素键白名单（按容器键取对应 Schema 键集）
+        checkObjectArray(root, "points", POINT_KEYS);
+        checkObjectArray(root, "channels", CHANNEL_KEYS);
+        checkObjectArray(root, "dataItems", DATA_ITEM_KEYS);
+        checkObjectArray(root, "formulas", FORMULA_KEYS);
+        checkObjectArray(root, "metrics", METRIC_KEYS);
+

+        Set<String> credentialKeys = new LinkedHashSet<>();
+        Set<String> credentialValues = new LinkedHashSet<>();
+        collectCredentialHits(root, "", credentialKeys, credentialValues);
+        if (!credentialKeys.isEmpty() || !credentialValues.isEmpty()) {
+            throw new BusinessException(ResultCode.PARAM_ERROR.getCode(),
+                    "快照终检失败：快照体含凭据关键词（ADR-0006：凭据独立存凭据管理，不入配置快照）: "
+                            + String.join("、", credentialKeys) + String.join("、", credentialValues));
+        }
+    }
+
+    /** 校验五对象数组元素的键 ⊆ 对应 Schema 键集 */
+    private void checkObjectArray(JsonNode root, String arrayKey, Set<String> objectKeys) {
+        JsonNode array = root.get(arrayKey);
+        if (array == null || !array.isArray()) {
+            return;
+        }
+        for (JsonNode element : array) {
+            if (!element.isObject()) {
+                throw new BusinessException(ResultCode.PARAM_ERROR.getCode(),
+                        "快照终检失败：" + arrayKey + " 数组元素必须是 JSON 对象");
+            }
+            Set<String> violations = new LinkedHashSet<>();
+            element.fieldNames().forEachRemaining(name -> {
+                if (!objectKeys.contains(name)) {
+                    violations.add(name);
+                }
+            });
+            if (!violations.isEmpty()) {
+                throw new BusinessException(ResultCode.PARAM_ERROR.getCode(),
+                        "快照终检失败：" + arrayKey + " 对象出现 Schema v1.0.1 之外字段: "
+                                + String.join("、", violations));
+            }
+        }
+    }
+
+    /**
+     * 全深度递归收集凭据命中（数组元素逐个递归——{"creds":[{"password":...}]} 类形状
+     * 不得逃逸双保险，照 0018 同款口径）；路径标注如 points[0].protocolAddr.password。
+     */
+    private void collectCredentialHits(JsonNode node, String path,
+                                       Set<String> keyHits, Set<String> valueHits) {
+        if (node.isObject()) {
+            node.fields().forEachRemaining(entry -> {
+                String name = entry.getKey();
+                String childPath = path.isEmpty() ? name : path + "." + name;
+                if (!SANCTIONED_MARKER_KEYS.contains(name)
+                        && containsAnyWord(name.toLowerCase(Locale.ROOT), FORBIDDEN_KEY_WORDS)) {
+                    keyHits.add(childPath);
+                }
+                collectCredentialHits(entry.getValue(), childPath, keyHits, valueHits);
+            });
+        } else if (node.isArray()) {
+            for (int i = 0; i < node.size(); i++) {
+                collectCredentialHits(node.get(i), path + "[" + i + "]", keyHits, valueHits);
+            }
+        } else if (node.isTextual()) {
+            String fieldName = fieldNameOf(path);
+            if (!SANCTIONED_MARKER_KEYS.contains(fieldName)
+                    && containsAnyWord(node.asText().toLowerCase(Locale.ROOT), FORBIDDEN_VALUE_WORDS)) {
+                valueHits.add(path);
+            }
+        }
+    }
+
+    /** 从路径尾段取字段名（数组下标回退——credentialTypes[0] → credentialTypes） */
+    private String fieldNameOf(String path) {
+        int dot = path.lastIndexOf('.');
+        String tail = dot >= 0 ? path.substring(dot + 1) : path;
+        int bracket = tail.indexOf('[');
+        return bracket >= 0 ? tail.substring(0, bracket) : tail;
+    }
+
+    private boolean containsAnyWord(String text, List<String> words) {
+        for (String word : words) {
+            if (text.contains(word)) {
+                return true;
+            }
+        }
+        return false;
+    }
+}

+     * @param tenantId  租户 id（子目录隔离）
+     * @param scopeType 范围 1 网关 2 计算
+     * @param scopeId   范围对象 ID（2 时为 0）
+     * @param version   版本号
+     * @param json      快照体 JSON（UTF-8）
+     * @return 相对路径 snapshot_uri（{@code <tenant_id>/<scope_type>-<scope_id>-<version>.json}）
+     * @throws IOException 目录创建/写入/落位失败
+     */
+    public String store(Long tenantId, Integer scopeType, Long scopeId, Long version, String json)
+            throws IOException {
+        Path target = resolveAbsolutePath(relativeUri(tenantId, scopeType, scopeId, version));
+        Files.createDirectories(target.getParent());
+        Path tmp = Files.createTempFile(target.getParent(), "snapshot-", ".tmp");
+        try {
+            Files.write(tmp, json.getBytes(StandardCharsets.UTF_8));
+            try {
+                Files.move(tmp, target, StandardCopyOption.REPLACE_EXISTING, StandardCopyOption.ATOMIC_MOVE);
+            } catch (IOException atomicUnsupported) {
+                // 部分文件系统不支持跨目录原子落位——降级非原子替换（仍先写临时文件，半截风险仅剩 move 窗口）
+                Files.move(tmp, target, StandardCopyOption.REPLACE_EXISTING);
+            }
+        } finally {
+            Files.deleteIfExists(tmp);
+        }
+        return relativeUri(tenantId, scopeType, scopeId, version);
+    }
+
+    /**
+     * 读取快照体字节（取回链路哈希核对用）
+     *
+     * @param relativeUri 落库 snapshot_uri（相对路径）
+     * @return 文件字节；文件不存在返回 null（取回侧转 500 快照缺失告警）
+     * @throws IOException 读取失败
+     */
+    public byte[] readBytes(String relativeUri) throws IOException {
+        Path path = resolveAbsolutePath(relativeUri);
+        if (!Files.exists(path)) {
+            return null;
+        }
+        return Files.readAllBytes(path);
+    }
+
+    /**
+     * 删除快照文件（发布落库失败的孤儿补偿，best-effort——删除失败仅告警不抛）
+     *
+     * @param relativeUri 落库 snapshot_uri（相对路径）
+     */
+    public void deleteQuietly(String relativeUri) {
+        try {
+            Files.deleteIfExists(resolveAbsolutePath(relativeUri));
+        } catch (IOException e) {
+            // best-effort 补偿：孤儿文件清理失败不影响主流程（下个同版本号发布会覆盖同名文件）
+        }
+    }
+
+    /** 相对路径 snapshot_uri（落库存值；租户子目录隔离，见类注释） */
+    public static String relativeUri(Long tenantId, Integer scopeType, Long scopeId, Long version) {
+        return tenantId + "/" + scopeType + "-" + scopeId + "-" + version + ".json";
+    }
+
+    /** 相对路径 → 绝对路径（根目录取 ot.publish.storage-path；相对根按进程工作目录解析） */
+    private Path resolveAbsolutePath(String relativeUri) {
+        Path root = Paths.get(properties.getStoragePath());
+        return root.isAbsolute() ? root.resolve(relativeUri) : Paths.get("").toAbsolutePath().resolve(root).resolve(relativeUri);
+    }
+
+    /**
+     * SHA-256 十六进制（小写 64 字符——iot_config_version.snapshot_hash varchar(64) 口径）。
+     *
+     * @param bytes 原文字节
+     * @return 64 位小写十六进制哈希
+     */
+    public static String sha256Hex(byte[] bytes) {
+        try {
+            MessageDigest digest = MessageDigest.getInstance("SHA-256");
+            byte[] hash = digest.digest(bytes);
+            StringBuilder sb = new StringBuilder(hash.length * 2);
+            for (byte b : hash) {
+                sb.append(Character.forDigit((b >> 4) & 0xF, 16));
+                sb.append(Character.forDigit(b & 0xF, 16));
+            }
+            return sb.toString();
+        } catch (NoSuchAlgorithmException e) {
+            // JDK 必带 SHA-256，不可达
+            throw new IllegalStateException("SHA-256 algorithm unavailable", e);
+        }
+    }

+    @Test
+    @DisplayName("回滚 - 新版本行 publish_status=1 复用 targetVersion 快照文件（uri/哈希同），版本单调递增，内容行回写 config_version")
+    void rollbackReusesTargetSnapshotFile() throws Exception {
+        // 准备 target 快照（v1 已发布、文件在存储；内容含点位 301——回写 id 从复用快照解析）
+        String targetJson = "{\"schemaVersion\":\"1.0.1\",\"tenantId\":1,\"scopeType\":1,\"scopeId\":88,"
+                + "\"version\":1,\"points\":[{\"id\":301}],\"channels\":[]}";
+        String targetHash = SnapshotStorage.sha256Hex(targetJson.getBytes(StandardCharsets.UTF_8));
+        String targetUri = storage.store(TENANT, 1, GATEWAY_ID, 1L, targetJson);
+        IotConfigVersion target = new IotConfigVersion();
+        target.setId(1L);
+        target.setTenantId(TENANT);
+        target.setScopeType(1);
+        target.setScopeId(GATEWAY_ID);
+        target.setVersion(1L);
+        target.setSnapshotHash(targetHash);
+        target.setSnapshotUri(targetUri);
+        target.setPublishStatus(1);
+        when(versionMapper.selectOne(any())).thenReturn(target);
+        when(versionMapper.selectMaxVersion(TENANT, 1, GATEWAY_ID)).thenReturn(3L);
+
+        IotConfigVersion row = publishService.rollback(rollbackForm(1L));
+
+        assertEquals(4L, row.getVersion(), "版本单调递增（回滚也是新版本）");
+        assertEquals(targetUri, row.getSnapshotUri(), "快照复用旧文件（brief D5 裁决）");
+        assertEquals(targetHash, row.getSnapshotHash(), "哈希同旧快照");
+        assertEquals(1, row.getPublishStatus());
+        verify(versionMapper).insert(any(IotConfigVersion.class));
+        verify(pointMapper).update(isNull(), any());
+    }
