+    private List<DictImportRow> parseImportRows(MultipartFile file) throws IOException {
+        List<DictImportRow> rows = new ArrayList<>();
+        try (ExcelReader reader = ExcelUtil.getReader(file.getInputStream())) {
+            List<List<Object>> dataRows = reader.read(1);
+            int rowIndex = 2;
+            for (List<Object> cells : dataRows) {
+                DictImportRow row = new DictImportRow();
+                row.setRowIndex(rowIndex++);
+                row.setDataCode(cell(cells, 0));
+                row.setDataName(cell(cells, 1));
+                row.setCategory(cell(cells, 2));
+                row.setDataType(cell(cells, 3));
+                row.setDefaultUnit(cell(cells, 4));
+                rows.add(row);
+            }
+        }
+        return rows;
+    }

+    private String cell(List<Object> cells, int index) {
+        if (cells == null || cells.size() <= index) {
+            return null;
+        }
+        String value = Convert.toStr(cells.get(index));
+        return value == null ? null : value.trim();
+    }

+        ExcelWriter writer = ExcelUtil.getBigWriter();
+        try {
+            writer.write(rows);

+            writer.close();

+    @Override
+    @Transactional(rollbackFor = Exception.class)
+    public DictImportReceipt importDicts(List<DictImportRow> rows) {
+        // 批级前置拒绝（无回执——无可逐行标注的对象）：空文件/模板原样上传、超 5000 行
+        if (rows == null || rows.isEmpty()) {
+            throw paramError("导入文件无数据行");
+        }
+        List<DictImportRow> dataRows = new ArrayList<>();
+        for (DictImportRow row : rows) {
+            if (isSkippableRow(row)) {
+                continue;
+            }
+            dataRows.add(row);
+        }
+        if (dataRows.isEmpty()) {
+            throw paramError("导入文件无数据行（模板说明行与空行不计入）");
+        }
+        if (dataRows.size() > IMPORT_MAX_ROWS) {
+            throw paramError("单批导入上限 " + IMPORT_MAX_ROWS + " 行，当前 " + dataRows.size() + " 行");
+        }
+
+        // 第一遍：字段级校验（必填→超长→类型→值域）+ 批内互撞判定（先于落库，brief 裁决 4）
+        Map<String, Integer> firstSeenRow = new HashMap<>();
+        List<DictImportRowResult> rowResults = new ArrayList<>(dataRows.size());
+        List<DictImportRow> validRows = new ArrayList<>(dataRows.size());
+        for (DictImportRow row : dataRows) {
+            DictImportRowResult rr = new DictImportRowResult();
+            rr.setRowIndex(row.getRowIndex());
+            rr.setDataCode(row.getDataCode());
+            String[] fieldReason = validateFields(row);
+            if (fieldReason != null) {
+                rr.setSuccess(false);
+                rr.setReasonCode(fieldReason[0]);
+                rr.setMessage(fieldReason[1]);
+            } else {
+                Integer firstRow = firstSeenRow.get(row.getDataCode());
+                if (firstRow != null) {
+                    rr.setSuccess(false);
+                    rr.setReasonCode(DictImportRowResult.REASON_DUPLICATE_CODE);
+                    rr.setMessage("批内重复（与第 " + firstRow + " 行撞码）");
+                } else {
+                    firstSeenRow.put(row.getDataCode(), row.getRowIndex());
+                    rr.setSuccess(true);
+                    rr.setMessage("校验通过");
+                    validRows.add(row);
+                }
+            }
+            rowResults.add(rr);
+        }
+
+        // 第二遍：库内已有编码预检（活跃行，@TableLogic 过滤已删行；租户条件由插件追加；
+        // 与单接口 insertDict 的预检闸同口径——含已禁用行，编码永不复用）。一次 IN 查询，
+        // 避免逐行 count 的 5000 次往返。
+        if (!validRows.isEmpty()) {
+            Set<String> existing = selectExistingCodes(validRows);
+            if (!existing.isEmpty()) {
+                List<DictImportRow> stillValid = new ArrayList<>(validRows.size());
+                for (DictImportRowResult rr : rowResults) {
+                    if (Boolean.TRUE.equals(rr.getSuccess()) && existing.contains(rr.getDataCode())) {
+                        rr.setSuccess(false);
+                        rr.setReasonCode(DictImportRowResult.REASON_DUPLICATE_CODE);
+                        rr.setMessage("库内已存在同编码条目（含已禁用，编码不复用）");
+                    }
+                }
+                for (DictImportRow row : validRows) {
+                    if (!existing.contains(row.getDataCode())) {
+                        stillValid.add(row);
+                    }
+                }
+                validRows = stillValid;
+            }
+        }
+
+        // 回执汇总：成功行数=校验通过行数；persisted 标记整批是否真正落库
+        long failCount = rowResults.stream().filter(rr -> !Boolean.TRUE.equals(rr.getSuccess())).count();
+        DictImportReceipt receipt = new DictImportReceipt();
+        receipt.setTotalCount(dataRows.size());
+        receipt.setSuccessCount((int) (dataRows.size() - failCount));
+        receipt.setFailCount((int) failCount);
+        receipt.setRows(rowResults);

+            receipt.setPersisted(false);
+            throw new DictImportRejectException(

+                            + " 行 / 失败 " + receipt.getFailCount() + " 行），请按回执修正后重新导入",
+                    receipt);
+        }
+
+        // 全部通过：白名单构建实体逐行落库（照 insertDict 范式——status=1/deleted=0/tenant_id 会话覆盖；
+        // 导入行 DTO 结构性不含 status/deleted/id/审计字段，mass assignment 无注入通道）
+        try {

+            }

+            throw paramError("导入失败：编码并发冲突，整批未入库，请重试");
+        }

+    private String[] validateFields(DictImportRow row) {

+            missing.add("data_type");
+        }

+        }

+        }
+        if (row.getDataName().length() > MAX_LEN_DATA_NAME) {
+            tooLong.add("data_name(" + row.getDataName().length() + ">" + MAX_LEN_DATA_NAME + ")");
+        }
+        if (row.getCategory() != null && row.getCategory().length() > MAX_LEN_CATEGORY) {
+            tooLong.add("category(" + row.getCategory().length() + ">" + MAX_LEN_CATEGORY + ")");
+        }
+        if (row.getDefaultUnit() != null && row.getDefaultUnit().length() > MAX_LEN_DEFAULT_UNIT) {
+            tooLong.add("default_unit(" + row.getDefaultUnit().length() + ">" + MAX_LEN_DEFAULT_UNIT + ")");
+        }
+        if (!tooLong.isEmpty()) {
+            return new String[]{DictImportRowResult.REASON_TOO_LONG, "字段超长: " + String.join(", ", tooLong)};
+        }
+        if (!DATA_TYPE_PATTERN.matcher(row.getDataType()).matches()) {
+            return new String[]{DictImportRowResult.REASON_INVALID_DATA_TYPE,
+                    "数据类型非法: " + row.getDataType() + "（仅允许 DOUBLE/INT/BOOL/ENUM）"};
+        }
+        if (row.getCategory() != null && !CATEGORY_PATTERN.matcher(row.getCategory()).matches()) {
+            return new String[]{DictImportRowResult.REASON_INVALID_CATEGORY,
+                    "类别非法: " + row.getCategory() + "（仅允许 ELEC/WATER/GAS/STEAM/MATERIAL/ENV/OTHER）"};
+        }
+        return null;
+    }

+        receipt.setFailCount(failCount);
+        receipt.setPersisted(persisted);
+        List<DictImportRowResult> rows = new java.util.ArrayList<>();
+        for (int i = 0; i < successCount + failCount; i++) {
+            DictImportRowResult rr = new DictImportRowResult();
+            rr.setRowIndex(2 + i);
+            rr.setDataCode("CODE_" + i);
+            rr.setSuccess(i < successCount);
+            rr.setMessage(i < successCount ? "校验通过" : "失败");
+            rows.add(rr);
+        }
+        receipt.setRows(rows);
