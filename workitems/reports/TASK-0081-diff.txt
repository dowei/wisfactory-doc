+ * 确认升级数据面编排服务实现（TASK-0081，ADR-0008 确认链路最后一环）。
+ *
+ * <p><b>生效机制=拷贝传播</b>（计划评审裁决①）：applied 版本 content 基线 vs 目标版本 content
+ * 经 0015 {@link TemplateContentDiffEngine} 产出逐条目档位化 diff，按条目落到实例数据项行
+ * （iot_data_item）——ConfigSnapshotAssembler 读行即为新版本生效值，组装器零改动。
+ *
+ * <p><b>三方合并</b>（裁决①）：字段级 CHANGED 条目——取 applied content 的旧默认（模板态）
+ * 与 target content 的新默认（模板态），与实例行现值做三方比对。行值为空或等于旧默认的
+ * <b>实例态形态</b>（公式经 {@link RelativeRefExpander} 按本实例展开、calc_period 经
+ * {@link IotDataItemServiceImpl#mapCalcPeriod} 换算、source_type 复现生成兜底 null→1）
+ * 视为未覆盖 → 写新默认实例态（公式展开 + 引用边先删后插重建）；否则保留实例已覆盖值。
+ * 判定误向（显式覆盖为旧默认同值）方向保守，符合「模板批量一致」核心价值。
+ *
+ * <p><b>T1 补插</b>（brief 接口 2）：ADDED 数据项委托 {@code IotDataItemService#regenerateForInstance}
+ * ——0013 生成范式补插缺失行（相对引用展开 + 引用边落库；已占用含逻辑删行幂等跳过）。
+ *
+ * <p><b>T3 迁移</b>（裁决③）：REMOVED 数据项=逻辑删（SET deleted=1，@TableLogic 退出快照、
+ * 双 uk 不含 deleted——键永久占用防 full_key 复用污染 TDengine 历史标签；行自身承载历史）；
+ * valueKind/defaultSourceType CHANGED=结构迁移（重写 source_type/formula/rollup 并边重建）。
+ * 被引用入边与指标引用<b>不清理</b>（ADR 明示后果，逐台 diff 明示）。
+ *
+ * <p><b>留痕</b>：逐台返回 {@link TemplateUpgradeResultVO}（diff + 迁移动作结果）+ 逐台 INFO
+ * 日志（[UPGRADE-APPLY] 范式）；本类不另起事务边界——由调用方 confirm 事务包裹（单原子单位）。
+ *
+ * @author dowei
+ * @date 2026-11-23
+ */
+@Slf4j
+@Service
+public class IotUpgradeApplyServiceImpl implements IotUpgradeApplyService {
+
+    private static final ObjectMapper PARSE_MAPPER = new ObjectMapper();
+
+    private static final String OBJ_DATA_ITEM = TemplateChangeEntry.OBJECT_DATA_ITEM;
+    private static final String CHG_ADDED = TemplateChangeEntry.CHANGE_ADDED;
+    private static final String CHG_REMOVED = TemplateChangeEntry.CHANGE_REMOVED;
+    private static final String CHG_CHANGED = TemplateChangeEntry.CHANGE_CHANGED;
+
+    private static final int SOURCE_TYPE_BIND = 1;
+    private static final int SOURCE_TYPE_ROLLUP = 3;
+    private static final int REF_TYPE_DATA_ITEM = 1;
+
+    @Autowired private IotDataItemMapper dataItemMapper;
+    @Autowired private IotDataItemRefMapper dataItemRefMapper;
+    @Autowired private IotModelInstanceMapper instanceMapper;
+    @Autowired private IotModelTemplateMapper modelTemplateMapper;
+    @Autowired private IotDataItemTemplateMapper dataItemTemplateMapper;
+    @Autowired private IotModelTemplateVersionMapper versionMapper;
+    @Autowired private IotDataItemService dataItemService;
+
+    @Override
+    public TemplateUpgradeResultVO applyToVersion(IotModelInstance instance, Integer targetVersion) {
+        TemplateUpgradeResultVO result = new TemplateUpgradeResultVO();
+        result.setInstanceId(instance.getId());
+        result.setInstanceCode(instance.getInstanceCode());
+        result.setAppliedVersion(instance.getAppliedTemplateVersion());
+        result.setTargetVersion(targetVersion);
+
+        Integer appliedVersion = instance.getAppliedTemplateVersion();
+        Long templateId = instance.getTemplateId();
+        if (appliedVersion == null || templateId == null) {
+            // 脏数据防御——无基线可比对，零操作（等价于「空=零操作」边界）
+            result.setRowUpdated(false);
+            return result;
+        }
+
+        // 1) applied content 基线 + target content（版本历史行；无基线降级=零操作）
+        String appliedContent = loadContent(templateId, appliedVersion);
+        String targetContent = loadContent(templateId, targetVersion);
+        if (appliedContent == null || targetContent == null) {
+            log.warn("[UPGRADE-APPLY] content baseline missing instanceId={} templateId={} applied={} target={}",
+                    instance.getId(), templateId, appliedVersion, targetVersion);
+            result.setRowUpdated(false);
+            return result;
+        }
+        JsonNode appliedNode = parse(appliedContent);
+        JsonNode targetNode = parse(targetContent);
+
+        // 2) 0015 diff 引擎——档位化逐条目（diff 空=零操作）
+        List<TemplateChangeEntry> entries = TemplateContentDiffEngine.diff(appliedNode, targetNode);
+        List<TemplateChangeEntryVO> diffVos = entries.stream()
+                .map(TemplateChangeEntryVO::from).collect(Collectors.toList());
+        result.setDiff(diffVos);
+        result.setTierSummary(TemplateUpgradeResultVO.summarize(diffVos));
+        if (entries.isEmpty()) {
+            result.setRowUpdated(false);
+            return result;
+        }
+
+        // 3) 索引：两版模板项（dataCode → content JSON 对象）
+        Map<String, JsonNode> appliedItems = indexItems(appliedNode);
+        Map<String, JsonNode> targetItems = indexItems(targetNode);
+
+        // 4) 实例数据项行（@TableLogic 过滤已删行；仅本实例）
+        List<IotDataItem> rows = dataItemMapper.selectList(new LambdaQueryWrapper<IotDataItem>()
+                .eq(IotDataItem::getInstanceId, instance.getId()));
+        Map<String, IotDataItem> rowsByCode = new HashMap<>();
+        for (IotDataItem row : rows) {
+            rowsByCode.put(row.getDataCode(), row);
+        }
+
+        List<String> actions = result.getActions();
+        boolean rowUpdated = false;
+
+        // 5) 对象级 ADDED（T1）：委托 regenerate 补插缺失行（0013 生成范式）
+        List<String> addedCodes = entries.stream()
+                .filter(e -> OBJ_DATA_ITEM.equals(e.getObjectType()) && CHG_ADDED.equals(e.getChangeType()))
+                .map(TemplateChangeEntry::getObjectKey)
+                .distinct().collect(Collectors.toList());
+        if (!addedCodes.isEmpty()) {
+            dataItemService.regenerateForInstance(instance.getId());
+            actions.add("新增项补插: " + String.join(",", addedCodes) + "（0013 生成范式）");
+            rowUpdated = true;
+        }
+
+        // 6) 对象级 REMOVED（T3）：逻辑删停用（保留历史行自身，键永久占用）
+        List<String> removedCodes = entries.stream()
+                .filter(e -> OBJ_DATA_ITEM.equals(e.getObjectType()) && CHG_REMOVED.equals(e.getChangeType()))
+                .map(TemplateChangeEntry::getObjectKey)
+                .distinct().collect(Collectors.toList());
+        for (String code : removedCodes) {
+            IotDataItem row = rowsByCode.get(code);
+            if (row == null) {
+                continue; // 实例本就无该行——无需删
+            }
+            LambdaUpdateWrapper<IotDataItem> del = new LambdaUpdateWrapper<IotDataItem>()
+                    .eq(IotDataItem::getId, row.getId())
+                    .set(IotDataItem::getDeleted, 1);
+            dataItemMapper.update(null, del);
+            // 被删项自身出边随之清理（其公式不再生效）；入边与指标引用不清理（ADR 明示后果）
+            dataItemRefMapper.deleteByItemIdPhysical(TenantContext.getTenantId(), row.getId());
+            actions.add(code + " 逻辑删停用（键永久占用，历史行保留；入边与指标引用不清理）");
+            rowUpdated = true;
+        }
+
+        // 7) 字段级 CHANGED：三方合并（按 dataCode 分组，逐字段比对）
+        Map<String, List<TemplateChangeEntry>> changedByCode = entries.stream()
+                .filter(e -> OBJ_DATA_ITEM.equals(e.getObjectType()) && CHG_CHANGED.equals(e.getChangeType()))
+                .collect(Collectors.groupingBy(TemplateChangeEntry::getObjectKey, LinkedHashMap::new,
+                        Collectors.toList()));
+        for (Map.Entry<String, List<TemplateChangeEntry>> group : changedByCode.entrySet()) {
+            String code = group.getKey();
+            IotDataItem row = rowsByCode.get(code);
+            if (row == null) {
+                continue; // 新增项已由 regenerate 处理；无行可合并
+            }
+            JsonNode oldItem = appliedItems.get(code);
+            JsonNode newItem = targetItems.get(code);
+            if (oldItem == null || newItem == null) {
+                continue;
+            }
+            if (applyFieldChanges(instance, row, code, group.getValue(), oldItem, newItem, actions)) {
+                rowUpdated = true;
+            }
+        }
+
+        result.setRowUpdated(rowUpdated);
+        log.info("[UPGRADE-APPLY] instanceId={} templateId={} applied={} target={} summary={} actions={}",
+                instance.getId(), templateId, appliedVersion, targetVersion, result.getTierSummary(), actions);
+        return result;
+    }
+
+    @Override
+    public List<TemplateUpgradeResultVO> applyToVersionBatch(List<IotModelInstance> instances, Integer targetVersion) {
+        List<TemplateUpgradeResultVO> results = new ArrayList<>(instances.size());
+        for (IotModelInstance instance : instances) {
+            results.add(applyToVersion(instance, targetVersion));
+        }
+        return results;
+    }
+
+    /**
+     * 单数据项字段级三方合并：逐 CHANGED 字段判定「未覆盖→写新默认 / 已覆盖→保留」。
+     *
+     * @return 是否发生行写（任一字段写入即 true）
+     */
+    private boolean applyFieldChanges(IotModelInstance instance, IotDataItem row, String code,
+                                      List<TemplateChangeEntry> changes, JsonNode oldItem, JsonNode newItem,
+                                      List<String> actions) {
+        Long tenant = TenantContext.getTenantId();
+        IotDataItem patch = new IotDataItem();
+        patch.setId(row.getId());
+        boolean anyWrite = false;
+        boolean formulaChanged = false;
+        boolean sourceTypeChanged = false;
+        boolean formulaCleared = false;
+        LinkedHashSet<String> newRefs = new LinkedHashSet<>();
+
+        for (TemplateChangeEntry change : changes) {
+            String field = change.getField();
+            if (field == null) {
+                continue;
+            }
+            switch (field) {
+                case "defaultFormula" -> {
+                    String oldTemplate = text(oldItem, field);
+                    String newTemplate = text(newItem, field);
+                    String oldInstanceForm = expandFormula(oldTemplate, instance);
+                    if (isBlank(row.getFormula()) || eq(row.getFormula(), oldInstanceForm)) {
+                        String newInstanceForm = expandFormula(newTemplate, instance);
+                        patch.setFormula(newInstanceForm);
+                        anyWrite = true;
+                        formulaChanged = true;
+                        newRefs.addAll(expandRefs(newTemplate, instance));
+                        actions.add(code + " 公式继承新默认: " + newInstanceForm);
+                    }
+                }
+                case "defaultCalcPeriod" -> {
+                    Integer oldSeconds = safeMapPeriod(text(oldItem, field));
+                    Integer newSeconds = safeMapPeriod(text(newItem, field));
+                    if (row.getCalcPeriod() == null || row.getCalcPeriod().equals(oldSeconds)) {
+                        patch.setCalcPeriod(newSeconds == null ? 0 : newSeconds);
+                        anyWrite = true;
+                        actions.add(code + " 计算周期继承新默认: " + newSeconds + "s");
+                    }
+                }
+                case "defaultSourceType" -> {
+                    Integer oldDefault = intVal(oldItem, field);
+                    Integer newDefault = intVal(newItem, field);
+                    int oldFallback = oldDefault == null ? SOURCE_TYPE_BIND : oldDefault;
+                    int newFallback = newDefault == null ? SOURCE_TYPE_BIND : newDefault;
+                    if (row.getSourceType() == null || row.getSourceType() == oldFallback) {
+                        patch.setSourceType(newFallback);
+                        anyWrite = true;
+                        sourceTypeChanged = true;
+                        actions.add(code + " 取值方式结构迁移: " + oldFallback + "→" + newFallback);
+                        // 取值方式互斥：新默认非公式时清公式/汇总，是公式时写新公式展开
+                        if (newFallback == SOURCE_TYPE_BIND) {
+                            patch.setFormula(null);
+                            patch.setRollupMethod(null);
+                            formulaCleared = true;
+                            newRefs.clear();
+                        } else if (newFallback == SOURCE_TYPE_ROLLUP) {
+                            patch.setFormula(null);
+                            patch.setRollupMethod(text(newItem, "aggMethod"));
+                            formulaCleared = true;
+                            newRefs.clear();
+                        } else {
+                            String newFormula = expandFormula(text(newItem, "defaultFormula"), instance);
+                            patch.setFormula(newFormula);
+                            patch.setRollupMethod(null);
+                            newRefs.addAll(expandRefs(text(newItem, "defaultFormula"), instance));
+                        }
+                    }
+                }
+                case "aggMethod" -> {
+                    String oldAgg = text(oldItem, field);
+                    String newAgg = text(newItem, field);
+                    if (isBlank(row.getRollupMethod()) || eq(row.getRollupMethod(), oldAgg)) {
+                        patch.setRollupMethod(newAgg);
+                        anyWrite = true;
+                        actions.add(code + " 汇总方式继承新默认: " + newAgg);
+                    }
+                }
+                case "defaultValidMin" -> {
+                    BigDecimal oldV = decimal(oldItem, field);
+                    BigDecimal newV = decimal(newItem, field);
+                    if (row.getValidMin() == null || eq(row.getValidMin(), oldV)) {
+                        patch.setValidMin(newV);
+                        anyWrite = true;
+                        actions.add(code + " 有效下限继承新默认: " + newV);
+                    }
+                }
+                case "defaultValidMax" -> {
+                    BigDecimal oldV = decimal(oldItem, field);
+                    BigDecimal newV = decimal(newItem, field);
+                    if (row.getValidMax() == null || eq(row.getValidMax(), oldV)) {
+                        patch.setValidMax(newV);
+                        anyWrite = true;
+                        actions.add(code + " 有效上限继承新默认: " + newV);
+                    }
+                }
+                case "defaultAlarmLow" -> {
+                    BigDecimal oldV = decimal(oldItem, field);
+                    BigDecimal newV = decimal(newItem, field);
+                    if (row.getAlarmLow() == null || eq(row.getAlarmLow(), oldV)) {
+                        patch.setAlarmLow(newV);
+                        anyWrite = true;
+                        actions.add(code + " 告警低阈值继承新默认: " + newV);
+                    }
+                }
+                case "defaultAlarmHigh" -> {
+                    BigDecimal oldV = decimal(oldItem, field);
+                    BigDecimal newV = decimal(newItem, field);
+                    if (row.getAlarmHigh() == null || eq(row.getAlarmHigh(), oldV)) {
+                        patch.setAlarmHigh(newV);
+                        anyWrite = true;
+                        actions.add(code + " 告警高阈值继承新默认: " + newV);
+                    }
+                }
+                case "dataName" -> {
+                    String oldV = text(oldItem, field);
+                    String newV = text(newItem, field);
+                    if (isBlank(row.getDataName()) || eq(row.getDataName(), oldV)) {
+                        patch.setDataName(newV);
+                        anyWrite = true;
+                        actions.add(code + " 名称继承新默认: " + newV);
+                    }
+                }
+                case "category" -> {
+                    String oldV = text(oldItem, field);
+                    String newV = text(newItem, field);
+                    if (isBlank(row.getCategory()) || eq(row.getCategory(), oldV)) {
+                        patch.setCategory(newV);
+                        anyWrite = true;
+                    }
+                }
+                case "unit" -> {
+                    String oldV = text(oldItem, field);
+                    String newV = text(newItem, field);
+                    if (isBlank(row.getUnit()) || eq(row.getUnit(), oldV)) {
+                        patch.setUnit(newV);
+                        anyWrite = true;
+                        actions.add(code + " 单位继承新默认: " + newV);
+                    }
+                }
+                case "valueKind" -> {
+                    // valueKind（数据类型）不落实例行（0013「留模板不落实例」，实例表无该列）——
+                    // T3 迁移经继承判定覆盖的公式/结构字段重物化承载，此处仅记逐台 diff 明示。
+                    actions.add(code + " 数据类型变更（T3，模板态承载，经结构字段重物化承载）");
+                }
+                default -> {
+                    // 未登记字段（diff 引擎 fail-closed 归 T2）——保守不动实例行，仅记 diff 明示
+                    actions.add(code + " 字段 " + field + " 变更（无实例行载体，逐台 diff 明示）");
+                }
+            }
+        }
+
+        if (anyWrite) {
+            // 白名单更新集：仅写本轮判定的列（patch 中 null 列不进 SET，保留实例已覆盖值）
+            LambdaUpdateWrapper<IotDataItem> upd = lambdaUpdate(patch);
+            dataItemMapper.update(null, upd);
+            // 公式或取值方式变更 → 引用边先删后插重建（0013/0036 regenerate 范式）
+            // formulaCleared（取值方式切到直绑/汇总）=只删不插（无公式则无出边）
+            if (formulaChanged || sourceTypeChanged) {
+                dataItemRefMapper.deleteByItemIdPhysical(tenant, row.getId());
+                if (!formulaCleared) {
+                    insertRefs(tenant, row.getId(), newRefs);
+                }
+            }
+        }
+        return anyWrite;
+    }
+
+    /** 构建仅含非 null 字段的白名单更新包装器（id 作 WHERE 守卫） */
+    private LambdaUpdateWrapper<IotDataItem> lambdaUpdate(IotDataItem patch) {
+        LambdaUpdateWrapper<IotDataItem> upd = new LambdaUpdateWrapper<IotDataItem>()
+                .eq(IotDataItem::getId, patch.getId());
+        if (patch.getFormula() != null) {
+            upd.set(IotDataItem::getFormula, patch.getFormula());
+        } else if (patch.getSourceType() != null && (patch.getSourceType() == SOURCE_TYPE_BIND
+                || patch.getSourceType() == SOURCE_TYPE_ROLLUP)) {
+            // 取值方式切换离公式时显式清公式（互斥）——用 set 覆盖为 NULL
+            upd.set(IotDataItem::getFormula, null);
+        }
+        if (patch.getSourceType() != null) {
+            upd.set(IotDataItem::getSourceType, patch.getSourceType());
+        }
+        if (patch.getRollupMethod() != null) {
+            upd.set(IotDataItem::getRollupMethod, patch.getRollupMethod());
+        } else if (patch.getSourceType() != null && patch.getSourceType() == SOURCE_TYPE_BIND) {
+            upd.set(IotDataItem::getRollupMethod, null);
+        }
+        if (patch.getCalcPeriod() != null) {
+            upd.set(IotDataItem::getCalcPeriod, patch.getCalcPeriod());
+        }
+        if (patch.getValidMin() != null) {
+            upd.set(IotDataItem::getValidMin, patch.getValidMin());
+        }
+        if (patch.getValidMax() != null) {
+            upd.set(IotDataItem::getValidMax, patch.getValidMax());
+        }
+        if (patch.getAlarmLow() != null) {
+            upd.set(IotDataItem::getAlarmLow, patch.getAlarmLow());
+        }
+        if (patch.getAlarmHigh() != null) {
+            upd.set(IotDataItem::getAlarmHigh, patch.getAlarmHigh());
+        }
+        if (patch.getDataName() != null) {
+            upd.set(IotDataItem::getDataName, patch.getDataName());
+        }
+        if (patch.getCategory() != null) {
+            upd.set(IotDataItem::getCategory, patch.getCategory());
+        }
+        if (patch.getUnit() != null) {
+            upd.set(IotDataItem::getUnit, patch.getUnit());
+        }
+        return upd;
+    }

+    /** 引用边落库（ref_type=1；ref_item_id 按 full_key 查活跃行尽力回填，查不到置 NULL 不阻断） */
+    private void insertRefs(Long tenant, Long itemId, LinkedHashSet<String> refFullKeys) {
+        for (String fullKey : refFullKeys) {
+            IotDataItemRef ref = new IotDataItemRef();
+            ref.setTenantId(tenant);
+            ref.setItemId(itemId);
+            ref.setRefFullKey(fullKey);
+            ref.setRefType(REF_TYPE_DATA_ITEM);
+            IotDataItem target = dataItemMapper.selectOne(new LambdaQueryWrapper<IotDataItem>()
+                    .select(IotDataItem::getId).eq(IotDataItem::getFullKey, fullKey));
+            ref.setRefItemId(target == null ? null : target.getId());
+            dataItemRefMapper.insert(ref);
+        }
+    }

+    /** 模板公式 → 本实例展开态（相对引用 {SELF}/{PARENT} 按 0013 RelativeRefExpander） */
+    private String expandFormula(String templateFormula, IotModelInstance instance) {
+        if (isBlank(templateFormula)) {
+            return templateFormula;
+        }
+        String parentCode = null;
+        if (RelativeRefExpander.containsParentRef(templateFormula)) {
+            Long pid = instance.getParentId();
+            if (pid != null && pid != 0L) {
+                IotModelInstance parent = instanceMapper.selectById(pid);
+                parentCode = parent == null ? null : parent.getInstanceCode();
+            }
+        }
+        return RelativeRefExpander.expand(templateFormula, instance.getInstanceCode(), parentCode).expandedFormula();
+    }
+
+    /** 模板公式 → 解析后的被引用全键集（用于引用边重建） */
+    private LinkedHashSet<String> expandRefs(String templateFormula, IotModelInstance instance) {
+        if (isBlank(templateFormula)) {
+            return new LinkedHashSet<>();
+        }
+        String parentCode = null;
+        if (RelativeRefExpander.containsParentRef(templateFormula)) {
+            Long pid = instance.getParentId();
+            if (pid != null && pid != 0L) {
+                IotModelInstance parent = instanceMapper.selectById(pid);
+                parentCode = parent == null ? null : parent.getInstanceCode();
+            }
+        }
+        return new LinkedHashSet<>(RelativeRefExpander.expand(
+                templateFormula, instance.getInstanceCode(), parentCode).resolvedRefs());
+    }

+    private String loadContent(Long templateId, int version) {
+        IotModelTemplateVersion row = versionMapper.selectOne(new LambdaQueryWrapper<IotModelTemplateVersion>()
+                .eq(IotModelTemplateVersion::getTemplateId, templateId)
+                .eq(IotModelTemplateVersion::getVersion, version)
+                .last("LIMIT 1"));
+        return row == null ? null : row.getContent();
+    }
+
+    private Map<String, JsonNode> indexItems(JsonNode content) {
+        Map<String, JsonNode> map = new TreeMap<>();
+        JsonNode items = content == null ? null : content.get("dataItems");
+        if (items != null && items.isArray()) {
+            for (JsonNode item : items) {
+                if (item.hasNonNull("dataCode")) {
+                    map.put(item.get("dataCode").asText(), item);
+                }
+            }
+        }
+        return map;
+    }

+    private JsonNode parse(String json) {
+        try {
+            return PARSE_MAPPER.readTree(json);
+        } catch (Exception e) {
+            throw new IllegalStateException("模板内容快照解析失败: " + e.getMessage(), e);
+        }
+    }
+
