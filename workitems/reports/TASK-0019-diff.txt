+package net.otlab.admin.module.iot.credential.domain;

+/**
+ * 凭据掩码器（TASK-0019，列表出口唯一展示形态——ADR-0006/REQ-0011「列表永不返回密文」）
+ *
+ * <p>掩码口径（brief：格式自定如 {@code abcd****wxyz} 前四后四）：
+ * <ul>
+ * <li>长度 ≥ 12：前四 + {@code ****} + 后四（中段至少隐藏 4 字符）；</li>
+ * <li>长度 8-11：前四 + {@code ****}（尾段至少隐藏 4 字符）；</li>
+ * <li>长度 &lt; 8 或空：纯占位 {@code ****}（短凭据隐藏比例不足 4 字符，不露任何字符）。</li>
+ * </ul>
+ *
+ * <p>边界纪律：只要露段，隐藏段恒 ≥ 4 字符；掩码输入为<b>解密后的明文</b>（列表服务层
+ * 解密后取前后四），输出只允许掩码字符串——任何明文/密文子串不得出现在掩码之外；
+ * 解密失败（密钥不匹配/密文损坏）由调用方降级为纯占位，列表不崩、不泄底。
+ *
+ * @author dowei
+ * @date 2026-11-22
+ */
+public final class CredentialMask {
+
+    /** 纯占位符（短凭据/解密失败/空值通用） */
+    public static final String FULL_MASK = "****";
+
+    /** 前后四双露的最小明文长度（≥12：中段至少 4 字符隐藏） */
+    private static final int BOTH_ENDS_MIN_LENGTH = 12;
+
+    /** 单露首四的最小明文长度（8-11：尾段至少 4 字符隐藏） */
+    private static final int HEAD_ONLY_MIN_LENGTH = 8;
+
+    /** 首段长度 */
+    private static final int EDGE_LENGTH = 4;
+
+    private CredentialMask() {
+    }
+
+    /**
+     * 掩码化：明文（或任意待展示敏感串）→ 掩码字符串。null/空安全。
+     */
+    public static String mask(String plaintext) {
+        if (plaintext == null || plaintext.isEmpty()) {
+            return FULL_MASK;
+        }
+        int length = plaintext.length();
+        if (length >= BOTH_ENDS_MIN_LENGTH) {
+            return plaintext.substring(0, EDGE_LENGTH) + FULL_MASK
+                    + plaintext.substring(length - EDGE_LENGTH);
+        }
+        if (length >= HEAD_ONLY_MIN_LENGTH) {
+            return plaintext.substring(0, EDGE_LENGTH) + FULL_MASK;
+        }
+        return FULL_MASK;
+    }
+}

+            throw paramError("凭据明文不能为空");

+        IotChannelCredential upd = new IotChannelCredential();

+        upd.setCiphertext(ciphertext);

+            return INITIAL_KEY_VERSION;
+        }
+        try {
+            return String.valueOf(Long.parseLong(current.trim()) + 1);
+        } catch (NumberFormatException e) {
+            return INITIAL_KEY_VERSION;
+        }
+    }

+    public CredentialPullVO pullCredential(Long channelId, String credType) {
+        if (channelId == null) {

+        }
+        if (credType == null || credType.isBlank()) {
+            throw paramError("credType 必传");
+        }

+        IotChannel channel = channelMapper.selectById(channelId);
+        if (channel == null || channel.getStatus() == null || channel.getStatus() != STATUS_ENABLED) {
+            throw forbidden("通道不存在、已删除或已禁用");
+        }
+        // 凭据存在/启用：缺失与吊销统一 403，不区分原因（不给探测面）；未知 credType 走
+        // 同一 403（不枚举校验字典——避免类型枚举探测）；直接 mapper 调用（uk 保证 ≤1 行）
+        IotChannelCredential row = this.getBaseMapper().selectOne(
+                new LambdaQueryWrapper<IotChannelCredential>()
+                        .eq(IotChannelCredential::getChannelId, channelId)
+                        .eq(IotChannelCredential::getCredType, credType));
+        if (row == null || row.getStatus() == null || row.getStatus() != STATUS_ENABLED) {
+            throw forbidden("凭据不存在或已吊销");
+        }

+        if (!channel.getTenantId().equals(row.getTenantId())) {

+        }
+        // 服务端不解密传输：只回密文与 key_version，调用方内存解密（ADR-0006）
+        return CredentialPullVO.of(row);
+    }

+            throw paramError("凭据类型不能为空");

+        if (!CredentialTypes.isValid(credType)) {
+            throw paramError("凭据类型仅允许 DEVICE_TOKEN/MQTT_PASSWORD/HTTP_SECRET/CERT: " + credType);
+        }
+        return credType;
+    }

+    @DisplayName("SM4 往返 - encrypt 产 {v1} 前缀密文且 decrypt 还原明文（随机 IV 两次密文不同）")
+    void testSm4Roundtrip() {
+        String cipher1 = SmDataCrypto.encrypt(PLAINTEXT);
+        String cipher2 = SmDataCrypto.encrypt(PLAINTEXT);
+        assertTrue(cipher1.startsWith("{v1}"), "密文必须带 {v1} 版本前缀");
+        assertTrue(cipher1.contains(":"), "{v1}<iv>:<cipher> 结构");
+        assertTrue(SmDataCrypto.isCiphertext(cipher1));
+        assertNotEquals(cipher1, cipher2, "随机 IV——同一明文两次加密密文不同");
+        assertEquals(PLAINTEXT, SmDataCrypto.decrypt(cipher1));
+    }

+    void testList_MaskOnly() {
+        when(channelMapper.selectById(100L)).thenReturn(existingChannel());
+        when(baseMapper.selectList(any(Wrapper.class))).thenReturn(List.of(
+                storedRow(PLAINTEXT, "1", 1)));

+        assertEquals("1", item.getKeyVersion());
+        assertEquals(1, item.getStatus());
+        assertEquals(CredentialMask.mask(PLAINTEXT), item.getCredentialMask(),
+                "掩码必须取自明文前四后四口径");
+        assertEquals("mqtt****1234", item.getCredentialMask(), "掩码格式 abcd****wxyz");
+
+        // grep 级结构断言：VO 类字段集合中不存在密文/明文承载体（ciphertext/credential/plain）
+        List<String> fieldNames = java.util.Arrays.stream(CredentialItemVO.class.getDeclaredFields())
+                .map(java.lang.reflect.Field::getName).toList();
+        for (String banned : List.of("ciphertext", "credential", "plaintext", "cipher")) {
+            assertFalse(fieldNames.contains(banned), "列表 VO 不得携带 " + banned + " 字段: " + fieldNames);
+        }
+        String voText = item.toString();
+        assertFalse(voText.contains(PLAINTEXT), "列表响应不得含明文");
+        assertFalse(voText.startsWith("{v1}") || voText.contains("ciphertext="), "列表响应不得含密文");
+    }

+    void testMaskBoundaries() {
+        assertEquals("****", CredentialMask.mask(null));
+        assertEquals("****", CredentialMask.mask(""));
+        assertEquals("****", CredentialMask.mask("abc"));
