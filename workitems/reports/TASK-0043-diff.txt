+        // 第一遍：逐项字段校验（编码/名称/口径标注/日历锁/开关值域/比值齐备性——与 add 共用
+        // 同一校验链）+ 批内撞码判定（先于库内预检——撞码批连库内核对都无需消耗，0011 范式）；
+        // 字段级违规在本遍收敛（任一项失败即抛出，未落库任何行）
+        Set<String> seenCodes = new LinkedHashSet<>(items.size());
+        for (IotMetricCreateForm item : items) {
+            if (item.getMetricCode() == null || !seenCodes.add(item.getMetricCode())) {
+                throw duplicateCode4002("指标编码批内重复: " + item.getMetricCode());
+            }
+            validateMetricCode(item.getMetricCode());
+            if (item.getMetricName() == null || item.getMetricName().length() > MAX_LEN_METRIC_NAME) {
+                throw paramError("指标名称必填且长度不能超过 " + MAX_LEN_METRIC_NAME);
+            }
+            resolveAggType(item.getAggType());
+            resolvePeriod(item.getPeriod());
+            resolveCalendarType(item.getCalendarType());
+            resolveFlag(item.getYoyFlag(), "同比开关仅允许 0 否 / 1 是");
+            resolveFlag(item.getMomFlag(), "环比开关仅允许 0 否 / 1 是");
+            resolveCreateStatus(item.getStatus());
+            if (item.getAggType() != null && item.getAggType() == AGG_TYPE_RATIO
+                    && (item.getRatioNumItemId() == null || item.getRatioDenItemId() == null)) {
+                throw paramError("比值型指标必须同时提供分子与分母数据项（ratioNumItemId/ratioDenItemId 缺一不可）");
+            }
+        }
+
+        // 第二遍：库内级校验（批内一次 IN，避免逐行往返放大，照 0021 导入范式）——
+        // ① 绑定部门存在性（去重后一次 selectByIds）；② 引用数据项存在性/同实例/启用
+        // （比值分子分母 + source_item_id，批内一次 selectByIds）；③ 编码占用旁路预检
+        // （含已删行——编码删除归档不复用）
+        Set<Long> deptIds = items.stream().map(IotMetricCreateForm::getDeptId)
+                .filter(java.util.Objects::nonNull)
+                .collect(Collectors.toCollection(LinkedHashSet::new));
+        if (!deptIds.isEmpty()) {
+            Set<Long> existingDeptIds = deptMapper.selectByIds(deptIds).stream()
+                    .map(SysDept::getId)
+                    .collect(Collectors.toSet());
+            for (IotMetricCreateForm item : items) {
+                if (item.getDeptId() != null && !existingDeptIds.contains(item.getDeptId())) {
+                    throw paramError("绑定部门不存在或已删除: " + item.getDeptId());
+                }
+            }
+        }
+        Map<Long, IotDataItem> itemById = loadItemsByIds(collectReferencedItemIds(items, instance.getId()));
+        for (IotMetricCreateForm item : items) {
+            validateReferences(itemById, instance.getId(), item.getSourceItemId(),
+                    item.getRatioNumItemId(), item.getRatioDenItemId(), item.getAggType());
+        }
+        Set<String> codes = items.stream().map(IotMetricCreateForm::getMetricCode).collect(Collectors.toSet());
+        Map<String, IotMetricCodeRow> occupiedByCode = new LinkedHashMap<>();
+        for (IotMetricCodeRow row : metricCodeMapper.selectOccupiedByCodes(tenantId, codes)) {
+            occupiedByCode.putIfAbsent(row.getMetricCode(), row);
+        }
+        for (IotMetricCreateForm item : items) {
+            IotMetricCodeRow occupied = occupiedByCode.get(item.getMetricCode());
+            if (occupied != null) {
+                throw duplicateMetricCode(occupied);
+            }
+        }
+
+        // 全部通过：白名单构建实体逐项落库（instanceId 以入参覆盖；status/tenant_id/审计字段
+        // 口径同 add）。任一异常 rollbackFor 整批回滚——「逐项校验整批事务」brief 口径
+        try {
+            int count = 0;
+            for (IotMetricCreateForm item : items) {
+                this.save(validateAndBuild(tenantId, instance.getId(), item, seenCodes, itemById));
+                count++;
+            }
+            return count;
+        } catch (DuplicateKeyException e) {
+            // 兜底：预检与落库之间的并发撞码窗口（物理唯一索引），rollbackFor 整批回滚
+            throw duplicateCode4002("指标编码并发冲突，整批未入库，请重试");
+        }
+    }

+    /** 批内引用数据项 id 收集（比值分子分母按生效聚合方式 + source_item_id，一次 IN 预检用） */
+    private Set<Long> collectReferencedItemIds(List<IotMetricCreateForm> items, Long instanceId) {
+        Set<Long> ids = new LinkedHashSet<>();
+        for (IotMetricCreateForm item : items) {
+            collectItemIds(ids, item.getAggType(), item.getSourceItemId(),
+                    item.getRatioNumItemId(), item.getRatioDenItemId());
+        }
+        return ids;
+    }

+    @Override
+    @Transactional(rollbackFor = Exception.class)
+    public boolean updateMetric(IotMetricUpdateForm form) {
+        IotMetric existing = this.getById(form.getId());
+        if (existing == null) {
+            throw paramError("指标不存在或已删除");
+        }
+        // 写路径数据权限防线（brief 口径照 0021 修订轮范式）：存在性 → 数据权限 → 业务校验。
+        // 可见口径与列表裁剪同条件（dept_id ∈ 部门集 OR instance_id ∈ 可见集，NULL 行
+        // fail-closed；无 create_by 分支——资产类非个人创建物）；不可见 403 且文案不泄露指标标识
+        requireWritePerm(existing);
+        // metric_code 不可改（编码即指标锚点与看板取数标识）：携带不一致一律拒（携带同值为 no-op 不拦）
+        if (form.getMetricCode() != null && !form.getMetricCode().equals(existing.getMetricCode())) {
+            throw paramError("指标编码不允许修改: " + existing.getMetricCode());
+        }
+
+        // 生效组合（携带值优先，未携带取库内值——单边改也须不破坏口径/比值/归属约束）
+        Integer effAggType = resolveAggType(form.getAggType() != null ? form.getAggType() : existing.getAggType());
+        String effPeriod = resolvePeriod(form.getPeriod() != null ? form.getPeriod() : existing.getPeriod());
+        Integer effCalendar = resolveCalendarType(
+                form.getCalendarType() != null ? form.getCalendarType() : existing.getCalendarType());
+        resolveFlag(form.getYoyFlag(), "同比开关仅允许 0 否 / 1 是");
+        resolveFlag(form.getMomFlag(), "环比开关仅允许 0 否 / 1 是");
+        if (form.getMetricName() != null && form.getMetricName().length() > MAX_LEN_METRIC_NAME) {
+            throw paramError("指标名称超长（上限 " + MAX_LEN_METRIC_NAME + " 字符）");
+        }
+        // 生效实例归属校验（携带时先过存在性；创建侧防线不适用编辑——目标行可见性已防线，
+        // 挪实例不扩大调用者可见面，照 0021 编辑 boundInstanceId 同口径）
+        Long effInstanceId = form.getInstanceId() != null ? form.getInstanceId() : existing.getInstanceId();
+        if (form.getInstanceId() != null) {
+            resolveInstance(form.getInstanceId());
+        }
+        validateDept(form.getDeptId());
+
+        // 来源/比值引用按生效组合重跑（改实例时引用须属新实例；生效聚合方式切非比值型时
+        // ratio_* 强制清 NULL——「非比值型 ratio_* 置 NULL」对编辑同口径）
+        Long effSource = form.getSourceItemId() != null ? form.getSourceItemId() : existing.getSourceItemId();
+        Long effRatioNum = form.getRatioNumItemId() != null ? form.getRatioNumItemId() : existing.getRatioNumItemId();
+        Long effRatioDen = form.getRatioDenItemId() != null ? form.getRatioDenItemId() : existing.getRatioDenItemId();
+        if (effAggType == AGG_TYPE_RATIO && (effRatioNum == null || effRatioDen == null)) {
+            // 比值型齐备性对编辑同口径（生效组合为比值型而分子分母缺一 → 400 拒）
+            throw paramError("比值型指标必须同时提供分子与分母数据项（ratioNumItemId/ratioDenItemId 缺一不可）");
+        }
+        Map<Long, IotDataItem> itemById = loadItemsByIds(
+                collectItemIds(new LinkedHashSet<>(), effAggType, effSource, effRatioNum, effRatioDen));
+        validateReferences(itemById, effInstanceId, effSource, effRatioNum, effRatioDen, effAggType);
+
+        // 字段白名单构建更新（null 不进更新集，MP 自动跳过保持库内原值；不可变字段不回写、
+        // 连 no-op 都不进 SET）：仅 metricName/instanceId/sourceItemId/aggType/period/
+        // calendarType/yoyFlag/momFlag/deptId 可变——status 不进编辑集（启停走 changeStatus
+        // 独立接口，§3.1 p22 独立行）、metricCode/tenantId/审计字段/deleted 永不进管理面
+        // 更新集（杜绝 mass assignment）。比值型携带比值字段（生效=5 且请求携带）进 SET；
+        // 非比值型 ratio_* 一律强制清 NULL（UpdateWrapper 显式 set null——实体 updateById
+        // 的 null 跳过语义清不掉旧值）
+        IotMetric upd = new IotMetric();
+        upd.setId(existing.getId());
+        upd.setMetricName(form.getMetricName());
+        upd.setInstanceId(form.getInstanceId());
+        upd.setSourceItemId(form.getSourceItemId());
+        upd.setAggType(form.getAggType());
+        upd.setPeriod(form.getPeriod());
+        upd.setCalendarType(form.getCalendarType());
+        if (effAggType == AGG_TYPE_RATIO) {
+            upd.setRatioNumItemId(form.getRatioNumItemId());
+            upd.setRatioDenItemId(form.getRatioDenItemId());
+        }
+        upd.setYoyFlag(form.getYoyFlag());
+        upd.setMomFlag(form.getMomFlag());
+        upd.setDeptId(form.getDeptId());
+        this.updateById(upd);
+        if (effAggType != AGG_TYPE_RATIO) {
+            // 非比值型：显式清 ratio_*（「非比值型 ratio_* 置 NULL」——切换类型/误携带都收敛为 NULL）
+            this.update(new LambdaUpdateWrapper<IotMetric>()
+                    .eq(IotMetric::getId, existing.getId())
+                    .set(IotMetric::getRatioNumItemId, null)
+                    .set(IotMetric::getRatioDenItemId, null));
+        }
+        return true;
+    }

+    // ---------- 批量启停（§3.1 p22 独立行 changeStatus） ----------

+    @Override
+    @Transactional(rollbackFor = Exception.class)
+    public boolean changeMetricStatus(List<Long> ids, Integer status) {
+        if (status == null || !FLAG_VALUES.contains(status)) {
+            throw paramError("指标状态仅允许 0 禁用 / 1 启用");
+        }
+        if (ids == null || ids.isEmpty()) {
+            throw paramError("请选择要启停的指标");
+        }
+        // fail-closed：批内全部 id 必须存在且活跃（@TableLogic 过滤已删行），缺失即整批拒绝（不半批落库）
+        List<IotMetric> existingList = this.listByIds(ids);
+        if (existingList.size() < ids.size()) {
+            Map<Long, IotMetric> found = existingList.stream()
+                    .collect(Collectors.toMap(IotMetric::getId, Function.identity(), (a, b) -> a, LinkedHashMap::new));
+            List<Long> missing = ids.stream().filter(id -> !found.containsKey(id)).toList();
+            throw paramError("指标不存在或已删除: " + missing);
+        }
+
+        // 写路径数据权限防线（brief 口径照 0021 修订轮范式）：批内任一不可见即整批拒绝
+        // （fail-closed 不半批落库，403 文案不泄露指标标识）；可见口径与列表裁剪同条件
+        requireWritePerm(existingList);
+
+        // 批量启停：只更新状态字段（不夹带其他业务字段）。停用（置 0）=停止调度、已算结果
+        // 保留（REQ-0024 口径——无删除动作，结果表归计算侧）；置 0/置 1 逐指标发
+        // MetricStatusChangedEvent（启停事件占位——调度消费方归 FEAT-0013/0045，照 0013
+        // 范式只发布不建监听；同态 no-op 的重复事件由消费方幂等承接）
+        IotMetric upd = new IotMetric();
+        upd.setStatus(status);
+        for (Long id : ids) {
+            upd.setId(id);
+            this.updateById(upd);
+        }
+        for (IotMetric metric : existingList) {
+            publishStatusChangedEvent(metric, status);
+        }
+        return true;
+    }

+    // ---------- 校验链（add 与批量生成共用——「生成校验同 add」brief 口径） ----------
+
+    /**
+     * 单项校验 + 白名单构建（createMetric/generateMetrics 共用）：编码（非空/限长/字符集）
+     * → 名称（非空/限长）→ 口径强制标注（agg_type/period 必填合法）→ 日历锁 1 → 开关值域
+     * → ratio 齐备性（比值型缺一拒/非比值型置 NULL）→ 引用存在性/同实例/启用（4001）→
+     * 部门归属 → status 解析 → tenant 会话覆盖。批内撞码由调用方先判（generateMetrics
+     * 第一遍），此处不再重复。
+     *
+     * @param prefetchedItems 批内预载引用数据项（generateMetrics 第二遍产物；单条创建传 null
+     *                        ——引用校验按本次引用集逐项 selectById 补载）
+     */
+    private IotMetric validateAndBuild(Long tenantId, Long instanceId, IotMetricCreateForm item,
+                                       Set<String> seenCodes, Map<Long, IotDataItem> prefetchedItems) {
+        String metricCode = validateMetricCode(item.getMetricCode());
+        if (seenCodes != null) {
+            seenCodes.add(metricCode);
+        }
+        if (item.getMetricName() == null || item.getMetricName().length() > MAX_LEN_METRIC_NAME) {
+            throw paramError("指标名称必填且长度不能超过 " + MAX_LEN_METRIC_NAME);
+        }
+        Integer aggType = resolveAggType(item.getAggType());
+        String period = resolvePeriod(item.getPeriod());
+        Integer calendarType = resolveCalendarType(item.getCalendarType());
+        resolveFlag(item.getYoyFlag(), "同比开关仅允许 0 否 / 1 是");
+        resolveFlag(item.getMomFlag(), "环比开关仅允许 0 否 / 1 是");
+        Integer status = resolveCreateStatus(item.getStatus());
+        validateDept(item.getDeptId());
+
+        // 比值齐备性（比值型缺一 400 拒；非比值型 ratio_* 忽略请求值置 NULL）+ 引用校验（4001）
+        Long ratioNum = aggType == AGG_TYPE_RATIO ? item.getRatioNumItemId() : null;
+        Long ratioDen = aggType == AGG_TYPE_RATIO ? item.getRatioDenItemId() : null;
+        if (aggType == AGG_TYPE_RATIO && (ratioNum == null || ratioDen == null)) {
+            throw paramError("比值型指标必须同时提供分子与分母数据项（ratioNumItemId/ratioDenItemId 缺一不可）");
+        }
+        Map<Long, IotDataItem> itemById = resolveReferenceItems(prefetchedItems, aggType,
+                item.getSourceItemId(), ratioNum, ratioDen);
+        validateReferences(itemById, instanceId, item.getSourceItemId(), ratioNum, ratioDen, aggType);
+
+        // 字段白名单构建新增实体：id/审计字段不复制（框架自增 + MetaObjectHandler 回填）；
+        // tenant_id 服务层从会话覆盖（插件对 iot_ 前缀表 INSERT 自动补列与会话租户同值，
+        // 双保险口径同模板/实例/点位域）
+        IotMetric entity = new IotMetric();
+        entity.setTenantId(tenantId);
+        entity.setMetricCode(metricCode);
+        entity.setMetricName(item.getMetricName());
+        entity.setInstanceId(instanceId);
+        entity.setSourceItemId(item.getSourceItemId());
+        entity.setAggType(aggType);
+        entity.setPeriod(period);
+        entity.setCalendarType(calendarType);
+        entity.setRatioNumItemId(ratioNum);
+        entity.setRatioDenItemId(ratioDen);
+        entity.setYoyFlag(item.getYoyFlag() == null ? DEFAULT_FLAG : item.getYoyFlag());
+        entity.setMomFlag(item.getMomFlag() == null ? DEFAULT_FLAG : item.getMomFlag());
+        entity.setDeptId(item.getDeptId());
+        entity.setStatus(status);
+        return entity;
+    }

+    /**
+     * 引用数据项校验（4001 拒存，TC-0014 步骤 6）：引用携带时必须命中本租户活跃数据项
+     * （多租户插件注入会话租户条件即同租户保证）、instance_id 与指标生效实例一致（同实例）
+     * 且 enabled=1（启用）——引用不存在/跨实例/未启用 → 4001「来源引用不存在」拒存。
+     * source_item_id 与比值分子分母同口径；比值分子分母由齐备性校验保证非空（agg_type=5）。
+     */
+    private void validateReferences(Map<Long, IotDataItem> itemById, Long instanceId,
+                                    Long sourceItemId, Long ratioNumItemId, Long ratioDenItemId,
+                                    Integer aggType) {
+        for (Long itemId : collectItemIds(new LinkedHashSet<>(), aggType, sourceItemId, ratioNumItemId, ratioDenItemId)) {
+            IotDataItem item = itemById.get(itemId);
+            if (item == null) {
+                throw referenceNotFound("来源引用不存在或已删除: 数据项 " + itemId);
+            }
+            if (!instanceId.equals(item.getInstanceId())) {
+                throw referenceNotFound("来源引用与指标归属实例不一致: 数据项 " + itemId
+                        + "（须与指标同实例）");
+            }
+            if (item.getEnabled() == null || item.getEnabled() != 1) {
+                throw referenceNotFound("来源引用的数据项未启用: 数据项 " + itemId);
+            }
+        }
+    }

+    /** 引用 id 收集（source_item_id + 比值分子分母按生效聚合方式；比值型齐备性由调用方先保证） */
+    private Set<Long> collectItemIds(Set<Long> sink, Integer aggType,
+                                     Long sourceItemId, Long ratioNumItemId, Long ratioDenItemId) {
+        if (sourceItemId != null) {
+            sink.add(sourceItemId);
+        }
+        if (aggType != null && aggType == AGG_TYPE_RATIO) {
+            if (ratioNumItemId != null) {
+                sink.add(ratioNumItemId);
+            }
+            if (ratioDenItemId != null) {
+                sink.add(ratioDenItemId);
+            }
+        }
+        return sink;
+    }
+
+    /** 引用数据项批内预载（一次 selectByIds——多租户插件注入会话租户条件即同租户保证） */
+    private Map<Long, IotDataItem> loadItemsByIds(Set<Long> itemIds) {
+        if (itemIds.isEmpty()) {
+            return Map.of();
+        }
+        return dataItemMapper.selectByIds(itemIds).stream()
+                .collect(Collectors.toMap(IotDataItem::getId, Function.identity(), (a, b) -> a));
+    }
+
+    /**
+     * 引用校验数据集解析：批内预载（generateMetrics 传入）时合并缺失引用补载结果——预载命中
+     * 直接复用、未命中（如单条路径 / 预载遗漏）按本次引用集 selectById 补载；单条创建（传 null）
+     * 退化为按本次引用集一次 selectById。
+     */
+    private Map<Long, IotDataItem> resolveReferenceItems(Map<Long, IotDataItem> prefetchedItems, Integer aggType,
+                                                         Long sourceItemId, Long ratioNumItemId,
+                                                         Long ratioDenItemId) {
+        Set<Long> refIds = collectItemIds(new LinkedHashSet<>(), aggType, sourceItemId, ratioNumItemId, ratioDenItemId);
+        if (prefetchedItems == null || prefetchedItems.isEmpty()) {
+            return loadItemsByIds(refIds);
+        }
+        Set<Long> missing = refIds.stream()
+                .filter(id -> !prefetchedItems.containsKey(id))
+                .collect(Collectors.toCollection(LinkedHashSet::new));
+        if (missing.isEmpty()) {
+            return prefetchedItems;
+        }
+        Map<Long, IotDataItem> merged = new LinkedHashMap<>(prefetchedItems);
+        merged.putAll(loadItemsByIds(missing));
+        return merged;
+    }

+    // ---------- 内部辅助 ----------
+
+    /** 编码校验：非空、长度 ≤96、字符集 [A-Za-z0-9_]（表单 @Size/@Pattern 之外的服务层双保险） */
+    private String validateMetricCode(String metricCode) {
+        if (metricCode == null || metricCode.isBlank()) {
+            throw paramError("指标编码不能为空");
+        }
