+    private void validateFormula(String formula) {
+        if (formula == null || formula.isEmpty()) {
+            return;
+        }
+        if (formula.length() > MAX_LEN_FORMULA) {
+            throw paramError("默认公式长度不能超过 " + MAX_LEN_FORMULA);

+                - formula.chars().filter(c -> c == '}').count();

+            throw paramError("默认公式基础格式非法：括号/占位符不配平（Aviator 语法校验归 FEAT-0012，"
+                    + "本任务仅做基础格式校验）");
+        }
+    }

+        entity.setDefaultAlarmHigh(item.getDefaultAlarmHigh());

+            throw duplicateTemplateCode(entity.getTemplateCode());

+    @Override
+    @Transactional(rollbackFor = Exception.class)
+    public boolean updateTemplate(IotModelTemplate template) {
+        IotModelTemplate existing = this.getById(template.getId());
+        if (existing == null) {
+            throw paramError("模板不存在或已删除");
+        }
+        // template_code 创建后不可改（一期不开放改编码，REQ-0004 核心规则）：
+        // 请求携带的编码与库中不一致（含未携带）即拒绝（照 data_code 不可改范式）
+        if (template.getTemplateCode() == null || !template.getTemplateCode().equals(existing.getTemplateCode())) {
+            throw paramError("模板编码不允许修改");
+        }
+        validateNodeType(template.getNodeType());
+        if (template.getStatus() != null && template.getStatus() != 0 && template.getStatus() != 1) {
+            throw paramError("模板状态仅允许 0 禁用 / 1 启用");
+        }
+        if (template.getVersion() != null && template.getVersion() < 1) {
+            throw paramError("模板版本号必须为正整数");
+        }
+
+        // 字段白名单构建更新（不直接 updateById 请求实体；不可变字段不回写、连 no-op 都不进 SET）：
+        // 1) 仅 templateName/nodeType/version/status/remark 可变——templateCode 经上方不可改校验
+        //    后不进更新集（库内值本就不变，SET 不含不可变字段，语义更干净）；
+        // 2) tenant_id 不进更新集（插件 WHERE 条件已隔离租户，SET 也不得改写；模板不可跨租户挪移）；
+        // 3) createBy/createTime/updateBy/updateTime/deleted 等审计与逻辑删字段不进更新集，
+        //    杜绝客户端 mass assignment（updateBy/updateTime 由 MetaObjectHandler 按当前登录者回填）。
+        IotModelTemplate upd = new IotModelTemplate();
+        upd.setId(existing.getId());
+        upd.setTemplateName(template.getTemplateName());
+        upd.setNodeType(template.getNodeType());
+        upd.setVersion(template.getVersion());
+        upd.setStatus(template.getStatus());
+        upd.setRemark(template.getRemark());
+        return this.updateById(upd);
+    }

+    @Override
+    @Transactional(rollbackFor = Exception.class)
+    public boolean deleteTemplates(List<Long> ids) {
+        if (ids == null || ids.isEmpty()) {
+            throw paramError("请选择要删除的模板");
+        }
+        // fail-closed：批内全部 id 必须存在且活跃（@TableLogic 过滤已删行），
+        // 缺失即整批拒绝（不半删）
+        List<IotModelTemplate> existingList = this.listByIds(ids);
+        if (existingList.size() < ids.size()) {
+            Map<Long, IotModelTemplate> found = existingList.stream()
+                    .collect(Collectors.toMap(IotModelTemplate::getId, Function.identity()));
+            List<Long> missing = ids.stream().filter(id -> !found.containsKey(id)).toList();
+            throw paramError("模板不存在或已删除: " + missing);
+        }
+
+        // 删除保护：两类外部引用任一命中即整批拒绝并汇总来源（引用计数查询）
+        List<String> reasons = new ArrayList<>();
+
+        // ① 实例引用（iot_model_instance.template_id）：租户条件由插件追加、deleted 由
+        //    @TableLogic 运行时过滤——引用计数天然只数本租户活跃实例
+        LambdaQueryWrapper<IotModelInstance> instanceWrapper = new LambdaQueryWrapper<>();
+        instanceWrapper.select(IotModelInstance::getInstanceCode)
+                .in(IotModelInstance::getTemplateId, ids);
+        List<IotModelInstance> instanceRefs = modelInstanceMapper.selectList(instanceWrapper);
+        if (!instanceRefs.isEmpty()) {
+            List<String> instanceCodes = instanceRefs.stream()
+                    .map(IotModelInstance::getInstanceCode)
+                    .collect(Collectors.toList());
+            reasons.add("模板被实例引用，禁止删除（实例: " + joinLimit(instanceCodes) + "）");
+        }
+
+        // ② 模板数据项引用（iot_data_item_template.template_id）：模板下仍有数据项即删除
+        //    会产生孤儿数据项行（照网关/通道「其下有点位时拒绝」范式 fail-closed）
+        LambdaQueryWrapper<IotDataItemTemplate> itemWrapper = new LambdaQueryWrapper<>();
+        itemWrapper.select(IotDataItemTemplate::getTemplateId)
+                .in(IotDataItemTemplate::getTemplateId, ids);
+        List<IotDataItemTemplate> itemRefs = dataItemTemplateMapper.selectList(itemWrapper);
+        if (!itemRefs.isEmpty()) {
+            Map<Long, Long> countByTemplate = itemRefs.stream()
+                    .collect(Collectors.groupingBy(IotDataItemTemplate::getTemplateId, Collectors.counting()));
+            Map<Long, String> codeById = existingList.stream()
+                    .collect(Collectors.toMap(IotModelTemplate::getId, IotModelTemplate::getTemplateCode));
+            List<String> itemSources = countByTemplate.entrySet().stream()
+                    .map(e -> codeById.getOrDefault(e.getKey(), String.valueOf(e.getKey())) + "（" + e.getValue() + " 项）")
+                    .collect(Collectors.toList());
+            reasons.add("模板存在模板数据项，禁止删除（先清理数据项: " + joinLimit(itemSources) + "）");
+        }
+
+        if (!reasons.isEmpty()) {
+            throw paramError(String.join("；", reasons));
+        }
+
+        // 无引用：逻辑删（@TableLogic，deleted=1）
+        return this.removeByIds(ids);
+    }
+
+    private void validateNodeType(Integer nodeType) {
+        if (nodeType == null || nodeType < NODE_TYPE_MIN || nodeType > NODE_TYPE_MAX) {
+            throw paramError("节点类型仅允许 1-5（1 工厂 2 车间 3 产线 4 设备 5 组件）");
+        }
+    }
+
+    private boolean isOverLength(String value, int max) {
+        return value != null && value.length() > max;
+    }
+
+    /** 引用来源展示限长：最多展示前 5 个，超出以「等 N 个」汇总 */
+    private String joinLimit(List<String> sources) {
+        if (sources.size() <= MAX_REF_SOURCES_DISPLAYED) {
+            return String.join(", ", sources);
+        }
+        return String.join(", ", sources.subList(0, MAX_REF_SOURCES_DISPLAYED))
+                + " 等 " + sources.size() + " 个";
+    }
+
+    private BusinessException duplicateTemplateCode(String templateCode) {
+        return new BusinessException(ResultCode.DUPLICATE_CODE.getCode(), "模板编码已存在: " + templateCode);
+    }
+
+    private BusinessException paramError(String message) {
+        return new BusinessException(ResultCode.PARAM_ERROR.getCode(), message);
+    }
