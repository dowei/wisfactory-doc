+// Package itembridge 承载 TASK-0084 桥接生产者。
+//
+// 本文件实现 pointCode → (fullKey, instanceCode) 的权威映射（TASK-0084 修订轮 C-1）：
+// pointCode 是点位编码（现场层，下划线形态如 V29_NUM_01），fullKey 是数据项全键
+// （instance_code + "." + data_code）——两者是不同编码空间，不能从 pointCode 字符串结构臆造
+// instance。正确路径是经配置快照反查 pointCode → Point(ID) → DataItem(PointID==ID)。
+package itembridge
+
+import (
+	"fmt"
+	"strings"
+
+	"net.otlab/ingest/internal/snapshot"
+)
+
+// PointMapper pointCode → (数据项全键 fullKey, 主题实例段 instanceCode) 映射薄接口。
+// 生产实现 = SnapshotPointMapper（配置快照反查）；测试可用同接口假实现。
+// 未绑定/未登记 → 返回 error（桥接 WARN + 丢弃，不造假主题）。
+type PointMapper interface {
+	Map(pointCode string) (fullKey, instanceCode string, err error)
+}
+
+// SnapshotPointMapper 基于配置快照的映射实现（C-1 裁决）。
+//
+// 反查链：Cache.LookupPoint(pointCode) → Point.ID → 快照 DataItems 中 PointID==ID 的
+// DataItem → FullKey（原样发布）+ 从 FullKey 去末段取 instanceCode（此时 "." 才是契约定义的
+// 数据项全键分隔符，解析合法）。快照未装载 / pointCode 未登记 / 点位未绑数据项 / FullKey
+// 无实例段 → error（fail-soft 丢弃）。
+type SnapshotPointMapper struct {
+	cache *snapshot.Cache
+}
+
+// NewSnapshotPointMapper 构造（cache 由 main 的 snapCache 注入；nil 时 Map 恒失败）。
+func NewSnapshotPointMapper(cache *snapshot.Cache) *SnapshotPointMapper {
+	return &SnapshotPointMapper{cache: cache}
+}
+
+// Map 实现 PointMapper。
+func (m *SnapshotPointMapper) Map(pointCode string) (string, string, error) {
+	if m == nil || m.cache == nil {
+		return "", "", fmt.Errorf("itembridge: 配置快照缓存未装配，pointCode=%q 无法反查", pointCode)
+	}
+	p, ok := m.cache.LookupPoint(pointCode)
+	if !ok || p == nil {
+		return "", "", fmt.Errorf("itembridge: pointCode %q 未在配置快照点位中登记", pointCode)
+	}
+	snap := m.cache.Get()
+	if snap == nil {
+		return "", "", fmt.Errorf("itembridge: 配置快照未装载，pointCode=%q 无法反查", pointCode)
+	}
+	var item *snapshot.DataItem
+	for i := range snap.DataItems {
+		di := &snap.DataItems[i]
+		if di.PointID != nil && *di.PointID == p.ID {
+			item = di
+			break
+		}
+	}
+	if item == nil {
+		return "", "", fmt.Errorf("itembridge: 点位 %q(id=%d) 未绑定数据项", pointCode, p.ID)
+	}
+	full := item.FullKey
+	idx := strings.LastIndex(full, ".")
+	if idx <= 0 || idx == len(full)-1 {
+		return "", "", fmt.Errorf("itembridge: 数据项 fullKey %q 无法解析实例段（须 instance_code + '.' + data_code）", full)
+	}
+	return full, full[:idx], nil
+}
