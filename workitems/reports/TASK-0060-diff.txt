+    const prev = valueByTs.get(oneYearBefore(row.ts));
+    if (prev != null) lastYear.push([key, prev]);
+  }
+
+  const downsampleNote = tier.bucketMs > 0 ? `当前展示为降采样结果：${tier.label.replace(/^降采样结果：/, '')}` : '当前展示为原始粒度，未降采样';

+    ? `去年同期对比点 ${lastYear.length}/${current.length}（仅取真实同 ts 去年同期行）`
+    : '去年同期对比行缺失（不以比率反推历史值，不造假对比曲线）';
+  return { current, lastYear, annotation: `${downsampleNote} · ${compareNote}`, hasData: current.length > 0 };
