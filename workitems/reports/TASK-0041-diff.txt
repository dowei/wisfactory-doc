+/**
+ * 轻量表达式求值器（0041 执行器范围内最小口径）：四则运算 + 括号 + 一元负号 +
+ * 标识符引用（fullKey / 点位编码）+ 数值字面量；比较运算 == != > >= < <= 与逻辑
+ * && || !（返回 1.0/0.0）；三元 ?: 。
+ *
+ * <p><b>边界</b>：DELTA/INTEGRAL 等内置算子归 TASK-0042，本求值器不实现——遇到
+ * 函数调用抛 {@link UnsupportedOperationException}，由执行器按「单节点失败隔离」
+ * 处理（质量码 6，不打假 0）。真 Aviator 归 FEAT-0012；本类仅满足 0041 拓扑级联/
+ * 汇总/失败隔离契约测试所需的确定性求值。
+ *
+ * @author dowei
+ * @date 2026-11-24
+ */
+public final class ArithmeticEvaluator {
+
+    /** 标识符解析器（fullKey / 点位编码 → Double；缺失返回 null → 视为无值） */
+    @FunctionalInterface
+    public interface Resolver {
+        Double resolve(String identifier);
+    }
+
+    private final String src;
+    private final Resolver resolver;
+    private int pos;
+
+    private ArithmeticEvaluator(String src, Resolver resolver) {
+        this.src = src;
+        this.resolver = resolver;
+    }
+
+    /** 求值；任何解析/引用失败上抛 RuntimeException（由调用方隔离为单节点失败） */
+    public static double evaluate(String expression, Resolver resolver) {
+        ArithmeticEvaluator e = new ArithmeticEvaluator(expression, resolver);
+        double result = e.parseTernary();
+        e.skipWs();
+        if (e.pos < e.src.length()) {
+            throw new IllegalArgumentException("表达式尾部残留: " + e.src.substring(e.pos));
+        }
+        return result;
+    }
+
+    // ?: (lowest)
+    private double parseTernary() {
+        double cond = parseOr();
+        skipWs();
+        if (consume('?')) {
+            double a = parseTernary();
+            skipWs();
+            expect(':');
+            double b = parseTernary();
+            return cond != 0 ? a : b;
+        }
+        return cond;
+    }
+
+    private double parseOr() {
+        double v = parseAnd();
+        while (true) {
+            skipWs();
+            if (consume("||")) {
+                double r = parseAnd();
+                v = (v != 0 || r != 0) ? 1 : 0;
+            } else {
+                return v;
+            }
+        }
+    }
+
+    private double parseAnd() {
+        double v = parseEquality();
+        while (true) {
+            skipWs();
+            if (consume("&&")) {
+                double r = parseEquality();
+                v = (v != 0 && r != 0) ? 1 : 0;
+            } else {
+                return v;
+            }
+        }
+    }
+
+    private double parseEquality() {
+        double v = parseComparison();
+        while (true) {
+            skipWs();
+            if (consume("==")) {
+                double r = parseComparison();
+                v = r == v ? 1 : 0;
+            } else if (consume("!=")) {
+                double r = parseComparison();
+                v = r != v ? 1 : 0;
+            } else {
+                return v;
+            }
+        }
+    }
+
+    private double parseComparison() {
+        double v = parseAdditive();
+        while (true) {
+            skipWs();
+            if (consume(">=")) {
+                v = parseAdditive() >= v ? 1 : 0;
+            } else if (consume("<=")) {
+                v = parseAdditive() <= v ? 1 : 0;
+            } else if (consume(">")) {
+                double r = parseAdditive();
+                v = v > r ? 1 : 0;
+            } else if (consume("<")) {
+                double r = parseAdditive();
+                v = v < r ? 1 : 0;
+            } else {
+                return v;
+            }
+        }
+    }
+
+    private double parseAdditive() {
+        double v = parseMultiplicative();
+        while (true) {
+            skipWs();
+            if (consume('+')) {
+                v += parseMultiplicative();
+            } else if (consume('-')) {
+                v -= parseMultiplicative();
+            } else {
+                return v;
+            }
+        }
+    }
+
+    private double parseMultiplicative() {
+        double v = parseUnary();
+        while (true) {
+            skipWs();
+            if (consume('*')) {
+                v *= parseUnary();
+            } else if (consume('/')) {
+                double d = parseUnary();
+                if (d == 0) {
+                    throw new ArithmeticException("除零");
+                }
+                v /= d;
+            } else if (consume('%')) {
+                v %= parseUnary();
+            } else {
+                return v;
+            }
+        }
+    }
+
+    private double parseUnary() {
+        skipWs();
+        if (consume('!')) {
+            return parseUnary() == 0 ? 1 : 0;
+        }
+        if (consume('-')) {
+            return -parseUnary();
+        }
+        if (consume('+')) {
+            return parseUnary();
+        }
+        return parseAtom();
+    }
+
+    private double parseAtom() {
+        skipWs();
+        if (pos >= src.length()) {
+            throw new IllegalArgumentException("表达式意外结束");
+        }
+        char c = src.charAt(pos);

+            skipWs();

+            return v;
+        }

+            while (pos < src.length()

+                pos++;
+            }
+            return Double.parseDouble(src.substring(start, pos));
+        }
+        if (Character.isJavaIdentifierStart(c)) {
+            int start = pos;
+            while (pos < src.length()
+                    && (Character.isJavaIdentifierPart(src.charAt(pos)) || src.charAt(pos) == '.')) {
+                pos++;
+            }
+            String ident = src.substring(start, pos);
+            if (consume('(')) {

+                        + "（DELTA/INTEGRAL 归 TASK-0042）");
+            }
+            Double value = resolver.resolve(ident);
+            if (value == null) {
+                throw new IllegalStateException("无值输入: " + ident);
+            }
+            return value;
+        }
+        throw new IllegalArgumentException("非法字符: " + c + " @ " + pos);
+    }
+
+    private void skipWs() {
+        while (pos < src.length() && Character.isWhitespace(src.charAt(pos))) {
+            pos++;
+        }
+    }
+
+    private boolean consume(char c) {
+        skipWs();
+        if (pos < src.length() && src.charAt(pos) == c) {
+            pos++;
+            return true;
+        }
+        return false;
+    }
+

+        if (src.startsWith(s, pos)) {

+            return true;
+        }
+        return false;
+    }

+        if (!consume(c)) {
+            throw new IllegalArgumentException("期望 '" + c + "' @ " + pos + ": " + src);
+        }
+    }

+            throw new IllegalArgumentException("calc_period 不可为负: " + seconds);
+        }
+        return new CalcPeriod(Kind.FIXED, seconds);
+    }
+
+    /**
+     * 周期码 → CalcPeriod（与 backend mapCalcPeriod 同款口径 + MONTH/YEAR）。
+     * 未知码 fail-closed（不静默退化为到达驱动）。
+     */
+    public static CalcPeriod fromCode(String code) {
+        if (code == null || code.isBlank()) {
+            return arrival();
+        }
+        String value = code.trim();
+        if (value.matches("\\d+")) {
+            return fromSeconds((int) Long.parseLong(value));
+        }
+        return switch (value.toUpperCase(java.util.Locale.ROOT)) {
+            case "SECOND" -> new CalcPeriod(Kind.FIXED, 1);
+            case "MINUTE" -> new CalcPeriod(Kind.FIXED, 60);
+            case "HOUR" -> new CalcPeriod(Kind.FIXED, 3600);
+            case "DAY" -> new CalcPeriod(Kind.FIXED, 86400);
+            case "MONTH" -> new CalcPeriod(Kind.MONTH, 0);
+            case "YEAR" -> new CalcPeriod(Kind.YEAR, 0);
+            default -> throw new IllegalArgumentException("无法识别的计算周期码: " + code);
+        };
+    }
+
+    public Kind kind() {
+        return kind;
+    }
+
+    public boolean isArrival() {
+        return kind == Kind.ARRIVAL;
+    }
+
+    public long seconds() {
+        return seconds;
+    }
+
+    /** 计算给定时刻所在周期窗（UTC 对齐；固定秒数自 epoch 起算，月/年按日历） */
+    public Boundary boundaryOf(long epochMillis) {
+        if (epochMillis < 0) {
+            throw new IllegalArgumentException("epochMillis 不可为负: " + epochMillis);
+        }
+        java.time.ZonedDateTime t = java.time.Instant.ofEpochMilli(epochMillis)
+                .atZone(java.time.ZoneOffset.UTC);
+        java.time.ZonedDateTime start;
+        java.time.ZonedDateTime end;
+        switch (kind) {
+            case MONTH -> {
+                start = t.withDayOfMonth(1).truncatedTo(java.time.temporal.ChronoUnit.DAYS);
+                end = start.plusMonths(1);
+            }
+            case YEAR -> {
+                start = t.withDayOfYear(1).truncatedTo(java.time.temporal.ChronoUnit.DAYS);
+                end = start.plusYears(1);
+            }
+            case FIXED -> {
+                long span = seconds * 1000L;
+                long bucket = (epochMillis / span) * span;
+                return new Boundary(bucket, bucket + span);
+            }
+            default -> throw new IllegalStateException("到达驱动项无周期边界");
+        }
+        return new Boundary(start.toInstant().toEpochMilli(), end.toInstant().toEpochMilli());
+    }
+}

+                continue;

+                continue;

+                }

+                        results, failed, visitOrder);

+                trigger.kind(), visitOrder.size(), out.size(), failed.size(), visitOrder);

+    private CalcResult computeItem(DataItem item, GraphNode node, DependencyGraph graph, long ts) {
+        int sourceType = item.getSourceType() == null ? 1 : item.getSourceType();
+        String key = node.key();
+        return switch (sourceType) {
+            case 1 -> {
+                ValueSample sample = store.get(key);
+                if (sample == null) {
+                    yield CalcResult.failure(key, 1, "直绑项无值");
+                }
+                yield CalcResult.success(key, 1, sample);
+            }
+            case 2 -> {
+                String formula = item.getFormula();
+                if (formula == null || formula.isBlank()) {
+                    throw new IllegalStateException("公式项缺表达式");
+                }
+                double value = ArithmeticEvaluator.evaluate(formula,
+                        id -> {
+                            ValueSample s = store.get(id);
+                            return s == null ? null : s.value();
+                        });
+                yield CalcResult.success(key, 2, ValueSample.of(value, ts));
+            }
+            case 3 -> {
+                RollupOperator.Method method = RollupOperator.Method.parse(item.getRollupMethod());
+                List<RollupOperator.Child> children = childSamples(key, graph);
+                RollupOperator.RollupResult r = rollup.evaluate(method, children, ts);
+                yield CalcResult.success(key, 3, r.sample());
+            }
+            default -> throw new IllegalStateException("未知 sourceType: " + sourceType);
+        };
+    }

+    private List<RollupOperator.Child> childSamples(String rollupKey, DependencyGraph graph) {
+        List<RollupOperator.Child> children = new ArrayList<>();
+        for (GraphEdge edge : graph.getEdges()) {
+            if (edge.type() == EdgeType.ROLLUP && edge.from().equals(rollupKey)) {
+                GraphNode child = graph.node(edge.to());
+                boolean enabled = child == null || child.item() == null
+                        || child.item().getEnabled() == null || child.item().getEnabled() != 0;
+                children.add(new RollupOperator.Child(edge.to(), store.get(edge.to()), enabled));
+            }
+        }
+        return children;
+    }

+    private Set<String> selectTargets(Trigger trigger, DependencyGraph graph) {
+        Set<String> all = new LinkedHashSet<>();
+        if (!trigger.isArrival()) {
+            // 周期通道：整图（点位/指标节点在计算中自然跳过）
+            all.addAll(graph.getTopoOrder());
+            return all;
+        }
+        // 到达通道：触发项 + 其下游可达子图（收敛影响面）
+        Set<String> downstream = downstreamOf(trigger.fullKey(), graph);
+        all.addAll(downstream);
+        return all;
+    }
+
+    /** 从根出发，按「from 依赖 to」反向求下游（谁依赖我） */
+    static Set<String> downstreamOf(String root, DependencyGraph graph) {
+        Map<String, Set<String>> dependents = new LinkedHashMap<>();
+        for (GraphEdge edge : graph.getEdges()) {
+            dependents.computeIfAbsent(edge.to(), k -> new LinkedHashSet<>()).add(edge.from());
+        }
+        Set<String> seen = new LinkedHashSet<>();
+        seen.add(root);
+        java.util.Deque<String> queue = new java.util.ArrayDeque<>();
+        queue.add(root);
+        while (!queue.isEmpty()) {
+            String cur = queue.remove();
+            for (String next : dependents.getOrDefault(cur, Set.of())) {
+                if (seen.add(next)) {
+                    queue.add(next);
+                }
+            }
+        }
+        return seen;
+    }
+
+    /** 结果时间戳：周期项=周期窗右边界；到达项=事件 ts（brief 决策 1） */
+    private long resultTimestamp(Trigger trigger, DependencyGraph graph) {
+        if (!trigger.isArrival()) {
+            long periodSec = Math.max(1L, trigger.periodMs() / 1000L);
+            CalcPeriod.Boundary b = CalcPeriod.fromSecondsLong(periodSec).boundaryOf(trigger.at());
+            return b.endMillis();
+        }
+        return trigger.at();
+    }

+                ? 1 : node.item().getSourceType();
+        results.put(key, CalcResult.failure(key, sourceType, reason));
+        failed.add(key);
+        log.error("[CALC-EXEC] 单节点失败（隔离，不发假 0）key={} reason={}", key, reason);
+    }
+
+    /** 到达驱动项（calc_period 0/null）——周期通道跳过 */
+    private static boolean isArrivalDriven(DataItem item) {
+        Long period = item.getCalcPeriod();
+        return period == null || period <= 0;
+    }

+    /** 硬依赖（排除汇总边——子级缺数走 quality=7，不算失败传播） */
+    private static Set<String> hardDependencies(String key, DependencyGraph graph) {
+        Set<String> out = new LinkedHashSet<>();
+        for (GraphEdge edge : graph.getEdges()) {
+            if (edge.from().equals(key) && edge.type() != EdgeType.ROLLUP) {
+                out.add(edge.to());
+            }
+        }
+        return out;
+    }
+
+    private static boolean disjoint(Set<String> a, Set<String> b) {
+        for (String x : a) {
+            if (b.contains(x)) {
+                return false;
+            }
+        }
+        return true;
+    }
+
+    /** 供调度器构造固定秒数周期边界 */
+    public static long boundaryEnd(long periodMs, long atMillis) {
+        if (periodMs <= 0) {
+            throw new IllegalArgumentException("periodMs 须 > 0");
+        }
+        long bucket = (atMillis / periodMs) * periodMs;
+        return bucket + periodMs;
+    }
+
+    /** 供调度器构造固定秒数周期窗键 */
+    public static String windowKey(String fullKey, long periodMs, long atMillis) {
+        return fullKey + "@" + (atMillis / Math.max(1L, periodMs));
+    }

+ * 静态表达式，无法表达「按数据项配置的 N 秒周期 + 运行时换图即时生效」；自实现线程

+ * 下一个 tick 命中）；命中判据=当前时刻跨过周期边界（at % period == 0 的边界点），
+ * 用 {@link WindowIdempotencyGuard} 保证同一周期窗只触发一次（进程重启后由窗口键
+ * 去重——多实例化登记演进位）。

+ * @date 2026-11-24

+
+    private static final Logger log = LoggerFactory.getLogger(PeriodicScheduler.class);
+
+    private final GraphHolder holder;
+    private final DualTriggerExecutor executor;
+    private final WindowIdempotencyGuard guard;
+    private final long tickSeconds;
+    private final Map<String, Long> lastFiredBucket = new ConcurrentHashMap<>();
+    private volatile ScheduledExecutorService scheduler;
+
+    public PeriodicScheduler(GraphHolder holder, DualTriggerExecutor executor,
+                             WindowIdempotencyGuard guard, long tickSeconds) {
+        if (tickSeconds < 1) {
+            throw new IllegalArgumentException("tickSeconds 须 >= 1");
+        }
+        this.holder = holder;
+        this.executor = executor;
+        this.guard = guard;
+        this.tickSeconds = tickSeconds;
+    }
+
+    public synchronized void start() {
+        if (scheduler != null) {
+            return;
+        }
+        scheduler = Executors.newSingleThreadScheduledExecutor(r -> {
+            Thread t = new Thread(r, "calc-periodic-scheduler");
+            t.setDaemon(true);
+            return t;
+        });
+        scheduler.scheduleAtFixedRate(this::safeTick, tickSeconds, tickSeconds, TimeUnit.SECONDS);
+        log.info("[CALC-SCHED] 周期调度启动 tick={}s（整点触发 + 窗边界对齐）", tickSeconds);
+    }
+
+    public synchronized void stop() {
+        if (scheduler != null) {
+            scheduler.shutdownNow();
+            scheduler = null;
+            log.info("[CALC-SCHED] 周期调度已停止");
+        }
+    }
+
+    private void safeTick() {
+        try {
+            tick(System.currentTimeMillis());
+        } catch (RuntimeException e) {
+            log.error("[CALC-SCHED] tick 未预期异常（fail-soft，下个 tick 重试）", e);
+        }
+    }
+
+    /** 单次调度扫描（可测试入口）：按当前图的周期项检查是否跨过边界 */
+    public void tick(long nowMillis) {
+        net.otlab.calc.runtime.AppliedGraph applied = holder.current();
+        if (applied == null) {
+            return;
+        }
+        DependencyGraph graph = applied.graph();
+        for (String key : graph.getTopoOrder()) {
+            var node = graph.node(key);
+            if (node == null || node.kind() != net.otlab.calc.graph.NodeKind.ITEM || node.item() == null) {
+                continue;
+            }
+            DataItem item = node.item();
+            if (item.getCalcPeriod() == null || item.getCalcPeriod() <= 0) {
+                continue; // 到达驱动项不参与周期调度
+            }
+            if (item.getEnabled() != null && item.getEnabled() == 0) {
+                continue;
+            }
+            long periodMs = item.getCalcPeriod() * 1000L;
+            long boundary = DualTriggerExecutor.boundaryEnd(periodMs, nowMillis);
+            // 整点触发：当前时刻所在周期窗（bucket）与上次触发窗口不同即触发一次——
+            // tick 粒度保证在新窗开始后的下个 tick 命中；结果时间戳仍由执行器对齐到窗右边界。
+            long bucket = nowMillis / periodMs;
+            Long last = lastFiredBucket.get(key);
+            if (last != null && last == bucket) {
+                continue;
+            }
+            // 幂等由执行器的 WindowIdempotencyGuard 按 windowKey 统一完成（同窗只算一次）；
+            // 此处 bucket 比较只控制「不再重复提交同一窗的触发请求」。
+            lastFiredBucket.put(key, bucket);
+            String windowKey = DualTriggerExecutor.windowKey(key, periodMs, nowMillis);
+            Trigger trigger = Trigger.periodic(windowKey, periodMs, nowMillis);
+            log.info("[CALC-SCHED] 周期触发 key={} period={}ms boundary={} bucket={}",
+                    key, periodMs, boundary, bucket);
+            executor.execute(trigger, graph);
+        }
+    }
+
+    /** 启动后是否已 Running（观测/测试） */
+    public boolean isRunning() {
+        return scheduler != null && !scheduler.isShutdown();
+    }

+            if (c.enabled() && c.sample() != null) {
+                maxTs = Math.max(maxTs, c.sample().ts());
+            }
+        }
+        ValueSample sample;
+        if (included.isEmpty()) {
+            // 无可用子级：不写 0，quality=7 汇总不完整（CONV-02）
+            sample = new ValueSample(0.0, 7, maxTs);

+                case SUM -> included.stream().mapToDouble(Double::doubleValue).sum();
+                case AVG -> included.stream().mapToDouble(Double::doubleValue).average().orElse(0.0);
+                case MAX -> included.stream().mapToDouble(Double::doubleValue).max().orElse(0.0);
+                case MIN -> included.stream().mapToDouble(Double::doubleValue).min().orElse(0.0);
+            };
+            sample = new ValueSample(value, incomplete ? 7 : 0, maxTs);
+        }
+        return new RollupResult(sample, incomplete, q7, disabled, missing);

+/**
+ * 窗口幂等 + 同秒合并去重的进程内有界 TTL 登记表（brief 决策 2）。
+ *
+ * <p>语义：同一 {@code windowKey}（数据项全键 + 周期窗）只允许第一次触发执行，后续
+ * 到达判为重复（节流合并 + 幂等——只产出一次结果）。登记表有界（FIFO 淘汰最旧）
+ * 且带 TTL（默认 5 分钟），避免长期运行内存无界增长。
+ *
+ * <p>演进位：多实例部署时本表需 Redis 化（SET NX EX windowKey 实现跨实例幂等），
+ * 接口不变、实现可零改动替换；本期为进程内假实现（brief 决策 2 明示）。
+ *
+ * @author dowei
+ * @date 2026-11-24
+ */
+public final class WindowIdempotencyGuard {
+
+    /** 触发决策 */
+    public enum Decision {
+        /** 本窗口首次——执行 */
+        FIRST,
+        /** 本窗口已登记——节流合并/幂等跳过（不重复产出） */
+        DUPLICATE
+    }
+
+    private final int maxEntries;
+    private final long ttlMillis;
+    private final LinkedHashMap<String, Long> entries = new LinkedHashMap<>();
+
+    public WindowIdempotencyGuard() {
+        this(4096, 5L * 60L * 1000L);
+    }
+
+    public WindowIdempotencyGuard(int maxEntries, long ttlMillis) {
+        if (maxEntries < 1) {
+            throw new IllegalArgumentException("maxEntries 须 >= 1");
+        }
+        if (ttlMillis < 1) {
+            throw new IllegalArgumentException("ttlMillis 须 >= 1");
+        }
+        this.maxEntries = maxEntries;
+        this.ttlMillis = ttlMillis;
+    }
+
+    /**
+     * 尝试占用窗口键。首次返回 {@link Decision#FIRST}；重复/未过期返回
+     * {@link Decision#DUPLICATE}。过期条目视为新窗口（TTL 到期后可再执行）。
+     */
+    public synchronized Decision tryAcquire(String windowKey, long nowMillis) {
+        Long at = entries.get(windowKey);
+        if (at != null) {
+            if (nowMillis - at < ttlMillis) {
+                return Decision.DUPLICATE;
+            }
+            entries.remove(windowKey);
+        }
+        // 有界：FIFO 淘汰最旧（LinkedHashMap 插入序）
+        while (entries.size() >= maxEntries) {
+            Iterator<String> it = entries.keySet().iterator();
+            it.next();
+            it.remove();
+        }
+        entries.put(windowKey, nowMillis);
+        return Decision.FIRST;
+    }
+
+    /** 当前登记条目数（观测/测试用） */
+    public synchronized int size() {
+        return entries.size();
+    }
+
+    public int maxEntries() {
+        return maxEntries;
+    }
+
+    public long ttlMillis() {
+        return ttlMillis;
+    }

+        AtomicReference<ValueSample> seen = new AtomicReference<>();
+        RedisArrivalSubscriber sub = new RedisArrivalSubscriber(addr, "", 0, tenant, (key, sample) -> {

+        try {

+                    + ts + ",\"value\":42.5,\"quality\":0}";
+            publish(host, port, channel, payload);
+            assertTrue(latch.await(5, TimeUnit.SECONDS), "收到真 Pub/Sub 事件");
+            assertEquals("P41_LIVE", seenKey.get());
+            assertEquals(42.5, seen.get().value(), 1e-6);
+            assertEquals(0, seen.get().quality());
+            assertEquals(ts, seen.get().ts());

+            sub.close();
+        }

+    private static void publish(String host, int port, String channel, String payload) throws Exception {
+        try (Socket s = new Socket()) {
+            s.connect(new InetSocketAddress(host, port), 3000);

+            String[] args = {"PUBLISH", channel, payload};
+            StringBuilder sb = new StringBuilder("*").append(args.length).append("\r\n");
+            for (String a : args) {

+            }
