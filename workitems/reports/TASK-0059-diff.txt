+  value: number;
+  quality: number | null;

+/** 降采样结果（含必显式展示的标注） */

+/**
+ * 对已取回的点位曲线做时间桶聚合（ADR-0010 留痕）。
+ *
+ * 输入 ts 为 epoch 毫秒或可解析时间字符串；无法解析的行跳过（不造假点）。
+ * 60s/300s 档取桶内最后一条（保留最新质量码）；3600s 档取桶内值的算术均值
+ * （质量码取桶内最差：超上限=1/超下限=2 优先，其次断线=3，再其次其他非 0）。
+ * 原始档仅做排序与过滤，标注为「未降采样」。
+ */
+export function downsampleSeries(
+  points: Array<{ ts: string | null; value: number | null; quality: number | null }>,
+  tier: DownsampleTier
+): DownsampledSeries {
+  const parsed: CurveSample[] = [];
+  for (const p of points) {
+    if (p.ts == null) continue;
+    const ts = typeof p.ts === 'number' ? p.ts : Date.parse(p.ts);
+    if (!Number.isFinite(ts) || p.value == null || !Number.isFinite(p.value)) continue;
+    parsed.push({ ts, value: p.value, quality: p.quality ?? null });
+  }
+  parsed.sort((a, b) => a.ts - b.ts);
+
+  if (tier.bucketMs <= 0) {
+    return {
+      samples: parsed,
+      annotation: tier.method === '原始粒度' ? '当前展示为原始粒度，未降采样' : tier.label,
+      tier,
+    };
+  }
+
+  const buckets = new Map<number, CurveSample[]>();
+  for (const p of parsed) {
+    const key = Math.floor(p.ts / tier.bucketMs) * tier.bucketMs;
+    const list = buckets.get(key);
+    if (list) list.push(p);
+    else buckets.set(key, [p]);
+  }
+
+  const samples: CurveSample[] = [];
+  for (const [key, list] of [...buckets.entries()].sort((a, b) => a[0] - b[0])) {
+    if (tier.method === '均值') {
+      const avg = list.reduce((sum, p) => sum + p.value, 0) / list.length;
+      samples.push({ ts: key, value: avg, quality: worstQuality(list.map((p) => p.quality)) });
+    } else {
+      const last = list[list.length - 1];
+      samples.push({ ts: key, value: last.value, quality: last.quality });
+    }
+  }
+
+  return {
+    samples,
+    annotation: `当前展示为降采样结果：${tier.label.replace(/^降采样结果：/, '')}`,
+    tier,
+  };
+}
+
+/** 桶内最差质量码（超限 1/2 > 断线 3 > 其他非 0 > 正常 0；未知码非 0 视为异常） */
+function worstQuality(qualities: Array<number | null>): number | null {
+  let worst: number | null = null;
+  for (const q of qualities) {
+    if (q == null) continue;
+    if (worst == null) worst = q;
+    else if ((q === 1 || q === 2) && worst !== 1 && worst !== 2) worst = q;
+    else if (q === 3 && worst === 0) worst = q;
+    else if (q !== 0 && worst === 0) worst = q;
+  }
+  return worst;
+}

+/** 时间展示（epoch 毫秒 ISO 本地化；无效值回退原文，不编造） */
+export function formatTs(ts: number | string | null | undefined): string {
+  if (ts == null) return '—';
+  if (typeof ts === 'string') return ts;
+  if (!Number.isFinite(ts)) return '—';
+  const d = new Date(ts);
+  return Number.isNaN(d.getTime()) ? '—' : d.toLocaleString('zh-CN', { hour12: false });
+}
+
+/** 持续时长（ms→可读中文；changeTime 为 null 时展示「未知」，不编造起始时刻） */
+export function formatDuration(changeTime: number | null | undefined, now = Date.now()): string {
+  if (changeTime == null || !Number.isFinite(changeTime)) return '未知';
+  const ms = Math.max(0, now - changeTime);
+  const sec = Math.floor(ms / 1000);
+  if (sec < 60) return `${sec} 秒`;
+  const min = Math.floor(sec / 60);
+  if (min < 60) return `${min} 分 ${sec % 60} 秒`;
+  const hour = Math.floor(min / 60);
+  if (hour < 24) return `${hour} 小时 ${min % 60} 分`;
+  const day = Math.floor(hour / 24);
+  return `${day} 天 ${hour % 24} 小时`;
+}
