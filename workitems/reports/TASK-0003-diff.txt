+-- 存量库升级脚本：租户建库触发与配额统计任务（TASK-0003 / FEAT-0001 / REQ-0001）
+--
+-- 用途：
+--   1) sys_tenant 加列 db_created tinyint NOT NULL DEFAULT 0——租户开通后置待建标志，
+--      定时扫描任务（sys_job）扫 status=1 且 db_created=0 的租户调 TDengine 建库
+--      （iot_<tenant_code>，KEEP 1825d 等，见 resources/contract/tdengine-super-tables.md 附录 B），
+--      成功置 1；失败 WARN 留待下轮（建库异步化+重试，TASK-0003 brief 配套裁决）；
+--   2) 默认租户 default 置 db_created=1——私有化约定初始库由部署脚本创建（REQ-0001 §0.3），
+--      扫描任务不再对其建库；
+--   3) sys_job 播种两条定时任务（框架动态调度，SysJobServiceImpl 启动注册 status=0 行）：
+--      a) 租户库建库扫描（建议 60s 间隔，cron 0 * * * * ?）；
+--      b) 租户配额存储统计（建议 5min 间隔，cron 0 0/5 * * * ?）——逐启用租户查存储用量，
+--         超阈值 WARN 告警（ADR-0011 TD-0035/TD-0041 定版：软限只告警不停写）。
+--
+-- 适用环境：
+--   已执行 20261121_sys_tenant.sql（sys_tenant 在位）与 01_base.sql（sys_job 在位）的存量库
+--   及全新库均可直接执行。
+--
+-- 幂等口径：
+--   加列以 information_schema 判存（MySQL 8.0 无 ADD COLUMN IF NOT EXISTS，条件化存储过程，
+--   与 20261121_tenant_id_columns.sql 同款）；UPDATE 幂等（重复执行不改变结果）；
+--   sys_job 行以 invoke_target 判存（sys_job 无唯一键，用 INSERT ... SELECT ... WHERE NOT EXISTS）。
+--   可安全重复执行。
+--
+-- 注意（兼容 backend/scripts/sql-exec.sh）：
+--   存储过程体压缩为单行（执行器按行尾分号切块）；语句前可挂 -- 注释行（执行器逐行剥除），
+--   但为稳妥起见语句间只以空行分隔，说明集中在本头部与文件尾。
+-- ============================================================
+
+SET NAMES utf8mb4;
+
+DROP PROCEDURE IF EXISTS ot_tdb_addcol;
+CREATE PROCEDURE ot_tdb_addcol(IN tn VARCHAR(64), IN cn VARCHAR(64), IN ddl VARCHAR(1000)) BEGIN DECLARE n INT DEFAULT 0; SELECT COUNT(*) INTO n FROM information_schema.COLUMNS WHERE TABLE_SCHEMA = DATABASE() AND TABLE_NAME = tn AND COLUMN_NAME = cn; IF n = 0 THEN SET @s = ddl; PREPARE st FROM @s; EXECUTE st; DEALLOCATE PREPARE st; END IF; END;
+CALL ot_tdb_addcol('sys_tenant', 'db_created', 'ALTER TABLE `sys_tenant` ADD COLUMN `db_created` tinyint NOT NULL DEFAULT 0 COMMENT ''TDengine 建库标志 0 待建 1 已建（定时扫描任务维护，TASK-0003）''');
+DROP PROCEDURE IF EXISTS ot_tdb_addcol;

+INSERT INTO `sys_job` (`job_name`, `invoke_target`, `cron_expression`, `status`, `remark`)
+SELECT '租户库建库扫描', 'tenantDbProvisionServiceImpl.scanPendingDatabases', '0 * * * * ?', 0, '扫 status=1 且 db_created=0 的租户调 TDengine 建库 iot_<tenant_code>（KEEP 1825d，附录 B 参数）；失败 WARN 留待下轮重试，不阻断开通主流程（TASK-0003）'

+
+INSERT INTO `sys_job` (`job_name`, `invoke_target`, `cron_expression`, `status`, `remark`)
+SELECT '租户配额存储统计', 'iotQuotaStatsServiceImpl.collectAndWarn', '0 0/5 * * * ?', 0, '逐启用租户查 TDengine 存储用量，超 ot.quota.storage-warn-ratio 阈值 WARN（[QUOTA-WARN] 格式可 grep）；软限只告警不停写（ADR-0011 TD-0035/TD-0041，TASK-0003）'
+WHERE NOT EXISTS (SELECT 1 FROM `sys_job` WHERE `invoke_target` = 'iotQuotaStatsServiceImpl.collectAndWarn' AND `deleted` = 0);

+--   SELECT id, tenant_code, db_created FROM sys_tenant   （default 应为 1，其余存量租户 0 待扫描）
