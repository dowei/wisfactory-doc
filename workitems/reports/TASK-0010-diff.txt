+/**
+ * 数据编码字典核对状态（TASK-0010 模板引用公共校验器）
+ *
+ * <p>校验器纯查询语义的返回三态：只报告核对结论、不抛异常——异常转换与 4002 错误码
+ * 映射由调用方（TASK-0011 模板保存流程）决定，便于单测与多场景复用。
+ * 口径来源 REQ-0003 / FEAT-0003：模板数据项保存时按 (tenant_id, data_code) 校验
+ * 存在且启用，不在字典或已禁用返回校验错误不许落库。
+ *
+ * <ul>
+ * <li>{@link #OK}：已登记且启用，允许引用；</li>
+ * <li>{@link #NOT_REGISTERED}：未登记（租户内查无此码；deleted 行按本口径——
+ * {@code @TableLogic} 自动过滤后等价于从未登记，编码语义不漂移）；</li>
+ * <li>{@link #DISABLED}：已登记但已停用（status=0；行仍在、编码不复用，
+ * 既有引用可解析、新引用被拒）。</li>
+ * </ul>
+ *
+ * <p>报错文案区分「未登记/已停用」（任务正文硬要求）：两份文案前缀常量供调用方
+ * 直接复用（拼上编码即完整文案），格式在本枚举内唯一维护、避免调用方各自硬编码漂移。
+ *
+ * @author dowei
+ * @date 2026-11-22
+ */
+public enum DictCodeStatus {
+
+    /** 已登记且启用，允许引用 */
+    OK,
+
+    /** 未登记（租户内查无此码；deleted 行按本口径） */
+    NOT_REGISTERED,
+
+    /** 已登记但已停用（status=0），新引用被拒 */
+    DISABLED;
+
+    /** 未登记文案前缀（完整文案 = 前缀 + 数据编码），供调用方报错直接复用 */
+    public static final String MSG_NOT_REGISTERED = "数据编码未在字典登记: ";
+
+    /** 已停用文案前缀（完整文案 = 前缀 + 数据编码），供调用方报错直接复用 */
+    public static final String MSG_DISABLED = "数据编码已停用: ";
+
+    /**
+     * 按状态拼接完整报错文案。
+     *
+     * @param dataCode 数据编码
+     * @return NOT_REGISTERED → {@code 数据编码未在字典登记: <code>}；
+     *         DISABLED → {@code 数据编码已停用: <code>}；OK → null（无报错文案）
+     */
+    public String messageFor(String dataCode) {
+        return switch (this) {
+            case NOT_REGISTERED -> MSG_NOT_REGISTERED + dataCode;
+            case DISABLED -> MSG_DISABLED + dataCode;
+            case OK -> null;
+        };
+    }
+}

+    // ---------- 模板引用公共校验器（TASK-0010） ----------
+
+    @Override
+    public Map<String, DictCodeStatus> validateForTemplateRef(Long tenantId, Collection<String> dataCodes) {
+        // 纯查询语义（brief 裁决 2）：只返回状态不抛异常——异常与 4002 映射由调用方
+        // （TASK-0011 模板保存流程）决定。返回 Map 保序（LinkedHashMap），调用方可按
+        // 模板数据项顺序取首个失败项报错。
+        Map<String, DictCodeStatus> result = new LinkedHashMap<>();
+        if (dataCodes == null || dataCodes.isEmpty()) {
+            return result;
+        }
+        // fail-closed 边界：tenantId 为 null 时无法确认登记，一律判未登记且不查库
+        // （校验器不做会话假设——若此处放开条件改为依赖插件注入，则违背纯查询语义，
+        // 且 standalone/异步上下文无会话时会跨租户漏判）；批内 null/空串编码同口径
+        // 直判未登记，不进 IN 条件。
+        if (tenantId == null) {
+            for (String code : dataCodes) {
+                result.put(code, DictCodeStatus.NOT_REGISTERED);
+            }
+            return result;
+        }
+        // 批量入口一次 IN 预检（brief 裁决 1）：模板一次保存 N 个数据项，去重后单查
+        // 回填，不逐个 count。
+        Set<String> codes = new LinkedHashSet<>();
+        for (String code : dataCodes) {
+            if (code == null || code.isEmpty()) {
+                result.put(code, DictCodeStatus.NOT_REGISTERED);
+                continue;
+            }
+            codes.add(code);
+        }
+        if (!codes.isEmpty()) {
+            // 显式 tenant_id 条件进查询（与 0002 旁路方法同款显式条件风格）：校验器不做
+            // 会话假设，跨租户编码不可见由本条件保证；运行时 iot_ 白名单表插件再注入
+            // 同值会话条件，双保险且语义自洽。@TableLogic 自动过滤 deleted 行（按未登记
+            // 口径），status=0 即已停用。
+            LambdaQueryWrapper<IotDataDict> wrapper = new LambdaQueryWrapper<>();
+            wrapper.select(IotDataDict::getDataCode, IotDataDict::getStatus)
+                    .eq(IotDataDict::getTenantId, tenantId)
+                    .in(IotDataDict::getDataCode, codes);
+            Map<String, Integer> statusByCode = new HashMap<>();
+            for (IotDataDict item : this.list(wrapper)) {
+                // uk_tenant_data_code 保证 (tenant_id, data_code) 至多一行；status 理论
+                // 非空（服务层白名单恒写 1），null 防御性兜底按不可用（DISABLED）处理
+                Integer status = item.getStatus();
+                statusByCode.put(item.getDataCode(), status != null ? status : 0);
+            }
+            for (String code : codes) {
+                Integer status = statusByCode.get(code);
+                if (status == null) {
+                    result.put(code, DictCodeStatus.NOT_REGISTERED);
+                } else {
+                    result.put(code, status == 1 ? DictCodeStatus.OK : DictCodeStatus.DISABLED);
+                }
+            }
+        }
+        return result;
+    }
