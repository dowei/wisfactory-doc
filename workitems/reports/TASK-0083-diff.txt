+     * 排除，保持旧版本并留在 0016 清单。

+     * <p><b>事务与失败语义</b>：不另开事务——加入 publish 的 {@code @Transactional}；apply 抛
+     * 异常（数据面失败/上下文缺失/坏默认 fail-closed）直接逃出 → 整批回滚（版本行/快照/下发

+     * @param tenantId  当前租户（插件注入 + 显式会话读）
+     * @param scopeType 发布范围（1 网关 / 2 计算——仅计算触发跟随）
+     */
+    private void followUnlockedInstances(Long tenantId, Integer scopeType) {
+        if (scopeType == null || scopeType != SCOPE_COMPUTE) {
+            return; // 网关 scope 与模板实例无关（brief 决策 2）
+        }
+        List<IotModelTemplate> templates = templateMapper.selectList(new LambdaQueryWrapper<>());
+        if (templates == null || templates.isEmpty()) {
+            return;
+        }
+        for (IotModelTemplate template : templates) {
+            Integer current = template.getVersion();
+            if (template.getId() == null || current == null) {
+                continue;
+            }
+            // 未锁定（locked=0）+ 滞后（applied<current）实例；租户条件由插件注入（同租户）
+            List<IotModelInstance> lagging = instanceMapper.selectList(new LambdaQueryWrapper<IotModelInstance>()
+                    .eq(IotModelInstance::getTemplateId, template.getId())
+                    .eq(IotModelInstance::getLocked, 0)
+                    .lt(IotModelInstance::getAppliedTemplateVersion, current)
+                    .orderByAsc(IotModelInstance::getId));
+            if (lagging == null || lagging.isEmpty()) {
+                continue;
+            }
+            String currentContent = loadTemplateVersionContent(template.getId(), current);
+            if (currentContent == null) {
+                log.warn("[PUBLISH-FOLLOW] current content missing templateId={} version={} skipped",
+                        template.getId(), current);
+                continue;
+            }
+            JsonNode currentNode = parseFollowContent(currentContent);
+            if (currentNode == null) {
+                continue;
+            }
+            for (IotModelInstance inst : lagging) {
+                Integer applied = inst.getAppliedTemplateVersion();
+                // 内存二次防线（筛选已在 SQL 层 locked=0 / applied<current；此处防御桩返回/并发漂移）
+                if (applied == null || applied >= current
+                        || (inst.getLocked() != null && inst.getLocked() == 1)) {
+                    continue;
+                }
+                String appliedContent = loadTemplateVersionContent(template.getId(), applied);
+                if (appliedContent == null) {
+                    log.warn("[PUBLISH-FOLLOW] applied content missing instanceId={} templateId={} applied={} skipped",
+                            inst.getId(), template.getId(), applied);
+                    continue;
+                }
+                JsonNode appliedNode = parseFollowContent(appliedContent);
+                if (appliedNode == null) {
+                    continue;
+                }
+                // T3 永不自动：applied→current 净 diff 含 T3 条目则跳过该实例（仍走 0016 逐台确认）
+                List<TemplateChangeEntry> diff = TemplateContentDiffEngine.diff(appliedNode, currentNode);
+                boolean hasTier3 = diff.stream().anyMatch(e -> e.getTier() == TemplateChangeTier.TIER_3);
+                if (hasTier3) {
+                    log.info("[PUBLISH-FOLLOW] skipped (T3 never auto) instanceId={} templateId={} applied={} current={}",
+                            inst.getId(), template.getId(), applied, current);
+                    continue;
+                }
+                // 数据面应用 + 乐观锁簿记推进（同事务；失败抛异常整批回滚）
+                upgradeApplyService.applyToVersion(inst, current);
+                int rows = instanceMapper.update(null, new LambdaUpdateWrapper<IotModelInstance>()
+                        .eq(IotModelInstance::getId, inst.getId())
+                        .eq(IotModelInstance::getAppliedTemplateVersion, applied)
+                        .set(IotModelInstance::getAppliedTemplateVersion, current));
+                if (rows == 0) {
+                    throw new BusinessException(ResultCode.DUPLICATE_CODE.getCode(),
+                            "发布自动跟随版本并发冲突，请重试: instanceId=" + inst.getId());
+                }
+                log.info("[PUBLISH-FOLLOW] instanceId={} templateId={} applied {} -> {} (userId={})",
+                        inst.getId(), template.getId(), applied, current, SecurityUtils.getUserId());
+            }
+        }
+    }
