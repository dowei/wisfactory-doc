+// escapeIdentifier 把任意字符串确定性映射为 TDengine 合法标识符（字母或下划线开头、
+// 仅 [A-Za-z0-9_]、≤192 字节）：
+//   - 全安全字节（[A-Za-z0-9_]）且非数字开头：恒等映射（契约示例编码 group_a、
+//     F01_WS01_EQ001_M01 等不受影响）；
+//   - 含非安全字节：逐字节 '_XX' 十六进制转义，并追加 '_' + 原文 FNV-64 后缀——
+//     后缀使转义结果与恒等集区分（如「F01-M01」不会撞上字面编码「F01_2dM01…」），
+//     且逐字节转义本身单射，同一批非安全编码之间互不冲突；
+//   - 首字符为数字：加前缀 "t_"（TDengine 标识符不能以数字开头）；
+//   - 超长：截断 + 哈希后缀保持 ≤192 字节与确定性。
+func escapeIdentifier(s string) string {
+	if s == "" {
+		return ""
+	}
+	allSafe := true
