+
+import java.util.List;

+/**
+ * 租户服务实现类（TASK-0001）
+ *
+ * <p>编码唯一性双道闸：先对活跃行（@TableLogic 过滤 deleted=0）预检，友好报 4002；
+ * 再捕获 {@link DuplicateKeyException} 兜底——uk_tenant_code 是物理唯一索引，
+ * 逻辑删不释放编码，兜底覆盖「已删除行仍占用编码」场景（删除后编码不复用，
+ * 详细设计 §1.1 既定纪律）。
+ *
+ * @author dowei
+ * @date 2026-11-21
+ */
+@Service
+public class SysTenantServiceImpl extends ServiceImpl<SysTenantMapper, SysTenant> implements SysTenantService {
+
+    @Override
+    public Page<SysTenant> selectTenantList(Page<SysTenant> page, SysTenant tenant) {
+        LambdaQueryWrapper<SysTenant> wrapper = new LambdaQueryWrapper<>();
+        if (tenant.getTenantCode() != null) {
+            wrapper.like(SysTenant::getTenantCode, SqlLikeUtils.escape(tenant.getTenantCode()));
+        }
+        if (tenant.getTenantName() != null) {
+            wrapper.like(SysTenant::getTenantName, SqlLikeUtils.escape(tenant.getTenantName()));
+        }
+        if (tenant.getStatus() != null) {
+            wrapper.eq(SysTenant::getStatus, tenant.getStatus());
+        }
+        wrapper.orderByDesc(SysTenant::getId);
+        return this.page(page, wrapper);
+    }
+
+    @Override
+    @Transactional(rollbackFor = Exception.class)
+    public boolean insertTenant(SysTenant tenant) {
+        // 预检活跃行：友好报 4002（逻辑删除会自动过滤已删行）
+        LambdaQueryWrapper<SysTenant> wrapper = new LambdaQueryWrapper<>();
+        wrapper.eq(SysTenant::getTenantCode, tenant.getTenantCode());
+        if (this.count(wrapper) > 0) {
+            throw duplicateCode(tenant.getTenantCode());
+        }
+        try {
+            return this.save(tenant);
+        } catch (DuplicateKeyException e) {
+            // 兜底：已删除行仍占用编码（物理唯一索引不被逻辑删除释放）
+            throw duplicateCode(tenant.getTenantCode());
+        }
+    }
+
+    @Override
+    @Transactional(rollbackFor = Exception.class)
+    public boolean updateTenant(SysTenant tenant) {
+        SysTenant existing = this.getById(tenant.getId());
+        if (existing == null) {
+            throw new BusinessException(ResultCode.PARAM_ERROR.getCode(), "租户不存在或已删除");
+        }
+        // 创建后编码不可改：请求携带的编码与库中不一致即拒绝
+        if (tenant.getTenantCode() == null || !tenant.getTenantCode().equals(existing.getTenantCode())) {
+            throw new BusinessException(ResultCode.PARAM_ERROR.getCode(), "租户编码不允许修改");
+        }
+        return this.updateById(tenant);
+    }
+
+    @Override
+    @Transactional(rollbackFor = Exception.class)
+    public boolean updateTenantStatus(Long id, Integer status) {
+        if (status == null || (status != 0 && status != 1)) {
+            throw new BusinessException(ResultCode.PARAM_ERROR.getCode(), "状态值仅允许 0 或 1");
+        }
+        if (this.getById(id) == null) {
+            throw new BusinessException(ResultCode.PARAM_ERROR.getCode(), "租户不存在或已删除");
+        }
+        // 只更新状态字段，不夹带其他业务字段
+        SysTenant upd = new SysTenant();
+        upd.setId(id);
+        upd.setStatus(status);
+        return this.updateById(upd);
+    }
+
+    @Override
+    @Transactional(rollbackFor = Exception.class)
+    public boolean deleteTenantByIds(List<Long> ids) {
+        return this.removeByIds(ids);
+    }
+
+    private BusinessException duplicateCode(String tenantCode) {
+        return new BusinessException(ResultCode.DUPLICATE_CODE.getCode(), "租户编码已存在: " + tenantCode);
+    }
