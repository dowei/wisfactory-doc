+/*
+ * 租户管理（智慧工厂）
+ *
+ * 对接后端 SysTenantController（TASK-0001 交付，路径前缀 /tenant，
+ * 权限位 iot:tenant:*，与 sys_menu 4002-4007 六个按钮行一一对应）。
+ * TASK-0055：控制台骨架阶段唯一有真实后端接口的 iot API 封装；
+ * 其余页面（字典/模板/实例/点位/网关/通道/指标/发布/导入）的封装
+ * 随各自后端任务交付接口时按本样式补建，不预写空壳。
+ */
+import { deleteRequest, getRequest, postRequest, putRequest } from '/@/lib/axios';
+
+export const tenantApi = {
+  /**
+   * 分页查询租户列表（tenantCode/tenantName 模糊，status 精确）
+   */
+  queryTenantList: (params?) => {
+    return getRequest('/tenant/list', params);
+  },
+
+  /**
+   * 根据ID查询租户
+   */
+  queryTenantById: (id) => {
+    return getRequest(`/tenant/${id}`, undefined);
+  },
+
+  /**
+   * 新增租户
+   */
+  addTenant: (data) => {
+    return postRequest('/tenant', data);
+  },
+
+  /**
+   * 修改租户（tenant_code 创建后不可改，编辑表单禁输该字段）
+   */
+  updateTenant: (data) => {
+    return putRequest('/tenant', data);
+  },
+
+  /**
+   * 租户启停（status 0 禁用 1 启用）
+   */
+  changeTenantStatus: (id, status) => {
+    return putRequest(`/tenant/${id}/status`, { status });
+  },
+
+  /**
+   * 删除租户（支持批量，逗号分隔；逻辑删除不释放编码）
+   */
+  deleteTenant: (ids) => {
+    return deleteRequest(`/tenant/${ids}`);
+  },
+};
