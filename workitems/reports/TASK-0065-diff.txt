+    /** 取方法上第一个写 HTTP 映射注解对应的动词；无写映射返回 null */
+    private static String writeVerb(Method method) {
+        Map<Class<? extends Annotation>, String> verbs = new LinkedHashMap<>();
+        verbs.put(PostMapping.class, "POST");
+        verbs.put(PutMapping.class, "PUT");
+        verbs.put(DeleteMapping.class, "DELETE");
+        verbs.put(PatchMapping.class, "PATCH");
+        for (Map.Entry<Class<? extends Annotation>, String> e : verbs.entrySet()) {
+            if (method.getAnnotation(e.getKey()) != null) {
+                return e.getValue();
+            }
+        }
+        return null;
+    }
+
+    /** 取方法首个 HTTP 映射注解的首个路径（用于排除前缀判定） */
+    private static String firstMappingPath(Method method) {
+        for (Annotation a : method.getAnnotations()) {
+            if (a instanceof PostMapping) return first(((PostMapping) a).value());
+            if (a instanceof PutMapping) return first(((PutMapping) a).value());
+            if (a instanceof DeleteMapping) return first(((DeleteMapping) a).value());
+            if (a instanceof PatchMapping) return first(((PatchMapping) a).value());
+        }
+        return "";
+    }

+    private static String first(String[] arr) {
+        return (arr != null && arr.length > 0) ? arr[0] : "";
+    }
+
+    private static boolean isExcluded(String path) {
+        for (String prefix : EXCLUDED_PATH_PREFIXES) {
+            if (path.startsWith(prefix)) {
+                return true;
+            }
+        }
+        return false;
+    }

+    /** 反射取唯一同名方法（防歧义） */
+    private static Method uniqueMethod(Class<?> clazz, String name) throws Exception {
+        List<Method> hits = new ArrayList<>();
+        for (Method m : clazz.getDeclaredMethods()) {
+            if (m.getName().equals(name)) {
+                hits.add(m);
+            }
+        }
+        assertEquals(1, hits.size(), clazz.getSimpleName() + "#" + name + " 必须恰一个方法（防歧义）");
+        return hits.get(0);
+    }

+"""TASK-0065 机械覆盖审计：iot/openapi 管理面写端点（POST/PUT/DELETE/PATCH）
+vs @OperateLog 标注集差集。照 0064 export-design-perms.py 范式（纯机械）。
+
+口径：
+  - 对每个 controller，按 'public Result' 等返回签名行定位方法，向前取注解窗口
+    （从上一方法体结束/成员声明到本签名行），识别 @OperateLog 与 HTTP 映射注解。
+  - 类级 @RequestMapping 前缀拼入路径。
+  - 写端点 = POST/PUT/DELETE/PATCH；只读 GET 不审（凭据明文 view 单列）。
+  - 内部令牌端点 /internal/v1/**、对外 /openapi/v1/** 不挂管理面操作日志（brief 决策4）。
+"""
+import re
+import sys
+from pathlib import Path
+
+ROOT = Path(sys.argv[1]) if len(sys.argv) > 1 else Path(r"D:/workspace/WisFactory")
+MOD = ROOT / "backend/ot-admin/src/main/java/net/otlab/admin/module"
+
+METHOD_SIG = re.compile(
+    r'^\s*public\s+(?:static\s+)?[\w<>,.\[\]\s?]+\s+(\w+)\s*\(')
+MAP = re.compile(r'@(Get|Post|Put|Delete|Patch)Mapping\b')
+PERM = re.compile(r'@SaCheckPermission\(\s*"([^"]+)"')
+OP = re.compile(r'@OperateLog\s*\(')
+
+
+def strip_comments(src):
+    out = []
+    i = 0
+    n = len(src)
+    ins = False
+    esc = False
+    while i < n:
+        c = src[i]
+        if ins:
+            out.append(c)
+            if esc:
+                esc = False
+            elif c == '\\':
+                esc = True
+            elif c == '"':
+                ins = False
+            i += 1
+            continue
+        if c == '"':
+            ins = True; out.append(c); i += 1; continue
+        if c == '/' and i+1 < n and src[i+1] == '/':
+            while i < n and src[i] != '\n':
+                i += 1
+            continue
+        if c == '/' and i+1 < n and src[i+1] == '*':
+            i += 2
+            while i+1 < n and not (src[i] == '*' and src[i+1] == '/'):
+                i += 1
+            i += 2
+            continue
+        out.append(c); i += 1
+    return ''.join(out)
+
+
+def analyze(path):
+    src = strip_comments(path.read_text(encoding='utf-8'))
+    bm = re.search(r'@RequestMapping\(\s*"([^"]+)"\s*\)', src)
+    base = bm.group(1) if bm else ''
+    lines = src.split('\n')
+    rows = []
+    # 定位所有方法签名行
+    sigs = [i for i, ln in enumerate(lines) if METHOD_SIG.match(ln)]
+    for i in sigs:
+        name = METHOD_SIG.match(lines[i]).group(1)
+        # 注解窗口：从上一签名行之后到本签名行（含）；不跨上一方法体（以 '}' 行截断）
+        start = 0
+        for j in range(i-1, -1, -1):
+            if METHOD_SIG.match(lines[j]):
+                break
+            if lines[j].strip() == '}':
+                start = j + 1
+                break
+        win = '\n'.join(lines[start:i+1])
+        maps = MAP.findall(win)
+        if not maps:
+            continue
+        verbs = [m.lower() for m in maps]
+        # 每个 mapping 的路径（取紧跟其后的第一个字符串字面量）
+        paths = []
+        for ln in win.split('\n'):
+            for m in MAP.finditer(ln):
+                tail = ln[m.end():]
+                pm = re.search(r'"([^"]*)"', tail)
+                paths.append(pm.group(1) if pm else '')
+        # 若一条 Mapping 注解跨行（多路径数组），把窗口内所有字符串补全
+        if len(paths) < len(verbs):
+            all_str = re.findall(r'"([^"]*)"', win)
+            # 去掉权限注解字符串
+            paths = [p for p in all_str if not p.startswith('iot:') and not p.startswith('openapi:')]
+        opm = OP.search(win)
+        flags = ''
+        if opm:
+            tail = win[opm.end()-1:opm.end()+220]
+            flags = ','.join(f for f in (
+                'saveParam=false' if re.search(r'saveParam\s*=\s*false', tail) else '',
+                'saveResponse=false' if re.search(r'saveResponse\s*=\s*false', tail) else '') if f)
+        rows.append({
+            'file': str(path.relative_to(ROOT)).replace('\\', '/'),
+            'class': path.stem,
+            'method': name,
+            'verbs': verbs,
+            'paths': paths,
+            'base': base,
+            'audited': bool(opm),
+            'flags': flags,
+            'perms': ','.join(PERM.findall(win)),
+        })
+    return rows
+
+
+def fmt_paths(r):
+    return '|'.join((r['base'] + p) for p in r['paths']) or '-'
+

+    writes = [r for r in all_rows if any(v in ('post', 'put', 'delete', 'patch') for v in r['verbs'])]
