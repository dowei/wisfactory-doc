+// Package gwtoken 承载网关令牌链（TASK-0076 brief 决策 3）：
+//
+//	网关编码 → 通道定位（GatewayChannelResolver）→ 凭据拉取（CredentialPuller，0019
+//	/internal/v1/credential/pull，静态内部令牌 X-Internal-Token 配置化）→ 本地缓存
+//	（TTL 配置化）→ 校验器（上报/认证回调携带令牌 vs 拉取凭据比对）。
+//
+// fail-closed 纪律：网关未知、凭据拉取失败（后端不可达 / 403 吊销禁用）、令牌不匹配
+// 一律拒绝——FEAT-0009 处理顺序第 1 步「验令牌与租户」的令牌侧实现，后端不可达不放行。
+// 租户必须来自令牌链解析产物（GatewayIdentity.TenantCode），不从消息自带 tenant 取。
+//
+// mock 口径（brief 裁决）：凭据明文比对占位——拉取的 ciphertext 原样作为期望令牌做
+// 常量时间比对，不做 SM4 解密；真网关令牌体系（校验令牌归属通道、密文解密、轮换并行）
+// 归 TASK-0020/0027 正式认领时定版，本包以三个接口留位：
+// GatewayChannelResolver / CredentialPuller / Validator，替换实现不改调用方。
+package gwtoken
+
+import (
+	"context"
+	"crypto/subtle"
+	"fmt"
+	"sync"
+	"time"
+)
+
+// 凭据类型（REQ-0011：DEVICE_TOKEN/MQTT_PASSWORD/HTTP_SECRET/CERT）。
+const (
+	CredTypeDeviceToken  = "DEVICE_TOKEN"
+	CredTypeMQTTPassword = "MQTT_PASSWORD"
+	CredTypeHTTPSecret   = "HTTP_SECRET"
+	CredTypeCert         = "CERT"
+)
+
+// DefaultCacheTTL 凭据本地缓存默认 TTL（可配置覆盖）。
+const DefaultCacheTTL = 30 * time.Second
+
+// GatewayIdentity 令牌链校验产物：租户选库（iot_<tenant_code>）与处理顺序第 1 步
+// 的租户来源，只信任此结构（令牌侧），不信任消息自带 tenant。
+type GatewayIdentity struct {
+	GatewayCode string
+	TenantCode  string
+	ChannelID   int64
+}
+
+// ResolvedChannel 网关编码解析产物：凭据所在通道 + 令牌侧租户。
+type ResolvedChannel struct {
+	ChannelID  int64
+	TenantCode string
+}
+
+// GatewayChannelResolver 网关编码 → 通道定位。mock 口径为静态映射（StaticResolver）；
+// 真实现（网关注册表/快照通道，校验令牌归属通道）归 TASK-0020/0027，接口留位。
+type GatewayChannelResolver interface {
+	Resolve(gatewayCode string) (ResolvedChannel, bool)
+}
+
+// StaticResolver 静态映射实现（配置/测试用）。
+type StaticResolver map[string]ResolvedChannel
+
+// Resolve 实现 GatewayChannelResolver。
+func (r StaticResolver) Resolve(gatewayCode string) (ResolvedChannel, bool) {
+	rc, ok := r[gatewayCode]
+	return rc, ok
+}
+
+// Credential 凭据拉取产物（0019 CredentialPullVO 契约形态：只回密文与 key_version，
+// 服务端不解密传输——调用方内存解密，ADR-0006）。
+type Credential struct {
+	ChannelID  int64
+	CredType   string
+	Ciphertext string // {v1} 密文原样（mock 口径：明文比对占位，不解密）
+	KeyVersion string
+}
+
+// CredentialPuller 凭据拉取抽象。HTTPPuller 为 0019 锚点实现；测试可注入假实现。
+type CredentialPuller interface {
+	Pull(ctx context.Context, channelID int64, credType string) (Credential, error)
+}
+
+// Validator 网关令牌校验器（fail-closed 语义）。上报端点与 EMQX 认证回调共用；
+// 真网关令牌体系替换实现时接口零改动。
+type Validator interface {
+	// Validate 校验（凭据类型, 网关编码, 携带令牌）。任何失败（网关未知/拉取失败/
+	// 不匹配/空参数）返回错误——调用方必须拒绝放行，不得降级。
+	Validate(ctx context.Context, credType, gatewayCode, token string) (GatewayIdentity, error)
+}
+
+// TokenChainValidator 令牌链实现：Resolver → Puller（TTL 缓存）→ 常量时间比对。
+type TokenChainValidator struct {
+	Resolver GatewayChannelResolver // 必填
+	Puller   CredentialPuller       // 必填
+	// TTL 凭据本地缓存 TTL（配置化）；<=0 → DefaultCacheTTL。
+	TTL time.Duration
+
+	now func() time.Time // 可替换时钟（测试用；非导出）
+
+	mu    sync.Mutex
+	cache map[cacheKey]cacheEntry
+}
+
+type cacheKey struct {
+	channelID int64
+	credType  string
+}
+
+type cacheEntry struct {
+	cred    Credential
+	expires time.Time
+}
+
+// NewTokenChainValidator 构造令牌链校验器。resolver/puller 为 nil 时返回 nil
+// （装配方自查依赖），与 0073 组件「接口注入」口径一致。
+func NewTokenChainValidator(resolver GatewayChannelResolver, puller CredentialPuller, ttl time.Duration) *TokenChainValidator {
+	if resolver == nil || puller == nil {
+		return nil
+	}
+	if ttl <= 0 {
+		ttl = DefaultCacheTTL
+	}
+	return &TokenChainValidator{
+		Resolver: resolver,
+		Puller:   puller,
+		TTL:      ttl,
+		now:      time.Now,
+		cache:    make(map[cacheKey]cacheEntry),
+	}
+}
+
+// Validate 实现 Validator：fail-closed 五闸（空参数 → 网关未知 → 拉取失败 → 空凭据 →
+// 令牌不匹配），任一失败即拒绝；成功返回令牌侧解析产物。
+func (v *TokenChainValidator) Validate(ctx context.Context, credType, gatewayCode, token string) (GatewayIdentity, error) {
+	if gatewayCode == "" || token == "" {
+		return GatewayIdentity{}, fmt.Errorf("gwtoken: 网关编码或令牌为空，fail-closed 拒绝")
+	}
+	rc, ok := v.Resolver.Resolve(gatewayCode)
+	if !ok {
+		return GatewayIdentity{}, fmt.Errorf("gwtoken: 网关 %q 未知（通道定位未命中），fail-closed 拒绝", gatewayCode)
+	}
+	cred, err := v.credential(ctx, rc.ChannelID, credType)
+	if err != nil {
+		return GatewayIdentity{}, fmt.Errorf("gwtoken: 网关 %q 凭据拉取失败（后端不可达/吊销），fail-closed 拒绝: %w", gatewayCode, err)
+	}
+	if cred.Ciphertext == "" {
+		return GatewayIdentity{}, fmt.Errorf("gwtoken: 网关 %q 拉取到空凭据，fail-closed 拒绝", gatewayCode)
+	}
+	if subtle.ConstantTimeCompare([]byte(token), []byte(cred.Ciphertext)) != 1 {
+		return GatewayIdentity{}, fmt.Errorf("gwtoken: 网关 %q 令牌不匹配，fail-closed 拒绝", gatewayCode)
+	}
+	return GatewayIdentity{GatewayCode: gatewayCode, TenantCode: rc.TenantCode, ChannelID: rc.ChannelID}, nil
+}
+
+// credential 带本地缓存的凭据拉取（TTL 配置化）。持锁拉取（串行化同一键的并发回源，
+// 防后端惊群）；拉取失败不缓存（fail-closed：下次校验重新回源）。
+func (v *TokenChainValidator) credential(ctx context.Context, channelID int64, credType string) (Credential, error) {
+	key := cacheKey{channelID: channelID, credType: credType}
+	v.mu.Lock()
+	defer v.mu.Unlock()
+	if entry, ok := v.cache[key]; ok && v.now().Before(entry.expires) {
+		return entry.cred, nil
+	}
+	cred, err := v.Puller.Pull(ctx, channelID, credType)
+	if err != nil {
+		return Credential{}, err
+	}
+	v.cache[key] = cacheEntry{cred: cred, expires: v.now().Add(v.TTL)}
+	return cred, nil
+}

+package gwtoken
+
+import (
+	"context"
+	"encoding/json"
+	"fmt"
+	"io"
+	"net/http"
+	"net/url"
+	"strconv"
+	"sync"
+	"time"
+)
+
+// 0019 凭据拉取锚点（响应形状以后端代码为准）。
+const (
+	// CredentialPullPath 凭据拉取路径：GET /internal/v1/credential/pull（0019）。
+	CredentialPullPath = "/internal/v1/credential/pull"
+	// InternalTokenHeader 内部静态令牌请求头（后端 InternalAuthChecker.TOKEN_HEADER 同名对齐）。
+	InternalTokenHeader = "X-Internal-Token"
+	// DefaultPullTimeout 拉取默认超时：后端不可达必须在有限时间内转为错误
+	//（上层据此 fail-closed 拒绝），不得悬挂令牌链。
+	DefaultPullTimeout = 10 * time.Second
+	// resultSuccessCode 后端框架 Result 成功业务码（ot-core ResultCode.SUCCESS=200）。
+	resultSuccessCode = 200
+)
+
+// HTTPPuller 0019 凭据拉取客户端：GET {BaseURL}/internal/v1/credential/pull
+// ?channelId=&credType=，请求头 X-Internal-Token（配置化——后端 InternalAuthChecker
+// 静态令牌 + IP 白名单口径；本客户端只承担令牌头，白名单在服务端）。
+type HTTPPuller struct {
+	// BaseURL 管理系统基址（如 http://mgmt-host:8080）。
+	BaseURL string
+	// InternalToken 内部静态令牌（配置化；空 = 不带头，服务端必拒——fail-closed）。
+	InternalToken string
+	// Timeout 单次拉取超时；<=0 → DefaultPullTimeout。
+	Timeout time.Duration
+
+	clientOnce sync.Once
+	client     *http.Client
+}
+
+// NewHTTPPuller 构造拉取客户端（timeout<=0 取默认）。
+func NewHTTPPuller(baseURL, internalToken string, timeout time.Duration) *HTTPPuller {
+	return &HTTPPuller{BaseURL: baseURL, InternalToken: internalToken, Timeout: timeout}
+}
+
+func (p *HTTPPuller) httpClient() *http.Client {
+	p.clientOnce.Do(func() {
+		timeout := p.Timeout
+		if timeout <= 0 {
+			timeout = DefaultPullTimeout
+		}
+		p.client = &http.Client{Timeout: timeout}
+	})
+	return p.client
+}
+
+// pullEnvelope 后端框架 Result<CredentialPullVO> 信封（响应形状以 0019 代码为准：
+// Result{code, message, data, timestamp}；data=CredentialPullVO{channelId, credType,
+// ciphertext, keyVersion}——只回密文不回明文，ADR-0006）。
+type pullEnvelope struct {
+	Code    int    `json:"code"`
+	Message string `json:"message"`
+	Data    *struct {
+		ChannelID  int64  `json:"channelId"`
+		CredType   string `json:"credType"`
+		Ciphertext string `json:"ciphertext"`
+		KeyVersion string `json:"keyVersion"`
+	} `json:"data"`
+}
+
+// Pull 拉取通道凭据（0019 锚点）。非 200 HTTP 状态、非 200 业务码、data 缺失、
+// 密文为空一律转错误（fail-closed 前置：失败不产出可用凭据）。
+func (p *HTTPPuller) Pull(ctx context.Context, channelID int64, credType string) (Credential, error) {
+	q := url.Values{}
+	q.Set("channelId", strconv.FormatInt(channelID, 10))
+	q.Set("credType", credType)
+	endpoint := p.BaseURL + CredentialPullPath + "?" + q.Encode()
+
+	req, err := http.NewRequestWithContext(ctx, http.MethodGet, endpoint, nil)
+	if err != nil {
+		return Credential{}, fmt.Errorf("gwtoken: 构造凭据拉取请求失败: %w", err)
+	}
+	req.Header.Set(InternalTokenHeader, p.InternalToken)
+
+	resp, err := p.httpClient().Do(req)
+	if err != nil {
+		return Credential{}, fmt.Errorf("gwtoken: 凭据拉取失败（后端不可达）: %w", err)
+	}
+	defer resp.Body.Close()
+	body, err := io.ReadAll(io.LimitReader(resp.Body, 1<<20))
+	if err != nil {
+		return Credential{}, fmt.Errorf("gwtoken: 读取凭据拉取响应失败: %w", err)
+	}
+
+	var env pullEnvelope
+	if err := json.Unmarshal(body, &env); err != nil {
+		return Credential{}, fmt.Errorf("gwtoken: 凭据拉取响应解析失败 status=%d body=%s: %w",
+			resp.StatusCode, truncateForLog(body), err)
+	}
+	if resp.StatusCode != http.StatusOK {
+		return Credential{}, fmt.Errorf("gwtoken: 凭据拉取失败 status=%d code=%d message=%q（REQ-0011：吊销/禁用 403）",
+			resp.StatusCode, env.Code, env.Message)
+	}
+	if env.Code != resultSuccessCode {
+		return Credential{}, fmt.Errorf("gwtoken: 凭据拉取失败 code=%d message=%q", env.Code, env.Message)
+	}
+	if env.Data == nil {
+		return Credential{}, fmt.Errorf("gwtoken: 凭据拉取响应缺 data（0019 CredentialPullVO 形态）")
+	}
+	if env.Data.Ciphertext == "" {
+		return Credential{}, fmt.Errorf("gwtoken: 凭据拉取响应密文为空（channelId=%d credType=%s）", channelID, credType)
+	}
+	return Credential{
+		ChannelID:  env.Data.ChannelID,
+		CredType:   env.Data.CredType,
+		Ciphertext: env.Data.Ciphertext,

+// 「EMQX 认证回调；内部令牌」）。
+//
+// 流程：端点自身鉴权（内部令牌，常量时间比较）→ 校验网关编码与设备令牌、租户与网关
+// status（令牌链，fail-closed）→ 返回 allow/deny 及该连接允许订阅/发布的主题范围
+// （只能收自己的 config、只能发自己的 data/status，REQ-0013）。
+//
+// 响应形状按 EMQX HTTP 认证文档惯例（{result: "allow"|"deny"}），主题范围为自定契约
+// 扩展（topics.publish/subscribe）；形状由测试锁定，真 EMQX 联调归 0068。
+type MQTTAuthHandler struct {
+	// Validator 网关令牌链校验器（fail-closed）。
+	Validator gwtoken.Validator
+	// InternalToken 回调端点自身鉴权的内部令牌（配置化）。空值 = 全部拒绝（fail-closed：
+	// 未配置保护时不暴露无鉴权端点）。
+	InternalToken string
+}
+
+// mqttAuthRequest EMQX HTTP 认证回调入参（username=gatewayCode，password=网关令牌，
+// brief 决策 2）。EMQX 其余回调字段（clientid/peerhost/sock 等）宽松忽略——回调载荷
+// 演进不打停接入面。
+type mqttAuthRequest struct {
+	Username string `json:"username"`
+	Password string `json:"password"`
+}

+// mqttTopics 主题范围：只能发自己的 data/status，只能收自己的 config（REQ-0013）。
+type mqttTopics struct {
+	Publish   []string `json:"publish"`
+	Subscribe []string `json:"subscribe"`
+}
+
+// mqttAuthResponse 响应：{result: allow|deny} + 主题范围（deny 时省略 topics）。
+type mqttAuthResponse struct {
+	Result string      `json:"result"`
+	Topics *mqttTopics `json:"topics,omitempty"`
+}
+
+const (
+	authResultAllow = "allow"
+	authResultDeny  = "deny"
+)
+
+// ServeHTTP 实现 http.Handler。
+func (h *MQTTAuthHandler) ServeHTTP(w http.ResponseWriter, r *http.Request) {
+	defer func() {
+		if rec := recover(); rec != nil {
+			writeAuth(w, http.StatusInternalServerError, mqttAuthResponse{Result: authResultDeny})
+		}
+	}()
+
+	// 端点自身鉴权（内部令牌；缺失/不匹配 401 + 响应体 deny 双信号——不依赖调用方
+	// 对非 200 的处理口径，fail-closed 收敛）。
+	presented := r.Header.Get(gwtoken.InternalTokenHeader)
+	if h.InternalToken == "" ||
+		subtle.ConstantTimeCompare([]byte(presented), []byte(h.InternalToken)) != 1 {
+		writeAuth(w, http.StatusUnauthorized, mqttAuthResponse{Result: authResultDeny})
+		return
+	}
+
+	var req mqttAuthRequest
+	if err := json.NewDecoder(http.MaxBytesReader(w, r.Body, MaxRequestBodyBytes)).Decode(&req); err != nil {
+		writeAuth(w, http.StatusOK, mqttAuthResponse{Result: authResultDeny})
+		return
+	}
+	// 认证三要素缺失即拒绝（不做部分匹配放行；不触发令牌链）。
+	if req.Username == "" || req.Password == "" {
+		writeAuth(w, http.StatusOK, mqttAuthResponse{Result: authResultDeny})
+		return
+	}
+
+	// 令牌链校验（fail-closed）：网关未知/凭据拉取失败/令牌不匹配一律 deny。
+	identity, err := h.Validator.Validate(r.Context(), gwtoken.CredTypeMQTTPassword, req.Username, req.Password)
+	if err != nil {
+		writeAuth(w, http.StatusOK, mqttAuthResponse{Result: authResultDeny})
+		return
+	}
+
+	writeAuth(w, http.StatusOK, mqttAuthResponse{
+		Result: authResultAllow,
+		Topics: &mqttTopics{
+			Publish:   []string{dataTopic(identity), statusTopic(identity)},
+			Subscribe: []string{configTopic(identity)},
+		},
+	})
+}

+// 主题范围（契约 §3.2 主题规划：iot/{tenant_code}/{gateway_code}/{data,status,config}）。
+func dataTopic(id gwtoken.GatewayIdentity) string    { return fmt.Sprintf("iot/%s/%s/data", id.TenantCode, id.GatewayCode) }

+func configTopic(id gwtoken.GatewayIdentity) string  { return fmt.Sprintf("iot/%s/%s/config", id.TenantCode, id.GatewayCode) }
+
+func writeAuth(w http.ResponseWriter, status int, resp mqttAuthResponse) {
+	w.Header().Set("Content-Type", "application/json; charset=utf-8")
+	w.WriteHeader(status)
+	_ = json.NewEncoder(w).Encode(resp)
+}

+	gotSlice, ok := got.([]any)

+	}

+		if gotSlice[i] != want[i] {
+			return false
+		}
+	}
+	return true

+// PipelineSink ReportSink 的内核流水线实现：经 pipeline.Submit 同步提交——与 MQTT
+// 消费循环（Run）同一 processOne 五步冻结顺序（统一流水线入口，HTTP 与 MQTT 同构结果），
+// 攒批缓冲、flusher、去重表与 Run 路径共享。租户取令牌侧解析产物。
+type PipelineSink struct {
+	// Pipe 已装配的内核流水线（Run 或 StartFlusher 须已启动，flusher 承担写库）。
+	Pipe *pipeline.Pipeline
+}
+
+// Report 实现 ReportSink。
+func (s *PipelineSink) Report(ctx context.Context, tenant string, payload []byte) (ReportOutcome, error) {
+	res := s.Pipe.Submit(ctx, pipeline.Inbound{Tenant: tenant, Payload: payload})
+	if res.RejectedWhole {
+		// 信封级整批拒绝理论上已被端点入口 400 拦截；此处为防御分支（直提交/口径漂移）。
+		return ReportOutcome{}, fmt.Errorf("httpapi: 信封级整批拒绝（入口纪律应已拦截）")
+	}
+	out := ReportOutcome{
+		Accepted:   res.Accepted,
+		Rejected:   res.Rejected,
+		Rejections: make([]Rejection, 0, len(res.Rejections)),
+		Duplicate:  res.Duplicate,
+	}
+	for _, rj := range res.Rejections {
+		out.Rejections = append(out.Rejections, Rejection{
+			PointCode: rj.PointCode,
+			Ts:        rj.Ts,
+			Reason:    string(rj.Reason),
+		})
+	}
+	return out, nil
+}

+// ServeHTTP 实现 http.Handler。
+func (h *ReportHandler) ServeHTTP(w http.ResponseWriter, r *http.Request) {
+	defer func() {
+		if rec := recover(); rec != nil {
+			writeJSON(w, http.StatusInternalServerError, errorResponse{Message: fmt.Sprintf("上报处理异常: %v", rec)})
+		}
+	}()
+
+	body, err := readBody(w, r)
+	if err != nil {
+		writeJSON(w, http.StatusBadRequest, errorResponse{Message: err.Error()})
+		return
+	}
+
+	// 信封级纪律（空 points 4001 / 超 500 整批 / 缺 tenant/batchId / JSON 非法）：
+	// 入口整批拒绝，不进流水线（契约文案与数值码照 message.Error）。
+	env, err := message.Parse(body)
+	if err != nil {
+		var msgErr *message.Error
+		if errors.As(err, &msgErr) {
+			writeJSON(w, http.StatusBadRequest, errorResponse{Code: msgErr.Code, Message: msgErr.Msg})
+			return
+		}
+		writeJSON(w, http.StatusBadRequest, errorResponse{Message: err.Error()})
+		return
+	}
+
+	// 处理顺序第 1 步：验令牌与租户（fail-closed）。网关编码取信封 gateway 声明
+	// （身份主张），租户只取校验产物；缺令牌头不发起校验。
+	token := r.Header.Get(GatewayTokenHeader)
+	if token == "" {
+		writeJSON(w, http.StatusUnauthorized, errorResponse{Message: "缺少网关令牌（X-Gateway-Token）"})
+		return
+	}
+	identity, err := h.Validator.Validate(r.Context(), h.credType(), env.Gateway, token)
+	if err != nil {
+		writeJSON(w, http.StatusUnauthorized, errorResponse{Message: "网关令牌校验失败（fail-closed）: " + err.Error()})
+		return
+	}
+
+	// 统一流水线入口（0073 pipeline；租户 = 令牌侧解析产物）。
+	out, err := h.Sink.Report(r.Context(), identity.TenantCode, body)
+	if err != nil {
+		writeJSON(w, http.StatusServiceUnavailable, errorResponse{Message: "流水线入口暂不可用: " + err.Error()})
+		return
+	}
+	rejections := out.Rejections
+	if rejections == nil {
+		rejections = make([]Rejection, 0)
+	}
+	writeJSON(w, http.StatusOK, reportResponse{
+		Accepted:   out.Accepted,
+		Rejected:   out.Rejected,
+		Rejections: rejections,
+		Duplicate:  out.Duplicate,
+	})
+}
+
+// readBody 读取请求体（带上限）。超限/读失败转 400。
+func readBody(w http.ResponseWriter, r *http.Request) ([]byte, error) {
+	body, err := io.ReadAll(http.MaxBytesReader(w, r.Body, MaxRequestBodyBytes))
+	if err != nil {
+		var maxErr *http.MaxBytesError
+		if errors.As(err, &maxErr) {
+			return nil, fmt.Errorf("请求体超过上限 %d 字节", MaxRequestBodyBytes)
+		}
+		return nil, fmt.Errorf("读取请求体失败: %w", err)
+	}
+	return body, nil
+}

+func writeJSON(w http.ResponseWriter, status int, v any) {
+	w.Header().Set("Content-Type", "application/json; charset=utf-8")
+	w.WriteHeader(status)
+	_ = json.NewEncoder(w).Encode(v)
+}

+// Exec 实现 Executor：POST {Endpoint}/rest/sql，Basic 认证，body = SQL 文本。
+// 全限定名纪律由 SQL 构造侧保证（InsertPointSQL / EnsureSchema——9750 教训）。
+// 响应 code=0 成功；24/918 → *EnvError（WARN + fail-soft）；其余非零 → *SQLError。
+func (c *RESTClient) Exec(ctx context.Context, sql string) error {
+	endpoint := strings.TrimRight(c.cfg.Endpoint, "/") + RESTPath
+	req, err := http.NewRequestWithContext(ctx, http.MethodPost, endpoint, strings.NewReader(sql))
+	if err != nil {
+		return fmt.Errorf("tdengine: 构造 REST 请求失败: %w", err)
+	}
+	req.SetBasicAuth(c.cfg.User, c.cfg.Password)
+	req.Header.Set("Content-Type", "text/plain")
+
+	resp, err := c.hc.Do(req)
+	if err != nil {
+		return fmt.Errorf("tdengine: REST 请求失败: %w", err)
+	}
+	defer resp.Body.Close()
+	body, err := io.ReadAll(io.LimitReader(resp.Body, 1<<20))
+	if err != nil {
+		return fmt.Errorf("tdengine: 读取 REST 响应失败: %w", err)
+	}
+
+	var env restEnvelope
+	if err := json.Unmarshal(body, &env); err != nil {
+		return fmt.Errorf("tdengine: REST 响应解析失败 status=%d body=%s: %w", resp.StatusCode, truncateSQL(string(body)), err)
+	}
+	if env.Code == 0 {
+		if resp.StatusCode != http.StatusOK {
+			return &SQLError{Code: resp.StatusCode, Desc: env.Desc}
+		}
+		return nil
+	}
+	if envErrorCodes[env.Code] {
+		// 环境级失败 fail-soft：WARN 告警 + 以 EnvError 返回（重试计数归流水线重排）。
+		c.logger.Printf("tdengine: 环境级失败（fail-soft 重试计数）code=%d desc=%q sql=%s", env.Code, env.Desc, truncateSQL(sql))
+		return &EnvError{Code: env.Code, Desc: env.Desc}
+	}
+	return &SQLError{Code: env.Code, Desc: env.Desc}
+}

+	return c.Exec(ctx, strings.Join(stmts, "\n"))

+// 契约依据：brief 决策 4 fail-soft——环境级失败（code 24）+ WARN + 重试计数，不炸管道：
+// 经 0073 攒批流水线（Writer{Ex: RESTClient}），持续环境级失败只消耗有界重试
+// （重排计数 → 超上限死信计数告警），流水线不 panic、不死循环、不无界增长。
+func TestContractRESTEnvFailSoftRetryCountThroughPipeline(t *testing.T) {
+	fake := &fakeTaosAdapter{respCode: 24, respDesc: "Too many open files"}
+	srv := httptest.NewServer(http.HandlerFunc(fake.handle))
+	defer srv.Close()
+
+	var logBuf bytes.Buffer
+	client := newClient(srv.URL, nil, log.New(&logBuf, "", 0))
+
+	const snapJSON = `{
+	  "schemaVersion": "1.0.1", "tenantId": 1, "scopeType": 1, "scopeId": 88, "version": 17,
+	  "points": [
+	    {"id":101,"tenantId":1,"pointCode":"F01_WS01_EQ001_M01","pointName":"P1","dataType":1,"valueType":"DOUBLE","deptId":3,"status":1}
+	  ]
+	}`
+	snap, err := snapshot.ParseSnapshot([]byte(snapJSON))
+	if err != nil {
+		t.Fatalf("快照解析失败: %v", err)
+	}
+	snapCache := snapshot.NewCache()
+	if err := snapCache.Replace(snap); err != nil {
+		t.Fatalf("快照替换失败: %v", err)
+	}
+
+	ch := make(chan pipeline.Inbound, 8)
+	p, err := pipeline.Assemble(pipeline.Config{BatchSize: 1, FlushInterval: 5 * time.Millisecond}, pipeline.Deps{
+		Consumer:  chanSourceOf(ch),
+		Parser:    message.Parser{},
+		Snapshots: snapCache,
+		Batches:   pipeline.NewMemoryBatchDeduper(0),
+		Points:    pipeline.NewMemoryPointDeduper(0),
+		Writer:    &tdengine.Writer{Ex: client},
+		Cache:     cache.NewMemoryPointCache(),
+	})
+	if err != nil {
+		t.Fatalf("Assemble error: %v", err)
+	}
+	ctx, cancel := context.WithCancel(context.Background())
+	defer cancel()
+	go func() { _ = p.Run(ctx) }()
+
+	ch <- pipeline.Inbound{Tenant: "group_a", Payload: []byte(
+		`{"tenant":"group_a","gateway":"GW_F01_01","batchId":"B-FAILSOFT","sentAt":1795212000000,` +
+			`"points":[{"pointCode":"F01_WS01_EQ001_M01","ts":1795211995000,"value":132.5,"quality":0}]}`)}
+
+	// 环境级失败持续：重试计数走满（初始下发 + 3 次重排后）转死信计数告警——不炸管道。
+	deadline := time.Now().Add(3 * time.Second)
+	for time.Now().Before(deadline) {
+		st := p.Stats()
+		if st.WriteRetriesExhausted == 1 && st.DeadLettered == 1 {
+			if st.WriteFailures != 4 {
+				t.Errorf("WriteFailures = %d, want 4（初始下发 + 3 次重排）", st.WriteFailures)
+			}
+			if st.RequeuedRows != 3 {
+				t.Errorf("RequeuedRows = %d, want 3（fail-soft 重试计数）", st.RequeuedRows)
+			}
+			if !strings.Contains(logBuf.String(), "环境级失败") {
+				t.Errorf("WARN 日志缺「环境级失败」: %q", logBuf.String())
+			}
+			return
+		}
+		time.Sleep(2 * time.Millisecond)
+	}
+	t.Fatalf("等待超时: 环境级失败未按 fail-soft 重试计数收敛（stats=%+v log=%q）", p.Stats(), logBuf.String())
+}
