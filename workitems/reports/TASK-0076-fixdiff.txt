+//
+// 背景（评审 C1 实证）：修复前 Seen/Mark 分离的 check-then-act 在并发 Submit 下，
+// 800 轮 × 24 goroutine 实验中同 batchId 被当新批处理 8-11 次、唯一 (pointCode, ts)

+// 契约依据：REQ-0013 验收 3「同 batchId 重发 → 整批只落一次」的并发面——
+// 同 batchId 并发双 Submit 恰一处理：多轮迭代断言零重复。
+func TestContractSubmitRaceSameBatchIdExactlyOnce(t *testing.T) {
+	const rounds = 30
+	const goroutines = 12
+	h := newHarness(t, pipeline.Config{BatchSize: 1, FlushInterval: 10 * time.Millisecond})
+	ctx, cancel := context.WithCancel(context.Background())
+	defer cancel()
+	go func() { _ = h.pipe.Run(ctx) }()
+
+	for round := 0; round < rounds; round++ {
+		payload := fmt.Sprintf(
+			`{"tenant":"group_a","gateway":"GW_F01_01","batchId":"B-RACE-%03d","sentAt":1795212000000,"points":[%s]}`,
+			round, pointJSON("F01_WS01_EQ001_M01", 1795211995000+int64(round), 132.5, 0))
+
+		results := make([]pipeline.SubmitResult, goroutines)
+		var wg sync.WaitGroup
+		start := make(chan struct{})
+		for i := 0; i < goroutines; i++ {
+			wg.Add(1)
+			go func(i int) {
+				defer wg.Done()
+				<-start // 同一刻齐发（放大竞态窗口；原子 MarkIfSeen 后仍须恰一处理）
+				results[i] = h.pipe.Submit(ctx, pipeline.Inbound{Tenant: "group_a", Payload: []byte(payload)})
+			}(i)
+		}
+		close(start)
+		wg.Wait()
+
+		var nonDup, dup int
+		for _, r := range results {
+			if r.RejectedWhole {
+				t.Fatalf("round %d: RejectedWhole 异常: %+v", round, r)
+			}
+			if r.Duplicate {
+				dup++
+				if r.Rejected != 0 || len(r.Rejections) != 0 || r.Accepted != 1 {
+					t.Fatalf("round %d: 重复批次结果异常: %+v（重报按成功无操作报全部点数）", round, r)
+				}
+			} else {
+				nonDup++
+				if r.Accepted != 1 || r.Rejected != 0 {
+					t.Fatalf("round %d: 首见批次结果异常: %+v", round, r)
+				}
+			}
+		}
+		if nonDup != 1 || dup != goroutines-1 {
+			t.Fatalf("round %d: 同 batchId 并发恰一处理被破坏——nonDup=%d dup=%d, want 1/%d",
+				round, nonDup, dup, goroutines-1)
+		}
+	}

+	st := h.pipe.Stats()

+	if st.DuplicateBatches != rounds*(goroutines-1) {
+		t.Errorf("DuplicateBatches = %d, want %d（修订轮 C1：并发整批重报零重复处理）",
+			st.DuplicateBatches, rounds*(goroutines-1))
+	}
