+    private static final class MenuRow {
+        final long id;
+        final String parentId;
+        final String menuType;
+        final String perms;

+        MenuRow(String id, String parentId, String menuType, String perms) {

+    /** 一条 INSERT 语句：显式列清单 + VALUES 体 */
+    private static final class Statement {
+        final List<String> columns;
+        final List<List<String>> tuples;
+        /** INSERT ... SELECT 形态（20260925 微信脚本范式，无 VALUES 子句）时的语句体原文 */
+        final String selectBody;
+
+        Statement(List<String> columns, List<List<String>> tuples, String selectBody) {
+            this.columns = columns;
+            this.tuples = tuples;
+            this.selectBody = selectBody;
+        }
+    }

+    /** INSERT ... SELECT 形态行值的权限码样式（三段及以上冒号分隔，如 iot:point:import） */
+    private static final java.util.regex.Pattern PERM_TOKEN =
+            java.util.regex.Pattern.compile("[a-z][a-z0-9_]*(?::[a-z0-9_*]+)+");
+
+    /**
+     * 解析脚本中指定表（sys_menu / sys_role_menu）的 INSERT 全部行。
+     * 先剥除 -- 注释行；列序从 INSERT 显式列清单感知（各脚本 module_id 有无不一，不依赖固定列位）；
+     * VALUES 形态逐元组解析；INSERT ... SELECT 形态（无 VALUES）收集语句体内匹配权限码样式的
+     * 引号字面量——WHERE NOT EXISTS 条件引用的正是本语句登记的码，并入语义正确。
+     */
+    private List<MenuRow> parseMenuRows(String sql, String table) {
+        List<MenuRow> rows = new ArrayList<>();
+        for (Statement stmt : statementsOf(stripCommentLines(sql), table)) {
+            if (stmt.selectBody != null) {
+                for (String quoted : quotedStrings(stmt.selectBody)) {
+                    if (PERM_TOKEN.matcher(quoted).matches()) {
+                        rows.add(new MenuRow("-1", null, null, quoted));
+                    }
+                }
+                continue;
+            }
+            int permsIdx = stmt.columns.indexOf("perms");
+            int idIdx = stmt.columns.indexOf("id");
+            int parentIdx = stmt.columns.indexOf("parent_id");
+            int typeIdx = stmt.columns.indexOf("menu_type");
+            assertTrue(permsIdx >= 0 && idIdx >= 0 && parentIdx >= 0 && typeIdx >= 0,
+                    "INSERT 列清单缺关键列：" + stmt.columns);
+            for (List<String> tuple : stmt.tuples) {
+                rows.add(new MenuRow(unquote(tuple.get(idIdx)), unquote(tuple.get(parentIdx)),
+                        unquote(tuple.get(typeIdx)), unquote(tuple.get(permsIdx))));
+            }
+        }
+        return rows;
+    }

+    private List<String[]> parseRoleMenuRows(String sql) {
+        List<String[]> grants = new ArrayList<>();

+            int roleIdx = stmt.columns.indexOf("role_id");
+            int menuIdx = stmt.columns.indexOf("menu_id");

+    /** 剥除 -- 注释行（照 sql-exec.sh 同口径） */
+    private static String stripCommentLines(String sql) {
+        StringBuilder sb = new StringBuilder();
+        for (String line : sql.split("\n", -1)) {
+            String t = line.strip();
+            if (!t.isEmpty() && !t.startsWith("--")) {
+                sb.append(line).append('\n');
+            }
+        }
+        return sb.toString();
+    }

+    /** 定位 INTO `table` 的每条 INSERT：拆显式列清单 + VALUES 全部顶层元组（引号感知到语句末分号） */
+    private static List<Statement> statementsOf(String sql, String table) {
+        List<Statement> out = new ArrayList<>();
+        String needle = "INTO `" + table + "`";
+        int pos = 0;
+        while (true) {
+            int at = sql.indexOf(needle, pos);
+            if (at < 0) {
+                break;
+            }
+            int parenOpen = sql.indexOf('(', at);
+            int parenClose = matchingParen(sql, parenOpen);
+            List<String> columns = new ArrayList<>();
+            for (String col : splitTopLevel(sql.substring(parenOpen + 1, parenClose))) {
+                columns.add(col.replace("`", "").strip().toLowerCase());
+            }
+            int stmtEnd = statementEnd(sql, parenClose);
+            int valuesAt = sql.indexOf("VALUES", parenClose);
+            if (valuesAt >= 0 && valuesAt < stmtEnd) {
+                out.add(new Statement(columns,
+                        tuplesOf(sql.substring(valuesAt + "VALUES".length(), stmtEnd)), null));
+            } else {
+                // INSERT ... SELECT 形态：无 VALUES 子句，语句体从列清单右括号到语句末分号
+                out.add(new Statement(columns, null, sql.substring(parenClose + 1, stmtEnd)));
+            }
+            pos = stmtEnd;
+        }
+        return out;
+    }
+
+    /** 引号感知提取语句体内全部单引号字符串字面量内容 */
+    private static List<String> quotedStrings(String body) {
+        List<String> out = new ArrayList<>();
+        StringBuilder current = new StringBuilder();
+        boolean inQuote = false;
+        for (int i = 0; i < body.length(); i++) {
+            char c = body.charAt(i);
+            if (c == '\'') {
+                if (inQuote) {
+                    out.add(current.toString());
+                    current.setLength(0);
+                }
+                inQuote = !inQuote;
+            } else if (inQuote) {
+                current.append(c);
+            }
+        }
+        return out;
+    }
+
+    /** VALUES 体扫描：收集全部顶层圆括号元组（引号感知），止于语句末分号或文本末尾 */
+    private static List<List<String>> tuplesOf(String body) {
+        List<List<String>> tuples = new ArrayList<>();
+        int depth = 0;
+        int start = -1;
+        boolean inQuote = false;
+        for (int i = 0; i < body.length(); i++) {
+            char c = body.charAt(i);
+            if (c == '\'') {
+                inQuote = !inQuote;
+            } else if (!inQuote) {
+                if (c == '(') {
+                    if (depth == 0) {
+                        start = i + 1;
+                    }
+                    depth++;
+                } else if (c == ')') {
+                    depth--;
+                    if (depth == 0 && start >= 0) {
+                        tuples.add(splitTopLevel(body.substring(start, i)));
+                        start = -1;
+                    }
+                } else if (c == ';' && depth == 0) {
+                    break;
+                }
+            }
+        }
+        return tuples;
+    }
+
+    /** 从 offset 起扫到语句末分号（引号感知） */
+    private static int statementEnd(String sql, int offset) {
+        boolean inQuote = false;
+        for (int i = offset; i < sql.length(); i++) {
+            char c = sql.charAt(i);
+            if (c == '\'') {
+                inQuote = !inQuote;
+            } else if (c == ';' && !inQuote) {
+                return i;
+            }
+        }
+        return sql.length();
+    }
+
+    /** 找匹配右括号（引号感知） */
+    private static int matchingParen(String sql, int openAt) {
+        int depth = 0;
+        boolean inQuote = false;
+        for (int i = openAt; i < sql.length(); i++) {
+            char c = sql.charAt(i);
+            if (c == '\'') {
+                inQuote = !inQuote;
+            } else if (!inQuote) {
+                if (c == '(') {
+                    depth++;
+                } else if (c == ')') {
+                    depth--;
+                    if (depth == 0) {
+                        return i;
+                    }
+                }
+            }
+        }
+        throw new IllegalStateException("括号不匹配 at " + openAt);
+    }
+
+    /** 按顶层逗号切分（引号感知） */
+    private static List<String> splitTopLevel(String text) {
+        List<String> parts = new ArrayList<>();
+        StringBuilder current = new StringBuilder();
+        boolean inQuote = false;
+        for (int i = 0; i < text.length(); i++) {
+            char c = text.charAt(i);
+            if (c == '\'') {
+                inQuote = !inQuote;
+                current.append(c);
+            } else if (c == ',' && !inQuote) {
+                parts.add(current.toString());
+                current.setLength(0);
+            } else {
+                current.append(c);
+            }
+        }
+        parts.add(current.toString());
+        return parts;
+    }

+    private static String unquote(String v) {
+        String s = v.strip();
+        if (s.length() >= 2 && s.startsWith("'") && s.endsWith("'")) {
+            return s.substring(1, s.length() - 1);
+        }
+        return "NULL".equalsIgnoreCase(s) ? null : s;
+    }

+    /** 收集 net.otlab.admin.module 下全部 @RestController 的 @SaCheckPermission 值（类级+方法级） */
+    private Set<String> collectModuleAnnotatedPerms() throws ClassNotFoundException {
+        ClassPathScanningCandidateComponentProvider scanner =
+                new ClassPathScanningCandidateComponentProvider(false);
+        scanner.addIncludeFilter(new AnnotationTypeFilter(RestController.class));
+        Set<String> perms = new TreeSet<>();
+        int controllers = 0;
+        for (BeanDefinition bd : scanner.findCandidateComponents("net.otlab.admin.module")) {

+            SaCheckPermission classLevel = clazz.getAnnotation(SaCheckPermission.class);
+            if (classLevel != null) {

+            }
+            for (Method method : clazz.getDeclaredMethods()) {
+                SaCheckPermission annotation = method.getAnnotation(SaCheckPermission.class);
+                if (annotation != null) {
+                    perms.addAll(Arrays.asList(annotation.value()));
+                }
+            }
+        }
+        assertTrue(controllers > 50, "控制器扫描异常偏少（" + controllers + "），扫描基包或类路径不可信");
+        return perms;
+    }

+        assertTrue(Files.exists(script), "升级脚本不存在：" + script);
+        return Files.readString(script, StandardCharsets.UTF_8);
+    }

+    /** 反射取指定控制器指定（唯一同名）方法的 @SaCheckPermission 首值（无注解/歧义即断言失败） */
+    private String methodPerm(Class<?> clazz, String methodName) {
+        List<Method> hits = new ArrayList<>();
+        for (Method m : clazz.getDeclaredMethods()) {
+            if (m.getName().equals(methodName)) {
+                hits.add(m);
+            }
+        }
+        assertEquals(1, hits.size(), clazz.getSimpleName() + "#" + methodName + " 方法必须恰一个（防歧义）");
+        SaCheckPermission annotation = hits.get(0).getAnnotation(SaCheckPermission.class);
+        assertNotNull(annotation, clazz.getSimpleName() + "#" + methodName + " 缺 @SaCheckPermission");
+        assertTrue(annotation.value().length >= 1, "@SaCheckPermission 必须声明权限值");
+        return annotation.value()[0];
+    }

+    private Set<String> difference(Set<String> minuend, Set<String> subtrahend) {
+        Set<String> out = new LinkedHashSet<>(minuend);
+        out.removeAll(subtrahend);
+        return out;
+    }
