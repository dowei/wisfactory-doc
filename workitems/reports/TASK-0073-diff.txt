+// Package cache 承载 Redis 点位缓存抽象（FEAT-0009 处理顺序第 5 步「+ 更新 Redis 点位缓存」）。
+//
+// 值形态契约：统一消息模型 v1.1.0 §3.5「GET /openapi/v1/items/realtime —— 优先读 Redis，
+// 返回 v / q / ts」，缓存 JSON 恰为 {"v","q","ts"} 三键。
+//
+// 留白与口径：
+//   - 缓存键 scheme 上游未定版，本实现拟定 iot:{tenant}:point:{pointCode}（TASK-0029 接线
+//     真 Redis 时如需调整只改 Key 函数与对应读取方）；
+//   - 本包只提供抽象与内存假实现；真 Redis 客户端实现归 TASK-0029，接口零改动替换。
+//     测试环境不连真实 Redis（后端占用 db0，隔离口径见 TASK-0073 报告）。
+package cache
+
+import (
+	"context"
+	"encoding/json"
+	"errors"
+	"fmt"
+	"sync"
+)
+
+// Value 点位缓存值：实时值 v、质量码 q（CONV-02 0-10）、采样时间 ts（毫秒）。
+type Value struct {
+	V  float64 `json:"v"`
+	Q  int     `json:"q"`
+	Ts int64   `json:"ts"`
+}
+
+// ErrCacheMiss 点位缓存未命中。
+var ErrCacheMiss = errors.New("cache: 点位缓存未命中")
+
+// PointCache 点位缓存抽象。
+type PointCache interface {
+	// SetPoint 覆盖写入点位最新值。
+	SetPoint(ctx context.Context, tenant, pointCode string, v Value) error
+	// GetPoint 读取点位最新值；未命中返回 ErrCacheMiss（未知编码/无数据不得静默给零值）。
+	GetPoint(ctx context.Context, tenant, pointCode string) (Value, error)
+}
+
+// Key 点位缓存键（拟定 scheme：iot:{tenant}:point:{pointCode}；见包注释留白声明）。
+func Key(tenant, pointCode string) string {
+	return fmt.Sprintf("iot:%s:point:%s", tenant, pointCode)
+}
+
+// MemoryPointCache 内存假实现：内部以 JSON 字节存储（与真 Redis 的字符串语义一致），
+// 并发安全（RWMutex）。键空间随配置点位规模有界，无需淘汰。
+type MemoryPointCache struct {
+	mu sync.RWMutex
+	m  map[string][]byte
+}
+
+// NewMemoryPointCache 创建空缓存。
+func NewMemoryPointCache() *MemoryPointCache {
+	return &MemoryPointCache{m: make(map[string][]byte)}
+}
+
+// SetPoint 覆盖写入（JSON 编码 v/q/ts 三键形态）。
+func (c *MemoryPointCache) SetPoint(_ context.Context, tenant, pointCode string, v Value) error {
+	b, err := json.Marshal(v)
+	if err != nil {
+		return fmt.Errorf("cache: 编码点位值失败: %w", err)
+	}
+	c.mu.Lock()
+	c.m[Key(tenant, pointCode)] = b
+	c.mu.Unlock()
+	return nil
+}
+
+// GetPoint 读取并解码；未命中返回 ErrCacheMiss。
+func (c *MemoryPointCache) GetPoint(_ context.Context, tenant, pointCode string) (Value, error) {
+	c.mu.RLock()
+	b, ok := c.m[Key(tenant, pointCode)]
+	c.mu.RUnlock()
+	if !ok {
+		return Value{}, ErrCacheMiss
+	}
+	var v Value
+	if err := json.Unmarshal(b, &v); err != nil {
+		return Value{}, fmt.Errorf("cache: 解码点位值失败: %w", err)
+	}
+	return v, nil
+}

+// Package message 承载统一消息模型（冻结契约 v1.1.0 §3.2）的线上形态与解析。
+//
+// 契约纪律（§3.2）：
+//   - 消息只认 pointCode，不认点位 id——id 是管理系统内部概念；
+//   - ts 为设备侧采样时间（毫秒），与 sentAt（发送时间）分开，迟到数据按 ts 落库；
+//   - 单消息 points 不超过 500 条，超限网关侧拆包；batchId 用于整批幂等与对账；
+//   - quality 由网关能判断的先填，超限类质量码由接入服务按点位 min/max 二次判定后覆盖写入（判定挂点见 internal/quality 与流水线，完整逻辑归 TASK-0028）。
+//
+// 解析容错：对未知字段宽松（忽略）——网关端演进字段不得打停接入面；
+// 信封级纪律（tenant/batchId 必填、空 points、超限）按契约整批拒绝。
+package message
+
+import (
+	"encoding/json"
+	"errors"
+	"fmt"
+)
+
+// MaxPointsPerMessage 契约批量上限：单消息 points 不超过 500 条（v1.1.0 §3.2）。
+const MaxPointsPerMessage = 500
+
+// CodeEmptyPoints 空 points 的拒绝码（REQ-0013：空数据=空 points 4001；TC-0009 步骤 6）。
+const CodeEmptyPoints int64 = 4001
+
+// Point 统一消息模型点位条目（§3.2 示例对象形态）。
+type Point struct {
+	PointCode string  `json:"pointCode"` // 消息只认编码不认 id
+	Ts        int64   `json:"ts"`        // 设备侧采样时间（毫秒），与 sentAt 分开
+	Value     float64 `json:"value"`
+	Quality   int     `json:"quality"` // 网关先填；超限类由接入按 min/max 二次判定覆盖（TASK-0028）
+}
+
+// Envelope 统一消息模型信封（§3.2 上行消息 JSON）。
+type Envelope struct {
+	Tenant  string  `json:"tenant"`
+	Gateway string  `json:"gateway"`
+	BatchID string  `json:"batchId"` // 整批幂等与对账键
+	SentAt  int64   `json:"sentAt"`  // 发送时间（毫秒），与 ts 分开
+	Points  []Point `json:"points"`
+}
+
+// Error 解析/纪律拒绝错误。Code 为契约已定数值码（4001=空 points）；
+// 契约未定数值码的拒绝（超限等）Code=0，HTTP 响应映射归 TASK-0027，不臆造码值。
+type Error struct {
+	Code int64
+	Msg  string
+}
+
+func (e *Error) Error() string {
+	if e.Code != 0 {
+		return fmt.Sprintf("message: code %d: %s", e.Code, e.Msg)
+	}
+	return "message: " + e.Msg
+}
+
+// CodeOf 返回 err 的契约数值码；非 *Error 或未定数值码返回 0。
+func CodeOf(err error) int64 {
+	var e *Error
+	if errors.As(err, &e) {
+		return e.Code
+	}
+	return 0
+}
+
+var (
+	// ErrEmptyPoints 空 points（code 4001）。
+	ErrEmptyPoints = &Error{Code: CodeEmptyPoints, Msg: "空 points：空数据整批拒绝"}
+	// ErrTooManyPoints 单消息 points 超过 500 条上限，整批拒绝（契约未给数值码，Code=0）。
+	ErrTooManyPoints = &Error{Msg: fmt.Sprintf("points 超过单消息上限 %d 条，整批拒绝", MaxPointsPerMessage)}
+	// ErrMissingTenant 缺 tenant：租户是选库（iot_<tenant_code>）与隔离的根，缺则无法处理。
+	ErrMissingTenant = &Error{Msg: "缺少 tenant 字段"}
+	// ErrMissingBatchID 缺 batchId：整批幂等与对账依赖 batchId，缺则整批拒绝。
+	ErrMissingBatchID = &Error{Msg: "缺少 batchId 字段（batchId 用于整批幂等与对账）"}
+)
+
+// Parse 解析统一消息模型线上形态 JSON 并执行信封级纪律校验。
+// 整批级拒绝返回 (nil, err)；JSON 非法同样整批拒绝。
+func Parse(data []byte) (*Envelope, error) {
+	var env Envelope
+	if err := json.Unmarshal(data, &env); err != nil {
+		return nil, &Error{Msg: "JSON 解析失败: " + err.Error()}
+	}
+	if env.Tenant == "" {
+		return nil, ErrMissingTenant
+	}
+	if env.BatchID == "" {
+		return nil, ErrMissingBatchID
+	}
+	// 空 points：空数据 4001（REQ-0013 / TC-0009 步骤 6）。points 键省略视同空。
+	if len(env.Points) == 0 {
+		return nil, ErrEmptyPoints
+	}
+	// 批量上限：单消息 points 不超过 500 条，超限整批拒绝（TC-0009 步骤 7）。
+	if len(env.Points) > MaxPointsPerMessage {
+		return nil, ErrTooManyPoints
+	}
+	return &env, nil
+}

+// Parser 是 Parse 的可注入形态，供流水线组装与 TASK-0027 接线替换。
+type Parser struct{}
+
+// Parse 实现 pipeline.Parser 语义。
+func (Parser) Parse(data []byte) (*Envelope, error) { return Parse(data) }

+//
+//	Consumer(假 EMQX=内存 channel) → Parser → [查快照→未知死信→禁用丢弃→去重→质量判定挂点]
+//	→ Batcher(有界缓冲/水位/丢最旧/失败重排) → Writer(假 TDengine 执行器) → PointCache 更新
+//
+// 组件间只经接口交互（Consumer/Parser/SnapshotLookup/BatchDeduper/PointDeduper/
+// DeadLetterSink/Writer/PointCache + QualityJudge 函数挂点），TASK-0027~0029 接线时
+// 以真实实现零改动替换假实现。
+//
+// 并发模型（brief 决策 5，起步口径）：
+//   - 单消费循环：Run 内单个 goroutine 顺序处理消息（解析→校验→入攒批缓冲），
+//     不并行处理批次——有界队列 = Consumer 提供的有界消息通道（假 EMQX 为有界 channel；
+//     真实 MQTT 订阅同样共享有界缓冲）。队列满时消费循环阻塞（背压），

+//   - 单 flusher goroutine：按水位（缓冲 ≥ BatchSize）或定时（FlushInterval，FEAT-0009
+//     「攒批 100-500 条或 200ms」）触发，串行写库并更新缓存。

+// Assemble 校验依赖并应用配置默认值。
+func Assemble(cfg Config, deps Deps) (*Pipeline, error) {
+	if deps.Consumer == nil {
+		return nil, errors.New("pipeline: 缺 Consumer")
+	}
+	if deps.Parser == nil {
+		return nil, errors.New("pipeline: 缺 Parser")
+	}
+	if deps.Snapshots == nil {
+		return nil, errors.New("pipeline: 缺 SnapshotLookup（配置快照缓存）")
+	}
+	if deps.Batches == nil {
+		return nil, errors.New("pipeline: 缺 BatchDeduper")
+	}
+	if deps.Points == nil {
+		return nil, errors.New("pipeline: 缺 PointDeduper")
+	}
+	if deps.Writer == nil {
+		return nil, errors.New("pipeline: 缺 Writer")
+	}
+	if deps.Cache == nil {
+		return nil, errors.New("pipeline: 缺 PointCache")
+	}
+	if cfg.BatchSize == 0 {
+		cfg.BatchSize = 500
+	}
+	if cfg.FlushInterval == 0 {
+		cfg.FlushInterval = 200 * time.Millisecond
+	}
+	if cfg.BufferCap == 0 {
+		cfg.BufferCap = 4096
+	}
+	if cfg.DedupCap == 0 {
+		cfg.DedupCap = 65536
+	}
+	if cfg.BatchSize < 1 || cfg.BatchSize > 500 {
+		return nil, fmt.Errorf("pipeline: BatchSize=%d 超出 1-500（FEAT-0009 攒批口径）", cfg.BatchSize)
+	}
+	if cfg.FlushInterval <= 0 {
+		return nil, fmt.Errorf("pipeline: FlushInterval=%v 必须为正", cfg.FlushInterval)
+	}
+	if cfg.BufferCap < 1 {
+		return nil, fmt.Errorf("pipeline: BufferCap=%d 必须为正", cfg.BufferCap)
+	}
+	if deps.Judge == nil {
+		deps.Judge = PassthroughJudge
+	}
+	if deps.Dead == nil {
+		deps.Dead = DeadLetterFunc(func(DeadLetterReason, string, string, *message.Point) {})
+	}
+	return &Pipeline{
+		cfg:     cfg,
+		deps:    deps,
+		judge:   deps.Judge,
+		flushCh: make(chan struct{}, 1),
+	}, nil
+}

+// SetJudge 替换质量判定挂点（并发安全；Run 前设置，Run 中替换亦可见）。
+func (p *Pipeline) SetJudge(j QualityJudge) {
+	p.judgeMu.Lock()
+	p.judge = j
+	p.judgeMu.Unlock()
+}

+// Run 启动单消费循环与 flusher，阻塞直到 ctx 取消或消息通道关闭。
+// 一次 Pipeline 实例只运行一个消费循环（并发模型见包注释）。
+func (p *Pipeline) Run(ctx context.Context) error {
+	go p.flusher(ctx)
+	msgs := p.deps.Consumer.Messages()
+	for {
+		select {
+		case <-ctx.Done():
+			return nil // 优雅停机：缓冲中未写出行随进程退出丢弃（内核口径，真实部署由 0027 收口）
+		case in, ok := <-msgs:
+			if !ok {
+				return nil
+			}
+			p.processOne(ctx, in)
+		}
+	}
+}

+// ---- 处理顺序（冻结五步）----
+
+// processOne 处理一条入站消息（单消费循环内串行执行）。
+func (p *Pipeline) processOne(ctx context.Context, in Inbound) {
+	p.stats.ReceivedBatches.Add(1)
+
+	// 第 1 步：验令牌与租户（挂点）。租户只来自令牌解析产物；缺租户整批拒绝。
+	if in.Tenant == "" {
+		p.rejectWholeBatch(in, "")
+		return
+	}
+
+	env, err := p.deps.Parser.Parse(in.Payload)
+	if err != nil {
+		// 信封级整批拒绝：空 points 4001 / 超 500 / 缺 batchId / JSON 非法。
+		p.rejectWholeBatch(in, "")
+		return
+	}
+
+	// batchId 整批幂等（标记即接受，内核简化口径；崩溃安全归 TASK-0028）。
+	if p.deps.Batches.SeenBatch(env.BatchID) {
+		p.stats.DuplicateBatches.Add(1)
+		return
+	}
+	p.deps.Batches.MarkBatch(env.BatchID)
+
+	judge := p.currentJudge()
+	rows := make([]tdengine.PointRow, 0, len(env.Points))
+	for i := range env.Points {
+		pt := &env.Points[i]
+
+		// 第 2、3 步：按 pointCode 查配置快照；未知编码死信计数告警、不落库。
+		cfgPt, ok := p.deps.Snapshots.LookupPoint(pt.PointCode)
+		if !ok {
+			p.stats.DeadLettered.Add(1)
+			p.deps.Dead.OnDeadLetter(DeadLetterUnknownCode, in.Tenant, env.BatchID, pt)
+			continue
+		}
+		if cfgPt.Disabled() {
+			p.stats.DisabledDiscarded.Add(1)
+			p.deps.Dead.OnDeadLetter(DeadLetterDisabledPoint, in.Tenant, env.BatchID, pt)
+			continue
+		}
+
+		// (pointCode, ts) 幂等键（内核简化口径同 batchId）。
+		if p.deps.Points.SeenPoint(pt.PointCode, pt.Ts) {
+			p.stats.DuplicatePoints.Add(1)
+			continue
+		}
+		p.deps.Points.MarkPoint(pt.PointCode, pt.Ts)
+
+		// 第 4 步：质量判定（挂点；CONV-02 质量码口径见 internal/quality）。
+		q := judge(cfgPt, pt)
+
+		kind, ok := tdengine.KindFor(cfgPt.DataType)
+		if !ok {
+			p.stats.DeadLettered.Add(1)
+			p.deps.Dead.OnDeadLetter(DeadLetterInvalidConfig, in.Tenant, env.BatchID, pt)
+			continue
+		}
+
+		// 第 5 步前半：按 data_type 分发构造写入行（标签取自配置快照）。
+		rows = append(rows, tdengine.PointRow{
+			TenantCode: in.Tenant, // 选库键：令牌解析租户（防写错靠写方选库）
+			TenantID:   cfgPt.TenantID,
+			PointCode:  pt.PointCode,
+			Kind:       kind,
+			DeptID:     deptIDOf(cfgPt),
+			Ts:         pt.Ts,
+			Value:      pt.Value,
+			Quality:    q,
+		})
+	}
+
+	for _, row := range rows {
+		p.push(row)
+	}
+}

+func (p *Pipeline) rejectWholeBatch(in Inbound, batchID string) {
+	p.stats.RejectedBatches.Add(1)
+	p.deps.Dead.OnDeadLetter(DeadLetterInvalidEnvelope, in.Tenant, batchID, nil)
+}

+func (p *Pipeline) currentJudge() QualityJudge {
+	p.judgeMu.RLock()
+	defer p.judgeMu.RUnlock()
+	return p.judge
+}

+func deptIDOf(p *snapshot.Point) int64 {
+	if p.DeptID != nil {
+		return *p.DeptID
+	}
+	return 0
+}

+// ---- 攒批缓冲（有界，满丢最旧并计数）与 flusher ----
+
+// push 入缓冲；满时丢最旧（FIFO，ADR-0014 口径）并计数；达到水位触发立即 flush。
+func (p *Pipeline) push(row tdengine.PointRow) {
+	p.bufMu.Lock()
+	if len(p.buf) >= p.cfg.BufferCap {
+		p.buf = p.buf[1:]
+		p.stats.DroppedOldest.Add(1)
+	}
+	p.buf = append(p.buf, row)
+	full := len(p.buf) >= p.cfg.BatchSize
+	p.bufMu.Unlock()
+	p.stats.AcceptedPoints.Add(1)
+	if full {
+		select {
+		case p.flushCh <- struct{}{}:
+		default: // 已有待处理信号
+		}
+	}
+}
+
+// flusher 定时 + 水位双触发（单 goroutine，串行写）。
+func (p *Pipeline) flusher(ctx context.Context) {
+	ticker := time.NewTicker(p.cfg.FlushInterval)
+	defer ticker.Stop()
+	for {
+		select {
+		case <-ctx.Done():
+			return
+		case <-ticker.C:
+			p.flush(ctx)
+		case <-p.flushCh:
+			p.flush(ctx)
+		}
+	}
+}
+
+// flush 取缓冲头部（最旧）至多 BatchSize 行下发；失败整批重排回缓冲头部。
+func (p *Pipeline) flush(ctx context.Context) {
+	p.bufMu.Lock()
+	n := p.cfg.BatchSize
+	if len(p.buf) < n {
+		n = len(p.buf)
+	}
+	if n == 0 {
+		p.bufMu.Unlock()
+		return
+	}
+	batch := make([]tdengine.PointRow, n)
+	copy(batch, p.buf[:n])
+	p.buf = p.buf[n:]
+	p.bufMu.Unlock()
+
+	if err := p.deps.Writer.WritePoints(ctx, batch); err != nil {
+		p.stats.WriteFailures.Add(1)
+		p.requeueFront(batch)
+		return
+	}
+	p.stats.WrittenBatches.Add(1)
+	p.stats.WrittenPoints.Add(int64(len(batch)))
+	p.updateCache(ctx, batch)
+}
+
+// requeueFront 失败重排：整批放回缓冲头部重试；总量超上限时丢最旧并计数（不无界增长）。
+func (p *Pipeline) requeueFront(batch []tdengine.PointRow) {
+	p.bufMu.Lock()
+	defer p.bufMu.Unlock()
+	combined := make([]tdengine.PointRow, 0, len(batch)+len(p.buf))
+	combined = append(combined, batch...)
+	combined = append(combined, p.buf...)
+	if over := len(combined) - p.cfg.BufferCap; over > 0 {
+		combined = combined[over:]
+		p.stats.DroppedOldest.Add(int64(over))
+	}
+	p.buf = combined
+	p.stats.RequeuedRows.Add(int64(len(batch)))
+}

+// ---- 内存去重假实现（FIFO 淘汰，对齐 ADR-0014 丢最旧口径）----
+
+// MemoryBatchDeduper batchId 幂等假实现：有界 FIFO，超容量淘汰最早标记项。
+type MemoryBatchDeduper struct {
+	mu    sync.Mutex
+	cap   int
+	seen  map[string]struct{}
+	order []string
+}
+
+// NewMemoryBatchDeduper 创建容量 cap 的假实现（cap<=0 取 65536）。
+func NewMemoryBatchDeduper(cap int) *MemoryBatchDeduper {
+	if cap <= 0 {
+		cap = 65536
+	}
+	return &MemoryBatchDeduper{cap: cap, seen: make(map[string]struct{}, cap)}
+}
+
+// SeenBatch 报告 batchId 是否已标记。
+func (d *MemoryBatchDeduper) SeenBatch(batchID string) bool {
+	d.mu.Lock()
+	defer d.mu.Unlock()
+	_, ok := d.seen[batchID]
+	return ok
+}
+
+// MarkBatch 标记 batchId（FIFO 淘汰最早项）。
+func (d *MemoryBatchDeduper) MarkBatch(batchID string) {
+	d.mu.Lock()
+	defer d.mu.Unlock()
+	if _, ok := d.seen[batchID]; ok {
+		return
+	}
+	if len(d.order) >= d.cap {
+		old := d.order[0]
+		d.order = d.order[1:]
+		delete(d.seen, old)
+	}
+	d.seen[batchID] = struct{}{}
+	d.order = append(d.order, batchID)
+}
+
+// MemoryPointDeduper (pointCode, ts) 幂等假实现：有界 FIFO。
+type MemoryPointDeduper struct {
+	mu    sync.Mutex
+	cap   int
+	seen  map[string]struct{}
+	order []string
+}
+
+// NewMemoryPointDeduper 创建容量 cap 的假实现（cap<=0 取 65536）。
+func NewMemoryPointDeduper(cap int) *MemoryPointDeduper {
+	if cap <= 0 {
+		cap = 65536
+	}
+	return &MemoryPointDeduper{cap: cap, seen: make(map[string]struct{}, cap)}
+}
+
+func pointDedupKey(pointCode string, ts int64) string {
+	return fmt.Sprintf("%s\x00%d", pointCode, ts)
+}
+
+// SeenPoint 报告 (pointCode, ts) 是否已标记。
+func (d *MemoryPointDeduper) SeenPoint(pointCode string, ts int64) bool {
+	d.mu.Lock()
+	defer d.mu.Unlock()
+	_, ok := d.seen[pointDedupKey(pointCode, ts)]
+	return ok
+}
+
+// MarkPoint 标记 (pointCode, ts)（FIFO 淘汰最早项）。
+func (d *MemoryPointDeduper) MarkPoint(pointCode string, ts int64) {
+	d.mu.Lock()
+	defer d.mu.Unlock()
+	k := pointDedupKey(pointCode, ts)
+	if _, ok := d.seen[k]; ok {
+		return
+	}
+	if len(d.order) >= d.cap {
+		old := d.order[0]
+		d.order = d.order[1:]
+		delete(d.seen, old)
+	}
+	d.seen[k] = struct{}{}
+	d.order = append(d.order, k)
+}

+// Package quality 承载 CONV-02 质量码唯一存储口径：数值枚举 0-10 与产品文本码映射。
+//
+// 契约依据：
+//   - workitems/conventions.md CONV-02（质量码数值枚举 0-10 为唯一存储口径）
+//   - TDengine 超级表契约 v1.0.0「质量码枚举注释（CONV-02 修订叠加）」
+//
+// 完整的质量判定逻辑（量程判定覆盖、时间校验纠偏、JUMP 检测等）归 TASK-0028；
+// 本包只提供枚举常量、文本码映射与取值域判断，作为流水线的判定挂点基础。
+package quality
+
+// 质量码数值枚举 0-10（CONV-02 唯一存储口径）。
+const (
+	Good              = 0  // 正常
+	OverMax           = 1  // 超上限
+	UnderMin          = 2  // 超下限
+	StaleOffline      = 3  // 断线/超时补标
+	DeviceAbnormal    = 4  // 设备异常值
+	Manual            = 5  // 手工补录
+	CalcFailed        = 6  // 计算失败
+	RollupIncomplete  = 7  // 汇总不完整
+	JumpPendingReview = 8  // 突变待核查（一期检测关闭，码位预留，ADR-0010）
+	CrossPeriodUsage  = 9  // 跨周期用量
+	CompareDeviation  = 10 // 比对偏差待核查（原值不删、计入完整度缺失）
+)
+
+// Valid 报告 code 是否落在 0-10 存储口径内。
+func Valid(code int) bool {
+	return code >= Good && code <= CompareDeviation
+}
+
+// FromTextCode 把产品文本码映射为质量码。
+//
+// CONV-02 文本码映射：GOOD→0、OVER_RANGE→1/2、STALE/OFFLINE→3、JUMP→8、MANUAL→5。
+// OVER_RANGE 依赖方向：超上限（overMax=true）→1，超下限→2。
+//
+// CALC_ERR 不落时序行（归 iot_calc_log），返回 ok=false，调用方不得将其写入时序数据。
+// 其余未知文本码同样返回 ok=false，由调用方按各自的告警口径处理。
+func FromTextCode(text string, overMax bool) (int, bool) {
+	switch text {
+	case "GOOD":
+		return Good, true
+	case "OVER_RANGE":
+		if overMax {
+			return OverMax, true
+		}
+		return UnderMin, true
+	case "STALE", "OFFLINE":
+		return StaleOffline, true
+	case "JUMP":
+		return JumpPendingReview, true
+	case "MANUAL":
+		return Manual, true
+	default:
+		// CALC_ERR 及一切未知文本码：不映射、不落时序行。
+		return Good, false
+	}
+}

+	// ChangesPath 增量变更查询：GET /internal/v1/config/changes?sinceVersion=

+// Scope 快照拉取范围。
+type Scope struct {
+	Type    int   // scopeType：1 网关范围（装载通道+点位） 2 计算范围（装载数据项+公式+指标，scopeId=0）
+	ID      int64 // scopeId：scopeType=1 时为 gateway_id
+	Version int64 // 0 = 不带 version 参数（返回最新版）；>0 = 返回指定版
+}

+// FetchResult 拉取结果：响应信封三部分（§3.3：响应含 version、snapshotHash 与快照体）。
+type FetchResult struct {
+	Snapshot     *Snapshot
+	SnapshotHash string
+	Version      int64 // 响应信封 version，与快照体顶层 version 同值（§3.3）
+}

+// SnapshotFetcher 快照拉取客户端抽象。假实现 = 本地 httptest 端点回放契约样例；
+// TASK-0027 接线时替换为指向管理系统的真实实现，本接口零改动。
+type SnapshotFetcher interface {
+	Fetch(ctx context.Context, scope Scope) (*FetchResult, error)
+}
+
+// HTTPFetcher SnapshotFetcher 的 HTTP 实现（§3.3 GET 全量拉取）。
+type HTTPFetcher struct {
+	// BaseURL 管理系统基址（如 http://mgmt-host:8080）。
+	BaseURL string
+	// Client HTTP 客户端；nil 时用 http.DefaultClient。
+	Client *http.Client
+}

+// envelope 响应信封。包装键名 "snapshot" 为本实现拟定（契约仅承诺 version、snapshotHash
+// 与快照体三部分内容，未定信封键名；见包注释留白声明）。
+type envelope struct {
+	Version      int64           `json:"version"`
+	SnapshotHash string          `json:"snapshotHash"`
+	Snapshot     json.RawMessage `json:"snapshot"`
+}
+
+// Fetch 拉取全量快照并执行信封纪律校验：
+// 信封 version 与快照体顶层 version 必须同值（§3.3），不同值拒收。
+func (f *HTTPFetcher) Fetch(ctx context.Context, scope Scope) (*FetchResult, error) {
+	q := url.Values{}
+	q.Set("scopeType", strconv.Itoa(scope.Type))
+	q.Set("scopeId", strconv.FormatInt(scope.ID, 10))
+	if scope.Version > 0 {
+		q.Set("version", strconv.FormatInt(scope.Version, 10))
+	}
+	endpoint := f.BaseURL + SnapshotPath + "?" + q.Encode()
+
+	client := f.Client
+	if client == nil {
+		client = http.DefaultClient
+	}
+	req, err := http.NewRequestWithContext(ctx, http.MethodGet, endpoint, nil)
+	if err != nil {
+		return nil, fmt.Errorf("snapshot: 构造请求失败: %w", err)
+	}
+	resp, err := client.Do(req)
+	if err != nil {
+		return nil, fmt.Errorf("snapshot: 拉取失败: %w", err)
+	}
+	defer resp.Body.Close()
+	body, err := io.ReadAll(resp.Body)
+	if err != nil {
+		return nil, fmt.Errorf("snapshot: 读取响应失败: %w", err)
+	}
+	if resp.StatusCode != http.StatusOK {
+		return nil, fmt.Errorf("snapshot: 拉取失败 status=%d body=%s", resp.StatusCode, truncate(body, 256))
+	}
+
+	var env envelope
+	if err := json.Unmarshal(body, &env); err != nil {
+		return nil, fmt.Errorf("snapshot: 响应信封解析失败: %w", err)
+	}
+	snap, err := ParseSnapshot(env.Snapshot)
+	if err != nil {
+		return nil, fmt.Errorf("snapshot: 快照体校验失败: %w", err)
+	}
+	if env.Version != snap.Version {
+		return nil, fmt.Errorf("snapshot: 信封 version(%d) 与快照体 version(%d) 不同值（§3.3 要求同值），拒收", env.Version, snap.Version)
+	}
+	return &FetchResult{Snapshot: snap, SnapshotHash: env.SnapshotHash, Version: env.Version}, nil
+}

+// truncate 截断调试输出。
+func truncate(b []byte, n int) string {
+	if len(b) <= n {
+		return string(b)
+	}
+	return string(b[:n]) + "…"
+}

+// pointIndex 惰性构建的 pointCode → 点位索引，供流水线按编码查配置（§3.2 处理顺序第 2 步）。
+type pointIndex map[string]*Point
+
+// Cache 配置快照本地缓存：原子替换（新快照校验失败则继续用旧版并回执失败）。
+// 并发模型：读走 atomic.Value 快照 + 只读索引；Replace 全量换入，无逐条锁。
+type Cache struct {
+	cur atomic.Pointer[snapshotWithIndex]
+}
+
+type snapshotWithIndex struct {
+	snap  *Snapshot
+	index pointIndex
+}
+
+// NewCache 空缓存（尚未拉取快照时 LookupPoint 一律未命中）。
+func NewCache() *Cache {
+	return &Cache{}
+}
+
+// Replace 校验并原子换入新快照；校验失败返回错误且保持旧版（§3.3 接入消费方式）。
+func (c *Cache) Replace(s *Snapshot) error {
+	if s == nil {
+		return errors.New("snapshot: Replace 收到 nil 快照")
+	}
+	idx := make(pointIndex, len(s.Points))
+	for i := range s.Points {
+		p := &s.Points[i]
+		if err := p.Validate(); err != nil {
+			return fmt.Errorf("snapshot: 新快照校验失败（version=%s）: %w", strconv.FormatInt(s.Version, 10), err)
+		}
+		idx[p.PointCode] = p
+	}
+	c.cur.Store(&snapshotWithIndex{snap: s, index: idx})
+	return nil
+}
+
+// Get 返回当前快照（可能为 nil，表示尚未拉取）。
+func (c *Cache) Get() *Snapshot {
+	if v := c.cur.Load(); v != nil {
+		return v.snap
+	}
+	return nil
+}
+
+// LookupPoint 按 pointCode 查点位配置；未命中 = 未知编码（流水线进死信）。
+func (c *Cache) LookupPoint(pointCode string) (*Point, bool) {
+	v := c.cur.Load()
+	if v == nil {
+		return nil, false
+	}
+	p, ok := v.index[pointCode]
+	return p, ok
+}

+// 四张超级表名（附录 B 逐字）。
+const (
+	StablePointNum    = "st_point_num"
+	StablePointState  = "st_point_state"
+	StableItemValue   = "st_item_value"
+	StableMetricValue = "st_metric_value"

+// kind 标签值（§2.2：num 表 INSTANT/CUMUL；state 表 STATE/SWITCH）。
+const (
+	KindInstant = "INSTANT"
+	KindCumul   = "CUMUL"
+	KindState   = "STATE"
+	KindSwitch  = "SWITCH"
+)
+
+// KindFor 数据种类（快照 dataType）→ kind 标签值：
+// 1 瞬时量 INSTANT、2 累计量 CUMUL、3 状态量 STATE、4 开关量 SWITCH。
+func KindFor(dataType int) (string, bool) {
+	switch dataType {
+	case 1:
+		return KindInstant, true
+	case 2:
+		return KindCumul, true
+	case 3:
+		return KindState, true
+	case 4:
+		return KindSwitch, true
+	default:
+		return "", false
+	}
+}

+// StableFor 数据种类（快照 dataType）→ 接入服务目标超级表：
+// 1,2 → st_point_num；3,4 → st_point_state（§2.2 按 data_type 分发）。
+func StableFor(dataType int) (string, bool) {
+	switch dataType {
+	case 1, 2:
+		return StablePointNum, true
+	case 3, 4:
+		return StablePointState, true
+	default:
+		return "", false
+	}
+}

+// DBName 每租户一库（附录 B）：iot_<tenant_code>。
+func DBName(tenantCode string) string {
+	return "iot_" + tenantCode
+}

+// PointRow 点位写入行（st_point_num / st_point_state）。
+// 标签值由流水线从配置快照填充（标签纪律：标签只放低基数维度，取自快照不取自消息）。
+type PointRow struct {
+	TenantCode  string  // 选库键（写方从令牌解析租户选库；iot_<tenant_code>）
+	TenantID    int64   // 标签 tenant_id
+	PointCode   string  // 标签 point_code，兼作子表名（实现口径见包注释）
+	Kind        string  // 标签 kind：INSTANT/CUMUL/STATE/SWITCH
+	InstanceKey string  // 标签 instance_key（快照 v1.0.1 无对应字段，暂空，见包注释）
+	DeptID      int64   // 标签 dept_id
+	Ts          int64   // 毫秒时间戳（设备侧采样时间，迟到数据按 ts 落库）
+	Value       float64 // num 表 DOUBLE；state 表 INT 枚举（按 INT 数值写入）
+	Quality     int     // 质量码（CONV-02 0-10）
+}

+// CreateDatabaseSQL 附录 B 建库语句（按租户参数化）。
+func CreateDatabaseSQL(tenantCode string) string {
+	return fmt.Sprintf("CREATE DATABASE IF NOT EXISTS %s KEEP 1825d DURATION 10d BUFFER 256;", DBName(tenantCode))
+}

+// UseSQL 附录 B 的 USE 语句（按租户参数化）。
+func UseSQL(tenantCode string) string {
+	return "USE " + DBName(tenantCode) + ";"
+}

+// CreateStableSQL 附录 B 四张超级表 DDL（语句逐字，仅随 USE 语境落库；顺序与附录 B 一致）。
+func CreateStableSQL(_ string) []string {
+	return []string{
+		"CREATE STABLE IF NOT EXISTS " + StablePointNum + " (ts TIMESTAMP, value DOUBLE, quality INT)\n" +
+			"  TAGS (tenant_id BIGINT, point_code BINARY(96), kind BINARY(16), instance_key BINARY(96), dept_id BIGINT);",
+		"CREATE STABLE IF NOT EXISTS " + StablePointState + " (ts TIMESTAMP, value INT, quality INT)\n" +
+			"  TAGS (tenant_id BIGINT, point_code BINARY(96), kind BINARY(16), instance_key BINARY(96), dept_id BIGINT);",
+		"CREATE STABLE IF NOT EXISTS " + StableItemValue + " (ts TIMESTAMP, value DOUBLE, quality INT)\n" +
+			"  TAGS (tenant_id BIGINT, full_key BINARY(180), instance_code BINARY(96), data_code BINARY(64), category BINARY(32), source_type INT);",
+		"CREATE STABLE IF NOT EXISTS " + StableMetricValue + " (ts TIMESTAMP, value DOUBLE, yoy DOUBLE, mom DOUBLE)\n" +
+			"  TAGS (tenant_id BIGINT, metric_code BINARY(96), instance_code BINARY(96), period BINARY(8));",
+	}

+// safeIdentifier 报告标识符（库名/子表名）是否安全：字母或下划线开头，仅字母数字下划线。
+func safeIdentifier(s string) bool {
+	if s == "" {
+		return false
+	}
+	for i, r := range s {
+		switch {
+		case r >= 'a' && r <= 'z', r >= 'A' && r <= 'Z', r == '_':
+			// 首字符与后续字符均合法
+		case r >= '0' && r <= '9':
+			if i == 0 {
+				return false // 标识符不能以数字开头
+			}
+		default:
+			return false
+		}
+	}
+	return true
+}
+
+// childTable 子表名（实现口径：点位编码即子表名，租户内每点位一张）。
+func childTable(row PointRow) (string, error) {
+	if !safeIdentifier(row.PointCode) {
+		return "", fmt.Errorf("tdengine: pointCode %q 不是安全子表名", row.PointCode)
+	}
+	return row.PointCode, nil
+}
+
+// InsertPointSQL 构造写入语句：同一子表多行合并单条 INSERT（USING 超级表 TAGS (...)）。
+// rows 必须同租户、同点位（Writer 已分组）；否则返回错误，绝不混写。
+func InsertPointSQL(rows []PointRow) (string, error) {
+	if len(rows) == 0 {
+		return "", fmt.Errorf("tdengine: 空 rows")
+	}
+	first := rows[0]
+	db := DBName(first.TenantCode)
+	if !safeIdentifier(first.TenantCode) {
+		return "", fmt.Errorf("tdengine: tenantCode %q 不是安全库名成分", first.TenantCode)
+	}
+	child, err := childTable(first)
+	if err != nil {
+		return "", err
+	}
+	stable := stableForKind(first.Kind)
+	if stable == "" {
+		return "", fmt.Errorf("tdengine: kind %q 不在 INSTANT/CUMUL/STATE/SWITCH 内", first.Kind)
+	}
+	var sb strings.Builder
+	fmt.Fprintf(&sb, "INSERT INTO %s.%s USING %s.%s TAGS (%d, '%s', '%s', '%s', %d) VALUES ",
+		db, child, db, stable,
+		first.TenantID, first.PointCode, first.Kind, first.InstanceKey, first.DeptID)
+	for i := range rows {
+		r := rows[i]
+		if r.TenantCode != first.TenantCode || r.PointCode != first.PointCode || r.Kind != first.Kind {
+			return "", fmt.Errorf("tdengine: rows 混入不同租户/点位/kind（%s/%s/%s），必须先分组", r.TenantCode, r.PointCode, r.Kind)
+		}
+		fmt.Fprintf(&sb, "(%d, %s, %d)", r.Ts, formatValue(r.Value), r.Quality)
+	}
+	return sb.String(), nil
+}
+
+// stableForKind kind → 超级表（与 KindFor/StableFor 一致的内部映射）。
+func stableForKind(kind string) string {
+	switch kind {
+	case KindInstant, KindCumul:
+		return StablePointNum
+	case KindState, KindSwitch:
+		return StablePointState
+	default:
+		return ""
+	}
+}

+// formatValue 数值列格式化：DOUBLE/INT 统一最短浮点表示（1 → "1"，98231.4 → "98231.4"）。
+func formatValue(v float64) string {
+	return strconv.FormatFloat(v, 'g', -1, 64)
+}

+// Executor SQL 执行器抽象。真 TDengine 连接归 TASK-0029 接线；
+// 假实现 FakeExecutor 只记录调用（brief 决策 3），零改动替换。
+type Executor interface {
+	Exec(ctx context.Context, sql string) error
+}
+
+// ExecCall 一次执行的记录：何时（At）以什么参数（SQL）调了什么。
+type ExecCall struct {
+	At  time.Time
+	SQL string
+}
+
+// FakeExecutor 假 TDengine 执行器：记录调用、不真执行。
+// failRemaining>0 时前 failRemaining 次 Exec 返回注入错误（攒批失败重排测试用）。
+type FakeExecutor struct {
+	mu           sync.Mutex
+	calls        []ExecCall
+	failRemaining int
+}
+
+// NewFakeExecutor 创建假执行器；failTimes 为连续注入失败的次数（0 = 永不失败）。
+func NewFakeExecutor(failTimes int) *FakeExecutor {
+	return &FakeExecutor{failRemaining: failTimes}
+}
+
+// Reset 重置失败注入额度与调用记录（测试编排用）。
+func (f *FakeExecutor) Reset(failTimes int) {
+	f.mu.Lock()
+	defer f.mu.Unlock()
+	f.calls = nil
+	f.failRemaining = failTimes
+}
+
+// Exec 记录并按注入策略返回。
+func (f *FakeExecutor) Exec(_ context.Context, sql string) error {
+	f.mu.Lock()
+	defer f.mu.Unlock()
+	f.calls = append(f.calls, ExecCall{At: time.Now(), SQL: sql})
+	if f.failRemaining > 0 {
+		f.failRemaining--
+		return fmt.Errorf("tdengine: 注入执行失败（剩余额度外调用，sql=%s）", truncateSQL(sql))
+	}
+	return nil
+}
+
+// Calls 返回已记录调用的副本。
+func (f *FakeExecutor) Calls() []ExecCall {
+	f.mu.Lock()
+	defer f.mu.Unlock()
+	out := make([]ExecCall, len(f.calls))
+	copy(out, f.calls)
+	return out
+}
+
+// Writer 攒批写出口：把写入行按（租户，子表）分组，逐组下发单条 INSERT。
+// 分组下发保证「攒批写入、禁止逐条写库」；任一组失败即返回错误（重排归流水线攒批层）。
+type Writer struct {
+	// Ex SQL 执行器（假实现或真 TDengine 连接）。
+	Ex Executor
+}
+
+type groupKey struct {
+	tenant string
+	child  string
+	kind   string
+}
+
+// WritePoints 写入一批点位行。按（租户，子表，kind）分组，组间顺序稳定（字典序）。
+// 空 rows 直接返回 nil。
+func (w *Writer) WritePoints(ctx context.Context, rows []PointRow) error {
+	if len(rows) == 0 || w == nil || w.Ex == nil {
+		if w == nil || w.Ex == nil {
+			return fmt.Errorf("tdengine: Writer 未配置执行器")
+		}
+		return nil
+	}
+	groups := make(map[groupKey][]PointRow)
+	order := make([]groupKey, 0, len(rows))
+	for _, r := range rows {
+		child, err := childTable(r)
+		if err != nil {
+			return err
+		}
+		k := groupKey{tenant: r.TenantCode, child: child, kind: r.Kind}
+		if _, seen := groups[k]; !seen {
+			order = append(order, k)
+		}
+		groups[k] = append(groups[k], r)
+	}
+	// 稳定序：按租户、子表排序，保证失败重排与重试的语句顺序可预期。
+	sort.Slice(order, func(i, j int) bool {
+		if order[i].tenant != order[j].tenant {
+			return order[i].tenant < order[j].tenant
+		}
+		return order[i].child < order[j].child
+	})
+	for _, k := range order {
+		sql, err := InsertPointSQL(groups[k])
+		if err != nil {
+			return err
+		}
+		if err := w.Ex.Exec(ctx, sql); err != nil {
+			return err
+		}
+	}
+	return nil
+}
+
+// truncateSQL 截断 SQL 用于错误信息。
+func truncateSQL(sql string) string {
+	if len(sql) <= 120 {
+		return sql
+	}
+	return sql[:120] + "…"
+}

+	ex := tdengine.NewFakeExecutor(1) // 首次调用失败，其后成功
+	if err := ex.Exec(context.Background(), "INSERT INTO t VALUES (1)"); err == nil {
+		t.Error("首次 Exec error = nil, want 注入失败")
+	}
+	if err := ex.Exec(context.Background(), "INSERT INTO t VALUES (2)"); err != nil {
+		t.Errorf("第二次 Exec error = %v, want nil（失败额度耗尽后成功）", err)
+	}
+	calls := ex.Calls()
+	if len(calls) != 2 {
+		t.Fatalf("记录条数 = %d, want 2", len(calls))
+	}
+	if !strings.Contains(calls[0].SQL, "INSERT INTO t VALUES (1)") {
+		t.Errorf("call[0].SQL = %q", calls[0].SQL)
+	}
+}
