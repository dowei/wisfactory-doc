+            return 0; // 非法主题：静默忽略

+            return 0;
+        } catch (IllegalArgumentException e) {
+            log.warn("[OPENAPI-SUB] 指标事件载荷非法丢弃 topic={}: {}", topic, e.getMessage());
+            return 0;
+        }
+    }

+import java.util.Deque;

+/**
+ * 订阅事件环形缓冲 + 幂等登记（TASK-0053，brief 决策 3）。
+ *
+ * <p>幂等口径严格按「指标全键 + 窗口」（{@link MetricProducedEventVO#idempotencyKey()}），
+ * <b>不用 hash</b>——hash 含结果值，0046 修正重算后变化，作幂等键会把重发当成新事件。
+ * 同键重发以最新载荷覆盖（值/伴随值反映修正结果），不重复占用缓冲位。
+ *
+ * <p>事件仅作触发信号：本缓冲不写任何业务表，指标结果以 0045 st_metric_value 物化表为准
+ * （ADR-0010，不重复写业务表）。有界 LinkedHashMap 维持最近容量条（按最近一次接受排序），
+ * 读口取最新/遍历全序列。无订阅者不报错（accept 恒真，调用方决定是否投递）。
+ *
+ * @author dowei
+ * @date 2026-11-24
+ */
+public class SubscriptionEventStore {
+
+    private final int capacity;
+    private final Map<String, MetricProducedEventVO> metrics = new LinkedHashMap<>();
+    private final Map<String, ItemChangeEventVO> items = new LinkedHashMap<>();
+
+    public SubscriptionEventStore(int capacity) {
+        this.capacity = capacity <= 0 ? 1 : capacity;
+    }
+
+    /** 接受指标事件：同全键+窗口覆盖（重发不新增），恒返回 true（无订阅者亦不报错） */
+    public synchronized boolean acceptMetric(MetricProducedEventVO event) {
+        if (event == null) {
+            return false;
+        }
+        upsert(metrics, event.idempotencyKey(), event);
+        return true;
+    }
+
+    /** 接受数据项事件：同 fullKey+ts+quality 覆盖，恒返回 true */
+    public synchronized boolean acceptItem(ItemChangeEventVO event) {
+        if (event == null) {
+            return false;
+        }
+        String key = event.getFullKey() + "|" + event.getTs() + "|" + event.getQuality();
+        upsert(items, key, event);
+        return true;
+    }
+
+    private <T> void upsert(Map<String, T> map, String key, T value) {
+        map.remove(key);
+        map.put(key, value); // 最近接受排到末尾；同键重发只占一个位
+        while (map.size() > capacity) {
+            String oldest = map.keySet().iterator().next();
+            map.remove(oldest);
+        }
+    }
+
+    public synchronized int metricCount() {
+        return metrics.size();
+    }
+
+    public synchronized int itemCount() {
+        return items.size();
+    }
+
+    public synchronized MetricProducedEventVO latestMetric() {
+        return last(metrics);
+    }
+
+    public synchronized ItemChangeEventVO latestItem() {
+        return last(items);
+    }
+
+    private <T> T last(Map<String, T> map) {
+        if (map.isEmpty()) {
+            return null;
+        }
+        java.util.Iterator<T> it = map.values().iterator();
+        T result = null;
+        while (it.hasNext()) {
+            result = it.next();
+        }
+        return result;
+    }
+
+    /** 指标环形缓冲快照（最旧在前、最新在后，逐条只读） */
+    public synchronized Deque<MetricProducedEventVO> metricRing() {
+        return new ArrayDeque<>(metrics.values());
+    }
+
+    /** 数据项环形缓冲快照 */
+    public synchronized Deque<ItemChangeEventVO> itemRing() {
+        return new ArrayDeque<>(items.values());
+    }
