+    public static List<MetricDef> affectedInTopoOrder(net.otlab.calc.graph.DependencyGraph graph,
+                                                      String changedMetric, List<MetricDef> defs) {
+        Map<String, MetricDef> byCode = new LinkedHashMap<>();
+        for (MetricDef def : defs) {
+            byCode.put(def.metricCode(), def);
+        }
+        // 反依赖：to 的下级变更 → from 受影响
+        Map<String, Set<String>> dependents = new LinkedHashMap<>();
+        for (GraphEdge edge : graph.getEdges()) {
+            dependents.computeIfAbsent(edge.to(), k -> new LinkedHashSet<>()).add(edge.from());
+        }
+        Set<String> affected = new LinkedHashSet<>();
+        java.util.ArrayDeque<String> queue = new java.util.ArrayDeque<>();
+        queue.add(changedMetric);
+        while (!queue.isEmpty()) {
+            String cur = queue.poll();
+            for (String dep : dependents.getOrDefault(cur, Set.of())) {
+                if (affected.add(dep)) {
+                    queue.add(dep);
+                }
+            }
+        }
+        List<MetricDef> ordered = new ArrayList<>();
+        for (String key : graph.getTopoOrder()) {
+            if (affected.contains(key) && byCode.containsKey(key)) {
+                ordered.add(byCode.get(key));
+            }
+        }
+        return ordered;
+    }

+    @Test
+    void cascadePropagatesCorrectedChildValueThroughTopologyMultiLevel() {
+        MetricValueRecorder recorder = new MetricValueRecorder("t46");
+        // 三级链：M_LOW 读原始 I.A；M_HIGH 读 M_LOW 物化结果；M_TOP 读 M_HIGH 物化结果
+        MetricRuntime rt = runtime(recorder, List.of(
+                new MetricDef("M_LOW", "DAY", 2, List.of("I.A"), 1),
+                new MetricDef("M_HIGH", "DAY", 2, List.of("M_LOW"), 4),
+                new MetricDef("M_TOP", "DAY", 2, List.of("M_HIGH"), 4)));
+
+        long now = Instant.parse("2026-11-23T06:00:00Z").toEpochMilli();

+        long dataTs = winStart + 3600_000L;
+
+        // 初始产出：按拓扑序（下级先）逐个 onDue
+        rt.source().put("I.A", 5.0, dataTs);
+        for (MetricDef def : rt.defsForTest()) {
+            rt.onDue(rt.previousWindow(def, now));
+        }
+        assertEquals(5.0, lastValueOf(recorder, "M_LOW").value(), 1e-9, "下级初始=5");
+        assertEquals(5.0, lastValueOf(recorder, "M_HIGH").value(), 1e-9,
+                "上级必须从下级物化结果读值=5（旧实现从 values 表读不到 M_LOW → null）");

+                "F-01 核心：上级反映下级修正后的物化新值=7（非旧值5、非null）");
+        assertEquals(7.0, lastValueOf(recorder, "M_TOP").value(), 1e-9,
