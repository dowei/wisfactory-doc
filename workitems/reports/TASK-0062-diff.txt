+/**
+ * TASK-0062 D3：告警状态值域与流转守卫（0 待确认 → 1 已确认 → 2 已处理；允许 0→2 直接处理）。
+ *
+ * <p>非法跃迁（如 2→0）由服务层显式拒绝（409），不静默改、不把已确认告警打回待确认。

+ * @author dowei

+
+    public static final int PENDING = 0;
+    public static final int CONFIRMED = 1;
+    public static final int HANDLED = 2;
+
+    private AlarmStatus() {}
+
+    public static String text(Integer status) {
+        if (status == null) {
+            return "未知";
+        }
+        return switch (status) {
+            case PENDING -> "待确认";
+            case CONFIRMED -> "已确认";
+            case HANDLED -> "已处理";
+            default -> "未知";
+        };
+    }
+
+    public static boolean canConfirm(Integer status) {
+        return status != null && status == PENDING;
+    }
+
+    public static boolean canHandle(Integer status) {
+        return status != null && (status == PENDING || status == CONFIRMED);
+    }

+/*
+ * TASK-0062：系统事件告警管理面端点封装（已登录态 axios）。
+ *
+ * 仅调 D3 管理面端点：
+ * - GET  /api/v1/iot/alarm/list?status=&type=&from=&to=&pageNum=&pageSize=
+ * - POST /api/v1/iot/alarm/{id}/confirm
+ * - POST /api/v1/iot/alarm/{id}/handle?note=
+ *
+ * 安全红线：本页**不持有 appKey/secret、不直连 /openapi/v1/** HMAC、不自造代理**；
+ * tenant 由后端会话强制注入，前端不传也不接受 tenant 参数（跨租户明细 403）。
+ *
+ * @Author: dowei
+ * @Date: 2026-11-24
+ */
+import { getRequest, postParamRequest, postRequest } from '/@/lib/axios';
+import type { AlarmRowVO } from '/@/views/monitor/alarm/helper';
+
+/** 后端 Result<T> 包络（code!==200 已被 axios 拦截器 reject，这里只取 data） */
+interface ResultEnvelope<T> {
+  code: number;
+  message?: string;
+  data: T;
+}
+
+/** 后端 AlarmListVO */
+export interface AlarmListVO {
+  rows: AlarmRowVO[];
+  total: number;
+}
+
+export interface AlarmListQuery {
+  status?: number;
+  type?: string;
+  from?: number;
+  to?: number;
+  pageNum?: number;
+  pageSize?: number;
+}
+
+const asArray = <T>(value: T[] | null | undefined): T[] => (Array.isArray(value) ? value : []);
+
+export const alarmApi = {
+  /** 告警分页列表（跨租户 403 由后端拒绝/裁剪，前端不暴露越权入口） */
+  async list(query: AlarmListQuery): Promise<AlarmListVO> {
+    const res: ResultEnvelope<AlarmListVO> = await getRequest('/api/v1/iot/alarm/list', query);
+    const data = res?.data;
+    return { rows: asArray(data?.rows), total: typeof data?.total === 'number' ? data.total : 0 };
+  },
+
+  /** 确认告警（0→1；非法跃迁 409、跨租户 403） */
+  async confirm(id: number): Promise<number> {
+    const res: ResultEnvelope<number> = await postRequest(`/api/v1/iot/alarm/${id}/confirm`, undefined);
+    return res?.data ?? 0;
+  },
+
+  /** 处理告警（0/1→2；note 必填留痕；非法跃迁 409、跨租户 403） */
+  async handle(id: number, note: string): Promise<number> {
+    const res: ResultEnvelope<number> = await postParamRequest(`/api/v1/iot/alarm/${id}/handle`, { note });
+    return res?.data ?? 0;
+  },
+};

+  if (status === ALARM_STATUS.HANDLED) return 'green';

+export function formatAlarmTime(ms?: number | null): string {
+  if (ms == null || !Number.isFinite(ms)) return '-';
