+/**
+ * TASK-0063 修订轮 I-1：iot_calc DDL 脚本「每条语句独占一行」守卫（对旧脚本必红）。
+ *
+ * <p>根因：{@code backend/scripts/sql-exec.sh} 以 {@code ;\s*\n}（分号 + 换行）切分语句——
+ * 同行中出现多个 {@code ;}（如 {@code PREPARE ...; EXECUTE ...; DEALLOCATE ...;} 写在一行）
+ * 会被当作一条语句交给 JDBC 执行，MySQL 报 {@code SQLERROR near 'PREPARE ...'}，建号段落中止、
+ * 留下无账号半套 schema（评审 F-1，只读探针实测）。本测试机械锁定「按 sql-exec.sh 切分后，
+ * 每段至多一条可执行语句」——覆盖脚本内所有多行语句，不只建号处。只文本分析，不连库。
+ *
+ * @author dowei
+ * @date 2026-11-24
+ */
+class IotCalcDdlSplitContractTest {
+
+    private static final String REL = "resources/sql/upgrade/20261124_iot_calc_schema.sql";
+
+    private static Path repoRoot() {
+        Path dir = Paths.get("").toAbsolutePath();
+        while (dir != null && !Files.exists(dir.resolve("resources/sql/01_base.sql"))) {
+            dir = dir.getParent();
+        }
+        assertTrue(dir != null, "未定位仓库根（resources/sql/01_base.sql）");
+        return dir;
+    }
+
+    private static List<String> executableSegments(String ddl) {
+        // 复刻 sql-exec.sh 的切分：split(";\\s*\\n")，逐行剥 -- 注释与空行
+        List<String> segments = new ArrayList<>();
+        for (String seg : ddl.split(";\\s*\\n")) {
+            StringBuilder sb = new StringBuilder();
+            for (String line : seg.split("\n")) {
+                String t = line.strip();
+                if (!t.isEmpty() && !t.startsWith("--")) {
+                    sb.append(line).append("\n");
+                }
+            }
+            String s = sb.toString().strip();
+            if (!s.isEmpty()) {
+                segments.add(s);
+            }
+        }
+        return segments;
+    }
+
+    @Test
+    @DisplayName("DDL 每条语句独占一行：按 sql-exec.sh 的 ;\\n 切分后，任一段不含行内第二处分号")
+    void everyStatementIsOnItsOwnLine() throws Exception {
+        Path p = repoRoot().resolve(REL);
+        assertTrue(Files.exists(p), "缺 DDL 脚本: " + REL);
+        String ddl = Files.readString(p, StandardCharsets.UTF_8);
+        List<String> segments = executableSegments(ddl);
+        assertTrue(segments.size() >= 8, "切分段数异常偏少（" + segments.size() + "），解析口径不可信");
+
+        List<String> bad = new ArrayList<>();
+        for (String seg : segments) {
+            // 段内若仍含分号（; 已被切分符消费，残余分号只能是同行多语句），即为 F-1 形态。
+            // 注意：字符串字面量里的分号（如 GRANTEE 常量）不会出现在本脚本——脚本内所有语句均无。
+            if (seg.contains(";")) {
+                bad.add(seg.replace("\n", " ⏎ ").strip());
+            }
+        }
+        assertEquals(List.of(), bad,
+                "以下切分段仍含同行分号（sql-exec.sh 会作为一条语句交给 JDBC，导致 SQLERROR）: " + bad);
+    }
+
+    @Test
+    @DisplayName("DDL 建号段落形态：PREPARE / EXECUTE / DEALLOCATE 各自独占一行（F-1 锚点）")
+    void accountStatementsEachOnOwnLine() throws Exception {
+        String ddl = Files.readString(repoRoot().resolve(REL), StandardCharsets.UTF_8);
+        for (String kw : new String[]{"PREPARE rw_stmt FROM", "EXECUTE rw_stmt", "DEALLOCATE PREPARE rw_stmt",
+                "PREPARE ro_stmt FROM", "EXECUTE ro_stmt", "DEALLOCATE PREPARE ro_stmt"}) {
+            long hit = java.util.Arrays.stream(ddl.split("\\R"))
+                    .filter(line -> line.strip().startsWith(kw)).count();
+            assertEquals(1, hit, "建号语句须独占一行（且仅一行）: " + kw);
+        }
+    }
+}
