+ * 读竞争在 x86 上不保证稳定复现异常，故本测试双管齐下：

+ *   <li><b>真实并发压力（行为面）</b>：多写线程 put + 多读线程 get，断言无异常且最终值可见。</li>

+    @Test
+    void threadSafetyContractIsBackedByImplementation() throws Exception {
+        boolean getSynchronized = Modifier.isSynchronized(
+                ValueStore.class.getMethod("get", String.class).getModifiers());
+        boolean concurrentMap = false;
+        Field field = ValueStore.class.getDeclaredField("values");
+        field.setAccessible(true);
+        Object map = field.get(new ValueStore());
+        if (map instanceof ConcurrentMap) {

+        }
+        assertTrue(getSynchronized || concurrentMap,

+    void concurrentPutGetHasNoExceptionAndValuesVisible() throws Exception {
+        ValueStore store = new ValueStore();
+        int writers = 4;
+        int readers = 4;
+        int perWriter = 500;
+        AtomicReference<Throwable> error = new AtomicReference<>();
+        CountDownLatch start = new CountDownLatch(1);
+        List<Thread> threads = new ArrayList<>();
+

+            final int id = w;

+                try {

+                    }

+                }

+        }
