+// Package itembridge 承载 TASK-0084 桥接生产者：订阅 Redis 新值事件（iot:newvalue:{tenant}），
+// 按 pointCode 映射 fullKey/instanceCode，发布到 MQTT 数据项订阅主题
+// iot/open/{tenant_code}/{instance_code}/item（统一消息模型 4 字段 fullKey/ts/value/quality）。
+//
+// 契约纪律（brief 决策 1-6，冻结契约 v1.1.0 §3.5 推送行）：
+//   - payload 恰 4 字段、camelCase、ts 毫秒（0053 定版）——tenant 只走主题段，不注入 payload；
+//   - pointCode 即 fullKey 核心（「现场只认 pointCode 不认 id」字段纪律不改）：fullKey = pointCode
+//     原样；instanceCode 取 pointCode 去掉末段数据码后的设备段（与 snapshot.DataItem.FullKey
+//     的 instance_code + "." + data_code 定义互逆；F01.WS01.EQ001.M01 → F01.WS01.EQ001）；
+//   - 映射失败（无分隔符 / 空段 / channel 租户不可解析 / payload 非法 JSON / payload tenant
+//     与主题 tenant 不一致）→ WARN + 丢弃，不造假主题、不发空字段；
+//   - opt-in：INGEST_ITEM_BRIDGE_ENABLED 默认 false；开启后独立 goroutine，ctx 取消退出；
+//   - MQTT QoS1 fire-and-forget（无订阅者不报错）；桥接失败只 WARN，不影响 ingest 主流水线。

+// NewValueSource Redis 新值事件订阅源薄接口（生产=RedisNewValueSource；测试=内存假源）。
+// Run 阻塞消费直至 ctx 取消；每条事件回调 (channel, payload)。重连/退避由源自身负责。
+type NewValueSource interface {
+	Run(ctx context.Context, onEvent func(channel, payload string)) error

+// ItemPublisher MQTT 数据项主题发布薄接口（生产=PahoItemPublisher；测试=记录式假发布器）。
+type ItemPublisher interface {
+	Publish(ctx context.Context, topic string, payload []byte) error

+// ItemChangeEvent 数据项变更事件载荷（契约 §3.5 推送行 4 字段逐字，字段序即线上序）。
+type ItemChangeEvent struct {

+	Ts      int64   `json:"ts"`
+	Value   float64 `json:"value"`
+	Quality int     `json:"quality"`

+// MarshalItemChange 有序序列化为契约 4 字段 JSON（camelCase / ts 毫秒原样 / quality 透传）。
+func MarshalItemChange(ev ItemChangeEvent) ([]byte, error) {
+	return json.Marshal(ev)

+// MapPointCode pointCode → (fullKey, instanceCode)。fullKey 恒等于 pointCode 原样（字段纪律：
+// pointCode 即 fullKey 核心）；instanceCode = 去掉最后一段数据码后的前缀（设备段）。
+// 契约合法域要求至少两段且每段非空；否则映射失败（丢弃+WARN，不造假主题）。
+func MapPointCode(pointCode string) (string, string, error) {
+	idx := strings.LastIndex(pointCode, ".")
+	if idx <= 0 || idx == len(pointCode)-1 {
+		return "", "", fmt.Errorf("itembridge: pointCode %q 无法解析 instanceCode（须含非空实例段与数据码段）", pointCode)
+	}
+	instance := pointCode[:idx]
+	// 中间段不得为空（F01..M01 之类），避免造出含空段的实例路径。
+	for _, seg := range strings.Split(instance, ".") {
+		if seg == "" {
+			return "", "", fmt.Errorf("itembridge: pointCode %q 实例路径含空段", pointCode)
+		}
+	}
+	return pointCode, instance, nil
+}

+// NewValuePattern Redis PSUBSCRIBE 模式（brief 逐字频道前缀 + 通配租户段）。
+func NewValuePattern() string { return "iot:newvalue:*" }

+const newValuePrefix = "iot:newvalue:"

+// TenantFromNewValueChannel 从频道名取 tenant（仅 iot:newvalue:{tenant} 形态且 tenant 非空合法）。
+func TenantFromNewValueChannel(channel string) (string, bool) {
+	if !strings.HasPrefix(channel, newValuePrefix) {
+		return "", false
+	}
+	tenant := strings.TrimPrefix(channel, newValuePrefix)
+	if tenant == "" || strings.Contains(tenant, ":") || strings.ContainsAny(tenant, " \t\r\n") {
+		return "", false
+	}
+	return tenant, true
+}

+func ItemTopic(tenant, instance string) string {

+// BridgeStats 桥接观测计数（原子；生产扇出只 WARN 不影响主流水线）。

+// Bridge 桥接主体（单 goroutine 事件循环；无内部缓冲——事件到即映射即发）。
+type Bridge struct {

+	logger *log.Logger

+func NewBridge(src NewValueSource, pub ItemPublisher, logger *log.Logger) *Bridge {

+// Stats 观测快照（原子加载）。
+func (b *Bridge) Stats() BridgeStats {
+	return BridgeStats{

+		Dropped:       b.dropped.Load(),
+		PublishFailed: b.publishFailed.Load(),

+// Run 启动桥接事件循环（阻塞直至 ctx 取消）；源退出后按指数退避重连（fail-soft，0027 范式）。
+func (b *Bridge) Run(ctx context.Context) error {
+	backoff := 200 * time.Millisecond
+	const maxBackoff = 5 * time.Second
+	for {
+		if ctx.Err() != nil {
+			return ctx.Err()
+		}
+		err := b.src.Run(ctx, b.onEvent)
+		if ctx.Err() != nil {
+			return ctx.Err()
+		}
+		b.logger.Printf("itembridge: 订阅源退出（%v），%v 后重连", err, backoff)
+		timer := time.NewTimer(backoff)
+		select {
+		case <-ctx.Done():
+			if !timer.Stop() {
+				<-timer.C
+			}
+			return ctx.Err()
+		case <-timer.C:
+		}
+		if backoff < maxBackoff {
+			backoff *= 2
+			if backoff > maxBackoff {
+				backoff = maxBackoff
+			}
+		}
+	}
+}

+		Quality   int     `json:"quality"`
+	}

+		return
+	}

+		b.dropped.Add(1)
+		return
+	}
+	fullKey, instance, err := MapPointCode(src.PointCode)

+		b.logger.Printf("itembridge: WARN pointCode 映射失败，丢弃 channel=%q pointCode=%q err=%v", channel, src.PointCode, err)
+		b.dropped.Add(1)
+		return
+	}
+	body, err := MarshalItemChange(ItemChangeEvent{
+		FullKey: fullKey, Ts: src.Ts, Value: src.Value, Quality: src.Quality,
+	})
+	if err != nil {
+		b.logger.Printf("itembridge: WARN 数据项事件编码失败（fail-soft）channel=%q err=%v", channel, err)
+		b.dropped.Add(1)
+		return
+	}
+	topic := ItemTopic(tenant, instance)
+	pubCtx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
+	err = b.pub.Publish(pubCtx, topic, body)
+	cancel()
+	if err != nil {
+		// QoS1 fire-and-forget：无订阅者/瞬时故障不报错、不影响后续事件（fail-soft）。
+		b.logger.Printf("itembridge: WARN MQTT 发布失败（fail-soft，不影响主链）topic=%q err=%v", topic, err)
+		b.publishFailed.Add(1)
+		return
+	}
+	b.published.Add(1)

+func (b *fixtureBroker) addr() string { return b.ln.Addr().String() }

+func (b *fixtureBroker) accept(t *testing.T) {
+	defer b.wg.Done()
+	for {
+		conn, err := b.ln.Accept()
+		if err != nil {
+			return
+		}
+		b.wg.Add(1)
+		go b.handle(conn)
+	}

+func (b *fixtureBroker) handle(conn net.Conn) {
+	defer b.wg.Done()
+	defer conn.Close()
+	br := bufio.NewReader(conn)
+	for {
+		b0, err := br.ReadByte()
+		if err != nil {
+			return
+		}
+		rem, err := readMqttLen(br)
+		if err != nil {
+			return
+		}
+		body := make([]byte, rem)
+		if _, err := io.ReadFull(br, body); err != nil {
+			return
+		}
+		pktType := b0 >> 4
+		switch pktType {
+		case 1: // CONNECT → CONNACK
+			if _, err := conn.Write([]byte{0x20, 0x02, 0x00, 0x00}); err != nil {
+				return
+			}
+		case 8: // SUBSCRIBE → 记录 filter + SUBACK
+			if len(body) < 2 {
+				return
+			}
+			pid := body[:2]
+			rest := body[2:]
+			filters := []string{}
+			for len(rest) >= 3 {
+				n := int(binary.BigEndian.Uint16(rest[:2]))
+				if len(rest) < 2+n+1 {
+					break
+				}
+				filters = append(filters, string(rest[2:2+n]))
+				rest = rest[2+n+1:]
+			}
+			b.mu.Lock()
+			b.subs[conn] = filters
+			b.mu.Unlock()
+			ack := []byte{0x90, byte(2 + len(filters)), pid[0], pid[1]}
+			for range filters {
+				ack = append(ack, 0x01) // granted QoS1
+			}
+			if _, err := conn.Write(ack); err != nil {
+				return
+			}
+		case 3: // PUBLISH → PUBACK(QoS1) + 转发匹配订阅者
+			if len(body) < 2 {
+				return
+			}
+			tl := int(binary.BigEndian.Uint16(body[:2]))
+			if len(body) < 2+tl {
+				return
+			}
+			topic := string(body[2 : 2+tl])
+			idx := 2 + tl
+			var pid []byte
+			var payload []byte
+			if b0&0x06 == 0x02 { // QoS1（flags 位 1-2 值 1；b0&0x06>>1 == 1）
+				if len(body) < idx+2 {
+					return
+				}
+				pid = body[idx : idx+2]
+				idx += 2
+				_, _ = conn.Write([]byte{0x40, 0x02, pid[0], pid[1]}) // PUBACK
+			}
+			payload = body[idx:]
+			b.fanout(topic, payload)
+		case 12: // PINGREQ → PINGRESP
+			_, _ = conn.Write([]byte{0xD0, 0x00})
+		case 14: // DISCONNECT
+			return
+		}
+	}
+}

+func (b *fixtureBroker) fanout(topic string, payload []byte) {
+	b.mu.Lock()
+	type target struct{ c net.Conn }
+	targets := []target{}
+	for c, filters := range b.subs {
+		for _, f := range filters {
+			if mqttMatch(f, topic) {
+				targets = append(targets, target{c})
+				break
+			}
+		}
+	}
+	b.mu.Unlock()
+	// 构造 QoS0 PUBLISH 给订阅者（简化：无 packet-id）。
+	var body []byte
+	body = append(body, 0x00, byte(len(topic)))
+	body = append(body, topic...)
+	body = append(body, payload...)

+func mqttMatch(filter, topic string) bool {
+	fs, ts := strings.Split(filter, "/"), strings.Split(topic, "/")
+	for i := range fs {
+		if i >= len(ts) {
+			return false
+		}
+		if fs[i] == "#" {
+			return true
+		}
+		if fs[i] != "+" && fs[i] != ts[i] {
+			return false
+		}
+	}
+	return len(fs) == len(ts)
+}

+func readMqttLen(br *bufio.Reader) (int, error) {
+	mult := 1
+	val := 0
+	for {
+		bb, err := br.ReadByte()
+		if err != nil {
+			return 0, err
+		}
+		val += int(bb&0x7f) * mult
+		if bb&0x80 == 0 {
+			return val, nil
+		}
+		mult *= 128
+	}
+}
+
+func mqttLen(n int) []byte {
+	var out []byte
+	for {
+		bb := byte(n % 128)
+		n /= 128
+		if n > 0 {
+			bb |= 0x80
+		}
+		out = append(out, bb)
+		if n == 0 {
+			return out
+		}
+	}
+}

+func dialSubscriber(t *testing.T, broker string, filter string) *rawSubscriber {
+	conn, err := net.DialTimeout("tcp", broker, 3*time.Second)
+	if err != nil {
+		t.Fatal(err)
+	}
+	br := bufio.NewReader(conn)
+	// CONNECT（clean session, keepalive 60, clientId=sub-live）
+	cid := "sub-live"
+	vh := append([]byte{0x00, 0x04}, []byte("MQTT")...)
+	vh = append(vh, 0x04, 0x02, 0x00, 0x3c)
+	vh = append(vh, 0x00, byte(len(cid)))
+	vh = append(vh, []byte(cid)...)
+	body := vh
+	pkt := append([]byte{0x10}, mqttLen(len(body))...)
+	pkt = append(pkt, body...)
+	_, _ = conn.Write(pkt)
+	if b0, err := br.ReadByte(); err != nil || b0 != 0x20 {
+		t.Fatalf("CONNACK 失败 b0=%v err=%v", b0, err)
+	}
+	if _, err := readMqttLen(br); err != nil {
+		t.Fatal(err)
+	}
+	if _, err := br.ReadByte(); err != nil { // 会话存在标志/返回码（2 字节）
+		t.Fatal(err)
+	}
+	if _, err := br.ReadByte(); err != nil {
+		t.Fatal(err)
+	}
+	// SUBSCRIBE（packetId=1）
+	sb := []byte{0x00, 0x01, 0x00, byte(len(filter))}
+	sb = append(sb, []byte(filter)...)
+	sb = append(sb, 0x01)
+	pkt = append([]byte{0x82}, mqttLen(len(sb))...)
+	pkt = append(pkt, sb...)
+	_, _ = conn.Write(pkt)
+	b0, err := br.ReadByte()
+	if err != nil || b0 != 0x90 {
+		t.Fatalf("SUBACK 失败 b0=%v err=%v", b0, err)
+	}
+	rl, err := readMqttLen(br)
+	if err != nil {
+		t.Fatal(err)
+	}
+	ack := make([]byte, rl)
+	if _, err := io.ReadFull(br, ack); err != nil {
+		t.Fatal(err)
+	}
+	return &rawSubscriber{conn: conn, br: br}
+}
+
+// readMsg 读一条 PUBLISH（QoS0 简化），返回 topic + payload；超时上返。
+func (s *rawSubscriber) readMsg(timeout time.Duration) (string, []byte, error) {
+	_ = s.conn.SetReadDeadline(time.Now().Add(timeout))
+	b0, err := s.br.ReadByte()
+	if err != nil {
+		return "", nil, err
+	}
+	if b0>>4 != 3 {
+		return "", nil, fmt.Errorf("非 PUBLISH 包 b0=%x", b0)
+	}
+	rem, err := readMqttLen(s.br)
+	if err != nil {
+		return "", nil, err
+	}
+	body := make([]byte, rem)
+	if _, err := io.ReadFull(s.br, body); err != nil {
+		return "", nil, err
+	}
+	tl := int(binary.BigEndian.Uint16(body[:2]))
+	return string(body[2 : 2+tl]), body[2+tl:], nil
+}

+// Connect 初连（失败快速上返，由调用方决定是否重试；生产装配启动 WARN 不阻断 ingest）。
+func (p *PahoItemPublisher) Connect() error {
+	timeout := p.opts.ConnectTimeout
+	if timeout <= 0 {
+		timeout = 5 * time.Second
+	}
+	cid := fmt.Sprintf("%s-%d-%d", p.opts.ClientIDPrefix, os.Getpid(), p.seq.Add(1))
+	opts := pahomqtt.NewClientOptions().
+		AddBroker(p.opts.BrokerURL).
+		SetClientID(cid).
+		SetUsername(p.opts.Username).
+		SetPassword(p.opts.Password).
+		SetCleanSession(true). // 发布端无需持久会话（消息本体 QoS1 由 broker 承载）
+		SetConnectRetry(false).
+		SetConnectTimeout(timeout)
+	client := pahomqtt.NewClient(opts)
+	tok := client.Connect()
+	if !tok.WaitTimeout(timeout + 2*time.Second) {
+		return fmt.Errorf("itembridge: MQTT 连接超时（%v）", timeout)
+	}
+	if err := tok.Error(); err != nil {
+		return fmt.Errorf("itembridge: MQTT 连接失败: %w", err)
+	}
+	p.client = client
+	return nil
+}

+// Publish QoS1 发布（ctx 取消/超时上返；无订阅者由 broker 正常回执，不视为错误）。
+func (p *PahoItemPublisher) Publish(ctx context.Context, topic string, payload []byte) error {
+	if p.client == nil {
+		return fmt.Errorf("itembridge: MQTT 发布器未连接")
+	}
+	tok := p.client.Publish(topic, 1, false, payload)
+	done := make(chan struct{})
+	go func() {
+		tok.Wait()
+		close(done)
+	}()
+	select {
+	case <-done:
+		return tok.Error()
+	case <-ctx.Done():
+		return ctx.Err()
+	}
+}

+	sourceDefaultCmdTimeout  = 3 * time.Second
+	sourceMaxLine            = 1 << 20 // 回复行上限（1 MiB）
+	sourceMaxBulk            = 1 << 20 // payload 上限（1 MiB）
+	sourceMaxArray           = 8       // push 数组元素上限（本源只消费 3/4 元素）

+// NewRedisNewValueSource 构造（惰性建连，Run 内循环）。
+func NewRedisNewValueSource(opts RedisSourceOptions) *RedisNewValueSource {

+// Run 阻塞订阅直至 ctx 取消；断线指数退避重连。
+func (s *RedisNewValueSource) Run(ctx context.Context, onEvent func(channel, payload string)) error {
+	backoff := 200 * time.Millisecond
+	for {
+		if ctx.Err() != nil {
+			return ctx.Err()
+		}
+		err := s.runOnce(ctx, onEvent)
+		if ctx.Err() != nil {
+			return ctx.Err()
+		}
+		timer := time.NewTimer(backoff)
+		select {
+		case <-ctx.Done():
+			if !timer.Stop() {
+				<-timer.C
+			}
+			return ctx.Err()
+		case <-timer.C:
+		}
+		_ = err // 退避重连（err 只用于内层日志；Break 由 ctx 决定）
+		if backoff < sourceMaxBackoff {
+			backoff *= 2
+			if backoff > sourceMaxBackoff {
+				backoff = sourceMaxBackoff
+			}
+		}
+	}
+}

+// runOnce 单次连接生命周期：拨号+握手+PSUBSCRIBE+逐条读 push。
+func (s *RedisNewValueSource) runOnce(ctx context.Context, onEvent func(channel, payload string)) error {
+	d := s.opts.DialTimeout
+	if dl, ok := ctx.Deadline(); ok {
+		if remain := time.Until(dl); remain < d {
+			d = remain
+		}
+	}
+	conn, err := net.DialTimeout("tcp", s.opts.Addr, d)
+	if err != nil {
+		return fmt.Errorf("itembridge: redis 连接失败 %s: %w", s.opts.Addr, err)
+	}
+	defer conn.Close()
+
+	br := bufio.NewReader(conn)
+	cmds := make([][]string, 0, 3)
+	if s.opts.Password != "" {
+		cmds = append(cmds, []string{"AUTH", s.opts.Password})
+	}
+	cmds = append(cmds, []string{"SELECT", strconv.Itoa(s.opts.DB)})
+	cmds = append(cmds, []string{"PSUBSCRIBE", NewValuePattern()})
+	for _, args := range cmds {
+		if err := writeRESP(conn, args); err != nil {
+			return fmt.Errorf("itembridge: 订阅握手写失败: %w", err)
+		}
+	}
+	for range cmds {
+		if _, err := sourceReadReply(br); err != nil {
+			return fmt.Errorf("itembridge: 订阅握手读失败: %w", err)
+		}
+	}
+
+	// 订阅期用独立 goroutine 监听 ctx 取消→关连接，使阻塞 Read 立即返回（不加锁，conn.Close 并发安全）。
+	done := make(chan struct{})
+	var once sync.Once
+	go func() {
+		select {
+		case <-ctx.Done():
+			_ = conn.Close()
+		case <-done:
+		}
+	}()
+	defer once.Do(func() { close(done) })
+
+	for {
+		reply, err := sourceReadReply(br)
+		if err != nil {
+			if ctx.Err() != nil {
+				return ctx.Err()
+			}
+			return fmt.Errorf("itembridge: 订阅读失败（将重连）: %w", err)
+		}
+		arr, ok := reply.([]any)
+		if !ok || len(arr) < 4 {
+			continue // 非 pmessage push（如 psubscribe 确认），忽略
+		}
+		kind := sourceAsText(arr[0])
+		channel := sourceAsText(arr[2])
+		payloadBytes, _ := arr[3].([]byte)
+		if kind != "pmessage" {
+			continue
+		}
+		if _, valid := TenantFromNewValueChannel(channel); !valid {
+			continue // 频道形态不符（异常 push）直接忽略，不进 onEvent
+		}
+		onEvent(channel, string(payloadBytes))
+	}
+}
+
+// sourceAsText bulk（[]byte）与简单串（string）统一取文本（RESP2 pmessage 的数组元素为 bulk）。
+func sourceAsText(v any) string {
+	switch t := v.(type) {
+	case string:
+		return t
+	case []byte:
+		return string(t)
+	default:
+		return ""
+	}
+}
+
+// writeRESP 编码一条 RESP2 命令数组（与 cache.appendRESPCommand 同形态）。
+func writeRESP(conn net.Conn, args []string) error {
+	var buf []byte
+	buf = append(buf, '*')
+	buf = append(buf, strconv.Itoa(len(args))...)
+	buf = append(buf, '\r', '\n')
+	for _, a := range args {
+		buf = append(buf, '$')
+		buf = append(buf, strconv.Itoa(len(a))...)
+		buf = append(buf, '\r', '\n')
+		buf = append(buf, a...)
+		buf = append(buf, '\r', '\n')
+	}
+	_, err := conn.Write(buf)
+	return err
+}
+
+// sourceReadReply RESP2 回复解析（行/数组有界，拒绝畸形长度）。
+func sourceReadReply(br *bufio.Reader) (any, error) {
+	line, err := sourceReadLine(br)
+	if err != nil {
+		return nil, err
+	}
+	if len(line) == 0 {
+		return nil, fmt.Errorf("itembridge: 空回复行")
+	}
+	switch line[0] {
+	case '+', '-':
+		return string(line[1:]), nil
+	case ':':
+		return strconv.ParseInt(string(line[1:]), 10, 64)
+	case '$':
+		n, err := strconv.Atoi(string(line[1:]))
+		if err != nil || n < -1 || n > sourceMaxBulk {
+			return nil, fmt.Errorf("itembridge: bulk 长度非法 %q", line)
+		}
+		if n == -1 {
+			return nil, nil
+		}
+		body := make([]byte, n+2)
+		if _, err := readFull(br, body); err != nil {
+			return nil, err
+		}
+		return body[:n], nil
+	case '*':
+		n, err := strconv.Atoi(string(line[1:]))
+		if err != nil || n < 0 || n > sourceMaxArray {
+			return nil, fmt.Errorf("itembridge: 数组长度非法 %q", line)
+		}
+		out := make([]any, 0, n)
+		for i := 0; i < n; i++ {
+			r, err := sourceReadReply(br)
+			if err != nil {
+				return nil, err
+			}
+			out = append(out, r)
+		}
+		return out, nil
+	default:
+		return nil, fmt.Errorf("itembridge: 回复类型未知 %q", line)
+	}
+}
+
+func sourceReadLine(br *bufio.Reader) ([]byte, error) {
+	var acc []byte
+	for {
+		chunk, err := br.ReadSlice('\n')
+		if len(acc)+len(chunk) > sourceMaxLine {
+			return nil, fmt.Errorf("itembridge: 回复行超上限")
+		}
+		acc = append(acc, chunk...)
+		if err == bufio.ErrBufferFull {
+			continue
+		}
+		if err != nil {
+			return nil, err
+		}
+		break
+	}
+	for len(acc) > 0 && (acc[len(acc)-1] == '\n' || acc[len(acc)-1] == '\r') {
+		acc = acc[:len(acc)-1]
+	}
+	return acc, nil
+}
+
+func readFull(br *bufio.Reader, buf []byte) (int, error) {
+	total := 0
+	for total < len(buf) {
+		n, err := br.Read(buf[total:])
+		total += n
+		if err != nil {
+			return total, err
+		}
+	}
+	return total, nil
+}
