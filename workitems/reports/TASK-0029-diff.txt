+func (s *fakeRedis29) handle(conn net.Conn) {
+	defer conn.Close()
+	br := bufio.NewReader(conn)
+	for {
+		args, err := readRESPCmd29(br)

+		switch strings.ToUpper(args[0]) {

+			_, _ = io.WriteString(conn, "+OK\r\n")
+		case "GET":
+			_, _ = io.WriteString(conn, "$-1\r\n")

+func readRESPCmd29(br *bufio.Reader) ([]string, error) {
+	line, err := br.ReadString('\n')
+	if err != nil {
+		return nil, err
+	}
+	line = strings.TrimRight(line, "\r\n")
+	if len(line) == 0 || line[0] != '*' {
+		return nil, fmt.Errorf("非法命令行 %q", line)
+	}
+	var n int
+	if _, err := fmt.Sscanf(line[1:], "%d", &n); err != nil {
+		return nil, err
+	}
+	args := make([]string, 0, n)
+	for i := 0; i < n; i++ {
+		hdr, err := br.ReadString('\n')
+		if err != nil {
+			return nil, err
+		}
+		hdr = strings.TrimRight(hdr, "\r\n")
+		var l int
+		if _, err := fmt.Sscanf(hdr[1:], "%d", &l); err != nil {
+			return nil, err
+		}
+		buf := make([]byte, l+2)
+		if _, err := io.ReadFull(br, buf); err != nil {
+			return nil, err
+		}
+		args = append(args, string(buf[:l]))
+	}
+	return args, nil
+}

+package cache

+import (
+	"bufio"
+	"context"
+	"encoding/json"
+	"errors"
+	"fmt"
+	"net"

+// RedisOptions Redis 连接配置。
+type RedisOptions struct {
+	// Addr host:port（INGEST_REDIS_ADDR）。
+	Addr string
+	// Password 空 = 匿名（无 AUTH）。
+	Password string
+	// DB 逻辑库编号（SELECT db）。生产 ingest db0；测试 db1+（Go 道纪律）。
+	DB int
+	// DialTimeout 建连超时；<=0 取 3s。
+	DialTimeout time.Duration
+	// CmdTimeout 单命令超时；<=0 取 3s。
+	CmdTimeout time.Duration
+}
+
+const (
+	defaultRedisDialTimeout = 3 * time.Second
+	defaultRedisCmdTimeout  = 3 * time.Second
+)
+
+// RedisClient 极简 RESP2 客户端（string 命令面：SET/GET/PUBLISH/PING/AUTH/SELECT）。
+// 与真 Redis 的 RESP2 协议兼容（RESP3 服务器对 RESP2 客户端保持兼容）。
+type RedisClient struct {
+	opts RedisOptions
+
+	mu   sync.Mutex
+	conn net.Conn
+	br   *bufio.Reader
+}
+
+// NewRedisClient 构造客户端（惰性建连：首次命令时拨号 + AUTH + SELECT）。
+func NewRedisClient(opts RedisOptions) *RedisClient {
+	if opts.DialTimeout <= 0 {
+		opts.DialTimeout = defaultRedisDialTimeout
+	}
+	if opts.CmdTimeout <= 0 {
+		opts.CmdTimeout = defaultRedisCmdTimeout
+	}
+	return &RedisClient{opts: opts}
+}
+
+// Close 关闭底层连接（幂等）。
+func (c *RedisClient) Close() error {
+	c.mu.Lock()
+	defer c.mu.Unlock()
+	if c.conn != nil {
+		err := c.conn.Close()
+		c.conn = nil
+		c.br = nil
+		return err
+	}
+	return nil
+}
+
+// Ping 健康探测（PING → PONG）。
+func (c *RedisClient) Ping(ctx context.Context) error {
+	_, err := c.Do(ctx, "PING")
+	return err
+}
+
+// Do 执行单条命令（互斥串行；连接故障时断连、下次调用重连自愈）。
+// 返回值形态：简单串 string / 整数 int64 / bulk []byte（nil=缺失）/ 数组 []any。
+func (c *RedisClient) Do(ctx context.Context, args ...string) (any, error) {
+	c.mu.Lock()
+	defer c.mu.Unlock()
+	if err := c.ensureConnLocked(ctx); err != nil {
+		return nil, err
+	}
+	reply, err := c.execLocked(ctx, [][]string{args})
+	if err != nil {
+		c.dropConnLocked()
+		return nil, err
+	}
+	if len(reply) == 1 {
+		return reply[0], nil
+	}
+	return nil, errors.New("cache: redis 响应数不符")
+}
+
+// Pipelined 管线化执行一批命令（一次写入、逐条读回——批量 SET/PUBLISH 的队头约束
+// 批量化落点，0027 遗留 ②）。任一命令失败即断连并返回错误（SET 幂等覆盖写，重试安全）。
+func (c *RedisClient) Pipelined(ctx context.Context, cmds [][]string) ([]any, error) {
+	c.mu.Lock()
+	defer c.mu.Unlock()
+	if err := c.ensureConnLocked(ctx); err != nil {
+		return nil, err
+	}
+	reply, err := c.execLocked(ctx, cmds)
+	if err != nil {
+		c.dropConnLocked()
+		return nil, err
+	}
+	return reply, nil
+}
+
+// ensureConnLocked 惰性建连 +（可选）AUTH + SELECT（连接建立序列）。
+func (c *RedisClient) ensureConnLocked(ctx context.Context) error {
+	if c.conn != nil {
+		return nil
+	}
+	d := c.opts.DialTimeout
+	if dl, ok := ctx.Deadline(); ok {
+		if remain := time.Until(dl); remain < d {
+			d = remain
+		}
+	}
+	conn, err := net.DialTimeout("tcp", c.opts.Addr, d)
+	if err != nil {
+		return fmt.Errorf("cache: redis 连接失败 %s: %w", c.opts.Addr, err)
+	}
+	c.conn = conn
+	c.br = bufio.NewReader(conn)
+	handshake := make([][]string, 0, 2)
+	if c.opts.Password != "" {
+		handshake = append(handshake, []string{"AUTH", c.opts.Password})
+	}
+	handshake = append(handshake, []string{"SELECT", strconv.Itoa(c.opts.DB)})
+	if _, err := c.execLocked(ctx, handshake); err != nil {
+		c.dropConnLocked()
+		return fmt.Errorf("cache: redis 建连握手失败: %w", err)
+	}
+	return nil
+}
+
+// execLocked 写入命令（管线：一次写全部）并逐条读回。失败时断连（下次重连自愈）。
+func (c *RedisClient) execLocked(ctx context.Context, cmds [][]string) ([]any, error) {
+	var buf []byte
+	for _, args := range cmds {
+		buf = appendRESPCommand(buf, args)
+	}
+	if dl, ok := ctx.Deadline(); ok {
+		_ = c.conn.SetDeadline(dl)
+	} else {
+		_ = c.conn.SetDeadline(time.Now().Add(c.opts.CmdTimeout))
+	}
+	if _, err := c.conn.Write(buf); err != nil {
+		return nil, fmt.Errorf("cache: redis 写命令失败: %w", err)
+	}
+	replies := make([]any, 0, len(cmds))
+	for range cmds {
+		r, err := readRESPReply(c.br)
+		if err != nil {
+			return nil, fmt.Errorf("cache: redis 读响应失败: %w", err)
+		}
+		if e, ok := r.(redisError); ok {
+			return nil, fmt.Errorf("cache: redis 错误回执: %s", string(e))
+		}
+		replies = append(replies, r)
+	}
+	return replies, nil
+}
+
+// dropConnLocked 断连（下次调用重连自愈）。
+func (c *RedisClient) dropConnLocked() {
+	if c.conn != nil {
+		_ = c.conn.Close()
+	}
+	c.conn = nil
+	c.br = nil
+}
+
+// redisError RESP2 错误回执（-ERR ...）。
+type redisError string
+
+// appendRESPCommand 编码一条 RESP2 命令数组。
+func appendRESPCommand(buf []byte, args []string) []byte {
+	buf = append(buf, '*')
+	buf = append(buf, strconv.Itoa(len(args))...)
+	buf = append(buf, '\r', '\n')
+	for _, a := range args {
+		buf = append(buf, '$')
+		buf = append(buf, strconv.Itoa(len(a))...)
+		buf = append(buf, '\r', '\n')
+		buf = append(buf, a...)
+		buf = append(buf, '\r', '\n')
+	}
+	return buf
+}
+
+// readRESPReply 读取一条 RESP2 回复。
+func readRESPReply(br *bufio.Reader) (any, error) {
+	line, err := readReplyLine(br)
+	if err != nil {
+		return nil, err
+	}
+	if len(line) == 0 {
+		return nil, errors.New("cache: redis 空回复行")
+	}
+	switch line[0] {
+	case '+': // 简单串
+		return string(line[1:]), nil
+	case '-': // 错误
+		return redisError(string(line[1:])), nil
+	case ':': // 整数
+		n, err := strconv.ParseInt(string(line[1:]), 10, 64)
+		if err != nil {
+			return nil, fmt.Errorf("cache: redis 整数回执非法 %q: %w", line, err)
+		}
+		return n, nil
+	case '$': // bulk string（$-1 = nil 缺失）
+		n, err := strconv.Atoi(string(line[1:]))
+		if err != nil || n < -1 {
+			return nil, fmt.Errorf("cache: redis bulk 长度非法 %q", line)
+		}
+		if n == -1 {
+			return nil, nil
+		}
+		body := make([]byte, n+2)
+		if _, err := ioReadFull(br, body); err != nil {
+			return nil, err
+		}
+		return body[:n], nil
+	case '*': // 数组
+		n, err := strconv.Atoi(string(line[1:]))
+		if err != nil || n < -1 {
+			return nil, fmt.Errorf("cache: redis 数组长度非法 %q", line)
+		}
+		out := make([]any, 0, max(n, 0))
+		for i := 0; i < n; i++ {
+			r, err := readRESPReply(br)
+			if err != nil {
+				return nil, err
+			}
+			out = append(out, r)
+		}
+		return out, nil
+	default:
+		return nil, fmt.Errorf("cache: redis 回复类型未知 %q", line)
+	}
+}
+
+func readReplyLine(br *bufio.Reader) ([]byte, error) {
+	line, err := br.ReadString('\n')
+	if err != nil {
+		return nil, fmt.Errorf("cache: redis 读行失败: %w", err)
+	}
+	// 去掉 \r\n（容忍只有 \n 的非规范服务端）。
+	for len(line) > 0 && (line[len(line)-1] == '\n' || line[len(line)-1] == '\r') {
+		line = line[:len(line)-1]
+	}
+	return []byte(line), nil
+}
+
+func ioReadFull(br *bufio.Reader, buf []byte) (int, error) {
+	total := 0
+	for total < len(buf) {
+		n, err := br.Read(buf[total:])
+		total += n
+		if err != nil {
+			return total, fmt.Errorf("cache: redis 读体失败: %w", err)
+		}
+	}
+	return total, nil
+}

+// ---- RedisPointCache（PointCache 真实现，零改动替换内存假实现）----
+
+// PointSet 批量写的单点条目（同租户）。
+type PointSet struct {
+	PointCode string
+	Value     Value
+}
+
+// BatchPointSetter 可选批量写接口（RESP 管线一次往返写整批）。流水线按租户分组调用；
+// 未实现该接口的缓存实现退回逐点 SetPoint（接口零改动替换原则——可选能力面）。
+type BatchPointSetter interface {
+	// SetPoints 同租户一批点位覆盖写（管线化）。任一条失败返回错误（调用方 fail-soft）。
+	SetPoints(ctx context.Context, tenant string, items []PointSet) error
+}
+
+// Backfill 未命中回源源（读-through 懒回源；ingestd 装配注入 TDengine 最近值查询）。
+type Backfill interface {
+	// LatestPointValue 查点位最近落库值；found=false 表示源无数据。
+	LatestPointValue(ctx context.Context, tenant, pointCode string) (Value, bool, error)
+}
+
+// BackfillFunc 函数适配器。
+type BackfillFunc func(ctx context.Context, tenant, pointCode string) (Value, bool, error)
+
+// LatestPointValue 实现 Backfill。
+func (f BackfillFunc) LatestPointValue(ctx context.Context, tenant, pointCode string) (Value, bool, error) {
+	return f(ctx, tenant, pointCode)
+}
+
+// RedisPointCache Redis 点位缓存真实现：SET/GET JSON（v/q/ts 三键）+ 未命中读-through。
+type RedisPointCache struct {
+	rc  *RedisClient
+	src Backfill // 可选：nil → 未命中直接 ErrCacheMiss
+}
+
+// NewRedisPointCache 构造（src 可 nil——不回源）。
+func NewRedisPointCache(opts RedisOptions, src Backfill) *RedisPointCache {
+	return &RedisPointCache{rc: NewRedisClient(opts), src: src}
+}
+
+// SetPoint 覆盖写入（SET iot:pt:{tenant}:{code} {"v","q","ts"}）。错误上返——
+// 落库成功后的缓存失败不阻塞（fail-soft 归流水线计数）。
+func (c *RedisPointCache) SetPoint(ctx context.Context, tenant, pointCode string, v Value) error {
+	b, err := json.Marshal(v)
+	if err != nil {
+		return fmt.Errorf("cache: 编码点位值失败: %w", err)
+	}
+	_, err = c.rc.Do(ctx, "SET", Key(tenant, pointCode), string(b))
+	return err
+}
+
+// GetPoint 读取并解码；未命中回源（读-through：源命中 → SET 写穿 → 返回）；
+// 源无数据或无源 → ErrCacheMiss。读路径错误上返（不静默给零值）。
+func (c *RedisPointCache) GetPoint(ctx context.Context, tenant, pointCode string) (Value, error) {
+	reply, err := c.rc.Do(ctx, "GET", Key(tenant, pointCode))
+	if err != nil {
+		return Value{}, err
+	}
+	b, ok := reply.([]byte)
+	if !ok || b == nil {
+		return c.backfill(ctx, tenant, pointCode)
+	}
+	var v Value
+	if err := json.Unmarshal(b, &v); err != nil {
+		return Value{}, fmt.Errorf("cache: 解码点位值失败: %w", err)
+	}
+	return v, nil
+}
+
+// SetPoints 实现 BatchPointSetter：RESP 管线一次往返写整批（同租户）。
+func (c *RedisPointCache) SetPoints(ctx context.Context, tenant string, items []PointSet) error {
+	if len(items) == 0 {
+		return nil
+	}
+	cmds := make([][]string, 0, len(items))
+	for _, it := range items {
+		b, err := json.Marshal(it.Value)
+		if err != nil {
+			return fmt.Errorf("cache: 编码点位值失败: %w", err)
+		}
+		cmds = append(cmds, []string{"SET", Key(tenant, it.PointCode), string(b)})
+	}
+	_, err := c.rc.Pipelined(ctx, cmds)
+	return err
+}
+
+// Delete 清理键（测试/运维用；lint：live 集成测试测后清理）。
+func (c *RedisPointCache) Delete(ctx context.Context, tenant, pointCode string) error {
+	_, err := c.rc.Do(ctx, "DEL", Key(tenant, pointCode))
+	return err
+}
+
+// backfill 未命中回源：源命中 → 写穿缓存 → 返回；源无数据/无源 → ErrCacheMiss。
+func (c *RedisPointCache) backfill(ctx context.Context, tenant, pointCode string) (Value, error) {
+	if c.src == nil {
+		return Value{}, ErrCacheMiss
+	}
+	v, found, err := c.src.LatestPointValue(ctx, tenant, pointCode)
+	if err != nil {
+		return Value{}, fmt.Errorf("cache: 回源失败: %w", err)
+	}
+	if !found {
+		return Value{}, ErrCacheMiss
+	}
+	if err := c.SetPoint(ctx, tenant, pointCode, v); err != nil {
+		// 回源写穿失败：值已到手，仍返回值（读路径优先服务读方；缓存下次再补）。
+		return v, nil
+	}
+	return v, nil
+}

+func (s *fakeRedisServer) serve() {
+	for {
+		conn, err := s.ln.Accept()

+func (s *fakeRedisServer) handle(conn net.Conn) {
+	defer conn.Close()
+	var pend []byte
+	buf := make([]byte, 64*1024)
+	for {
+		n, err := conn.Read(buf)
+		if err != nil {
+			return
+		}
+		s.mu.Lock()
+		s.segments = append(s.segments, string(buf[:n])) // 原始 TCP 读段（管线化断言依据）
+		s.mu.Unlock()
+		pend = append(pend, buf[:n]...)
+		for {
+			args, consumed, perr := parseRESPCommand(pend)
+			if perr != nil {
+				return
+			}
+			if consumed == 0 {
+				break // 不完整命令，等下一个读段
+			}
+			pend = pend[consumed:]
+			s.mu.Lock()
+			s.commands = append(s.commands, args)
+			fail := s.failNextN > 0
+			if fail {
+				s.failNextN--
+			}
+			s.mu.Unlock()
+			if fail {
+				_, _ = io.WriteString(conn, "-ERR injected failure\r\n")
+				continue
+			}
+			_, _ = io.WriteString(conn, s.reply(args))
+		}
+	}
+}

+// reply 按命令语义回放（SET/GET/PUBLISH/PING/AUTH/SELECT/DEL）。
+func (s *fakeRedisServer) reply(args []string) string {
+	switch strings.ToUpper(args[0]) {
+	case "PING":
+		return "+PONG\r\n"
+	case "AUTH", "SELECT":
+		if len(args) >= 2 && args[1] == "BAD" {
+			return "-ERR invalid password\r\n"
+		}
+		return "+OK\r\n"
+	case "SET":
+		if len(args) < 3 {
+			return "-ERR wrong number of arguments\r\n"
+		}
+		s.mu.Lock()
+		s.data[args[1]] = args[2]
+		s.mu.Unlock()
+		return "+OK\r\n"
+	case "GET":
+		if len(args) < 2 {
+			return "-ERR wrong number of arguments\r\n"
+		}
+		s.mu.Lock()
+		v, ok := s.data[args[1]]
+		s.mu.Unlock()
+		if !ok {
+			return "$-1\r\n"
+		}
+		return fmt.Sprintf("$%d\r\n%s\r\n", len(v), v)
+	case "PUBLISH":
+		return ":1\r\n"
+	case "DEL":
+		s.mu.Lock()
+		delete(s.data, args[1])
+		s.mu.Unlock()
+		return ":1\r\n"
+	default:
+		return "-ERR unknown command '" + args[0] + "'\r\n"
+	}
+}

+// parseRESPCommand 从缓冲解析一条完整 RESP2 命令：返回 (args, 消费字节数, 错误)；
+// consumed=0 表示命令不完整（等更多字节）。
+func parseRESPCommand(b []byte) ([]string, int, error) {
+	pos := 0
+	line, n, err := readBufLine(b, pos)
+	if err != nil {
+		return nil, 0, err
+	}
+	if n == 0 {
+		return nil, 0, nil
+	}
+	pos += n
+	if len(line) == 0 || line[0] != '*' {
+		return nil, 0, fmt.Errorf("fakeRedis: 期望命令数组，收到 %q", line)
+	}
+	var count int
+	if _, err := fmt.Sscanf(string(line[1:]), "%d", &count); err != nil {
+		return nil, 0, fmt.Errorf("fakeRedis: 数组长度非法 %q", line)
+	}
+	args := make([]string, 0, count)
+	for i := 0; i < count; i++ {
+		hdr, n, err := readBufLine(b, pos)
+		if err != nil {
+			return nil, 0, err
+		}
+		if n == 0 {
+			return nil, 0, nil
+		}
+		pos += n
+		if len(hdr) == 0 || hdr[0] != '$' {
+			return nil, 0, fmt.Errorf("fakeRedis: 期望 bulk 头，收到 %q", hdr)
+		}
+		var l int
+		if _, err := fmt.Sscanf(string(hdr[1:]), "%d", &l); err != nil {
+			return nil, 0, fmt.Errorf("fakeRedis: bulk 长度非法 %q", hdr)
+		}
+		if l < 0 {
+			args = append(args, "")
+			continue
+		}
+		if len(b)-pos < l+2 {
+			return nil, 0, nil // 体不完整
+		}
+		args = append(args, string(b[pos:pos+l]))
+		pos += l + 2 // 数据 + \r\n
+	}
+	return args, pos, nil
+}
+
+// readBufLine 从 b[pos:] 读取一行（到 \n 为止，含去 \r）；行不完整返回 n=0。
+func readBufLine(b []byte, pos int) (line []byte, n int, err error) {
+	i := pos
+	for i < len(b) && b[i] != '\n' {
+		i++
+	}
+	if i >= len(b) {
+		return nil, 0, nil // 不完整
+	}
+	line = b[pos:i]
+	for len(line) > 0 && line[len(line)-1] == '\r' {
+		line = line[:len(line)-1]
+	}
+	return line, i - pos + 1, nil
+}

+type fakePubServer struct {

+	mu       sync.Mutex
+	segments []string
+	commands [][]string
+	failNext int

+func newFakePubServer(t *testing.T) *fakePubServer {
+	t.Helper()
+	ln, err := net.Listen("tcp", "127.0.0.1:0")
+	if err != nil {
+		t.Fatalf("假 Redis 监听失败: %v", err)
+	}
+	s := &fakePubServer{addr: ln.Addr().String(), ln: ln}
+	go func() {
+		for {
+			conn, err := ln.Accept()
+			if err != nil {
+				return
+			}
+			go s.handle(conn)
+		}
+	}()
+	t.Cleanup(func() { _ = ln.Close() })
+	return s
+}

+func (s *fakePubServer) handle(conn net.Conn) {
+	defer conn.Close()
+	var pend []byte
+	buf := make([]byte, 64*1024)
+	for {
+		n, err := conn.Read(buf)
+		if err != nil {
+			return
+		}
+		s.mu.Lock()
+		s.segments = append(s.segments, string(buf[:n]))
+		s.mu.Unlock()
+		pend = append(pend, buf[:n]...)
+		for {
+			args, consumed, perr := parseResp(pend)
+			if perr != nil {
+				return
+			}
+			if consumed == 0 {
+				break
+			}
+			pend = pend[consumed:]
+			s.mu.Lock()
+			s.commands = append(s.commands, args)
+			fail := s.failNext > 0
+			if fail {
+				s.failNext--
+			}
+			s.mu.Unlock()
+			if fail {
+				_, _ = io.WriteString(conn, "-ERR injected\r\n")
+				continue
+			}
+			if strings.EqualFold(args[0], "AUTH") || strings.EqualFold(args[0], "SELECT") {
+				_, _ = io.WriteString(conn, "+OK\r\n")
+				continue
+			}
+			_, _ = io.WriteString(conn, ":1\r\n")
+		}
+	}
+}

+func parseResp(b []byte) ([]string, int, error) {
+	pos := 0
+	line, n, err := bufLine(b, pos)
+	if err != nil {
+		return nil, 0, err
+	}
+	if n == 0 {
+		return nil, 0, nil
+	}
+	pos += n
+	if len(line) == 0 || line[0] != '*' {
+		return nil, 0, fmt.Errorf("fakePub: 非法 RESP 命令形态")
+	}
+	var count int
+	if _, err := fmt.Sscanf(string(line[1:]), "%d", &count); err != nil {
+		return nil, 0, fmt.Errorf("fakePub: 非法 RESP 命令形态")
+	}
+	args := make([]string, 0, count)
+	for i := 0; i < count; i++ {
+		hdr, n, err := bufLine(b, pos)
+		if err != nil {
+			return nil, 0, err
+		}
+		if n == 0 {
+			return nil, 0, nil
+		}
+		pos += n
+		if len(hdr) == 0 || hdr[0] != '$' {
+			return nil, 0, fmt.Errorf("fakePub: 非法 RESP 命令形态")
+		}
+		var l int
+		if _, err := fmt.Sscanf(string(hdr[1:]), "%d", &l); err != nil {
+			return nil, 0, fmt.Errorf("fakePub: 非法 RESP 命令形态")
+		}
+		if len(b)-pos < l+2 {
+			return nil, 0, nil
+		}
+		args = append(args, string(b[pos:pos+l]))
+		pos += l + 2
+	}
+	return args, pos, nil
+}
+
+func bufLine(b []byte, pos int) (line []byte, n int, err error) {
+	i := pos
+	for i < len(b) && b[i] != '\n' {
+		i++
+	}
+	if i >= len(b) {
+		return nil, 0, nil
+	}
+	line = b[pos:i]
+	for len(line) > 0 && line[len(line)-1] == '\r' {
+		line = line[:len(line)-1]
+	}
+	return line, i - pos + 1, nil
+}

+	return segments, commands
+}

+func (f *fakeQueryAdapter) handle(w http.ResponseWriter, r *http.Request) {
+	body := make([]byte, 8192)
+	n, _ := r.Body.Read(body)
+	f.mu.Lock()
+	f.sqls = append(f.sqls, string(body[:n]))
+	resp := f.resp
+	status := f.status
+	f.mu.Unlock()
+	if status != 0 {
+		w.WriteHeader(status)
+	}
+	_ = json.NewEncoder(w).Encode(resp)
+}

+	f.mu.Lock()
+	defer f.mu.Unlock()
+	out := make([]string, len(f.sqls))

+	return out
