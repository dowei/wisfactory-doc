+/**
+ * 对外调用审计异步提交器（TASK-0051，REQ-0027：一切调用含 403 统一记录）。
+ *
+ * <p>过滤器在请求收尾（finally）组装 {@link OpenApiCallLogRow} 后交专用线程池异步落库；
+ * 队列满走 CallerRuns（照 ot-core operateLogExecutor 范式——审计不丢优先于零阻塞）。
+ * 落库失败只 WARN 不上抛：审计写失败绝不顶掉响应报文。
+ *
+ * @author dowei
+ * @date 2026-11-22
+ */
+@Slf4j
+@Component
+public class OpenApiAuditWriter {
+
+    private final ThreadPoolTaskExecutor executor;
+    private final OpenApiCallLogMapper callLogMapper;
+
+    public OpenApiAuditWriter(@Qualifier("openapiAuditExecutor") ThreadPoolTaskExecutor executor,
+                              OpenApiCallLogMapper callLogMapper) {
+        this.executor = executor;
+        this.callLogMapper = callLogMapper;
+    }
+
+    /**
+     * 异步落库（submit 前快照行对象——请求线程回收后对象仍自洽）。
+     * executor 拒绝策略 CallerRuns 兜底（队列满退化为同步写，不丢行）。
+     */
+    public void submit(OpenApiCallLogRow row) {
+        if (row == null) {
+            return;
+        }
+        try {
+            executor.execute(() -> {
+                try {
+                    callLogMapper.insert(row);
+                } catch (Exception e) {
+                    // 审计失败不顶响应：WARN 留痕（appKey 为业务标识可落日志）
+                    log.warn("[OPENAPI] 审计落库失败 appKey={} path={} code={}",
+                            row.getAppKey(), row.getApiPath(), row.getResultCode(), e);
+                }
+            });
+        } catch (Exception e) {
+            // 线程池已 shutdown 等极端场景：退化为同步尽力写，仍不顶响应
+            try {
+                callLogMapper.insert(row);
+            } catch (Exception inner) {
+                log.warn("[OPENAPI] 审计落库失败（同步兜底亦失败）appKey={} path={}",
+                        row.getAppKey(), row.getApiPath(), inner);
+            }
+        }
+    }

+ * 全局唯一定位，已删=不存在）→ 状态（禁用立即生效——验签前先查状态）→ 到期 →
+ * appKey 级可选 IP 白名单（ADR-0013 A7，NULL=不限）→ 签名比对。签名密钥 =
+ * secret_encrypted SM4 解密得的 secret 原文（HMAC 需服务端持原文——上游 secret_hash 列
+ * 无法支持，「与上游差异」登记）；secret_hash 保留列本期不参与验签（「SHA256(提交 secret)
+ * 快速校验」在纯 HMAC 通道无提交 secret 可算——登记报告，轮换期双密钥消费归 0052）。
+ *
+ * <p>比对恒时序（{@link MessageDigest#isEqual}）防逐字节试错；解密失败/密文缺失一律
+ * 6001 fail-closed（不降级 500、不给 500 探测面）。
+ *
+ * @author dowei
+ * @date 2026-11-22
+ */
+@Slf4j
+@Component
+public class HmacAuthStrategy implements OpenApiAuthStrategy {
+
+    /** HMAC 算法名（JDK 标准实现） */
+    private static final String HMAC_ALGO = "HmacSHA256";
+
+    private final OpenApiClientMapper clientMapper;
+    private final OpenApiProperties properties;
+
+    public HmacAuthStrategy(OpenApiClientMapper clientMapper, OpenApiProperties properties) {
+        this.clientMapper = clientMapper;
+        this.properties = properties;
+    }
+
+    @Override
+    public OpenApiClientIdentity authenticate(OpenApiCredentials credentials) {
+        requireText(credentials.appKey(), "缺少 " + OpenApiCodes.HEADER_APP_KEY + " 请求头");
+        requireText(credentials.timestamp(), "缺少 " + OpenApiCodes.HEADER_TIMESTAMP + " 请求头");
+        requireText(credentials.signature(), "缺少 " + OpenApiCodes.HEADER_SIGNATURE + " 请求头");
+        long timestamp = parseTimestamp(credentials.timestamp());
+
+        long drift = Math.abs(System.currentTimeMillis() - timestamp);
+        if (drift > properties.getReplayWindowSeconds() * 1000L) {
+            throw new OpenApiAuthException("签名时间戳超出防重放窗口（±" + properties.getReplayWindowSeconds() + " 秒）");
+        }
+
+        OpenApiClientRow client = clientMapper.selectByAppKey(credentials.appKey());
+        if (client == null || Integer.valueOf(1).equals(client.getDeleted())) {
+            throw new OpenApiAuthException("调用方不存在");
+        }
+        if (client.getStatus() == null || client.getStatus() == 0) {
+            // 禁用立即生效（REQ-0035）：验签前先查状态，不给禁用调用方任何验签通道
+            throw new OpenApiAuthException("调用方已禁用");
+        }
+        if (client.getExpireTime() != null && client.getExpireTime().isBefore(LocalDateTime.now())) {
+            throw new OpenApiAuthException("调用方已到期");
+        }
+        checkIpWhitelist(client, credentials.clientIp());
+
+        String secret = decryptSecret(client);
+        String expected = hmacSha256Hex(secret, credentials.signaturePayload());
+        boolean same = MessageDigest.isEqual(
+                expected.getBytes(StandardCharsets.UTF_8),
+                credentials.signature().trim().toLowerCase().getBytes(StandardCharsets.UTF_8));
+        if (!same) {
+            throw new OpenApiAuthException("签名不匹配");
+        }
+
+        OpenApiClientIdentity identity = new OpenApiClientIdentity();
+        identity.setClientId(client.getId());
+        identity.setTenantId(client.getTenantId());
+        identity.setAppKey(client.getAppKey());
+        identity.setScopeJson(client.getScope());
+        identity.setRateLimit(client.getRateLimit() == null ? 0 : client.getRateLimit());
+        return identity;
+    }
+
+    /** appKey 级可选 IP 白名单（ADR-0013 A7）：NULL/空白=不限；CSV 精确匹配（trim 容错） */
+    private void checkIpWhitelist(OpenApiClientRow client, String clientIp) {
+        String whitelist = client.getIpWhitelist();
+        if (StrUtil.isBlank(whitelist)) {
+            return;
+        }
+        if (clientIp != null) {
+            for (String entry : whitelist.split(",")) {
+                if (clientIp.equals(entry.trim())) {
+                    return;
+                }
+            }
+        }
+        // 白名单内 IP 不回显明细（脱敏）：报文不带来源 IP，供运营方对配置
+        throw new OpenApiAuthException("调用方 IP 白名单未放行");
+    }
+
+    /** secret_encrypted SM4 解密取原文（{v1} 密文，ot.crypto.data-key）；缺失/损坏 fail-closed 6001 */
+    private String decryptSecret(OpenApiClientRow client) {
+        if (StrUtil.isBlank(client.getSecretEncrypted())) {
+            throw new OpenApiAuthException("调用方密钥未配置");
+        }
+        try {
+            String secret = SmDataCrypto.decrypt(client.getSecretEncrypted());
+            if (StrUtil.isBlank(secret)) {
+                throw new OpenApiAuthException("调用方密钥未配置");
+            }
+            return secret;
+        } catch (OpenApiAuthException e) {
+            throw e;
+        } catch (Exception e) {
+            // 解密失败（密钥不匹配/密文损坏）：fail-closed 按 6001，不暴露 500 探测面
+            log.warn("[OPENAPI] secret 解密失败（appKey 脱敏，length={}）", StrUtil.length(client.getSecretEncrypted()));
+            throw new OpenApiAuthException("调用方密钥校验失败");
+        }
+    }
+
+    /** HMAC-SHA256(secret, appKey+timestamp+bodyHash) → 十六进制小写 */
+    static String hmacSha256Hex(String secret, String payload) {
+        try {
+            Mac mac = Mac.getInstance(HMAC_ALGO);
+            mac.init(new SecretKeySpec(OpenApiCredentials.utf8(secret), HMAC_ALGO));
+            return HexFormat.of().formatHex(mac.doFinal(OpenApiCredentials.utf8(payload)));
+        } catch (Exception e) {
+            throw new IllegalStateException("HMAC-SHA256 计算失败", e);
+        }
+    }
+
+    private long parseTimestamp(String timestamp) {
+        try {
+            return Long.parseLong(timestamp.trim());
+        } catch (NumberFormatException e) {
+            throw new OpenApiAuthException(OpenApiCodes.HEADER_TIMESTAMP + " 非法（须 epoch 毫秒）");
+        }
+    }
+
+    private void requireText(String value, String message) {
+        if (StrUtil.isBlank(value)) {
+            throw new OpenApiAuthException(message);
+        }
+    }
+}

+ * @author dowei
+ * @date 2026-11-22

+                                 String bodyHash, String clientIp, String path) {
+
+    /** 空请求体（六接口全 GET）的 SHA-256 十六进制摘要 */
+    public static String emptyBodyHash() {
+        return sha256Hex(new byte[0]);
+    }
+
+    /** 从请求提取凭据（body 已由缓存包装读出）；任一必填头缺失由过滤器按 6001 拒绝 */
+    public static OpenApiCredentials extract(HttpServletRequest request, byte[] body) {
+        return new OpenApiCredentials(
+                request.getHeader(OpenApiCodes.HEADER_APP_KEY),
+                request.getHeader(OpenApiCodes.HEADER_TIMESTAMP),
+                request.getHeader(OpenApiCodes.HEADER_SIGNATURE),
+                sha256Hex(body == null ? new byte[0] : body),
+                request.getRemoteAddr(),
+                request.getRequestURI());
+    }
+
+    /** HMAC-SHA256 签名串：appKey + timestamp + bodyHash（header 原文拼接，无分隔符——契约即此） */
+    public String signaturePayload() {
+        return (appKey == null ? "" : appKey)
+                + (timestamp == null ? "" : timestamp)
+                + (bodyHash == null ? "" : bodyHash);
+    }
+
+    /** SHA-256 → 十六进制小写（与调用方 hex 编码口径一致） */
+    public static String sha256Hex(byte[] data) {
+        try {
+            byte[] digest = MessageDigest.getInstance("SHA-256").digest(data);
+            StringBuilder sb = new StringBuilder(digest.length * 2);
+            for (byte b : digest) {
+                sb.append(Character.forDigit((b >> 4) & 0xF, 16));
+                sb.append(Character.forDigit(b & 0xF, 16));
+            }
+            return sb.toString();
+        } catch (NoSuchAlgorithmException e) {
+            // JDK 必带 SHA-256，理论不可达；fail-closed 不给吞错降级留口
+            throw new IllegalStateException("SHA-256 不可用", e);
+        }
+    }
+
+    /** UTF-8 字节（签名计算统一编码，杜绝平台默认字符集漂移） */
+    public static byte[] utf8(String value) {
+        return value == null ? new byte[0] : value.getBytes(StandardCharsets.UTF_8);
+    }

+ * <p>链路（brief 定版）：凭据提取 → 策略验签（6001 类失败 → HTTP 403 + body 6001）→
+ * 限流（超限 → HTTP 200 + body 6002）→ 放行 DispatcherServlet（控制器以请求属性取
+ * 验签产物做 scope 裁剪）→ finally 全量审计（403/6002/6003/200/500 统一异步落

+ * {@link OpenApiCodes#ATTR_RESULT_CODE} 回填业务码供审计（未回填时按传输状态兜底）。

+ * 在 {@code OpenApiAuthStrategy}，过滤器不感知具体策略）。

+ * @author dowei
+ * @date 2026-11-22

+@Slf4j
+public class OpenApiAuthFilter implements Filter {
+
+    /** 服务路径前缀（/openapi/v1 全覆盖；/openapi 下其它段未来接入时按同纪律显式扩） */
+    static final String PATH_PREFIX = "/openapi/v1/";
+
+    private final OpenApiAuthStrategy authStrategy;
+    private final OpenApiRateLimiter rateLimiter;
+    private final OpenApiAuditWriter auditWriter;
+
+    public OpenApiAuthFilter(OpenApiAuthStrategy authStrategy,
+                             OpenApiRateLimiter rateLimiter,
+                             OpenApiAuditWriter auditWriter) {
+        this.authStrategy = authStrategy;
+        this.rateLimiter = rateLimiter;
+        this.auditWriter = auditWriter;
+    }
+
+    @Override
+    public void doFilter(ServletRequest servletRequest, ServletResponse servletResponse, FilterChain chain)
+            throws IOException, ServletException {
+        HttpServletRequest request = (HttpServletRequest) servletRequest;
+        HttpServletResponse response = (HttpServletResponse) servletResponse;
+        String path = request.getRequestURI();
+        if (!path.startsWith(PATH_PREFIX)) {
+            // /openapi 其它段（未来）与其它过滤器链路径：直接放行
+            chain.doFilter(servletRequest, servletResponse);
+            return;
+        }
+
+        long start = System.currentTimeMillis();
+        CachedBodyRequestWrapper wrapped = new CachedBodyRequestWrapper(request);
+        OpenApiCredentials credentials = OpenApiCredentials.extract(wrapped, wrapped.cachedBody());
+        OpenApiCallLogRow auditRow = baseAuditRow(credentials, path, wrapped.getQueryString());
+        OpenApiClientIdentity identity = null;
+        Integer resultCode = null;
+        ContentCachingResponseWrapper responseWrapper = new ContentCachingResponseWrapper(response);
+        boolean bodyCopied = false;
+        try {
+            try {
+                identity = authStrategy.authenticate(credentials);
+            } catch (OpenApiAuthException e) {
+                // 认证类失败：HTTP 403（传输层 403 仅认证失败形态）+ body 6001
+                resultCode = e.getCode();
+                writeJson(responseWrapper, 403, Result.error(e.getCode(), e.getMessage()));
+                responseWrapper.copyBodyToResponse();
+                bodyCopied = true;
+                return;
+            }
+            request.setAttribute(OpenApiCodes.ATTR_CLIENT_IDENTITY, identity);
+            auditRow.setTenantId(identity.getTenantId());
+
+            if (!rateLimiter.tryAcquire(identity.getAppKey(), identity.getRateLimit())) {
+                // 限流：业务码形态（非认证失败，不占传输层 403）
+                resultCode = OpenApiCodes.RATE_LIMIT;

+                        "请求超出调用方限流上限（次/分钟），请稍后重试"));
+                responseWrapper.copyBodyToResponse();
+                bodyCopied = true;
+                return;
+            }
+
+            chain.doFilter(wrapped, responseWrapper);
+            // 业务码回读：统一 Result 体（GlobalExceptionHandler/控制器写回）的 code 字段优先，
+            // 属性/传输状态兜底——scope 外 6003（HTTP 200 + body 6003）等业务码形态由此如实入审计
+            resultCode = markedOrFallback(request, responseWrapper);
+        } catch (Exception e) {
+            // 未预期异常（控制器/框架抛穿，GlobalExceptionHandler 未拦截的形态）：如实记 500
+            resultCode = 500;
+            throw e;
+        } finally {
+            // 缓存体落真实响应（控制器/异常处理器写入的统一 Result 体）；仅落一次防重复写坏 Content-Length
+            if (!bodyCopied) {
+                responseWrapper.copyBodyToResponse();
+            }
+            auditRow.setResultCode(resultCode == null ? 200 : resultCode);
+            auditRow.setCostTime(System.currentTimeMillis() - start);
+            auditWriter.submit(auditRow);
+        }
+    }

+    private Integer markedOrFallback(HttpServletRequest request, ContentCachingResponseWrapper responseWrapper) {
+        try {
+            byte[] content = responseWrapper.getContentAsByteArray();
+            if (content.length > 0) {

+                if (code instanceof Number number) {
+                    return number.intValue();
+                }
+                if (code != null) {
+                    return Integer.parseInt(String.valueOf(code));
+                }
+            }
+        } catch (Exception e) {
+            // 非 Result 体（占位/下载形态等）：落兜底路径
+        }
+        Integer marked = (Integer) request.getAttribute(OpenApiCodes.ATTR_RESULT_CODE);
+        return marked != null ? marked : fallbackCode(responseWrapper);
+    }

+        OpenApiCallLogRow row = new OpenApiCallLogRow();
+        row.setAppKey(credentials.appKey());

+        if (credentials.bodyHash() != null && credentials.bodyHash().length() > 0
+                && !OpenApiCredentials.emptyBodyHash().equals(credentials.bodyHash())) {
+            if (sb.length() > 0) {
+                sb.append(';');
+            }
+            sb.append("body=").append(credentials.bodyHash(), 0, 16);
+        }
+        String digest = sb.toString();
+        return digest.length() > 500 ? digest.substring(0, 500) : digest;
+    }

+    }

+        }

+        }

+/**
+ * 对外接口限流器（TASK-0051，brief：rate_limit 次/分钟、0 不限——Redis 固定窗 60s）。
+ *
+ * <p>键 {@code rate_limit:openapi:<appKey>}（落在缓存监控既有 rate_limit 命名空间登记内，
+ * application.yml 无需新增登记项）；原子「INCR + 首次 PEXPIRE」脚本（RedisService.incrementWithTtl，
+ * BUG-0053 口径——进程崩溃不会留下无 TTL 永久 key 导致调用方被永久限流）。
+ *
+ * <p>身份绑定口径：限流在验签<b>成功后</b>执行——未验证身份的请求不占调用方额度，
+ * 攻击者无法凭伪造 appKey 洪泛把合法调用方顶到 6002（DoS 放大面收窄）；超限判据 =
+ * 窗口计数 > rate_limit。Redis 故障 fail-open（WARN 留痕放行）——限流是 QoS 控制而非
+ * 安全边界，Redis 不可用不应拒绝全部合法流量；安全边界由验签/scope 承担。
+ *
+ * @author dowei
+ * @date 2026-11-22
+ */
+@Slf4j
+@Component
+public class OpenApiRateLimiter {
+
+    /** 键前缀：与 CacheConstants.RATE_LIMIT_KEY 同域（缓存监控 rate_limit 命名空间） */
+    static final String KEY_PREFIX = "rate_limit:openapi:";
+
+    private final RedisService redisService;
+    private final OpenApiProperties properties;
+
+    public OpenApiRateLimiter(RedisService redisService, OpenApiProperties properties) {
+        this.redisService = redisService;
+        this.properties = properties;
+    }
+
+    /**
+     * 固定窗计数放行判定。
+     *
+     * @param appKey    调用方标识（已验签身份）
+     * @param rateLimit 每分钟上限（0=不限）
+     * @return true=放行；false=超出（业务码 6002）
+     */
+    public boolean tryAcquire(String appKey, int rateLimit) {
+        if (appKey == null || rateLimit <= 0) {
+            return true;
+        }
+        String key = KEY_PREFIX + appKey;
+        long count;
+        try {
+            count = redisService.incrementWithTtl(key, properties.getRateLimitWindowSeconds(), TimeUnit.SECONDS);
+        } catch (Exception e) {
+            log.warn("[OPENAPI] 限流计数失败（fail-open 放行，Redis 不可用）appKey={}", appKey, e);
+            return true;
+        }
+        return count <= rateLimit;
+    }

+        Map<Long, List<OpenApiInstanceRow>> byParent = new HashMap<>();
+        Set<Long> presentIds = new LinkedHashSet<>();
+        for (OpenApiInstanceRow node : nodes) {
+            Long parentId = node.getParentId() == null ? 0L : node.getParentId();
+            byParent.computeIfAbsent(parentId, k -> new ArrayList<>()).add(node);
+            presentIds.add(node.getId());
+        }
+        List<IotInstanceTreeVO> roots = new ArrayList<>();
+        for (OpenApiInstanceRow node : nodes) {
+            Long parentId = node.getParentId() == null ? 0L : node.getParentId();
+            if (parentId == 0L || !presentIds.contains(parentId)) {
+                roots.add(assemble(node, byParent, fullFieldIds));
+            }
+        }
+        return roots;
+    }

+        for (OpenApiInstanceRow child : byParent.getOrDefault(node.getId(), List.of())) {
+            children.add(assemble(child, byParent, fullFieldIds));
+        }
+        vo.setChildren(children);
+        return vo;
+    }

+    /** Redis 读（fail-soft）：故障/未命中/不可解析 → null */

+                json.getInt("q"),
