+<!--
+  * 通道连接配置 JSON 编辑器（TASK-0056，通道页公共组件）
+  *
+  * 协议类型联动 Schema 前置提示：按 ChannelProtocols 四值切换提示文案（后端
+  * classpath schema/channel/<协议>-v1.json 语义的前端镜像），并做基础格式提示
+  * （JSON 可解析、必含键、以 / 开头、凭据关键词扫描——递归对象/数组，镜像
+  * ChannelConnConfigValidator 黑名单）。提示仅供前置纠错，后端 Schema 校验为准
+  * （brief 决策 1：前端做基础格式提示，conn_config 校验以服务层为准）。
+-->
+<template>
+  <div>
+    <a-alert type="info" show-icon class="ot-margin-bottom10" :message="protocolHint" />
+    <a-textarea
+      :value="modelValue"
+      :rows="6"
+      placeholder='请输入 JSON 对象，如 {"broker":"tcp://127.0.0.1:1883","topic":"factory/gw01","qos":0}'
+      @update:value="onInput"
+    />
+    <div class="conn-config-check">
+      <template v-if="checkMessage">
+        <span class="conn-config-check-error">{{ checkMessage }}</span>
+      </template>
+      <template v-else-if="modelValue">
+        <span class="conn-config-check-ok">格式检查通过（最终以后端协议 Schema 校验为准）</span>
+      </template>
+    </div>
+  </div>
+</template>
+<script setup lang="ts">
+  import { computed } from 'vue';
+  import { IOT_CONN_CONFIG_HINTS } from './iot-consts';
+
+  const props = defineProps<{
+    // 协议类型 MQTT/HTTP/MODBUS/OPCUA（ChannelProtocols.ALL）
+    protocolType: string;
+    // JSON 文本（编辑态字符串；提交前 parse 为对象随表单提交）
+    modelValue: string;
+  }>();
+
+  const emit = defineEmits(['update:modelValue']);
+
+  const protocolHint = computed(() => {
+    return IOT_CONN_CONFIG_HINTS[props.protocolType] || '请先选择协议类型';
+  });
+
+  /** 凭据关键词黑名单（镜像 ChannelConnConfigValidator.CREDENTIAL_KEYWORDS，大小写不敏感子串匹配键名） */
+  const CREDENTIAL_KEYWORDS = ['credential', 'password', 'passwd', 'secret', 'token', 'apikey', 'accesskey', 'privatekey'];
+
+  /** 必含键（镜像各协议 -v1.json required；MODBUS/OPCUA 一期宽松骨架无 required） */
+  const REQUIRED_KEYS: Record<string, string[]> = {
+    MQTT: ['broker', 'topic'],
+    HTTP: ['reportPath'],
+    MODBUS: [],
+    OPCUA: [],
+  };
+
+  const checkMessage = computed(() => basicFormatCheck(props.modelValue, props.protocolType));
+
+  function onInput(value: string) {
+    emit('update:modelValue', value);
+  }
+
+  /**
+   * 基础格式检查（非权威）：空串=未配置（宽松骨架协议落 NULL；MQTT/HTTP 后端按空对象
+   * 参与校验、必含字段即拒，前置同口径提示）；非空时要求可解析为 JSON 对象、无凭据键、
+   * 协议必含键齐全；HTTP reportPath 以 / 开头。返回错误文案，''=通过。
+   */
+  function basicFormatCheck(text: string, protocolType: string): string {
+    if (!text || !text.trim()) {
+      const required = REQUIRED_KEYS[protocolType] || [];
+      if (required.length > 0) {
+        return `该协议必含字段 ${required.join('/')}（conn_config 不能为空，后端按空对象校验即拒）`;
+      }
+      return '';
+    }
+    let parsed: unknown;
+    try {
+      parsed = JSON.parse(text);
+    } catch (e) {
+      return `不是合法 JSON：${(e as Error).message}`;
+    }
+    if (parsed === null || typeof parsed !== 'object' || Array.isArray(parsed)) {
+      return 'conn_config 必须是 JSON 对象（{...}）';
+    }
+    const credentialPath = findCredentialKeyPath(parsed as Record<string, unknown>, '');
+    if (credentialPath) {
+      return `检测到疑似凭据字段 ${credentialPath}：凭据不入 conn_config（ADR-0006），后端将拒绝保存`;
+    }
+    if (protocolType === 'HTTP') {
+      const reportPath = (parsed as Record<string, unknown>).reportPath;
+      if (typeof reportPath === 'string' && !reportPath.startsWith('/')) {
+        return 'reportPath 必须以 / 开头（如 /api/v1/report）';
+      }
+    }
+    for (const key of REQUIRED_KEYS[protocolType] || []) {
+      const v = (parsed as Record<string, unknown>)[key];
+      if (v === undefined || v === null || (typeof v === 'string' && !v.trim())) {
+        return `该协议必含字段 ${key}`;
+      }
+    }
+    return '';
+  }
+
+  /** 凭据键名深度扫描（镜像后端：对象逐层 + 数组元素逐个递归，返回首个命中路径） */
+  function findCredentialKeyPath(node: unknown, path: string): string {
+    if (Array.isArray(node)) {
+      for (let i = 0; i < node.length; i++) {
+        const hit = findCredentialKeyPath(node[i], `${path}[${i}]`);
+        if (hit) {
+          return hit;
+        }
+      }
+      return '';

+const CONFLICT_KEYWORDS = ['乐观锁', '已被修改', '已被他人', '记录版本', 'version conflict', 'Data changed'];

+/** 乐观锁冲突提示开关（iot 域无 @Version 字段——登记不触发，brief 决策 3） */

+/**
+ * 写操作 catch 分支统一接入：命中乐观锁特征词时弹出刷新提示；始终原样上抛。
+ * @param error 响应拦截器 reject 的 AxiosResponse（body 在 error.data）或任意错误
+ */
+export function handleIotConflictError(error: unknown): unknown {
+  if (!IOT_OPTIMISTIC_LOCK_ENABLED) {
+    return error;
+  }
+  const body = (error as { data?: { message?: string; msg?: string } })?.data;
+  const text = `${body?.message || ''}${body?.msg || ''}`.toLowerCase();
+  if (CONFLICT_KEYWORDS.some((k) => text.includes(k.toLowerCase()))) {

+      okText: '知道了',
+    });
+  }
