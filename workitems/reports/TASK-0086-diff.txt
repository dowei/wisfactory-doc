+ * TASK-0086 · 试算求值引擎（纯计算沙箱）。
+ *
+ * <p><b>同引擎同口径</b>（brief D2 核心）：所有求值经真算的
+ * {@link ArithmeticEvaluator#evaluate(String, ArithmeticEvaluator.Resolver,
+ * ArithmeticEvaluator.FunctionResolver)}；原生算子经 {@link OperatorEvaluator} → DeltaOperator。
+ * 本类只做「请求数据 → resolver/functionResolver 适配 + 缺数收集 + 窗口逐点驱动」，

+ * （内部 baselines 是实例状态），不共享在线算子状态；不注入 CalcStateStore / JdbcTemplate /
+ * 事件发布 / 调度器，不查真库/Redis。

+ * @date 2026-11-24

+            Pattern.CASE_INSENSITIVE);

+        this(OperatorEvaluator::new);

+    /** 统一入口：window 存在 → 窗口曲线；否则单点求值 */
+    public TrialResponse evaluateSingle(TrialRequest req) {
+        if (req == null || req.formula() == null || req.formula().isBlank()) {
+            return TrialResponse.reject("公式空白（试算须提供 formula）");
+        }
+        FormulaExpressionValidator.ExpressionCheck check = FormulaExpressionValidator.validate(req.formula());
+        if (!check.ok()) {
+            return TrialResponse.reject(check.error());
+        }
+        if (req.window() != null) {
+            return evaluateCurve(req, check.identifiers());
+        }
+        return evaluatePoint(req, req.atTime(), check.identifiers());
+    }

+    // ---------- 单点 ----------
+
+    private TrialResponse evaluatePoint(TrialRequest req, Long atTime, List<String> identifiers) {
+        Map<String, Object> snapshot = snapshotAt(req, atTime, false);
+        long pointTs = atTime == null ? Long.MAX_VALUE : atTime;
+        EvalContext ctx = new EvalContext(req, operatorFactory.get(), identifiers, snapshot, pointTs);
+        try {
+            double value = ArithmeticEvaluator.evaluate(req.formula(), ctx::resolve, ctx::callFunction);
+            if (!ctx.missing().isEmpty()) {
+                return TrialResponse.failure(List.copyOf(ctx.missing()),
+                        "缺数: " + String.join(", ", ctx.missing()) + "（不以 0 代入）");
+            }
+            return TrialResponse.ok(value, ctx.lastQuality());
+        } catch (RuntimeException e) {
+            return TrialResponse.failure(List.copyOf(ctx.missing()), reasonOf(e));
+        }
+    }
+
+    // ---------- 窗口曲线 ----------
+
+    private TrialResponse evaluateCurve(TrialRequest req, List<String> identifiers) {
+        TrialRequest.TrialWindow w = req.window();
+        if (w.stepMs() <= 0 || w.end() < w.start()) {
+            return TrialResponse.reject("window 参数非法（stepMs>0 且 end>=start）");
+        }
+        List<TrialResponse.CurvePoint> curve = new ArrayList<>();
+        for (long ts = w.start(); ts <= w.end(); ts += w.stepMs()) {
+            // 每点新建算子实例——曲线各点基线互不污染（沙箱隔离）
+            EvalContext ctx = new EvalContext(req, operatorFactory.get(), identifiers, snapshotAt(req, ts, true), ts);
+            try {
+                double value = ArithmeticEvaluator.evaluate(req.formula(), ctx::resolve, ctx::callFunction);
+                if (ctx.missing().isEmpty()) {
+                    curve.add(new TrialResponse.CurvePoint(ts, value, ctx.lastQuality(), List.of(), ""));
+                } else {
+                    curve.add(new TrialResponse.CurvePoint(ts, null, TrialResponse.QUALITY_MISSING,
+                            List.copyOf(ctx.missing()),
+                            "缺数: " + String.join(", ", ctx.missing()) + "（不以 0 代入）"));
+                }
+            } catch (RuntimeException e) {
+                curve.add(new TrialResponse.CurvePoint(ts, null, TrialResponse.QUALITY_MISSING,
+                        List.copyOf(ctx.missing()), reasonOf(e)));
+            }
+            if (w.end() - ts < w.stepMs()) {
+                break; // 防长窗溢出
+            }
+        }
+        return TrialResponse.curve(curve);
+    }
+
+    /** 取 t 时刻可见的样本快照：variables 注入值 + history 中 ts<=t 的最新样本 */
+    private static Map<String, Object> snapshotAt(TrialRequest req, Long atTime, boolean curvePoint) {
+        long t = atTime != null ? atTime : Long.MAX_VALUE;
+        Map<String, Object> snapshot = new LinkedHashMap<>();
+        if (req.variables() != null) {
+            snapshot.putAll(req.variables());
+        }
+        for (Map.Entry<String, List<TrialRequest.VariableSample>> e : req.history().entrySet()) {
+            TrialRequest.VariableSample best = null;
+            if (e.getValue() != null) {
+                for (TrialRequest.VariableSample s : e.getValue()) {
+                    // 单点 atTime/缺省：取 <=t 最新样本；曲线点严格匹配 ts（缺数段不沿用旧值冒充）
+                    boolean visible = curvePoint ? s != null && s.ts() == t : s != null && s.ts() <= t;
+                    if (visible && (best == null || s.ts() >= best.ts())) {
+                        best = s;
+                    }
+                }
+            }
+            if (best != null) {
+                snapshot.put(e.getKey(), best); // history 覆盖同名 variables（时序数据优先）
+            }
+        }
+        return snapshot;
+    }
+
+    private static String reasonOf(RuntimeException e) {
+        String msg = e.getMessage();
+        return (msg == null || msg.isBlank()) ? e.getClass().getSimpleName() : msg;
+    }
+
+    // ---------- 求值上下文（resolver / functionResolver 适配器） ----------
+
+    private final class EvalContext {
+        private final TrialRequest req;
+        private final OperatorEvaluator operators;
+        private final Map<String, Object> snapshot;
+        private final Set<String> missing = new LinkedHashSet<>();
+        private final String deltaKey;
+        private final long currentTs;
+        private int lastQuality = 0;
+
+        EvalContext(TrialRequest req, OperatorEvaluator operators,
+                    List<String> identifiers, Map<String, Object> snapshot, long currentTs) {
+            this.req = req;
+            this.operators = operators;
+            this.snapshot = snapshot;
+            this.currentTs = currentTs;
+            this.deltaKey = extractDeltaKey(req.formula());
+            for (String id : identifiers) {
+                // history 中存在的标识符视为「已提供」（DELTA 从 history 取数，不在 variables 里）；
+                // 其余未提供的标识符 = 缺数（RD-缺数语义）
+                if (!snapshot.containsKey(id) && !req.history().containsKey(id)) {
+                    missing.add(id);
+                }
+            }
+        }
+
+        Set<String> missing() {
+            return missing;
+        }
+
+        int lastQuality() {
+            return lastQuality;
+        }
+
+        /** 标识符解析：number / sample / 缺失(null→抛异常由求值器隔离) */
+        Double resolve(String identifier) {
+            Object v = snapshot.get(identifier);
+            if (v == null) {
+                missing.add(identifier);
+                return null; // ArithmeticEvaluator 抛「无值输入」→ 转缺数响应
+            }
+            if (v instanceof Number n) {
+                return n.doubleValue();
+            }
+            if (v instanceof TrialRequest.VariableSample s) {
+                lastQuality = s.quality();
+                return s.value();
+            }
+            if (v instanceof Map<?, ?> m && m.get("value") instanceof Number n) {
+                Object q = m.get("quality");
+                lastQuality = q instanceof Number nq ? nq.intValue() : 0;
+                return n.doubleValue();
+            }
+            missing.add(identifier);
+            return null;
+        }
+
+        /**
+         * 函数求值：supports 交给 OperatorEvaluator；DELTA 按已提取的 key 从 history 取一窗样本，
+         * 在本次求值上下文内按 ts 升序逐点调用原生 delta(...) 得到差分（与真算同引擎口径）；
+         * INTEGRAL 手写已被白名单拒绝，其余函数返回 null → 求值器抛异常。
+         */
+        Double callFunction(String name, double arg) {
+            if (!operators.supports(name) || deltaKey == null) {
+                return null;
+            }
+            List<TrialRequest.VariableSample> samples = req.history().get(deltaKey);
+            if (samples == null || samples.isEmpty()) {
+                missing.add(deltaKey);
+                return null;
+            }
+            List<TrialRequest.VariableSample> sorted = new ArrayList<>();
+            for (TrialRequest.VariableSample s : samples) {
+                if (s != null && s.ts() <= currentTs) {
+                    sorted.add(s);
+                }
+            }
+            sorted.sort(java.util.Comparator.comparingLong(TrialRequest.VariableSample::ts));
+            if (sorted.isEmpty()) {
+                missing.add(deltaKey);
+                return null;
+            }
+            ValueSample result = null;
+            for (TrialRequest.VariableSample s : sorted) {
+                result = operators.delta(deltaKey,
+                        new ValueSample(s.value(), s.quality(), s.ts()), null, null, false);
+            }
+            if (result == null) {
+                missing.add(deltaKey);
+                return null;
+            }
+            lastQuality = result.quality();
+            return result.value();
+        }
+    }
+
+    /** 从已通过白名单的公式中提取 DELTA(key) 的 key（只识别参数名，不求值） */
+    private static String extractDeltaKey(String formula) {
+        Matcher m = DELTA_ARG.matcher(formula);
+        return m.find() ? m.group(1) : null;
+    }

+    /** 窗口曲线单点：逐点独立标注缺数（缺数段 value=null、missing 非空） */
+    public record CurvePoint(long ts, Double value, int quality, List<String> missing, String reason) {}
+
+    /** 与真算同口径的版本标识（算术求值器 + 原生算子桥同版本；随 calc 构建演进可更新） */
+    public static final String ENGINE_VERSION = "calc-engine/1.0.0/trial-v1";
+
+    /** 质量码：计算失败 / 缺数（CONV-02 码 6） */

+    public TrialResponse {

+        reason = reason == null ? "" : reason;
+    }
+
+    /** 单点成功 */
+    public static TrialResponse ok(double value, int quality) {
+        return new TrialResponse(value, quality, List.of(), "", null, ENGINE_VERSION, false);
+    }
+
+    /** 单点缺数/求值失败（value=null，绝不以 0 冒充；HTTP 200 + quality=6） */
+    public static TrialResponse failure(List<String> missing, String reason) {
+        return new TrialResponse(null, QUALITY_MISSING, missing, reason, null, ENGINE_VERSION, false);
+    }
+
+    /** 语法/白名单拒绝（如手写 INTEGRAL；HTTP 400） */
+    public static TrialResponse reject(String reason) {
+        return new TrialResponse(null, QUALITY_MISSING, List.of(), reason, null, ENGINE_VERSION, true);
+    }
