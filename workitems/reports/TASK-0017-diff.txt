+ * 内部认证检查器·配置静态令牌 + IP 白名单实现（TASK-0017 占位裁决，brief「内部认证裁决
+ * （占位）」）
+ *
+ * <p><b>静态令牌</b>：{@code ot.gateway.internal-token} 经环境变量注入（yml 占位键为
+ * 本任务唯一新增键），比对请求头 {@code X-Internal-Token}——常量时间比较（MessageDigest

+ * <p><b>IP 白名单</b>：{@code ot.gateway.internal-ip-whitelist}（默认 127.0.0.1，逗号分隔，
+ * 代码默认值——brief 只授权 yml 新增 internal-token 一个键，故白名单键经 @Value 默认值
+ * 交付，部署时可经环境变量 OT_GATEWAY_INTERNAL_IP_WHITELIST 覆盖）。来源 IP 取
+ * {@link OtIpUtils#getClientIp}（受 ot.ip.behind-proxy 开关控制，防伪造转发头）。
+ *
+ * <p>校验在拦截器层触发（{@link InternalAuthInterceptor}）；完整网关令牌链归 0019/0020。
+ *
+ * @author dowei
+ * @date 2026-11-22
+ */
+@Slf4j
+@Component
+public class StaticTokenInternalAuthChecker implements InternalAuthChecker {
+
+    /** 内部令牌请求头（brief 裁决：header X-Internal-Token 比对） */
+    public static final String TOKEN_HEADER = "X-Internal-Token";
+
+    /** 令牌配置键（application.yml 唯一新增键，env 注入） */
+    private static final String TOKEN_CONFIG_KEY = "${ot.gateway.internal-token:}";
+
+    /** IP 白名单配置键（代码默认 127.0.0.1；部署经环境变量覆盖） */
+    private static final String WHITELIST_CONFIG_KEY = "${ot.gateway.internal-ip-whitelist:127.0.0.1}";
+
+    @Value(TOKEN_CONFIG_KEY)
+    private String configuredToken;
+
+    @Value(WHITELIST_CONFIG_KEY)
+    private String ipWhitelistRaw;
+
+    @Override
+    public void check(HttpServletRequest request) {
+        checkToken(request);
+        checkIpWhitelist(request);
+    }
+
+    /** 令牌校验：配置为空 fail-closed；头缺失/不匹配 401（常量时间比较） */
+    private void checkToken(HttpServletRequest request) {
+        if (configuredToken == null || configuredToken.isBlank()) {
+            log.warn("[GATEWAY-INTERNAL] internal token not configured, reject all (fail-closed) ip={}",
+                    OtIpUtils.getClientIp(request));
+            throw new BusinessException(ResultCode.UNAUTHORIZED.getCode(), "内部令牌未配置，回写接口拒绝访问（fail-closed）");
+        }
+        String presented = request.getHeader(TOKEN_HEADER);
+        if (presented == null || presented.isBlank()
+                || !MessageDigest.isEqual(configuredToken.getBytes(StandardCharsets.UTF_8),
+                        presented.getBytes(StandardCharsets.UTF_8))) {
+            log.warn("[GATEWAY-INTERNAL] invalid internal token, ip={}", OtIpUtils.getClientIp(request));
+            throw new BusinessException(ResultCode.UNAUTHORIZED.getCode(), "内部令牌无效");
+        }
+    }
+
+    /** IP 白名单校验：来源 IP ∈ 白名单（逗号分隔，去空白），白名单外 403 */
+    private void checkIpWhitelist(HttpServletRequest request) {
+        Set<String> whitelist = parseWhitelist(ipWhitelistRaw);
+        String clientIp = OtIpUtils.getClientIp(request);
+        if (!whitelist.contains(clientIp)) {
+            log.warn("[GATEWAY-INTERNAL] ip not in internal whitelist, ip={} whitelist={}", clientIp, whitelist);
+            throw new BusinessException(ResultCode.FORBIDDEN.getCode(), "来源 IP 不在内部接口白名单");
+        }
+    }
+
+    /** 解析逗号分隔白名单（去空白、小写归一，IPv6 场景友好） */
+    private Set<String> parseWhitelist(String raw) {
+        Set<String> result = new HashSet<>();
+        if (raw == null || raw.isBlank()) {
+            return result;
+        }
+        Arrays.stream(raw.split(","))
+                .map(String::trim)
+                .filter(s -> !s.isEmpty())
+                .map(s -> s.toLowerCase(Locale.ROOT))
+                .forEach(result::add);
+        return result;
+    }
+}

+        // sys_tenant 非多租户插件覆盖表（租户主数据），显式查询；逻辑删行由 @TableLogic 过滤。
+        SysTenant tenant = tenantMapper.selectOne(new LambdaQueryWrapper<SysTenant>()
+                .eq(SysTenant::getTenantCode, form.getTenantCode()));
+        if (tenant == null) {
+            throw paramError("租户不存在: " + form.getTenantCode());
+        }
+        if (tenant.getStatus() == null || tenant.getStatus() != 1) {
+            throw paramError("租户已禁用: " + form.getTenantCode());
+        }

+        // 批量定位网关（fail-closed：任一未知编码整批拒绝，不半批落库）。无 Sa-Token 会话
+        // → 多租户插件跳过注入，显式 (tenant_id, gateway_code) 条件即定位语义（与物理唯一
+        // 索引作用域精确一致）；异常会话场景下插件条件与本条件叠加只会更严（防御纵深）。

+                .map(GatewayHeartbeatItem::getGatewayCode)

+                        .select(IotGateway::getId, IotGateway::getGatewayCode)
+                        .eq(IotGateway::getTenantId, tenant.getId())
+                        .in(IotGateway::getGatewayCode, codes))
+                .stream()
+                .collect(Collectors.toMap(IotGateway::getGatewayCode, Function.identity(),
+                        (a, b) -> a, LinkedHashMap::new));

+                .filter(code -> !gateways.containsKey(code))
+                .collect(Collectors.toList());
+        if (!missing.isEmpty()) {
+            throw paramError("网关编码不存在或已删除: " + missing);
+        }

+        // 不进 SET（MP 跳过），更新 WHERE 显式 (id, tenant_id, gateway_code) 定位。

+                    .eq(IotGateway::getTenantId, tenant.getId())
+                    .eq(IotGateway::getGatewayCode, item.getGatewayCode()));
+        }

+    }

+        SaCheckPermission classLevel = clazz.getAnnotation(SaCheckPermission.class);
+        if (classLevel != null) {
+            perms.addAll(Arrays.asList(classLevel.value()));
+        }
+        for (Method method : clazz.getDeclaredMethods()) {
+            SaCheckPermission annotation = method.getAnnotation(SaCheckPermission.class);
+            if (annotation != null) {
+                perms.addAll(Arrays.asList(annotation.value()));
+            }
+        }
+        return perms;
+    }

+        Matcher matcher = MENU_ROW_PATTERN.matcher(sqlText);
+        while (matcher.find()) {
+            rows.put(Long.parseLong(matcher.group(1)), matcher.group(3));
+        }
+        return rows;
+    }

+        assertTrue(Files.exists(script), "升级脚本不存在：" + script);
+        return Files.readString(script, StandardCharsets.UTF_8);
+    }
