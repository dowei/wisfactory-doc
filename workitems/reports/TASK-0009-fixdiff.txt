+        public void characters(char[] ch, int start, int length) {

+            }
+        }

+        public void endElement(String uri, String localName, String qName) {

+                default:
+                    break;

+        private void setCell(String value) {
+            if (cellColumn < 0) {
+                return;
+            }
+            // 空串归一 null：hutool 对 null 值写空 inlineStr（<t></t>），Excel 对空单元格同样可能留空串——
+            // 「显式空串」与「缺格」在导入语义上等价（五列均无空串合法值），统一为 null 后
+            // isSkippable（全空行）与 MISSING_REQUIRED（必填缺失）判定对两种编码口径行为一致
+            if (value != null && value.isEmpty()) {
+                value = null;
+            }
+            while (cells.size() <= cellColumn) {
+                cells.add(null);
+            }
+            cells.set(cellColumn, value);
+            if (cellColumn > maxCol) {
+                maxCol = cellColumn;
+            }
+        }
+
+        private String cellValue(String type, String v) {
+            if (v == null || v.isEmpty()) {
+                return null;
+            }
+            if ("s".equals(type)) {
+                // 共享字符串：v 为索引
+                int idx;
+                try {
+                    idx = Integer.parseInt(v.trim());
+                } catch (NumberFormatException e) {
+                    return null;
+                }
+                return idx >= 0 && idx < sharedStrings.size() ? sharedStrings.get(idx) : null;
+            }
+            if ("b".equals(type)) {
+                return "1".equals(v.trim()) ? "true" : "false";
+            }
+            if ("str".equals(type) || "e".equals(type)) {
+                return v; // 公式缓存字符串 / 错误值原样
+            }
+            return numericValue(v); // "n"（缺省类型）
+        }
+
+        /** 整数值（含 1002.0 形态）转十进制整数字符串；非整数保留原文；非法数值原样兜底。 */
+        private static String numericValue(String v) {
+            try {
+                double d = Double.parseDouble(v.trim());
+                if (Double.isFinite(d) && d == Math.rint(d) && Math.abs(d) < 9.0E15) {
+                    return String.valueOf((long) d);
+                }
+                return v;
+            } catch (NumberFormatException e) {
+                return v;
+            }
+        }
+
+        /** Excel 单元格引用（如 B12）→ 0-based 列号；无引用返回 -1。 */
+        private static int colIndex(String ref) {
+            if (ref == null || ref.isEmpty()) {
+                return -1;
+            }
+            int idx = 0;
+            for (int i = 0; i < ref.length(); i++) {
+                char ch = ref.charAt(i);
+                if (ch >= 'A' && ch <= 'Z') {
+                    idx = idx * 26 + (ch - 'A' + 1);
+                } else if (ch >= 'a' && ch <= 'z') {
+                    idx = idx * 26 + (ch - 'a' + 1);
+                } else {
+                    break; // 行号数字段开始，列字母段结束
+                }
+            }
+            return idx - 1; // A → 0
+        }
+
+        private void finishRow() {
+            long physical = rowNumber; // 无 r 属性的行已在 startElement 按「上一行 + 1」递增
+            if (physical <= 1) {
+                return; // 物理第 1 行 = 表头
+            }
+            DictImportRow row = new DictImportRow();
+            row.setRowIndex((int) physical);
+            row.setDataCode(trim(cells, 0));
+            row.setDataName(trim(cells, 1));
+            row.setCategory(trim(cells, 2));
+            row.setDataType(trim(cells, 3));
+            row.setDefaultUnit(trim(cells, 4));
+            rows.add(row);
+            if (!row.isSkippable() && ++dataCount > maxDataRows) {
+                // I-3 fail fast：RuntimeException 经 JAXP XMLReader.parse 原样透传出 parse()
+                throw new DictImportRowLimitException(dataCount);
+            }
+        }
+
+        private static String trim(List<String> cells, int index) {
+            String value = index < cells.size() ? cells.get(index) : null;
+            return value == null ? null : value.trim();
+        }
+    }

+    private List<DictImportRow> dataRowsOf(List<DictImportRow> rows) {
