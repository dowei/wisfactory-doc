+// OnDeadLetter 实现 DeadLetterSink：计数 → 采样告警 → 明细入环形缓冲 → 链式下游。
+func (c *DeadLetterCounter) OnDeadLetter(reason DeadLetterReason, tenant, batchID string, p *message.Point) {
+	k := deadKey{tenant: tenant, reason: reason}
+
+	c.mu.Lock()
+	// 有界 map：新键且容量满时按 FIFO 淘汰最旧键（order 恒为「未淘汰键的首入序」，
+	// 重复计数不改变位置；淘汰即整体前移一位——容量有界，前移开销可接受）。
+	if _, exists := c.counts[k]; !exists && len(c.order) >= c.opts.Cap {
+		old := c.order[0]
+		c.order = c.order[1:]
+		delete(c.counts, old)
+	}
+	c.counts[k]++
+	c.order = appendOrder(c.order, k)
+	c.ring[c.ringNext] = DeadLetterRecord{Reason: reason, Tenant: tenant, BatchID: batchID, Point: p}
+	c.ringNext = (c.ringNext + 1) % len(c.ring)
+	if c.ringFilled < len(c.ring) {
+		c.ringFilled++
+	}
+	c.total++
+	n := c.counts[k]
+	c.mu.Unlock()
+
+	// 告警采样：首次 + 每 N 条（防风暴；固定格式含租户/原因/计数）。
+	if n == 1 || n%int64(c.opts.WarnEvery) == 0 {
+		code := ""
+		if p != nil {
+			code = p.PointCode
+		}
+		c.opts.Logger.Printf("pipeline: [dead-letter] tenant=%s reason=%s count=%d pointCode=%s batchId=%s",

+	}

+	}

+// 淘汰序反映「首次入表序」，重复计数不改变位置）。
+func appendOrder(order []deadKey, k deadKey) []deadKey {
+	for i := range order {

+	}

+	t.Helper()
+	exec := tdengine.NewFakeExecutor(0)
+	dead := &recordingSink{}
+	tag := &tagRecorder{}
+	snapCache := preloadSnapshotJSON(t, snapJSON)
+	deps := pipeline.Deps{
+		Consumer:  newIdleChan(),
+		Parser:    message.Parser{},
+		Snapshots: snapCache,
+		Batches:   pipeline.NewMemoryBatchDeduper(65536),
+		Points:    pipeline.NewMemoryPointDeduper(65536),
+		Dead:      dead,
+		TimeTag:   tag.tag,
+		Now:       clockAt(frozenNowMs),
+		Writer:    &tdengine.Writer{Ex: exec},
+	}
+	ptCache := cache.NewMemoryPointCache()
+	deps.Cache = ptCache
+	cfg := pipeline.Config{BatchSize: 1, FlushInterval: 10 * time.Millisecond}
+	if mutate != nil {
+		mutate(&cfg, &deps)
+	}
+	p, err := pipeline.Assemble(cfg, deps)
+	if err != nil {
+		t.Fatalf("Assemble error: %v", err)
+	}
+	ctx, cancel := context.WithCancel(context.Background())
+	p.StartFlusher(ctx)
+	return &valPipe{t: t, pipe: p, snapCache: snapCache, exec: exec, ptCache: ptCache, dead: dead, tag: tag, cancel: cancel}
+}

+// submitBatch 同步提交一批（tenant 固定 group_a；points 为点位对象 JSON 片段）。
+func (s *valPipe) submitBatch(batchID string, points ...string) pipeline.SubmitResult {
+	var sb strings.Builder
+	fmt.Fprintf(&sb, `{"tenant":"group_a","gateway":"GW_F01_01","batchId":%q,"sentAt":%d,"points":[`, batchID, frozenNowMs)
+	for i, p := range points {
+		if i > 0 {
+			sb.WriteByte(',')
+		}
+		sb.WriteString(p)
+	}

+type TimeVerdict struct {
+	// Action 裁决动作。
+	Action TimeAction
+	// Ts 落库时间戳（毫秒）：Keep=原始 ts；Correct=接收时间（纠偏）；Reject 时无意义。

+	// Reason 拒绝原因码（仅 Reject 非空：ts_overflow / ts_too_old）。
+	Reason DeadLetterReason

+	maxAge       time.Duration
+}

+// judge 对消息点位 ts（毫秒）做三分裁决。now 为消息接收时间（处理时刻）。
+func (tp timePolicy) judge(now time.Time, tsMs int64) TimeVerdict {
+	ts := time.UnixMilli(tsMs)
+	switch {
+	case ts.After(now.Add(tp.futureReject)):
+		// 路径 1a：越未来上界 → 拒绝（ts_overflow）。
+		return TimeVerdict{Action: TimeReject, Reason: DeadLetterTsOverflow}
+	case ts.Before(now.Add(-tp.maxAge)):
+		// 路径 1b：越最大年龄 → 拒绝（ts_too_old）。
+		return TimeVerdict{Action: TimeReject, Reason: DeadLetterTsTooOld}
+	case ts.After(now):
+		// 路径 2：未来窄带（now < ts ≤ now+futureReject）→ 纠偏为接收时间并打标。
+		return TimeVerdict{Action: TimeCorrect, Ts: now.UnixMilli()}
+	default:
+		// 路径 3：补传豁免（now-maxAge ≤ ts ≤ now）→ 原始 ts 落库。
+		return TimeVerdict{Action: TimeKeep, Ts: tsMs}
+	}
+}

+//
+// 契约依据：
+//   - 统一消息模型 v1.1.0 §3.2（设计文档 §3.2 p24 逐字）：「quality 由网关能判断的先填
+//     （如设备异常码映射），超限类质量码由接入服务按点位配置的 min/max 二次判定后覆盖写入」；
+//   - 配置快照 Schema v1.0.1 点位对象：minValue/maxValue（DECIMAL(20,6)，可空）——
+//     「低于下限只打质量码标记、不丢弃；接入服务按此二次判定覆盖质量码」；
+//   - CONV-02（本包枚举，冻结）：1=超上限（OverMax）、2=超下限（UnderMin）。
+//     注：TASK-0028 brief 行文「value<min→1、value>max→2」与冻结枚举文字相反，
+//     以冻结枚举为准（报告已登记偏离）；「量程外打码 1/2 不丢弃」语义两侧一致。
+//   - ADR-0009 TD-0048 边界注记：人工补录/修正走管理面接口（质量码 MANUAL=5 记操作审计），
+//     ingest 通道为纯机器上报——ingest 不提供任何人工补录入口，本判定不产生 MANUAL 码，
+//     网关上报携带的 MANUAL 码按「网关能判断的先填」透传存储。
+package quality
+
+// MergeRange 按点位有效范围（min/max，可空）对上报值做量程二次判定，并按 §3.2 口径
+// 合并网关先填的质量码：
+//
+//   - value > maxValue → OverMax(1)；value < minValue → UnderMin(2)（超限类接入覆盖写，
+//     无论网关先填何值）；
+//   - 边界值（==min / ==max）不算越界；范围内值保留网关质量码（非超限类不覆盖）；
+//   - minValue/maxValue 为 nil（快照点位未配置该边界）时跳过该侧判定；
+//     双侧均缺失时整跳过量程判定，网关质量码原样透传；
+//   - 网关质量码越出 0-10 存储口径时防御收敛为 Good(0)（CONV-02 唯一存储口径，
+//     越域码不得写入时序行）；
+//   - 只产出质量码，不触碰数值本身——原值保留、不丢弃、不假 0（brief 裁决 1）。
+func MergeRange(gatewayQuality int, value float64, minValue, maxValue *float64) int {
+	// 超限覆盖写：优先级高于网关码与码域收敛（越域码遇上越限值仍打越限码）。
+	if maxValue != nil && value > *maxValue {
+		return OverMax
+	}
+	if minValue != nil && value < *minValue {
+		return UnderMin
+	}
+	// 非超限：网关码保留；越域防御收敛（CONV-02 0-10 唯一存储口径）。
+	if !Valid(gatewayQuality) {
+		return Good
+	}
+	return gatewayQuality
+}
