+ * 公式引用图循环依赖检测器（TASK-0036，brief 决策 8——DFS 纯函数件）
+ *
+ * <p><b>检测范围（brief 口径）</b>：单实例内 SELF/绝对键闭环 + 跨实例 PARENT 边闭环。
+ * 两类环在依赖图上同构——保存/生成链把 SELF./PARENT./绝对键引用统一落为
+ * iot_data_item_ref（ref_type=1）数据项边（ref_item_id 尽力回填，跨设备绝对引用指向
+ * 未建实例时为 NULL——无目标即无环，DFS 跳过），故本检测器只看一张
+ * {@code 数据项 id → 被引用数据项 id 集} 的有向图，实例边界不特殊处理（跨实例 PARENT
+ * 边闭环天然覆盖）。
+ *
+ * <p><b>调用方语义（saveFormula）</b>：候选项的新引用边「替换」其 DB 旧边后做可达性
+ * DFS——从候选项出发沿边游走，回到候选项即成环，返回环上项 id 列表（从候选项起、按
+ * 引用方向排序、起点不重复收尾）；调用方把列表拼进 4001「循环依赖」文案。环在别处但
+ * 不可达候选（下游环）不拒存——本接口口径是「本项可达自身」（brief：重建 ref 边后 DFS
+ * 本项可达自身 → 4001）。
+ *
+ * <p>算法：onPath 栈 + 全局 done 集（探索完毕剪枝），最坏 O(V+E)；公式链深度受
+ * FEAT-0012 嵌套深度上限约束（一期 DFS 递归深度=引用链长度，量级为配置面小图）。
+ *
+ * @author dowei
+ * @date 2026-11-22
+ */
+public final class FormulaCycleDetector {
+
+    private FormulaCycleDetector() {
+    }
+
+    /**
+     * 从 start 出发沿 edges 游走，检测是否存在回到 start 的路径。
+     *
+     * @param edges 有向图（数据项 id → 被引用数据项 id 集；无出边的项可缺键）
+     * @param start 候选项 id（本次公式保存的目标项）
+     * @return 环上项 id 列表（[start, …] 按引用方向，start 不重复收尾）；无环返回 null
+     */
+    public static List<Long> findCycle(Map<Long, Set<Long>> edges, long start) {
+        Deque<Long> path = new ArrayDeque<>();
+        Set<Long> onPath = new HashSet<>();
+        Set<Long> done = new HashSet<>();
+        if (dfs(start, start, edges, path, onPath, done)) {
+            return new ArrayList<>(path);
+        }
+        return null;
+    }
+
+    /** DFS：path/onPath 为当前栈，done 为已探索无解节点（剪枝防重复子树） */
+    private static boolean dfs(long current, long start, Map<Long, Set<Long>> edges,
+                               Deque<Long> path, Set<Long> onPath, Set<Long> done) {
+        if (done.contains(current)) {
+            return false;
+        }
+        path.addLast(current);
+        onPath.add(current);
+        for (long next : edges.getOrDefault(current, Set.of())) {
+            if (next == start) {
+                return true;
+            }
+            if (!onPath.contains(next) && dfs(next, start, edges, path, onPath, done)) {
+                return true;
+            }
+        }
+        path.removeLast();
+        onPath.remove(current);
+        done.add(current);
+        return false;
+    }

+    /** 一期封闭不开放函数（REQ-0016 非目标；仅随 REQ-0019 能源计量预置公式提供） */
+    public static final String CLOSED_INTEGRAL = "INTEGRAL";
+
+    /** 函数调用点形态：标识符紧跟左括号（容忍空白）；点分全键引用不含括号、天然不匹配 */
+    private static final Pattern FUNCTION_CALL = Pattern.compile("([A-Za-z_][A-Za-z0-9_]*)\\s*\\(");

+    }

+    /**
+     * 扫描公式中的函数调用点，返回首个违规描述；无违规返回 null。
+     *
+     * <p>INTEGRAL 命中给专门文案（「一期不开放」语义，区别于普通集合外函数）；扫描按公式
+     * 出现序取首个违规，便于前端定位（占位校验器一期只报首个——完整 issues 定位随
+     * FEAT-0012 validate 接口交付）。
+     *
+     * @param formula 展开后（裸键形态）公式，可 null/空白（视为无函数调用，放行）
+     * @return 违规描述（含函数名与白名单/不开放语义）；无违规 null
+     */
+    public static String findViolation(String formula) {
+        if (formula == null || formula.isBlank()) {
+            return null;
+        }
+        // 逐调用点扫描（保序）：先命中先报——INTEGRAL 优先级按出现序，不做全局重排
+        LinkedHashSet<String> seen = new LinkedHashSet<>();
+        Matcher matcher = FUNCTION_CALL.matcher(formula);
+        while (matcher.find()) {
+            String name = matcher.group(1);
+            if (!seen.add(name)) {
+                continue;
+            }
+            if (CLOSED_INTEGRAL.equals(name)) {
+                return "公式函数 INTEGRAL 一期不开放（仅随 REQ-0019 能源计量预置公式提供），保存拒绝";
+            }
+            if (!ALLOWED.contains(name)) {

+                        + "真实语法校验归 FEAT-0012）";
+            }
+        }
+        return null;
+    }

+    private LambdaUpdateWrapper<IotDataItem> capturedWrapper() {
+        LambdaUpdateWrapper<IotDataItem> uw = (LambdaUpdateWrapper<IotDataItem>) updateWrapperCaptor.getValue();
+        uw.getSqlSet();
+        uw.getSqlSegment();
+        return uw;
+    }
+
+    private List<String> capturedSetColumns() {
+        return Arrays.stream(capturedWrapper().getSqlSet().split(","))
+                .map(seg -> seg.substring(0, seg.indexOf('=')))
+                .toList();
+    }
+
+    private Object setParamValue(String column) {
+        LambdaUpdateWrapper<IotDataItem> uw = capturedWrapper();
+        for (String seg : uw.getSqlSet().split(",")) {
+            String col = seg.substring(0, seg.indexOf('='));
+            if (col.equals(column)) {
+                Matcher matcher = Pattern.compile("(MPGENVAL\\d+)").matcher(seg);
+                assertTrue(matcher.find(), () -> "sqlSet 段未含参数键: " + seg);
+                return uw.getParamNameValuePairs().get(matcher.group(1));
+            }
+        }
+        throw new AssertionError("sqlSet 中不存在列: " + column);
+    }
