+    @Override
+    public List<IotInstanceTreeVO> selectInstanceTree() {
+        VisibleScope scope = dataPermService.resolve(SecurityUtils.getUserId());
+
+        // createByOnly：实例树属数据查询语义 → 空树（ADR-0016 裁决 3/4：仅本人对数据类=空集；
+        // 可见集为空（无锚点命中/超限截断后空集）同口径短路，不查库）
+        if (!scope.allTenant() && (scope.createByOnly() || scope.instanceIds().isEmpty())) {
+            return List.of();
+        }
+
+        // 全树一次查出后按可见集内存过滤（brief 口径；多租户条件由插件注入，deleted 由
+        // @TableLogic 运行时过滤）；同级 sort 升序、id 升序保稳定序
+        List<IotModelInstance> all = this.list(new LambdaQueryWrapper<IotModelInstance>()
+                .orderByAsc(IotModelInstance::getSort)
+                .orderByAsc(IotModelInstance::getId));
+        if (all.isEmpty()) {
+            return List.of();
+        }
+
+        List<IotModelInstance> kept;
+        if (scope.allTenant()) {
+            kept = all;
+        } else {
+            // 只保留可见集节点及其到根的路径节点：路径节点仅作骨架展示、无权兄弟节点不返回
+            // （REQ-0002 验收「实例树接口无权节点不返回」）。ancestors 逗号路径整 token 解析
+            // （数据权限组件 expandSubtree 同范式，防字符串前缀误判）
+            Map<Long, IotModelInstance> byId = new LinkedHashMap<>();
+            for (IotModelInstance node : all) {
+                byId.put(node.getId(), node);
+            }
+            Set<Long> keep = new LinkedHashSet<>();
+            for (IotModelInstance node : all) {
+                if (scope.instanceIds().contains(node.getId())) {
+                    keep.add(node.getId());
+                    collectAncestorIds(node.getAncestors(), byId.keySet(), keep);
+                }
+            }
+            if (keep.isEmpty()) {
+                return List.of();
+            }
+            kept = all.stream().filter(node -> keep.contains(node.getId())).collect(Collectors.toList());
+        }
+        return buildTree(kept);
+    }
+
+    /** ancestors 逗号路径 → 存在于当前行的祖先 id 集合（防御性：已删祖先 id 不补骨架） */
+    private void collectAncestorIds(String ancestors, Set<Long> presentIds, Set<Long> keep) {
+        if (ancestors == null || ancestors.isBlank()) {
+            return;
+        }
+        for (String token : ancestors.split(",")) {
+            String trimmed = token.trim();
+            if (trimmed.isEmpty()) {
+                continue;
+            }
+            try {
+                Long ancestorId = Long.valueOf(trimmed);
+                if (presentIds.contains(ancestorId)) {
+                    keep.add(ancestorId);
+                }
+            } catch (NumberFormatException ignored) {
+                // 脏 token（非数字）跳过：骨架缺失不阻断树返回
+            }
+        }
+    }
+
+    /** 组装树：输入已按 sort,id 升序，groupingBy 保序 → 子节点天然有序；父不在场（骨架防御）的节点视作根 */
+    private List<IotInstanceTreeVO> buildTree(List<IotModelInstance> nodes) {
+        Map<Long, List<IotModelInstance>> byParent = new HashMap<>();
+        Set<Long> presentIds = new LinkedHashSet<>();
+        for (IotModelInstance node : nodes) {
+            Long parentId = node.getParentId() == null ? 0L : node.getParentId();
+            byParent.computeIfAbsent(parentId, k -> new ArrayList<>()).add(node);
+            presentIds.add(node.getId());
+        }
+        List<IotInstanceTreeVO> roots = new ArrayList<>();
+        for (IotModelInstance node : nodes) {
+            Long parentId = node.getParentId() == null ? 0L : node.getParentId();
+            if (parentId == 0L || !presentIds.contains(parentId)) {
+                roots.add(assemble(node, byParent));
+            }
+        }
+        return roots;
+    }
+
+    private IotInstanceTreeVO assemble(IotModelInstance node, Map<Long, List<IotModelInstance>> byParent) {
+        IotInstanceTreeVO vo = IotInstanceTreeVO.from(node);
+        List<IotInstanceTreeVO> children = new ArrayList<>();
+        for (IotModelInstance child : byParent.getOrDefault(node.getId(), List.of())) {
+            children.add(assemble(child, byParent));
+        }
+        vo.setChildren(children);
+        return vo;
+    }

+        String[] fieldNames = {"instanceCode", "parentId", "ancestors", "nodeType", "templateId"};
+        Object[] changedValues = {"F01.WS02", 9L, "0,9", 4, 77L};
+        for (int i = 0; i < fieldNames.length; i++) {
+            IotInstanceUpdateForm form = new IotInstanceUpdateForm();
+            form.setId(2L);
+            if (fieldNames[i].equals("instanceCode")) {
+                form.setInstanceCode("F01.WS02");
+            } else if (fieldNames[i].equals("parentId")) {
+                form.setParentId((Long) changedValues[i]);
+            } else if (fieldNames[i].equals("ancestors")) {
+                form.setAncestors((String) changedValues[i]);
+            } else if (fieldNames[i].equals("nodeType")) {
+                form.setNodeType((Integer) changedValues[i]);
+            } else {

+    private IotInstanceTreeVO flattenFind(List<IotInstanceTreeVO> roots, Long id) {
+        for (IotInstanceTreeVO vo : roots) {
+            if (vo.getId().equals(id)) {
+                return vo;
