+/**
+ * 状态规则条件表达式校验器（TASK-0048，brief「Expander 同口径语法」）
+ *
+ * <p><b>与 0013 {@code RelativeRefExpander} 同口径</b>（语法权威照 brief 指定）：相对引用
+ * 以<b>花括号包裹</b>——{@code {SELF.<code>}} / {@code {PARENT.<code>}}（关键字大小写
+ * 不敏感、编码字符集 [A-Za-z0-9_]）；花括号包裹的点分形态按绝对全键引用处理
+ * （{@code {F01.WS01.EQ002.ELEC_CONS}} 剥壳保留键）；两个正则与 Expander 的
+ * RELATIVE_REF / BRACE_REF 同形。<b>本类只做语法与引用收集</b>：
+ * <ul>
+ * <li>非空必填（NOT NULL 列 + 恒真空表达式无判定语义）；</li>
+ * <li>长度 ≤{@value #MAX_LEN_CONDITION}（condition_expr varchar(1000) 列宽口径）；</li>
+ * <li>括号配平：() 与 {} 各自计数配平（照 0011 默认公式一期降级校验同款基础格式面）；</li>
+ * <li>前缀白名单：相对引用关键字仅 SELF / PARENT（大小写不敏感）；其余花括号 token 按
+ *     绝对全键引用剥壳（Expander 同款），<b>展开/剥壳后残留花括号 = 畸形/超纲 token</b>
+ *     （如 {@code { SELF.X }} 带空格、{@code {SELF.ELEC-X}} 非法编码字符）→ 400 拒。</li>
+ * </ul>
+ * <b>相对引用存在性由服务层持模板数据项清单后核对</b>（SELF 编码 ⊆ 模板数据项——引用
+ * 不存在 4001 拒存）；PARENT 与绝对全键引用同 0036 一期口径<b>收集不核存在性</b>
+ * （被引实例可能后建；模板层无法解析父模板——报告登记）。
+ * <b>表达式真语义校验归计算引擎</b>（FEAT-0012/0049——0036 公式同款口径登记）。
+ *
+ * @author dowei
+ * @date 2026-11-22
+ */
+public final class ConditionExprValidator {
+
+    /** condition_expr 限长口径（与 iot_status_rule DDL condition_expr varchar(1000) 一致） */
+    public static final int MAX_LEN_CONDITION = 1000;
+
+    /** 相对引用 token：{SELF.<code>} / {PARENT.<code>}（与 RelativeRefExpander.RELATIVE_REF 同口径） */
+    private static final Pattern RELATIVE_REF =
+            Pattern.compile("\\{(SELF|PARENT)\\.([A-Za-z0-9_]+)\\}", Pattern.CASE_INSENSITIVE);
+
+    /** 任意花括号引用 token（含绝对全键引用）——与 RelativeRefExpander.BRACE_REF 同口径 */
+    private static final Pattern BRACE_REF = Pattern.compile("\\{([A-Za-z0-9_.]+)\\}");
+
+    /** 校验结果：SELF 编码集 + PARENT 编码集 + 绝对全键引用集（保序去重） */
+    public record Outcome(LinkedHashSet<String> selfCodes,
+                          LinkedHashSet<String> parentCodes,
+                          LinkedHashSet<String> absoluteRefs) {
+    }
+
+    private ConditionExprValidator() {
+    }
+
+    /**
+     * 条件表达式语法校验 + 引用收集。
+     *
+     * @param conditionExpr 模板态条件表达式（花括号占位形态；非空必填）
+     * @return 引用收集结果（SELF/PARENT 编码集 + 绝对全键引用集）
+     * @throws BusinessException 空白 / 超长 / 括号不配平 / 残留畸形花括号 token（400）
+     */
+    public static Outcome validate(String conditionExpr) {
+        if (conditionExpr == null || conditionExpr.isBlank()) {
+            throw new BusinessException(ResultCode.PARAM_ERROR.getCode(), "条件表达式不能为空");
+        }
+        String expr = conditionExpr.trim();
+        if (expr.length() > MAX_LEN_CONDITION) {
+            throw new BusinessException(ResultCode.PARAM_ERROR.getCode(),
+                    "条件表达式超长（上限 " + MAX_LEN_CONDITION + " 字符，实际 " + expr.length() + "）");
+        }
+        // 括号配平（照 0011 默认公式基础格式校验同款：() 与 {} 各自计数配平；{} 的形态
+        // 合法性由下方 token 提取 + 残留终检进一步收敛）
+        long parens = expr.chars().filter(c -> c == '(').count()
+                - expr.chars().filter(c -> c == ')').count();
+        long braces = expr.chars().filter(c -> c == '{').count()
+                - expr.chars().filter(c -> c == '}').count();
+        if (parens != 0 || braces != 0) {
+            throw new BusinessException(ResultCode.PARAM_ERROR.getCode(),
+                    "条件表达式基础格式非法：括号/占位符不配平（表达式真语义校验归计算引擎，"
+                            + "本任务仅做基础格式校验）");
+        }
+
+        // 第一遍：相对引用 token 提取（SELF/PARENT 前缀白名单，关键字大小写不敏感）
+        LinkedHashSet<String> selfCodes = new LinkedHashSet<>();
+        LinkedHashSet<String> parentCodes = new LinkedHashSet<>();
+        Matcher relative = RELATIVE_REF.matcher(expr);
+        StringBuilder stripped = new StringBuilder();
+        while (relative.find()) {
+            String kind = relative.group(1).toUpperCase();
+            String code = relative.group(2);
+            if ("SELF".equals(kind)) {
+                selfCodes.add(code);
+            } else {
+                parentCodes.add(code);
+            }
+            relative.appendReplacement(stripped, "");
+        }
+        relative.appendTail(stripped);
+
+        // 第二遍：剩余花括号 token 按绝对全键引用剥壳（Expander 同款口径）
+        LinkedHashSet<String> absoluteRefs = new LinkedHashSet<>();
+        Matcher absolute = BRACE_REF.matcher(stripped.toString());
+        StringBuilder bare = new StringBuilder();
+        while (absolute.find()) {
+            absoluteRefs.add(absolute.group(1));
+            absolute.appendReplacement(bare, "");
+        }
+        absolute.appendTail(bare);
+
+        // 残留花括号终检（0013 修订轮 I-1 同口径）：两个正则都不匹配的花括号段 = 畸形/超纲
+        // token（如 { SELF.X } 带空格、{SELF.ELEC-X} 非法编码字符、{FOO!} 等），坏表达式拒存
+        String rest = bare.toString();
+        if (rest.indexOf('{') >= 0 || rest.indexOf('}') >= 0) {
+            throw new BusinessException(ResultCode.PARAM_ERROR.getCode(),
+                    "条件表达式包含无法识别的引用 token（残留花括号）: 示例 " + firstResidualToken(rest));
+        }
+        return new Outcome(selfCodes, parentCodes, absoluteRefs);
+    }
+
+    /** 残留 token 示例（第一个花括号段起、截 40 字符防消息过长，0013 修订轮 I-1 同口径） */
+    private static String firstResidualToken(String rest) {
+        int start = rest.indexOf('{');
+        if (start < 0) {
+            start = rest.indexOf('}');
+        }
+        String token = rest.substring(Math.max(0, start));
+        return token.length() > 40 ? token.substring(0, 40) : token;
+    }
