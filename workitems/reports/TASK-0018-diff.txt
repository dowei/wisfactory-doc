+    private JsonSchema parseSchema(JsonNode node) {
+        JsonSchema schema = new JsonSchema();
+        schema.type = node.path("type").asText("object");
+        schema.additionalProperties = node.path("additionalProperties").asBoolean(true);
+        schema.required = new LinkedHashSet<>();
+        node.path("required").forEach(n -> schema.required.add(n.asText()));
+        schema.properties = new java.util.LinkedHashMap<>();
+        node.path("properties").fields().forEachRemaining(e -> schema.properties.put(e.getKey(), parseProperty(e.getValue())));
+        return schema;
+    }

+        PropertySchema p = new PropertySchema();
+        p.type = node.path("type").asText(null);
+        p.minLength = node.hasNonNull("minLength") ? node.get("minLength").asInt() : null;
+        p.maxLength = node.hasNonNull("maxLength") ? node.get("maxLength").asInt() : null;
+        p.pattern = node.hasNonNull("pattern") ? Pattern.compile(node.get("pattern").asText()) : null;
+        p.format = node.hasNonNull("format") ? node.get("format").asText() : null;
+        p.enumValues = new LinkedHashSet<>();
+        node.path("enum").forEach(n -> p.enumValues.add(n.asText()));
+        return p;
+    }

+    private void validateAgainstSchema(String path, JsonNode value, JsonSchema schema, List<String> errors) {
+        if (!"object".equals(schema.type)) {
+            errors.add(path + " Schema 词表暂不支持 type=" + schema.type);
+            return;
+        }
+        if (value == null || !value.isObject()) {
+            errors.add(path + " 必须是 JSON 对象");
+            return;
+        }
+        if (!schema.additionalProperties) {
+            value.fieldNames().forEachRemaining(name -> {
+                if (!schema.properties.containsKey(name)) {
+                    errors.add(path + "." + name + " 为未知字段（additionalProperties=false）");
+                }
+            });
+        }
+        for (Map.Entry<String, PropertySchema> entry : schema.properties.entrySet()) {
+            JsonNode child = value.get(entry.getKey());
+            if (child == null || child.isNull()) {
+                continue;
+            }
+            validateProperty(path + "." + entry.getKey(), child, entry.getValue(), errors);
+        }
+    }
+
+    private void validateProperty(String path, JsonNode value, PropertySchema p, List<String> errors) {
+        if ("string".equals(p.type)) {
+            if (!value.isTextual()) {
+                errors.add(path + " 必须是字符串");
+                return;
+            }
+            String text = value.asText();
+            if (p.minLength != null && text.length() < p.minLength) {
+                errors.add(path + " 长度不得小于 " + p.minLength);
+            }
+            if (p.maxLength != null && text.length() > p.maxLength) {
+                errors.add(path + " 长度不能超过 " + p.maxLength);
+            }
+            if (p.pattern != null && !p.pattern.matcher(text).find()) {
+                errors.add(path + " 不满足格式约定（" + p.pattern.pattern() + "）");
+            }
+            if ("uri".equals(p.format) && !isAbsoluteUri(text)) {
+                errors.add(path + " 必须是绝对 URI（如 tcp://host:1883）");
+            }
+            if (!p.enumValues.isEmpty() && !p.enumValues.contains(text)) {
+                errors.add(path + " 仅允许取值 " + p.enumValues);
+            }
+            return;
+        }
+        if ("integer".equals(p.type)) {
+            if (!value.isIntegralNumber()) {
+                errors.add(path + " 必须是整数");
+                return;
+            }
+            if (!p.enumValues.isEmpty() && !p.enumValues.contains(value.asText())) {
+                errors.add(path + " 仅允许取值 " + p.enumValues);
+            }
+            return;
+        }
+        // 未知词表属性类型：按宽松处理（与 draft 注解语义一致），不误伤二期扩展
+    }
+
+    private Set<String> keySet(JsonNode object) {
+        Set<String> keys = new LinkedHashSet<>();
+        object.fieldNames().forEachRemaining(keys::add);
+        return keys;
+    }
+
+    private void requireNoMissing(JsonSchema schema, Set<String> presentKeys) {
+        List<String> missing = schema.required.stream().filter(r -> !presentKeys.contains(r)).toList();
+        if (!missing.isEmpty()) {
+            throw paramError("conn_config 缺少必填字段: " + missing);
+        }
+    }
+
+    private boolean isAbsoluteUri(String text) {
+        try {
+            URI uri = new URI(text);
+            return uri.getScheme() != null && !uri.getScheme().isBlank();
+        } catch (Exception e) {
+            return false;
+        }
+    }
+

+
+    private void collectCredentialKeys(String path, JsonNode node, Set<String> hits) {
+        node.fieldNames().forEachRemaining(name -> {
+            String normalized = name.toLowerCase(Locale.ROOT);

+                hits.add(path.isEmpty() ? name : path + "." + name);
+            }
+        });

+            JsonNode child = e.getValue();
+            if (child.isObject()) {
+                collectCredentialKeys(path.isEmpty() ? e.getKey() : path + "." + e.getKey(), child, hits);
+            }
+        });
+    }

+        return new BusinessException(ResultCode.PARAM_ERROR.getCode(), message);
+    }

+        Set<String> enumValues = Set.of();
+    }
