+        boolean hasTriple = scopeType != null || scopeId != null || version != null;

+                form.getScopeId(), form.getVersion());

+                .eq(IotConfigPublishLog::getVersionId, version.getId())
+                .eq(IotConfigPublishLog::getTargetType, form.getTargetType())
+                .eq(IotConfigPublishLog::getTargetId, form.getTargetId()));

+                    + ", targetType=" + form.getTargetType() + ", targetId=" + form.getTargetId() + ")");
+        }

+                    .set(IotConfigPublishLog::getErrorMsg, null)

+            return false;

+    }

+            // ACK 生效竞态倒退（生效后 WHERE 不命中即跳过）
+            marked += publishLogMapper.update(null, new LambdaUpdateWrapper<IotConfigPublishLog>()
+                    .eq(IotConfigPublishLog::getId, row.getId())
+                    .eq(IotConfigPublishLog::getAckStatus, ACK_PENDING)
+                    .set(IotConfigPublishLog::getAckStatus, ACK_FAILED)
+                    .set(IotConfigPublishLog::getErrorMsg, message)
+                    .set(IotConfigPublishLog::getUpdateTime, LocalDateTime.now()));
+        }
+        log.info("[PUBLISH-ACK] timeout scan marked {} stale row(s) timeoutSeconds={} candidate={}",
+                marked, timeoutSeconds, stale.size());

+    }
