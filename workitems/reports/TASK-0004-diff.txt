+/**
+ * 运营方业务数据隔离守卫（TASK-0004 D2 / REQ-0001 验收 6）。
+ *
+ * <p>在控制器前（preHandle）生效：仅对 {@code hasRole("operator")} 的登录者启用，
+ * 白名单之外一律抛 {@link NotRoleException} → GlobalExceptionHandler 返回 403
+ * （统一文案「无权访问该资源」，不泄露资源标识）并发布 {@code PermissionDeniedEvent}
+ * 供既有 {@code PermissionDeniedAuditListener} 落库（BUG-0044 范式，审计事件不重造）。
+ *
+ * <p>admin（role 1）与租户管理员（user/tenant_admin）不持有 operator 角色，守卫直接放行，
+ * 仍按各自 {@code @SaCheckPermission} 口径鉴权，互不误伤。
+ *
+ * @author dowei
+ * @date 2026-11-24
+ */
+public class OperatorIsolationGuard implements HandlerInterceptor {
+
+    @Override
+    public boolean preHandle(HttpServletRequest request, HttpServletResponse response, Object handler) {
+        // 未登录 / 非运营方：不介入（登录校验由前一拦截器负责，注解鉴权由 SaInterceptor 负责）
+        if (!StpUtil.isLogin() || !StpUtil.hasRole(OperatorIsolationPolicy.OPERATOR_ROLE)) {
+            return true;
+        }
+
+        String uri = request.getRequestURI();
+        String contextPath = request.getContextPath();
+        if (contextPath != null && !contextPath.isEmpty() && uri.startsWith(contextPath)) {
+            uri = uri.substring(contextPath.length());
+        }
+        // 剥除 query（getRequestURI 本身不含 query，保险处理）
+        int q = uri.indexOf('?');
+        if (q >= 0) {
+            uri = uri.substring(0, q);
+        }
+
+        if (OperatorIsolationPolicy.isAllowed(request.getMethod(), uri)) {
+            return true;
+        }
+
+        // 抛 Sa-Token 原生 NotRoleException：GlobalExceptionHandler 统一 403 + 审计事件发布
+        throw new NotRoleException(OperatorIsolationPolicy.DENIED_ROLE_CODE);
+    }
+}

+import java.util.Locale;

+/**
+ * 运营方业务数据隔离白名单策略（TASK-0004 D2，REQ-0001 验收 6）。
+ *
+ * <p><b>默认拒绝（default-deny）</b>：只有精确列出的 (HTTP 方法, 路径) 组合才对
+ * {@code operator} 角色放行；任何新增业务接口未登记即自动受控返回 403，
+ * 无需逐个维护黑名单。白名单按实际控制器路径核对（system 模块为
+ * /user /role /menu /profile /auth，非 brief 示例中的 /sys/* 前缀）。
+ *
+ * <p>白名单逐条理由见 reports/TASK-0004-report.md。
+ *
+ * @author dowei
+ * @date 2026-11-24
+ */
+public final class OperatorIsolationPolicy {
+
+    /** 运营方角色码（与 sys_role.role_code 种子、告警平台角色语义对齐） */
+    public static final String OPERATOR_ROLE = "operator";
+
+    /** 隔离拒绝的「所需角色」标识：只描述策略，不携带被请求资源标识，403 文案不泄露路径 */
+    public static final String DENIED_ROLE_CODE = "operator:business-isolation";
+
+    /** 健康巡检只读：/api/v1/iot/inspect/{dashboard,history,latest} */
+    private static final Set<String> HEALTH_INSPECT_EXACT = Set.of(
+            "/api/v1/iot/inspect/dashboard",
+            "/api/v1/iot/inspect/history",
+            "/api/v1/iot/inspect/latest");
+
+    /** 用量计算只读：/api/v1/iot/calc/{state,log} */
+    private static final Set<String> HEALTH_CALC_EXACT = Set.of(
+            "/api/v1/iot/calc/state",
+            "/api/v1/iot/calc/log");
+
+    /** 租户管理：SysTenantController 实际挂载在 /iot/tenant（brief 要求按实际路径） */
+    private static final String TENANT_PREFIX = "/iot/tenant/";
+
+    /** 系统自管理前缀（用户/角色/菜单/个人中心/认证入口，均为实际 controller 路径） */
+    private static final Set<String> SYSTEM_PREFIXES = Set.of(
+            "/user/", "/role/", "/menu/", "/profile/", "/auth/");
+
+    private OperatorIsolationPolicy() {
+    }
+
+    /**
+     * 是否允许运营方访问该 (method, path)。
+     *
+     * @param method HTTP 方法（大小写不敏感）
+     * @param path   请求路径（不含 query string；含 contextPath 由调用方剥除）
+     */
+    public static boolean isAllowed(String method, String path) {
+        if (method == null || path == null) {
+            return false;
+        }
+        String m = method.toUpperCase(Locale.ROOT);
+        String p = path;
+
+        // ① 健康/用量：仅 GET，精确端点（防新增写口或子路径穿透）
+        if ("GET".equals(m)) {
+            if (HEALTH_INSPECT_EXACT.contains(p) || HEALTH_CALC_EXACT.contains(p)) {
+                return true;
+            }
+        }
+
+        // ② 租户管理：/iot/tenant/** 全方法（运营方需开户/启停/配额/初始化管理员），
+        //    含根路径 POST/PUT /iot/tenant（建租户/改租户——SysTenantController 实际映射）
+        if (p.startsWith(TENANT_PREFIX)
+                || (p.equals("/iot/tenant") && ("POST".equals(m) || "PUT".equals(m)))) {
+            return true;
+        }
+
+        // ③ 系统自管理：user/role/menu/profile/auth 前缀（子路径），含根路径 /profile、/user 等
+        for (String prefix : SYSTEM_PREFIXES) {
+            if (p.startsWith(prefix)) {
+                return true;
+            }
+        }
+        if (p.equals("/profile") || p.equals("/user") || p.equals("/role") || p.equals("/menu")) {
+            return true;
+        }
+
+        // ④ 其余一律拒绝（默认拒绝，新增业务接口自动受控）
+        return false;
+    }
+}

+        // sql-exec.sh 按 ";\n" 切分：每条语句独占一行，行内不得出现 ); 后还有下一条
+        for (String line : sql.split("\\R")) {
+            String trimmed = line.trim();
+            if (trimmed.startsWith("--") || trimmed.isEmpty()) {
+                continue;
+            }
+            int semi = trimmed.indexOf(';');
+            if (semi >= 0 && semi < trimmed.length() - 1) {
+                assertFalse(true, "语句未独占一行：" + trimmed);
+            }
+        }
+    }

+    /** 按 sql-exec.sh 口径（;\\n 切分）提取可执行语句，跳过注释 */
+    private java.util.List<String> splitStatements(String sql) {
+        java.util.List<String> statements = new java.util.ArrayList<>();
+        for (String line : sql.split("\\R")) {
+            String trimmed = line.trim();
+            if (trimmed.isEmpty() || trimmed.startsWith("--")) {
+                continue;
+            }
+            if (trimmed.endsWith(";")) {
+                statements.add(trimmed.substring(0, trimmed.length() - 1));
+            }
+        }
+        return statements;
+    }
