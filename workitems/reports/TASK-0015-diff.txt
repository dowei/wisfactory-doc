+                    .orderByAsc(IotModelTemplateVersion::getVersion)
+                    .last("LIMIT 1"));

+                baselineNode = parse(baselineRow.getContent());
+            }
+        }
+        JsonNode currentNode = parse(buildContentJson(template, items, rules));
+        // 待生效 diff = 相对未锁定跟随实例有效版本的内容变更；无跟随实例或历史基线缺失

+        RedisUtils.set(CONFIRM_KEY_PREFIX + tenantId + ":" + SCOPE_COMPUTE + ":" + SCOPE_COMPUTE_ID
+                + ":" + targetVersion, contentHash, ttlSeconds, TimeUnit.SECONDS);

+                + "summary={}", templateId, targetVersion, affected, locked, TemplateTierRuleTable.formatSummary(entries));

+        vo.setCurrentTemplateVersion(template.getVersion());
+        vo.setTargetVersion(targetVersion);

+        vo.setTierSummary(TemplateTierRuleTable.formatSummary(entries));

+/**
+ * 模板内容 diff 引擎（TASK-0015：「diff 由模板前后版本对比生成」）。
+ *
+ * <p>输入 = 两个版本的内容快照 JSON（{@code IotTemplateVersionService#buildContentJson}
+ * canonical 形态：顶层 template/dataItems/statusRules 三键；dataItems 按 dataCode 索引、
+ * statusRules 按 priority 索引、模板头单对象）。输出 = 变更条目列表
+ * （{@link TemplateChangeEntry}——对象类型/业务键/变更形态/字段/旧值/新值/档位），
+ * 每条独立判档（{@link TemplateTierRuleTable}）。
+ *
+ * <p>判空口径：JSON 缺键与 JSON null 等价（NON_NULL 序列化下 null 即缺键）——
+ * 「defaultAlarmLow 从 5 改为不填」与「改为 null」同判。
+ *
+ * @author dowei
+ * @date 2026-11-23
+ */
+public final class TemplateContentDiffEngine {
+
+    private TemplateContentDiffEngine() {
+    }
+
+    /**
+     * 对比两版内容快照，产出逐条目档位化变更清单（条目排序：objectType → objectKey →
+     * field，输出确定性）。
+     *
+     * @param prev 上一版内容 JSON（null/缺顶层键按空对象处理——首版无基线时全量计 ADDED）
+     * @param cur  当前版内容 JSON
+     */
+    public static List<TemplateChangeEntry> diff(JsonNode prev, JsonNode cur) {
+        List<TemplateChangeEntry> entries = new ArrayList<>();
+        diffHeader(nodeAt(prev, "template"), nodeAt(cur, "template"), entries);
+        diffObjectArray(nodeAt(prev, "dataItems"), nodeAt(cur, "dataItems"),
+                TemplateChangeEntry.OBJECT_DATA_ITEM, "dataCode", entries);
+        diffObjectArray(nodeAt(prev, "statusRules"), nodeAt(cur, "statusRules"),
+                TemplateChangeEntry.OBJECT_STATUS_RULE, "priority", entries);
+        entries.sort((a, b) -> {
+            int c = a.getObjectType().compareTo(b.getObjectType());
+            if (c != 0) {
+                return c;
+            }
+            c = a.getObjectKey().compareTo(b.getObjectKey());
+            if (c != 0) {
+                return c;
+            }
+            return String.valueOf(a.getField()).compareTo(String.valueOf(b.getField()));
+        });
+        return entries;
+    }
+
+    /** 模板头单对象 diff（objectKey=templateCode；ADDED/REMOVED 仅在整头缺失时出现） */
+    private static void diffHeader(JsonNode prev, JsonNode cur, List<TemplateChangeEntry> entries) {
+        String objectKey = cur != null && cur.hasNonNull("templateCode")
+                ? cur.get("templateCode").asText()
+                : (prev != null && prev.hasNonNull("templateCode")
+                        ? prev.get("templateCode").asText() : "template");
+        if (prev == null || prev.isNull() || !prev.isObject()) {
+            if (cur != null && cur.isObject() && cur.size() > 0) {
+                entries.add(new TemplateChangeEntry(TemplateChangeEntry.OBJECT_TEMPLATE,
+                        objectKey, TemplateChangeEntry.CHANGE_ADDED, null, null, null,
+                        TemplateTierRuleTable.tierOfAdded(TemplateChangeEntry.OBJECT_TEMPLATE)));
+            }
+            return;
+        }
+        if (cur == null || cur.isNull() || !cur.isObject()) {
+            entries.add(new TemplateChangeEntry(TemplateChangeEntry.OBJECT_TEMPLATE,
+                    objectKey, TemplateChangeEntry.CHANGE_REMOVED, null, null, null,
+                    TemplateTierRuleTable.tierOfRemoved(TemplateChangeEntry.OBJECT_TEMPLATE)));
+            return;
+        }
+        diffObjectFields(TemplateChangeEntry.OBJECT_TEMPLATE, objectKey, prev, cur, entries);
+    }
+
+    /** 对象数组 diff（按业务键索引：added/removed 对象级一条；changed 逐字段一条） */
+    private static void diffObjectArray(JsonNode prev, JsonNode cur, String objectType,
+                                        String keyField, List<TemplateChangeEntry> entries) {
+        Map<String, JsonNode> prevMap = indexBy(prev, keyField);
+        Map<String, JsonNode> curMap = indexBy(cur, keyField);
+        TreeSet<String> keys = new TreeSet<>();
+        keys.addAll(prevMap.keySet());
+        keys.addAll(curMap.keySet());
+        for (String key : keys) {
+            JsonNode oldObj = prevMap.get(key);
+            JsonNode newObj = curMap.get(key);
+            if (oldObj == null) {
+                entries.add(new TemplateChangeEntry(objectType, key,
+                        TemplateChangeEntry.CHANGE_ADDED, null, null, null,
+                        TemplateTierRuleTable.tierOfAdded(objectType)));
+            } else if (newObj == null) {
+                entries.add(new TemplateChangeEntry(objectType, key,
+                        TemplateChangeEntry.CHANGE_REMOVED, null, null, null,
+                        TemplateTierRuleTable.tierOfRemoved(objectType)));
+            } else {
+                diffObjectFields(objectType, key, oldObj, newObj, entries);
+            }
+        }
+    }
+
+    /** 对象字段级 diff：值不等（null 归一）即逐字段一条 CHANGED，档位查规则表 */
+    private static void diffObjectFields(String objectType, String objectKey,
+                                         JsonNode prevObj, JsonNode curObj,
+                                         List<TemplateChangeEntry> entries) {
+        Map<String, JsonNode> prevFields = fieldMap(prevObj);
+        Map<String, JsonNode> curFields = fieldMap(curObj);
+        TreeSet<String> fields = new TreeSet<>();
+        fields.addAll(prevFields.keySet());
+        fields.addAll(curFields.keySet());
+        for (String field : fields) {
+            JsonNode oldVal = prevFields.get(field);
+            JsonNode newVal = curFields.get(field);
+            if (!normalizedEquals(oldVal, newVal)) {
+                entries.add(new TemplateChangeEntry(objectType, objectKey,
+                        TemplateChangeEntry.CHANGE_CHANGED, field,
+                        oldVal == null ? null : oldVal.asText(),
+                        newVal == null ? null : newVal.asText(),
+                        TemplateTierRuleTable.tierOfChangedField(objectType, field)));
+            }
+        }
+    }
+
+    /** 数组按业务键索引（非对象/缺键元素跳过） */
+    private static Map<String, JsonNode> indexBy(JsonNode array, String keyField) {
+        Map<String, JsonNode> map = new TreeMap<>();
+        if (array == null || !array.isArray()) {
+            return map;
+        }
+        for (JsonNode element : array) {
+            if (element.isObject() && element.hasNonNull(keyField)) {
+                map.put(element.get(keyField).asText(), element);
+            }
+        }
+        return map;
+    }
+
+    /** 对象展平为字段 Map（TreeMap——字段遍历序确定性） */
+    private static Map<String, JsonNode> fieldMap(JsonNode object) {
+        Map<String, JsonNode> map = new TreeMap<>();
+        if (object != null && object.isObject()) {
+            Iterator<Map.Entry<String, JsonNode>> it = object.fields();
+            while (it.hasNext()) {
+                Map.Entry<String, JsonNode> e = it.next();
+                map.put(e.getKey(), e.getValue().isNull() ? null : e.getValue());
+            }
+        }
+        return map;
+    }
+
+    /** null 归一相等（缺键 ≡ null ≡ JSON null） */
+    private static boolean normalizedEquals(JsonNode a, JsonNode b) {
+        if (a == null || a.isNull()) {
+            return b == null || b.isNull();
+        }
+        if (b == null || b.isNull()) {
+            return false;
+        }
+        return a.equals(b);
+    }
+
+    private static JsonNode nodeAt(JsonNode root, String key) {
+        if (root == null || root.isNull() || !root.isObject()) {
+            return null;
+        }
+        JsonNode node = root.get(key);
+        return (node == null || node.isNull()) ? null : node;
+    }
