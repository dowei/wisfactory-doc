+    private static final int TARGET_GATEWAY = 1;

+    private static final int MQTT_QOS = 1;

+        this.tenantMapper = tenantMapper;

+    @Override
+    public void notify(IotConfigVersion version, Integer targetType, String targetId) {
+        // 载荷在调用点（事务内）组装捕获；afterCommit 回调内零 DB 读取
+        NotifyPayload payload = buildPayload(version);
+        if (payload == null) {
+            return;
+        }
+        String topic = topicOf(targetType, targetId, payload.tenant);
+        if (topic == null) {
+            return;
+        }
+        String json = serialize(payload);
+        if (TransactionSynchronizationManager.isSynchronizationActive()) {
+            // Minor ④ 决策 a：发送延迟到事务提交后（杜绝伪通知——见类注释）
+            TransactionSynchronizationManager.registerSynchronization(new TransactionSynchronization() {
+                @Override
+                public void afterCommit() {

+                }
+            });

+                            + "version={} target={}-{}", topic, version.getId(), version.getVersion(),
+                    targetType, targetId);
+        } else {
+            sendQuietly(topic, json, version, targetType, targetId);
+        }
+    }
+

+    private String topicOf(Integer targetType, String targetId, String tenantCode) {
+        if (targetType != null && targetType == TARGET_GATEWAY) {
+            if (targetId == null || targetId.isBlank()) {
+                // 落行侧已 fail-closed 网关码；防御分支仍兜底 fail-soft（不发空目标）
+                log.warn("[PUBLISH-NOTIFY] gateway target missing code (notify skipped, fail-soft) tenant={}",
+                        tenantCode);
+                return null;
+            }
+            return "iot/" + tenantCode + "/" + targetId + "/config";
+        }
+        return SHARED_TOPIC;
+    }
+
+    /**
+     * 载荷组装（五键）：tenant 由 tenant_id 反查 sys_tenant（平台级表无租户注入——会话
+     * 内外一致）；解析失败 WARN 跳过（fail-soft——载荷缺 tenant 不得外发，下游按 topic/
+     * scope 定位会失配）。
+     */
+    private NotifyPayload buildPayload(IotConfigVersion version) {
+        if (version == null) {
+            log.warn("[PUBLISH-NOTIFY] null version row (notify skipped, fail-soft)");
+            return null;
+        }
+        String tenantCode = resolveTenantCode(version.getTenantId());
+        if (tenantCode == null || tenantCode.isBlank()) {
+            log.warn("[PUBLISH-NOTIFY] tenant_code resolve failed (notify skipped, fail-soft) "
+                    + "tenantId={} versionId={} version={}", version.getTenantId(),
+                    version.getId(), version.getVersion());
+            return null;
+        }
+        return new NotifyPayload(tenantCode, version.getScopeType(), version.getScopeId(),
+                version.getVersion(), version.getSnapshotHash());
+    }
+
+    private String resolveTenantCode(Long tenantId) {
+        if (tenantId == null) {
+            return null;
+        }
+        SysTenant tenant = tenantMapper.selectById(tenantId);
+        return tenant == null ? null : tenant.getTenantCode();
+    }
+
+    private String serialize(NotifyPayload payload) {
+        try {
+            return PAYLOAD_MAPPER.writeValueAsString(payload);
+        } catch (JsonProcessingException e) {
+            // 载荷全为标量——序列化不可达；仍按 fail-soft 兜底（不阻断发布事务）
+            log.warn("[PUBLISH-NOTIFY] payload serialize failed (notify skipped, fail-soft): {}", e.getMessage());
+            return null;
+        }
+    }
+
+    /** 实际网络发送：任何异常 WARN 不抛（发布流程不阻断——REQ-0033 验收 2 拉取对账兜底） */
+    private void sendQuietly(String topic, String json, IotConfigVersion version,
+                             Integer targetType, String targetId) {
+        if (json == null) {
+            return;
+        }
+        try {
+            publisher.publish(topic, json, MQTT_QOS, false);
+            log.info("[PUBLISH-NOTIFY] config.changed sent topic={} versionId={} version={} target={}-{} hash={}",
+                    topic, version.getId(), version.getVersion(), targetType, targetId,
+                    version.getSnapshotHash());
+        } catch (Exception e) {
+            log.warn("[PUBLISH-NOTIFY] config.changed publish failed (fail-soft, 发布不阻断——"
+                    + "下游拉取对账哈希兜底 REQ-0033) topic={} versionId={} version={} target={}-{} err={}",
+                    topic, version.getId(), version.getVersion(), targetType, targetId, e.getMessage());
+        }
+    }
+
+    /**
+     * 通知载荷（§3.3 逐字五键；JSON 键序由 @JsonPropertyOrder 钉死）。
+     */
+    @JsonPropertyOrder({"tenant", "scopeType", "scopeId", "version", "hash"})
+    static class NotifyPayload {
+
+        /** 租户编码（sys_tenant.tenant_code） */
+        public final String tenant;
+        /** 范围 1 网关 2 计算 */
+        public final Integer scopeType;
+        /** 范围对象 ID（1=网关 id；2=0 租户粒度） */
+        public final Long scopeId;
+        /** 版本号（同 (tenant, scope) 单调递增；注意非版本行 id） */
+        public final Long version;
+        /** 快照 SHA-256（下游拉取快照后比对的锚） */
+        public final String hash;
+
+        NotifyPayload(String tenant, Integer scopeType, Long scopeId, Long version, String hash) {
+            this.tenant = tenant;
+            this.scopeType = scopeType;
+            this.scopeId = scopeId;
+            this.version = version;
+            this.hash = hash;
+        }
+    }
+}

+    }

+     * 进行，此处同步补一次 connect——已在重连中则 connect 抛 32110 由发送方 fail-soft）。
+     */
+    private synchronized MqttClient obtainClient() throws MqttException {
+        if (client != null && client.isConnected()) {
+            return client;
+        }
+        if (client == null) {
+            PublishProperties.Notify cfg = properties.getNotify();
+            MqttConnectOptions options = new MqttConnectOptions();
+            if (cfg.getUsername() != null && !cfg.getUsername().isBlank()) {
+                options.setUserName(cfg.getUsername());
+            }
+            if (cfg.getPassword() != null) {
+                options.setPassword(cfg.getPassword().toCharArray());
+            }
+            options.setKeepAliveInterval(cfg.getKeepAliveSeconds() == null ? 60 : cfg.getKeepAliveSeconds());
+            options.setConnectionTimeout(cfg.getTimeoutSeconds() == null ? 30 : cfg.getTimeoutSeconds());
+            options.setAutomaticReconnect(true);
+            options.setCleanSession(true);
+
+            String brokerUrl = cfg.getBrokerUrl();
+            if (brokerUrl == null || brokerUrl.isBlank()) {
+                // 未配置 broker：构造失败即刻上报（发送方 WARN fail-soft），不空转重连
+                throw new MqttException(MqttException.REASON_CODE_CLIENT_EXCEPTION);
+            }
+            try {
+                client = new MqttClient(brokerUrl, resolveClientId(cfg), new MemoryPersistence());
+                client.connect(options);
+                log.info("[PUBLISH-NOTIFY] mqtt publisher connected broker={} clientId={}",
+                        brokerUrl, client.getClientId());
+            } catch (MqttException e) {
+                client = null;
+                throw e;
+            }
+            return client;
+        }
+        // 句柄已存在但断连：同步补一次 connect（失败原样上报——发送方 fail-soft WARN）
+        MqttConnectOptions options = connectOptions();

+        return client;
+    }

+        MqttConnectOptions options = new MqttConnectOptions();

+        }

+            options.setPassword(cfg.getPassword().toCharArray());
+        }
+        options.setKeepAliveInterval(cfg.getKeepAliveSeconds() == null ? 60 : cfg.getKeepAliveSeconds());
+        options.setConnectionTimeout(cfg.getTimeoutSeconds() == null ? 30 : cfg.getTimeoutSeconds());
+        options.setAutomaticReconnect(true);
+        options.setCleanSession(true);
+        return options;
+    }

+            return cfg.getClientId();
+        }
+        return "ot-admin-notify-" + java.util.UUID.randomUUID().toString().substring(0, 8);
+    }

+    @DisplayName("changes - 相邻版本 diff 五段摘要正确（added/removed/changed 按 code/id 对齐）")
+    void changesDiffCoversFiveSegments() throws Exception {
+        // 计算 scope 五段全量差异：v1 基线 → v2 变更
+        String v1 = computeBody("P1:0", "C1", "D1", "200", "M1");
+        // v2：P1 变更 + P2 新增；C1 移除；D1 不变；公式 F2(201) 新增；指标 M1 移除
+        String v2 = computeBody("P1:5,P2:0", null, "D1", "200,201", null);
+        IotConfigVersion r1 = storedRow(TENANT, 2, 0L, 1, v1);
+        IotConfigVersion r2 = storedRow(TENANT, 2, 0L, 2, v2);
+        when(versionMapper.selectList(any())).thenReturn(List.of(r1, r2));
+
+        ChangesListVO vo = publishService.changesSince(2, 0L, 1L, null);
+
+        assertFalse(vo.isNeedFullSync());
+        assertEquals(2L, vo.getLatestVersion());
+        assertEquals(1, vo.getChanges().size(), "逐版 diff：一条（v2）");
+        ChangeEntryVO entry = vo.getChanges().get(0);
+        assertEquals(2L, entry.getVersion());
+        assertEquals(r2.getSnapshotHash(), entry.getHash());
+        assertNotNull(entry.getPublishTime());
+        Set<String> added = keysOf(entry, "added");
+        Set<String> removed = keysOf(entry, "removed");
+        Set<String> changed = keysOf(entry, "changed");
+        assertEquals(Set.of("point:P2", "formula:201"), added, "新增=点位 P2 + 公式边 201");
+        assertEquals(Set.of("channel:C1", "metric:M1"), removed, "移除=通道 C1 + 指标 M1");
+        assertEquals(Set.of("point:P1"), changed, "变更=点位 P1（minValue 0→5）");
+    }

+    /** 网关 scope 最小快照体（points 单段；值嵌入 minValue 便于断言 diff） */
+    private String body(long tenantId, long version, double minValue) {
+        return "{\"schemaVersion\":\"1.0.1\",\"tenantId\":" + tenantId + ",\"scopeType\":1,"
+                + "\"scopeId\":" + GATEWAY_ID + ",\"version\":" + version + ","
+                + "\"points\":[{\"id\":12020,\"pointCode\":\"P1\",\"minValue\":" + minValue + "}],"
+                + "\"channels\":[]}";
+    }
+
+    /**
+     * 计算 scope 最小快照体（五段齐备便于 diff 断言）：
+     * points 以 code:minValue 对（空段传 null）、channels/dataItems 传 code 清单、
+     * formulas 传 id 清单、metrics 传 metricCode 清单。
+     */
+    private String computeBody(String pointsSpec, String channels, String dataItems,
+                               String formulas, String metrics) {
+        StringBuilder sb = new StringBuilder();
+        sb.append("{\"schemaVersion\":\"1.0.1\",\"tenantId\":").append(TENANT)
+                .append(",\"scopeType\":2,\"scopeId\":0,\"version\":0,");
+        sb.append("\"points\":[");
+        if (pointsSpec != null) {
+            long id = 1;
+            for (String seg : pointsSpec.split(",")) {
+                String[] kv = seg.split(":");
+                if (id > 1) {
+                    sb.append(',');
+                }
+                sb.append("{\"id\":").append(id).append(",\"pointCode\":\"").append(kv[0])
+                        .append("\",\"minValue\":").append(kv[1]).append("}");
+                id++;
+            }
+        }
+        sb.append("],\"channels\":[");
+        if (channels != null) {
+            long id = 10;
+            for (String code : channels.split(",")) {
+                if (id > 10) {
+                    sb.append(',');
+                }
+                sb.append("{\"id\":").append(id).append(",\"channelCode\":\"").append(code).append("\"}");
+                id++;
+            }
+        }
+        sb.append("],\"dataItems\":[");
+        if (dataItems != null) {
+            long id = 100;
+            for (String code : dataItems.split(",")) {
+                if (id > 100) {
+                    sb.append(',');
+                }
+                sb.append("{\"id\":").append(id).append(",\"dataCode\":\"").append(code)
+                        .append("\",\"fullKey\":\"F01.WS1.EQ1.").append(code).append("\"}");
+                id++;
+            }
+        }
+        sb.append("],\"formulas\":[");
+        if (formulas != null) {
+            boolean first = true;
+            for (String idText : formulas.split(",")) {
+                if (!first) {
+                    sb.append(',');
+                }
+                sb.append("{\"id\":").append(idText).append(",\"itemId\":100}");
+                first = false;
+            }
+        }
+        sb.append("],\"metrics\":[");
+        if (metrics != null) {
+            long id = 300;
+            for (String code : metrics.split(",")) {
+                if (id > 300) {
+                    sb.append(',');
+                }
+                sb.append("{\"id\":").append(id).append(",\"metricCode\":\"").append(code).append("\"}");
+                id++;
+            }
+        }
+        sb.append("]}");
+        return sb.toString();
+    }

+    /** 落库行 + 快照文件真实落盘（哈希=文件内容 SHA-256，自洽） */
+    private IotConfigVersion storedRow(long tenantId, int scopeType, long scopeId,
+                                       long version, String json) throws Exception {
+        IotConfigVersion row = rowWithUri(tenantId, scopeType, scopeId, version,
+                tenantId + "/" + scopeType + "-" + scopeId + "-" + version + ".json");
+        Path file = tempDir.resolve(row.getSnapshotUri());
+        Files.createDirectories(file.getParent());
+        Files.write(file, json.getBytes(StandardCharsets.UTF_8));
+        row.setSnapshotHash(SnapshotStorage.sha256Hex(json.getBytes(StandardCharsets.UTF_8)));
+        return row;
+    }
+
+    private IotConfigVersion rowWithUri(long tenantId, int scopeType, long scopeId,
+                                        long version, String uri) {
+        IotConfigVersion row = new IotConfigVersion();
+        row.setId(version * 10);
+        row.setTenantId(tenantId);
+        row.setScopeType(scopeType);
+        row.setScopeId(scopeId);
+        row.setVersion(version);
+        row.setSnapshotUri(uri);
+        row.setPublishStatus(1);
+        row.setPublishTime(LocalDateTime.of(2026, 11, 23, 10, 0));
+        return row;
+    }

+    private SysTenant tenant(long id, String code) {
+        SysTenant t = new SysTenant();
+        t.setId(id);
+        t.setTenantCode(code);
+        return t;
+    }

+    /** 摘要三段转 "type:key" 集合（断言用） */
+    private Set<String> keysOf(ChangeEntryVO entry, String action) {
+        if ("added".equals(action)) {
+            return entry.getSummary().getAdded().stream()
+                    .map(i -> i.getType() + ":" + i.getKey()).collect(Collectors.toSet());
+        }
+        if ("removed".equals(action)) {
+            return entry.getSummary().getRemoved().stream()
+                    .map(i -> i.getType() + ":" + i.getKey()).collect(Collectors.toSet());
+        }
+        return entry.getSummary().getChanged().stream()
+                .map(i -> i.getType() + ":" + i.getKey()).collect(Collectors.toSet());
+    }
