+    private static final Logger log = LoggerFactory.getLogger(MetricEventPublisher.class);

+    @FunctionalInterface
+    public interface MqttChannel {

+    }
+
+    @FunctionalInterface

+        void publish(String topic, String payload);

+    private final KafkaChannel kafka;

+    private final boolean kafkaEnabled;

+        this.mqtt = mqtt;
+        this.kafka = kafka;
+        this.mqttEnabled = mqttEnabled;
+        this.kafkaEnabled = kafkaEnabled;
+    }
+
+    public void publish(MetricProducedEvent e) {
+        String payload = e.toJson();
+        if (mqttEnabled && mqtt != null) {
+            safePublish("MQTT", MetricProducedEvent.mqttTopic(e.tenant()), payload, mqtt::publish);
+        }
+        if (kafkaEnabled && kafka != null) {
+            safePublish("Kafka", MetricProducedEvent.kafkaTopic(e.tenant()), payload, kafka::publish);
+        }
+    }
+
+    private void safePublish(String channelName, String topic, String payload,
+                             java.util.function.BiConsumer<String, String> send) {
+        try {
+            send.accept(topic, payload); // QoS1 fire-and-forget：无订阅者不抛
+        } catch (RuntimeException ex) {

+        }

+        ZonedDateTime prev = switch (period.toUpperCase(java.util.Locale.ROOT)) {
+            case "HOUR" -> t.minusHours(1);
+            case "DAY" -> t.minusDays(1);
+            case "MONTH" -> t.minusMonths(1);
+            case "YEAR" -> t.minusYears(1);
+            default -> throw new IllegalArgumentException("不支持的指标周期: " + period);
+        };
+        // 同比 = 去年同期「同窗口」（本期 ts 减 1 年）；环比 = 上一周期（本期 ts 回退一周期），
