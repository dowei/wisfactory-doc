+        }
+        if (row == null) {
+            vo.setStatusName(code);
+            vo.setColor(null);
+            return;
+        }
+        vo.setStatusName(row.getStatusName() == null ? code : row.getStatusName());
+        vo.setColor(row.getColor());
+    }

+    /** rawStringRedisTemplate 直读（无 key-prefix）；故障 fail-soft 返回 null */
+    private String readRedis(String tenantCode, String instanceCode) {
+        if (tenantCode == null || tenantCode.isBlank() || instanceCode == null || instanceCode.isBlank()) {
+            return null;
+        }
+        try {
+            return rawStringRedisTemplate.opsForValue().get(STATE_KEY_PREFIX + tenantCode + ":" + instanceCode);
+        } catch (Exception e) {
+            return null;
+        }
+    }

+    private Long requireSessionTenant() {

+            return null;
+        }
+        List<OpenApiInstanceRow> rows = instanceMapper.selectByIdsForTenant(tenantId, List.of(instanceId));
+        return rows.isEmpty() ? null : rows.get(0).getInstanceCode();
+    }
+
+    private String tenantCode(Long tenantId) {
+        return tenantMapper == null || tenantId == null ? null : tenantMapper.selectTenantCodeById(tenantId);
+    }
+
+    private static void requireIds(Long instanceId) {
+        if (instanceId == null) {
+            throw new BusinessException(ResultCode.PARAM_ERROR.getCode(), "instanceId 不能为空");
+        }
+    }
