+/*
+ * 指标定义 API（智慧工厂）
+ *
+ * 对接后端 IotMetricController（TASK-0043 交付，路径前缀 /iot/metric；
+ * 权限位 iot:metric:* 对应 sys_menu 4058-4062）。管理面 /api 登录态，
+ * 不碰 appKey/secret/HMAC，不走 /openapi/v1/**。
+ *
+ * - 分页列表 GET /iot/metric/list（pageNum/pageSize/metricCode/metricName/
+ *   instanceId/period/status；数据权限由后端裁剪，前端不造假数据）
+ * - 详情 GET /iot/metric/{id}（getById / detail 同端点）
+ * - 新增 POST /iot/metric（metric_code 创建后不可改）
+ * - 批量生成 POST /iot/metric/generate?instanceId=（instanceId 为 query 参数，
+ *   IotMetricGenerateForm 为 body；模板载体未定义，按请求清单批量生成）
+ * - 修改 PUT /iot/metric（status 不进编辑集，启停走 changeStatus）
+ * - 批量启停 PUT /iot/metric/changeStatus（{ ids, status }）
+ * - 批量删除 DELETE /iot/metric/{ids}（路径参数逗号拼接，软删）
+ */
+import { deleteRequest, getRequest, postRequest, putRequest } from '/@/lib/axios';
+
+export const metricApi = {
+  /** 指标分页列表（GET /iot/metric/list，iot:metric:list） */
+  queryList: (params) => {
+    return getRequest('/iot/metric/list', params);
+  },
+
+  /** 指标详情（GET /iot/metric/{id}，iot:metric:query） */
+  getById: (id) => {
+    return getRequest(`/iot/metric/${id}`, undefined);
+  },
+
+  /** 指标详情别名（与 getById 同端点，语义化命名） */
+  detail: (id) => {
+    return getRequest(`/iot/metric/${id}`, undefined);
+  },
+
+  /** 新增指标（POST /iot/metric，iot:metric:add） */
+  add: (form) => {
+    return postRequest('/iot/metric', form);
+  },
+
+  /**
+   * 批量生成指标（POST /iot/metric/generate?instanceId=，iot:metric:add）
+   * instanceId 以 query 入参为准（覆盖项内值）；form 为 IotMetricGenerateForm。
+   */
+  generate: (instanceId, form) => {
+    return postRequest(`/iot/metric/generate?instanceId=${instanceId}`, form);
+  },
+
+  /** 修改指标（PUT /iot/metric，iot:metric:edit；metricCode 不可改，status 不进编辑集） */
+  update: (form) => {
+    return putRequest('/iot/metric', form);
+  },
+
+  /** 批量启停（PUT /iot/metric/changeStatus，iot:metric:edit；{ ids, status }） */
+  changeStatus: (form) => {
+    return putRequest('/iot/metric/changeStatus', form);
+  },
+
+  /** 批量删除（DELETE /iot/metric/{ids}，iot:metric:remove；ids 逗号拼接路径参数） */
+  remove: (ids) => {
+    const idsParam = Array.isArray(ids) ? ids.join(',') : ids;
+    return deleteRequest(`/iot/metric/${idsParam}`);
+  },
+};

+ * - 非比值型提交 ratio_* 置空、calendarType 锁 1、metricCode 创建后不可改（编辑传原值）；

+import { beforeEach, describe, expect, it, vi } from 'vitest';
+import { defineComponent, h, nextTick, ref } from 'vue';
+import { flushPromises, mount } from '@vue/test-utils';

+const queryList = vi.hoisted(() =>
+  vi.fn(() =>
+    Promise.resolve({

+const getDetail = vi.hoisted(() =>
+  vi.fn(() =>
+    Promise.resolve({
+      code: 200,
+      data: { id: 301, metricCode: 'INS1_DAY_AVG', metricName: '日平均', aggType: 1, period: 'DAY', status: 1 },
+    })
+  )
+);
+const addMetric = vi.hoisted(() => vi.fn((_form: Record<string, unknown>) => Promise.resolve({ code: 200, data: null })));
+const updateMetric = vi.hoisted(() => vi.fn((_form: Record<string, unknown>) => Promise.resolve({ code: 200, data: null })));
+const changeStatus = vi.hoisted(() => vi.fn(() => Promise.resolve({ code: 200, data: null })));
+const removeMetrics = vi.hoisted(() => vi.fn(() => Promise.resolve({ code: 200, data: null })));
+const generateMetrics = vi.hoisted(() => vi.fn(() => Promise.resolve({ code: 200, data: 2 })));
+const queryInstanceList = vi.hoisted(() =>
+  vi.fn(() => Promise.resolve({ code: 200, data: { records: [{ id: 1001, instanceCode: 'INS-001', instanceName: '1号设备' }] } }))
+);
+const queryItemList = vi.hoisted(() => vi.fn(() => Promise.resolve({ code: 200, data: [{ id: 11, itemCode: 'I-001', itemName: '用电量' }] })));
+const messageSuccess = vi.hoisted(() => vi.fn());
+const messageError = vi.hoisted(() => vi.fn());
+const captureError = vi.hoisted(() => vi.fn());

+vi.mock('/@/components/framework/ot-loading', () => ({ OtLoading: { show: vi.fn(), hide: vi.fn() } }));
+vi.mock('/@/components/iot/iot-optimistic-lock', () => ({ handleIotConflictError: (e: unknown) => e }));
+vi.mock('ant-design-vue', async () => {
+  const actual = await vi.importActual<typeof import('ant-design-vue')>('ant-design-vue');
+  return {
+    ...actual,
+    default: actual.default,
+    message: { success: messageSuccess, error: messageError },
+    Modal: {
+      ...actual.Modal,
+      confirm: (opts: { onOk: () => void }) => {

+        opts.onOk();

+import MetricPage from '/@/views/iot/metric/index.vue';
+import MetricFormModal from '/@/views/iot/metric/components/metric-form-modal.vue';
+import MetricDetailModal from '/@/views/iot/metric/components/metric-detail-modal.vue';

+const globalOpts = { plugins: [Antd], directives: { privilege: {} } };

+type ModalRef = {
+  showModal: (row?: Record<string, unknown>) => Promise<void>;
+  validateRatio: () => boolean;

+
+// 真实父组件 template ref 走 expose proxy（BUG-0008 回归教训）
+const FormParent = defineComponent({

+    const modalRef = ref<ModalRef | null>(null);
+    expose({ modalRef });
+    return () => h(MetricFormModal, { ref: modalRef });
+  },

+const DetailParent = defineComponent({
+  setup(_props, { expose }) {
+    const modalRef = ref<{ showModal: (id: number) => Promise<void> } | null>(null);
+    expose({ modalRef });
+    return () => h(MetricDetailModal, { ref: modalRef });
+  },
+});

+describe('指标管理页与表单校验（TASK-0095）', () => {
+  beforeEach(() => {
+    document.body.innerHTML = '';
+    vi.clearAllMocks();
+  });
+
+  it('列表加载真实后端数据、无 IotPlaceholder；查询参数透传', async () => {

+    await flushPromises();
+    expect(queryList).toHaveBeenCalledWith(expect.objectContaining({ pageNum: 1, pageSize: 10, metricCode: '', metricName: '', status: undefined }));
+    const text = wrapper.text();

+    expect(text).not.toContain('IotPlaceholder');
+    expect(text).not.toContain('TASK-0056 / TASK-0070');
+    wrapper.unmount();
+  });
+
+  it('详情按钮 → detail(id) → 详情弹窗展示（父 ref defineExpose）', async () => {
+    const wrapper = mount(MetricPage, { global: globalOpts });
+    await flushPromises();
+

+    const detailParent = mount(DetailParent, { global: globalOpts });
+    const detailRef = (detailParent.vm as unknown as { modalRef: { showModal: (id: number) => Promise<void> } }).modalRef;

+    expect(typeof modalRef.showModal).toBe('function');
+    expect(typeof modalRef.validateRatio).toBe('function');
+
+    await modalRef.showModal();
+    await flushPromises();
+    expect(queryInstanceList).toHaveBeenCalledWith(expect.objectContaining({ pageNum: 1, pageSize: 500 }));

+  it('新增提交：calendarType 锁 1，非比值型 ratio_* 置空，metricCode 原样透传', async () => {
+    const parent = mount(FormParent, { global: globalOpts });
+    const modalRef = (parent.vm as unknown as { modalRef: ModalRef }).modalRef;
+    await modalRef.showModal();
+    await flushPromises();

+    const vm = parent.findComponent(MetricFormModal).vm as unknown as {
+      form: Record<string, unknown>;
