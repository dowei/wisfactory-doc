+            }

+        }

+            });
+        }
+    }

+    }

+/**
+ * 状态规则集种子 JSON 解析（live 冒烟/演进位——冻结快照不装载 iot_status_rule，见 §11 对象集边界）。
+ *
+ * <p>形态（camelCase，与 0048 配置侧同名字段）：
+ * <pre>
+ * {"dict":{"RUNNING":{"seq":1},"FAULT":{"seq":3}},
+ *  "rules":[{"priority":10,"conditionExpr":"EQ1.FAULT_FLAG == 1","statusCode":"FAULT","enabled":1}],
+ *  "direct":{"EQ1.RUN_STATE":{"pointFullKey":"EQ1.RUN_STATE","enumToCode":{"1":"RUNNING"}}}}
+ * </pre>
+ * fail-closed：JSON 非法/缺字段抛 IllegalArgumentException（启动 fail-fast，不哑跑规则）。
+ *
+ * @author dowei
+ * @date 2026-11-24
+ */
+public final class StatusRuleSetParser {
+
+    private static final JsonMapper MAPPER = JsonMapper.builder().build();
+
+    public StatusRuleSet parse(String json) {
+        if (json == null || json.isBlank()) {
+            throw new IllegalArgumentException("状态规则种子 JSON 空白（status-enabled=true 须提供）");
+        }
+        JsonNode root = MAPPER.readTree(json);
+        Map<String, Integer> dict = new LinkedHashMap<>();
+        JsonNode dictNode = root.get("dict");
+        if (dictNode != null) {
+            dictNode.properties().forEach(entry -> dict.put(entry.getKey(), entry.getValue().get("seq").asInt()));
+        }
+        List<StatusRule> rules = new java.util.ArrayList<>();
+        JsonNode rulesNode = root.get("rules");
+        if (rulesNode != null) {
+            for (JsonNode r : rulesNode) {
+                rules.add(new StatusRule(r.get("priority").asInt(), r.get("conditionExpr").asString(),
+                        r.get("statusCode").asString(), r.has("enabled") ? r.get("enabled").asInt() : 1));
+            }
+        }
+        Map<String, StatusDirectBinding> direct = new LinkedHashMap<>();
+        JsonNode directNode = root.get("direct");
+        if (directNode != null) {
+            directNode.properties().forEach(entry -> {
+                JsonNode node = entry.getValue();
+                String point = node.get("pointFullKey").asString();
+                Map<Integer, String> enumToCode = new LinkedHashMap<>();
+                node.get("enumToCode").properties().forEach(e ->
+                        enumToCode.put(Integer.parseInt(e.getKey()), e.getValue().asString()));
+                direct.put(entry.getKey(), new StatusDirectBinding(point, enumToCode));
+            });
+        }
+        return new StatusRuleSet(rules, dict, direct);
+    }

+            int off = 0;
+            while (off < len) {
+                off += in.read(buf, off, len - off);
+            }
+            in.read();
+            in.read();
+            return new String(buf, java.nio.charset.StandardCharsets.UTF_8);
+        }
+        if (b == ':' || b == '+') {
+            return readLine(in);
+        }

+    }
