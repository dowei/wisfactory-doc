+ *
+ * <p><b>与状态判定同源（REQ-0029 验收 2）</b>：每个设备的 statusCode 一律经
+ * {@link StatusReadService#currentForClient(Long, Long)}（0050 同一读口、显式会话租户，
+ * 数据服务层不重复裁剪）——分类只对读口返回的 statusCode 做归类，绝不复制状态判定逻辑、
+ * 不读 Redis 原值、不硬编码状态名/色。

+ * <p><b>数据权限</b>：先会话租户守卫（无上下文 403），再 resolve(userId) 数据权限——
+ * factoryInstanceId 不在可见集即 403（通用文案不回显工厂标识）；实例树本身由
+ * {@link IotModelInstanceService#selectInstanceTree()} 按数据权限裁剪（无权节点/兄弟厂不返回），
+ * 本服务只在该树内做子树遍历，不二次补节点。读不审（0079 范式，与 status current 一致）。
+ *
+ * <p><b>fail-soft 三层</b>：单设备状态读失败 → unknown（不炸全厂）；空子树 → 全 0 真 0
+ * （available=true，新厂空态）；整厂聚合异常（树查询/外层遍历失败）→ 全 0 + available=false
+ * + 「状态数据接入中」（不把 0 冒充真 0）。
+ *
+ * @author dowei
+ * @date 2026-11-24
+ */
+@Service
+public class FactoryStatusSummaryServiceImpl implements FactoryStatusSummaryService {
+
+    private static final Logger log = LoggerFactory.getLogger(FactoryStatusSummaryServiceImpl.class);
+
+    /** 节点类型（与 IotInstanceTreeVO / DISPLAY_NODE_TYPE 对齐） */
+    private static final int NODE_TYPE_FACTORY = 1;
+    private static final int NODE_TYPE_DEVICE = 4;
+
+    private final IotModelInstanceService instanceService;
+    private final StatusReadService statusReadService;
+    private final DataPermService dataPermService;
+
+    public FactoryStatusSummaryServiceImpl(IotModelInstanceService instanceService,
+                                           StatusReadService statusReadService,
+                                           DataPermService dataPermService) {
+        this.instanceService = instanceService;
+        this.statusReadService = statusReadService;
+        this.dataPermService = dataPermService;
+    }
+
+    @Override
+    public FactoryStatusSummaryVO summary(Long factoryInstanceId) {
+        Long tenantId = requireTenant();
+        requireDataPerm(factoryInstanceId);
+        try {
+            List<IotInstanceTreeVO> tree = instanceService.selectInstanceTree();
+            IotInstanceTreeVO factory = findNode(tree, factoryInstanceId);
+            if (factory == null || !isFactory(factory)) {
+                throw forbidden();
+            }
+            // 子树内只取设备（nodeType=4）；组件（5）不单独计入
+            List<Long> deviceIds = collectDeviceIds(factory);
+            int running = 0;
+            int fault = 0;
+            int offline = 0;
+            int unknown = 0;
+            for (Long deviceId : deviceIds) {
+                String code = readStatusCode(tenantId, deviceId);
+                switch (code) {
+                    case "RUNNING" -> running++;
+                    case "FAULT" -> fault++;
+                    case "OFFLINE" -> offline++;
+                    default -> unknown++; // UNKNOWN/停机/未知自定义态一律 unknown
+                }
+            }
+            return new FactoryStatusSummaryVO(running, fault, offline, unknown, deviceIds.size());
+        } catch (BusinessException e) {
+            throw e; // 权限 403 不吞成 fail-soft
+        } catch (Exception e) {
+            log.warn("[FACTORY-SUMMARY] 整厂聚合异常 fail-soft：factory={}, err={}", factoryInstanceId, e.getMessage());
+            return FactoryStatusSummaryVO.unavailable();
+        }
+    }
+
+    /** 单设备状态读：复用 status current 取 statusCode；失败按 UNKNOWN 计（不炸全厂） */
+    private String readStatusCode(Long tenantId, Long deviceId) {
+        try {
+            StatusCurrentVO current = statusReadService.currentForClient(tenantId, deviceId);
+            return current == null || current.getStatusCode() == null || current.getStatusCode().isBlank()
+                    ? "UNKNOWN" : current.getStatusCode();
+        } catch (Exception e) {
+            log.debug("[FACTORY-SUMMARY] 单设备状态读失败按 unknown：device={}, err={}", deviceId, e.getMessage());
+            return "UNKNOWN";
+        }
+    }
+
+    /** 迭代式子树遍历（防深树栈溢出）：含根节点自身（工厂非设备，自然跳过）与全部后代设备 */
+    private List<Long> collectDeviceIds(IotInstanceTreeVO root) {
+        java.util.List<Long> out = new java.util.ArrayList<>();
+        Deque<IotInstanceTreeVO> stack = new ArrayDeque<>();
+        stack.push(root);
+        while (!stack.isEmpty()) {
+            IotInstanceTreeVO node = stack.pop();
+            if (node == null) {
+                continue;
+            }
+            if (node.getNodeType() != null && node.getNodeType() == NODE_TYPE_DEVICE && node.getId() != null) {
+                out.add(node.getId());
+            }
+            List<IotInstanceTreeVO> children = node.getChildren();
+            if (children != null) {
+                for (IotInstanceTreeVO child : children) {
+                    stack.push(child);
+                }
+            }
+        }
+        return out;
+    }
+
+    private IotInstanceTreeVO findNode(List<IotInstanceTreeVO> tree, Long id) {
+        if (tree == null || id == null) {
+            return null;
+        }
+        Deque<IotInstanceTreeVO> stack = new ArrayDeque<>(tree);
+        while (!stack.isEmpty()) {
+            IotInstanceTreeVO node = stack.pop();
+            if (node == null) {
+                continue;
+            }
+            if (id.equals(node.getId())) {
+                return node;
+            }
+            if (node.getChildren() != null) {
+                stack.addAll(node.getChildren());
+            }
+        }
+        return null;
+    }
+
+    private boolean isFactory(IotInstanceTreeVO node) {
+        return node.getNodeType() != null && node.getNodeType() == NODE_TYPE_FACTORY;
+    }
+
+    /** 会话租户：无租户上下文即 403 fail-closed（不猜默认租户、不跨租户读状态） */
+    private Long requireTenant() {
+        Long tenantId = TenantContext.getTenantId();
+        if (tenantId == null) {
+            throw forbidden();
+        }
+        return tenantId;
+    }
+
+    /** 数据权限：factoryInstanceId ∈ 可见集（allTenant 直过），否则 403 通用文案不回显标识 */
+    private void requireDataPerm(Long factoryInstanceId) {
+        VisibleScope scope = dataPermService.resolve(SecurityUtils.getUserId());
+        if (scope == null || (!scope.allTenant() && !scope.instanceIds().contains(factoryInstanceId))) {
+            throw forbidden();
+        }
+    }
+
+    private BusinessException forbidden() {
+        return new BusinessException(ResultCode.FORBIDDEN.getCode(), "无权访问该资源");
+    }
+}
