+import org.slf4j.LoggerFactory;
+
+import java.util.List;
+import java.util.Set;
+
+import static org.junit.jupiter.api.Assertions.assertEquals;
+import static org.junit.jupiter.api.Assertions.assertFalse;
+import static org.junit.jupiter.api.Assertions.assertTrue;
+import static org.mockito.Mockito.mockStatic;
+import static org.mockito.Mockito.verify;
+import static org.mockito.Mockito.verifyNoInteractions;
+import static org.mockito.Mockito.when;
+
+/**
+ * 数据权限翻译组件——入口租户上下文守卫单元测试（TASK-0005 修订轮，ADR-0016 增补第 9 条）。
+ *
+ * <p>无租户上下文时（异步线程 / 接线方漏设上下文）多租户插件对所有表跳过注入，
+ * 继续计算会静默产出跨租户全量——组件要防的最严重失效模式。定版：入口
+ * resolve/resolveClientScope 检查 TenantContext，无上下文返回空 VisibleScope
+ * （allTenant=false、集合全空）并 WARN {@code [DATAPERM] no-tenant-context user=<id 或 unknown>}。
+ *
+ * <p>standalone 测试以 mockStatic(TenantContext) 显式控制上下文有无：
+ * 未打桩时 getTenantId() 返回 null = 无上下文形态。
+ *
+ * @author dowei
+ * @date 2026-11-22
+ */
+class DataPermTenantContextGuardTest {
+
+    @Mock
+    private DataPermUserMapper userMapper;
+    @Mock
+    private DataPermRoleMapper roleMapper;
+    @Mock
+    private DataPermRoleDeptMapper roleDeptMapper;
+    @Mock
+    private DataPermDeptMapper deptMapper;
+    @Mock
+    private DataPermInstanceMapper instanceMapper;
+    @Mock
+    private VisibleScopeCache cache;
+
+    private DataPermProperties properties;
+    private DataPermServiceImpl service;
+
+    private ListAppender<ILoggingEvent> logAppender;
+
+    private MockedStatic<TenantContext> tenantContext;
+
+    private AutoCloseable mocks;
+
+    @BeforeEach
+    void setUp() {
+        mocks = MockitoAnnotations.openMocks(this);
+        // 默认不打桩：getTenantId() 返回 null = 无租户上下文（守卫用例主体）
+        tenantContext = mockStatic(TenantContext.class);
+        properties = new DataPermProperties();
+        service = new DataPermServiceImpl(userMapper, roleMapper, roleDeptMapper,
+                deptMapper, instanceMapper, cache, properties);
+        logAppender = new ListAppender<>();
+        logAppender.start();
+        ((Logger) LoggerFactory.getLogger(DataPermServiceImpl.class)).addAppender(logAppender);
+    }
+
+    @AfterEach
+    void tearDown() throws Exception {
+        tenantContext.close();
+        ((Logger) LoggerFactory.getLogger(DataPermServiceImpl.class)).detachAppender(logAppender);
+        mocks.close();
+    }
+
+    private DataPermRoleRow role(Long id, Integer dataScope) {
+        DataPermRoleRow row = new DataPermRoleRow();
+        row.setId(id);
+        row.setDataScope(dataScope);
+        return row;
+    }
+
+    /**
+     * 显式模拟无租户上下文。注意：mockStatic 未打桩时 Mockito 默认答案对包装类型 Long
+     * 返回 0 而非 null，与真实「无上下文返回 null」不符（会骗过守卫），必须显式置 null。
+     */
+    private void noTenantContext() {
+        tenantContext.when(TenantContext::getTenantId).thenReturn(null);
+    }
+
+    private List<String> warnMessages() {
+        return logAppender.list.stream()
+                .filter(e -> e.getLevel() == Level.WARN)
+                .map(ILoggingEvent::getFormattedMessage)
+                .toList();
+    }
+
+    @Test
+    @DisplayName("resolve 无租户上下文 - 返回空集 fail-closed + WARN 固定格式，不触达缓存与任何 Mapper")
+    void testResolve_NoTenantContext_DeniedWithWarn() {
+        noTenantContext();
+        VisibleScope scope = service.resolve(7L);
+
+        assertFalse(scope.allTenant());
+        assertFalse(scope.createByOnly());
+        assertEquals(Set.of(), scope.deptIds());
+        assertEquals(Set.of(), scope.instanceIds());
+        assertEquals(List.of("[DATAPERM] no-tenant-context user=7"), warnMessages(),
+                "守卫 WARN 格式为 ADR-0016 增补第 9 条硬约束（可 grep）");
+        verifyNoInteractions(cache, roleMapper, userMapper, roleDeptMapper, deptMapper, instanceMapper);
+    }
+
+    @Test
+    @DisplayName("resolveClientScope 无租户上下文 - 同样空集拒绝 + WARN（subject=clientId）")
+    void testResolveClientScope_NoTenantContext_DeniedWithWarn() {
+        noTenantContext();
+        VisibleScope scope = service.resolveClientScope(5L, "{\"instanceIds\":[1],\"subtree\":true}");
+
+        assertFalse(scope.allTenant());
+        assertEquals(Set.of(), scope.deptIds());
+        assertEquals(Set.of(), scope.instanceIds());
+        assertEquals(List.of("[DATAPERM] no-tenant-context user=5"), warnMessages());
+        verifyNoInteractions(instanceMapper, cache);
+    }
+
+    @Test
+    @DisplayName("subject 为 null - WARN 以 user=unknown 兜底（格式约定的 unknown 档）")
+    void testResolve_NullSubject_WarnUnknown() {
+        noTenantContext();
+        VisibleScope scope = service.resolve(null);
+
+        assertEquals(Set.of(), scope.deptIds());
+        assertEquals(Set.of(), scope.instanceIds());
+        assertEquals(List.of("[DATAPERM] no-tenant-context user=unknown"), warnMessages());
+    }
+
+    @Test
+    @DisplayName("有租户上下文 - 守卫放行，正常进入六步计算（不误伤合法调用）")
+    void testWithTenantContext_ComputationProceeds() {
+        tenantContext.when(TenantContext::getTenantId).thenReturn(1L);
+        when(roleMapper.selectEnabledRolesByUserId(7L)).thenReturn(List.of(role(1L, 1)));
+
+        VisibleScope scope = service.resolve(7L);
+
+        assertTrue(scope.allTenant(), "有上下文的合法调用不受守卫影响");
+        verify(roleMapper).selectEnabledRolesByUserId(7L);
+    }
+}
