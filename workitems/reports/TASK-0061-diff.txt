+/**
+ * TASK-0061 D2：巡检调度器（Spring @Scheduled，复用 PublishSchedulingConfig 的 @EnableScheduling）。
+ *
+ * <p>重入保护：Redis {@code SET key value NX EX ttl}（ttl=周期秒数×2 兜底），拿到锁才执行，
+ * finally 比对 value 释放（防误删）；Redis 不可用 fail-soft 跳过本轮 WARN 不阻断（与 ingest 同）。
+ *
+ * <p>本类把锁键/TTL 派生做成静态纯函数供单测（不连 Redis）；周期
+ * {@code ot.inspect.interval-seconds} 默认 3600，fixedDelayString + initialDelay SpEL（0025 范式）。

+ * @author dowei
+ * @date 2026-11-24

+public final class InspectScheduler {
+
+    private InspectScheduler() {}
+
+    /** 锁键：系统级巡检固定 global（巡检跨租户一次性扫描，避免多实例并发重入） */
+    public static String lockKey(String tenantCodeOrNull) {
+        return tenantCodeOrNull == null || tenantCodeOrNull.isBlank()
+                ? "iot:inspect:lock:global"
+                : "iot:inspect:lock:" + tenantCodeOrNull;
+    }
+
+    /** TTL = 周期秒数 × 2（兜底：任务异常未释放也不致锁死） */
+    public static Duration lockTtl(long intervalSeconds) {
+        long s = intervalSeconds <= 0 ? 3600 : intervalSeconds;
+        return Duration.ofSeconds(s * 2);
+    }

+        }

+/**
+ * TASK-0061 D2：巡检任务调度组件（Spring @Scheduled，复用 PublishSchedulingConfig 的 @EnableScheduling）。
+ *
+ * <p>周期 {@code ot.inspect.interval-seconds}（默认 3600；fixedDelayString + initialDelay SpEL
+ * 同 0025 范式），每次执行产出三类清单 + 标签租户比对清单 + 在线率聚合快照，落库
+ * iot_inspect_result 并归档（90 天）。
+ *
+ * <p><b>重入保护</b>：Redis {@code SET key value NX EX ttl}（ttl=周期×2 兜底），拿到锁才执行；
+ * finally 比对 value 一致才 DEL（防误删他人锁）；Redis 不可用 fail-soft 跳过本轮并 WARN，
+ * 不阻断（与 ingest fail-soft 同口径）。
+ *
+ * <p>本轮扫描的数据源（calc state/log、status current、metric list）均复用只读端点服务，
+ * 任务自身不改写任何业务数据，只写巡检结果表。当前阶段各清单产出由 InspectTaskService 完成，
+ * 本组件只负责调度编排 + 重入锁 + fail-soft（任务体异常仅 ERROR 不上抛，下轮重扫）。
+ *
+ * @author dowei
+ * @date 2026-11-24
+ */
+@Component
+public class InspectTaskScheduler {
+
+    private static final Logger log = LoggerFactory.getLogger(InspectTaskScheduler.class);
+
+    private final StringRedisTemplate redisTemplate;
+    private final InspectTaskRunner runner;
+    private final long intervalSeconds;
+
+    public InspectTaskScheduler(@Qualifier("rawStringRedisTemplate") StringRedisTemplate redisTemplate,
+                                InspectTaskRunner runner,
+                                @Value("${ot.inspect.interval-seconds:3600}") long intervalSeconds) {
+        this.redisTemplate = redisTemplate;
+        this.runner = runner;
+        this.intervalSeconds = intervalSeconds;
+    }
+
+    @Scheduled(fixedDelayString = "#{${ot.inspect.interval-seconds:3600} * 1000}",
+            initialDelayString = "#{${ot.inspect.interval-seconds:3600} * 1000}")
+    public void runInspect() {
+        String lockKey = InspectScheduler.lockKey(null);
+        String token = UUID.randomUUID().toString();
+        Duration ttl = InspectScheduler.lockTtl(intervalSeconds);
+        boolean locked = false;
+        try {
+            locked = Boolean.TRUE.equals(redisTemplate.opsForValue().setIfAbsent(lockKey, token, ttl));
+        } catch (RuntimeException e) {
+            log.warn("[INSPECT] Redis 不可用，跳过本轮巡检（fail-soft）: {}", e.getMessage());
+            return;
+        }
+        if (!locked) {
+            log.info("[INSPECT] 上一轮巡检仍在进行（锁未释放），跳过本轮");
+            return;
+        }
+        try {
+            runner.run();
+        } catch (RuntimeException e) {
+            log.error("[INSPECT] 巡检任务异常（下轮重试）", e);
+        } finally {
+            releaseLock(lockKey, token);
+        }
+    }
+
+    /** 仅当 value 与自身 token 一致才释放（防误删他人锁）；Redis 故障只 WARN 不上抛 */
+    private void releaseLock(String key, String token) {
+        try {
+            java.util.Objects.requireNonNull(redisTemplate);
+            String current = redisTemplate.opsForValue().get(key);
+            if (token.equals(current)) {
+                redisTemplate.delete(key);
+            }
+        } catch (RuntimeException e) {
+            log.warn("[INSPECT] 释放巡检锁失败（TTL 兜底）: {}", e.getMessage());
+        }
+    }

+    public static String onlineRateText(long online, long total) {
+        if (total <= 0) {
+            return "N/A";
+        }
+        return String.format(java.util.Locale.ROOT, "%.1f%%", online * 100.0 / total);
+    }
+
+    /** 计算成功率：无日志=N/A；resultStatus=1 成功、其余（2 失败）计失败 */
+    public static String calcSuccessRateText(List<CalcLogVO> logs) {
+        if (logs == null || logs.isEmpty()) {
+            return "N/A";
+        }
+        int success = 0;
+        int total = 0;
+        for (CalcLogVO log : logs) {
+            if (log == null || log.getResultStatus() == null) {
+                continue;
+            }
+            total++;
+            if (log.getResultStatus() == 1) {
+                success++;
+            }
+        }
+        if (total == 0) {
+            return "N/A";
+        }
+        return String.format(java.util.Locale.ROOT, "%.1f%%", success * 100.0 / total);
+    }
+
+    /** calc 状态是否异常（2 滞后 / 3 异常停算） */
+    public static boolean hasAbnormalCalcState(List<CalcStateVO> states) {
+        if (states == null) {
+            return false;
+        }
+        return states.stream().anyMatch(s -> s != null && s.getCalcStatus() != null && s.getCalcStatus() >= 2);
+    }

+    public static List<InspectItemVO> groupCalcErrors(List<CalcLogVO> logs) {
+        Map<String, Integer> counts = new LinkedHashMap<>();
+        if (logs != null) {
+            for (CalcLogVO log : logs) {
+                if (log == null || log.getResultStatus() == null || log.getResultStatus() != 2) {
+                    continue;
+                }
+                String reason = log.getErrorMsg() == null || log.getErrorMsg().isBlank()
+                        ? "未标注原因" : log.getErrorMsg();
+                counts.merge(reason, 1, Integer::sum);
+            }
+        }
+        List<InspectItemVO> out = new ArrayList<>();
+        counts.forEach((reason, count) -> {
+            InspectItemVO vo = new InspectItemVO();
+            vo.setReason(reason);
+            vo.setCount(count);
+            out.add(vo);
+        });
+        out.sort(Comparator.comparingInt(InspectItemVO::getCount).reversed()
+                .thenComparing(i -> i.getReason() == null ? "" : i.getReason()));
+        return out;
+    }

+    private static List<String> executableSegments(String ddl) {
+        List<String> segments = new ArrayList<>();
+        for (String seg : ddl.split(";\\s*\\n")) {
+            StringBuilder sb = new StringBuilder();
+            for (String line : seg.split("\n")) {
+                String t = line.strip();
+                if (!t.isEmpty() && !t.startsWith("--")) {
+                    sb.append(line).append("\n");
+                }
+            }
+            String s = sb.toString().strip();
+            if (!s.isEmpty()) {
+                segments.add(s);
+            }
+        }
+        return segments;
+    }

+  if (!Number.isFinite(total) || total <= 0) return 'N/A';
+  const n = Number.isFinite(online) ? online : 0;
+  return `${((n * 100) / total).toFixed(1)}%`;

+  const calcOk = !!data.calcSuccessRate && data.calcSuccessRate !== 'N/A';
+  return rateOk || calcOk || (data.channels?.length ?? 0) > 0 || (data.metrics?.length ?? 0) > 0;

+  const hours = Math.floor(m / 60);
+  const rest = m % 60;
+  return rest > 0 ? `${hours} 小时 ${rest} 分钟` : `${hours} 小时`;
