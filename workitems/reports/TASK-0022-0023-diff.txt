+     * @param xlsxInput   .xlsx 文件输入流（方法内消费并随 OPCPackage 关闭）
+     * @param maxDataRows 数据行（非可跳过行）上限，超过立即中止解析
+     * @return 行清单（按物理行顺序；空表/无行返回空清单，由服务层拒绝）
+     * @throws PointImportRowLimitException 数据行超过 maxDataRows（fail fast 中止）
+     * @throws IORuntimeException           文件损坏/非 xlsx 等解析失败（由全局异常处理器兜底）
+     */
+    public static List<PointImportRow> parse(InputStream xlsxInput, int maxDataRows) {
+        try (OPCPackage pkg = OPCPackage.open(xlsxInput)) {
+            List<String> sharedStrings = loadSharedStrings(pkg);
+            XSSFReader xssf = new XSSFReader(pkg);
+            Iterator<InputStream> sheets = xssf.getSheetsData();
+            if (!sheets.hasNext()) {
+                return new ArrayList<>();
+            }
+            try (InputStream sheet = sheets.next()) {
+                SheetRowHandler handler = new SheetRowHandler(sharedStrings, maxDataRows);
+                XMLReader xmlReader = XMLHelper.newXMLReader();
+                xmlReader.setContentHandler(handler);
+                xmlReader.parse(new InputSource(sheet));
+                return handler.rows;
+            }
+        } catch (PointImportRowLimitException e) {
+            throw e;
+        } catch (SAXException | IOException | javax.xml.parsers.ParserConfigurationException
+                | org.apache.poi.openxml4j.exceptions.OpenXML4JException e) {
+            throw new IORuntimeException("导入 Excel 解析失败: " + e.getMessage(), e);
+        }
+    }
+
+    /** ECMA-376 sharedStrings 关系类型（POI 5.2.5 的 PackageRelationshipTypes 无此常量） */
+    private static final String SHARED_STRINGS_REL =
+            "http://schemas.openxmlformats.org/officeDocument/2006/relationships/sharedStrings";
+
+    /**
+     * sharedStrings.xml 部件 SAX 展平（纯文本清单，无 JAXB 元素树）；部件缺失返回空清单。
+     * 关系挂载层与字典解析器同口径（sharedStrings 关系挂在 workbook 部件上而非包根）。
+     */
+    private static List<String> loadSharedStrings(OPCPackage pkg) throws SAXException, IOException,
+            javax.xml.parsers.ParserConfigurationException,
+            org.apache.poi.openxml4j.exceptions.InvalidFormatException {
+        List<String> list = new ArrayList<>();
+        PackagePart workbookPart = null;
+        for (PackageRelationship rel : pkg.getRelationshipsByType(PackageRelationshipTypes.CORE_DOCUMENT)) {
+            workbookPart = pkg.getPart(rel);
+            break;
+        }
+        if (workbookPart == null) {
+            return list;
+        }
+        for (PackageRelationship rel : workbookPart.getRelationshipsByType(SHARED_STRINGS_REL)) {
+            PackagePart part = workbookPart.getRelatedPart(rel);
+            if (part == null) {
+                continue;
+            }
+            try (InputStream in = part.getInputStream()) {
+                SharedStringsHandler handler = new SharedStringsHandler(list);
+                XMLReader xmlReader = XMLHelper.newXMLReader();
+                xmlReader.setContentHandler(handler);
+                xmlReader.parse(new InputSource(in));
+            }
+        }
+        return list;
+    }
+
+    /** {@code <si>} 内全部 {@code <t>} 文本拼接（富文本 run 合并为整串）。 */
+    private static final class SharedStringsHandler extends DefaultHandler {
+        private final List<String> list;
+        private final StringBuilder buf = new StringBuilder();
+        private boolean inSi = false;
+        private boolean inT = false;
+
+        SharedStringsHandler(List<String> list) {
+            this.list = list;
+        }
+
+        @Override
+        public void startElement(String uri, String localName, String qName, Attributes attributes) {
+            if ("si".equals(localName)) {
+                inSi = true;
+                buf.setLength(0);
+            } else if ("t".equals(localName) && inSi) {
+                inT = true;
+            }
+        }
+
+        @Override
+        public void characters(char[] ch, int start, int length) {
+            if (inT) {
+                buf.append(ch, start, length);
+            }
+        }
+
+        @Override
+        public void endElement(String uri, String localName, String qName) {
+            if ("t".equals(localName) && inT) {
+                inT = false;
+            } else if ("si".equals(localName) && inSi) {
+                inSi = false;
+                list.add(buf.toString());
+            }
+        }
+    }
+
+    /** 首个 sheet 的行级 ContentHandler：物理行号取 r 属性，逐行产出 {@link PointImportRow}。 */
+    private static final class SheetRowHandler extends DefaultHandler {
+        private final List<String> sharedStrings;
+        private final int maxDataRows;
+        private final List<PointImportRow> rows = new ArrayList<>();
+        private int dataCount = 0;
+
+        private long rowNumber = 0;          // 当前物理行号（r 属性，1-based；缺省时上一行 + 1）
+        private int cellColumn = -1;         // 当前单元格列号（r 属性推导，0-based）
+        private String cellType = null;      // 当前单元格 t 类型（null = "n" 数值）
+        private boolean inValue = false;     // 位于 <v> 内
+        private boolean inInlineText = false; // 位于 inlineStr 的 <is> 内
+        private final StringBuilder buf = new StringBuilder();
+        private List<String> cells = new ArrayList<>(); // 当前行按列位存放（稀疏补 null）
+        private int maxCol = -1;
+
+        SheetRowHandler(List<String> sharedStrings, int maxDataRows) {
+            this.sharedStrings = sharedStrings;
+            this.maxDataRows = maxDataRows;
+        }
+
+        @Override
+        public void startElement(String uri, String localName, String qName, Attributes attributes) {
+            switch (localName) {
+                case "row":
+                    String r = attributes.getValue("r");
+                    rowNumber = r != null ? Long.parseLong(r) : rowNumber + 1;
+                    cells = new ArrayList<>();
+                    maxCol = -1;
+                    break;
+                case "c":
+                    cellColumn = colIndex(attributes.getValue("r"));
+                    cellType = attributes.getValue("t");
+                    buf.setLength(0);
+                    break;
+                case "v":
+                    inValue = true;
+                    buf.setLength(0);
+                    break;
+                case "is":
+                    inInlineText = true;
+                    break;
+                case "t":
+                    if (inInlineText) {
+                        buf.setLength(0);
+                    }
+                    break;
+                default:
+                    break;
+            }
+        }
+
+        @Override
+        public void characters(char[] ch, int start, int length) {
+            if (inValue || inInlineText) {
+                buf.append(ch, start, length);
+            }
+        }
+
+        @Override
+        public void endElement(String uri, String localName, String qName) {
+            switch (localName) {
+                case "v":
+                    setCell(cellValue(cellType, buf.toString()));
+                    inValue = false;
+                    break;
+                case "t":
+                    if (inInlineText) {
+                        setCell(buf.toString());
+                    }
+                    break;
+                case "c":
+                    cellType = null;
+                    inValue = false;
+                    break;
+                case "is":
+                    inInlineText = false;
+                    break;
+                case "row":
+                    finishRow();
+                    break;
+                default:
+                    break;
+            }
+        }
+
+        private void setCell(String value) {
+            if (cellColumn < 0) {
+                return;
+            }
+            // 空串归一 null：与字典解析器同口径——「显式空串」与「缺格」在导入语义上等价
+            // （十三列均无空串合法值），统一为 null 后 isSkippable（全空行）与必填缺失判定
+            // 对两种编码口径行为一致
+            if (value != null && value.isEmpty()) {
+                value = null;
+            }
+            while (cells.size() <= cellColumn) {
+                cells.add(null);
+            }
+            cells.set(cellColumn, value);
+            if (cellColumn > maxCol) {
+                maxCol = cellColumn;
+            }
+        }
+
+        private String cellValue(String type, String v) {
+            if (v == null || v.isEmpty()) {
+                return null;
+            }
+            if ("s".equals(type)) {
+                // 共享字符串：v 为索引
+                int idx;
+                try {
+                    idx = Integer.parseInt(v.trim());
+                } catch (NumberFormatException e) {
+                    return null;
+                }
+                return idx >= 0 && idx < sharedStrings.size() ? sharedStrings.get(idx) : null;
+            }
+            if ("b".equals(type)) {
+                return "1".equals(v.trim()) ? "true" : "false";
+            }
+            if ("str".equals(type) || "e".equals(type)) {
+                return v; // 公式缓存字符串 / 错误值原样
+            }
+            return numericValue(v); // "n"（缺省类型）
+        }
+
+        /** 整数值（含 1002.0 形态）转十进制整数字符串；非整数保留原文；非法数值原样兜底。 */
+        private static String numericValue(String v) {
+            try {
+                double d = Double.parseDouble(v.trim());
+                if (Double.isFinite(d) && d == Math.rint(d) && Math.abs(d) < 9.0E15) {
+                    return String.valueOf((long) d);
+                }
+                return v;
+            } catch (NumberFormatException e) {
+                return v;
+            }
+        }
+
+        /** Excel 单元格引用（如 M12）→ 0-based 列号；无引用返回 -1。 */
+        private static int colIndex(String ref) {
+            if (ref == null || ref.isEmpty()) {
+                return -1;
+            }
+            int idx = 0;
+            for (int i = 0; i < ref.length(); i++) {
+                char ch = ref.charAt(i);
+                if (ch >= 'A' && ch <= 'Z') {
+                    idx = idx * 26 + (ch - 'A' + 1);
+                } else if (ch >= 'a' && ch <= 'z') {
+                    idx = idx * 26 + (ch - 'a' + 1);
+                } else {
+                    break; // 行号数字段开始，列字母段结束
+                }
+            }
+            return idx - 1; // A → 0
+        }
+
+        private void finishRow() {
+            long physical = rowNumber; // 无 r 属性的行已在 startElement 按「上一行 + 1」递增
+            if (physical <= 1) {
+                return; // 物理第 1 行 = 表头
+            }
+            PointImportRow row = new PointImportRow();
+            row.setRowIndex((int) physical);
+            row.setPointCode(trim(cells, 0));
+            row.setPointName(trim(cells, 1));
+            row.setDataType(trim(cells, 2));
+            row.setValueType(trim(cells, 3));
+            row.setUnit(trim(cells, 4));
+            row.setChannelCode(trim(cells, 5));
+            row.setBoundInstanceId(trim(cells, 6));
+            row.setDeptId(trim(cells, 7));
+            row.setMinValue(trim(cells, 8));
+            row.setMaxValue(trim(cells, 9));
+            row.setSamplePeriod(trim(cells, 10));
+            row.setDeadband(trim(cells, 11));
+            row.setRemark(trim(cells, 12));
+            rows.add(row);
+            if (!row.isSkippable() && ++dataCount > maxDataRows) {
+                // I-3 fail fast：RuntimeException 经 JAXP XMLReader.parse 原样透传出 parse()
+                throw new PointImportRowLimitException(dataCount);
+            }
+        }
+
+        private static String trim(List<String> cells, int index) {
+            String value = index < cells.size() ? cells.get(index) : null;
+            return value == null ? null : value.trim();
+        }
+    }
