+/**
+ * 计算状态机（TASK-0047）：水位推进 + 连续错误停算（§1.9 calc_status 1/2/3）。
+ *
+ * <ul>
+ *   <li>成功：last_calc_time=窗口 start、watermark=max(旧, 窗口 end)（单调不回退）、
+ *       error_count 归零、calc_status 回 1（手工重算恢复缝）；</li>
+ *   <li>失败：error_count 自增（连续），达阈值→calc_status=3 异常停算 + last_error；
+ *       中间成功打断连续计数；</li>
+ *   <li>滞后：evaluateLag 按 now-watermark 与 lagThresholdMs 判 2（不覆盖已停算的 3）。</li>
+ * </ul>
+ *
+ * <p>只管单 fullKey 状态迁移；子树隔离由 task 层沿依赖图向下游传播（§brief 决策 2）。
+ *
+ * @author dowei
+ * @date 2026-11-24
+ */
+public final class CalcStateMachine {
+
+    private final CalcStateStore store;
+    private final int errorStopThreshold;
+    private final long lagThresholdMs;
+
+    public CalcStateMachine(CalcStateStore store, int errorStopThreshold, long lagThresholdMs) {
+        this.store = store;
+        this.errorStopThreshold = Math.max(1, errorStopThreshold);
+        this.lagThresholdMs = Math.max(0, lagThresholdMs);
+    }
+
+    /** 成功完成一个窗口：推进水位、重置连续错误 */
+    public CalcState onSuccess(long tenantId, String fullKey, long windowStart, long windowEnd) {
+        CalcState old = store.get(tenantId, fullKey);
+        long wm = old == null ? windowEnd : Math.max(old.watermark(), windowEnd);
+        long lastCalc = old == null ? windowStart : Math.max(old.lastCalcTime(), windowStart);
+        CalcState next = CalcState.of(tenantId, fullKey)
+                .lastCalcTime(lastCalc).watermark(wm)
+                .calcStatus(1).errorCount(0).lastError(null)
+                .updateTime(System.currentTimeMillis()).build();
+        store.upsert(next);
+        return next;
+    }
+
+    /** 失败一次：error_count 连续自增，达阈值置 3 停算 */
+    public CalcState onFailure(long tenantId, String fullKey, long windowStart,
+                               String taskId, String error) {
+        CalcState old = store.get(tenantId, fullKey);
+        int count = (old == null ? 0 : old.errorCount()) + 1;
+        int status = count >= errorStopThreshold ? 3 : (old != null ? old.calcStatus() : 1);
+        // 已停算节点继续失败仍停算；未达阈值保持原状态（可能因滞后=2，不擅自回 1）
+        if (count >= errorStopThreshold) {
+            status = 3;
+        }
+        CalcState next = CalcState.of(tenantId, fullKey)
+                .lastCalcTime(old == null ? windowStart : old.lastCalcTime())
+                .watermark(old == null ? 0 : old.watermark())
+                .calcStatus(status).errorCount(count)
+                .lastError(error == null ? taskId : error)
+                .updateTime(System.currentTimeMillis()).build();
+        store.upsert(next);
+        return next;
+    }
+
+    /** 是否异常停算（task 层据此跳过该 fullKey 及其子树） */
+    public boolean isStopped(long tenantId, String fullKey) {
+        CalcState s = store.get(tenantId, fullKey);
+        return s != null && s.stopped();
+    }
+
+    /**
+     * 因上游停算而隔离该节点（子树隔离 §brief 决策 2）：标记 calc_status=3，
+     * 原因写明上游，同步停算态给状态页；不改 error_count（非自身连续失败）。
+     */

+        CalcState old = store.get(tenantId, fullKey);
+        if (old != null && old.stopped()) {
+            return old;
+        }
+        CalcState next = CalcState.of(tenantId, fullKey)
+                .lastCalcTime(old == null ? 0 : old.lastCalcTime())
+                .watermark(old == null ? 0 : old.watermark())
+                .calcStatus(3)
+                .errorCount(old == null ? 0 : old.errorCount())
+                .lastError("上游停算隔离: " + upstream)
+                .updateTime(System.currentTimeMillis()).build();
+        store.upsert(next);
+        return next;
+    }
+
+    /**
+     * 滞后判据（仅更新未停算节点）：now - watermark > lagThresholdMs → calc_status=2。
+     * 停算节点(3)不被滞后覆盖；未登记节点返回缺省 1。
+     */
+    public CalcState evaluateLag(long tenantId, String fullKey, long nowMillis) {
+        CalcState old = store.get(tenantId, fullKey);

+        }

+        }
+        CalcState next = CalcState.of(tenantId, fullKey)
+                .lastCalcTime(old.lastCalcTime()).watermark(old.watermark())
+                .calcStatus(status).errorCount(old.errorCount()).lastError(old.lastError())
+                .updateTime(System.currentTimeMillis()).build();
+        store.upsert(next);
+        return next;
+    }

+        if (fullKeys == null || fullKeys.isEmpty()) {
+            return List.of();

+        Set<String> included = new LinkedHashSet<>(fullKeys);
+        Map<String, GraphNode> nodes = new LinkedHashMap<>();
+        for (String key : fullKeys) {

+        }
+        List<GraphEdge> edges = new ArrayList<>();
+        for (Map.Entry<String, Set<String>> e : deps.entrySet()) {
+            if (!included.contains(e.getKey())) {
+                continue;
+            }
+            for (String up : e.getValue()) {
+                // from 依赖 up：边 from→up（to=up 先输出）
+                if (included.contains(up)) {
+                    edges.add(new GraphEdge(e.getKey(), up,
+                            net.otlab.calc.graph.EdgeType.METRIC_SOURCE, "recalc:" + e.getKey() + "<-" + up));
+                }
+            }
+        }
+        try {
+            return GraphBuilder.topoSort(nodes, edges);
+        } catch (RuntimeException e) {
+            // 环/内部矛盾 fail-closed：退提交序，不让补算任务丢失
+            return new ArrayList<>(fullKeys);
+        }
+    }

+    private void run(String taskId, String tenant, String triggerType, List<RecalcItem> items,
+                     Map<String, Set<String>> dependencies) {
+        tasks.put(taskId, new CalcTask(taskId, triggerType, tenant,
+                CalcTask.Status.RUNNING, items.size(), 0, 0, List.of()));
+        long tenantId = 1L; // 单租户形态（tenant_id 默认 1；0063 落库时由装配注入真实 id）
+        RecalcContext ctx = new RecalcContext(taskId, triggerType, true, tenantId, stateMachine);
+        int done = 0;
+        int failed = 0;
+        List<CalcTask.ItemResult> results = new ArrayList<>();
+        // 本次任务内新停算的节点（沿依赖图向下游子树传播，兄弟分支不受影响）
+        Set<String> stoppedThisRun = ConcurrentHashMap.newKeySet();
+        for (RecalcItem item : items) {
+            String stoppingUpstream = findStoppedUpstream(item.fullKey(), dependencies,
+                    stoppedThisRun, tenantId);
+            boolean alreadyStopped = stateMachine.isStopped(tenantId, item.fullKey());
+            if (alreadyStopped || stoppingUpstream != null) {
+                if (stoppingUpstream != null && !alreadyStopped) {
+                    stateMachine.stopByUpstream(tenantId, item.fullKey(), stoppingUpstream);
+                }
+                continue; // 停算节点/子树跳过（兄弟继续）
+            }
+            boolean ok;
+            try {
+                ok = compute.compute(item, ctx);
+            } catch (RuntimeException e) {
+                ok = false;
+                stateMachine.onFailure(tenantId, item.fullKey(), item.periodStart(), taskId,
+                        e.getMessage() == null ? e.getClass().getSimpleName() : e.getMessage());
+            }
+            if (ok) {
+                done++;
+                results.add(CalcTask.ItemResult.ok(item.fullKey(), item.periodStart(), item.periodEnd()));
+            } else {
+                failed++;
+                CalcState s = stateStore.get(tenantId, item.fullKey());
+                results.add(CalcTask.ItemResult.fail(item.fullKey(),
+                        s == null ? "failed" : String.valueOf(s.lastError())));
+                if (stateMachine.isStopped(tenantId, item.fullKey())) {
+                    stoppedThisRun.add(item.fullKey()); // 本次新停算→下游子树跳过
+                }
+            }
+            tasks.put(taskId, new CalcTask(taskId, triggerType, tenant,
+                    CalcTask.Status.RUNNING, items.size(), done, failed, results));
+        }
+        CalcTask.Status finalStatus = failed == 0 ? CalcTask.Status.COMPLETED : CalcTask.Status.FAILED;
+        tasks.put(taskId, new CalcTask(taskId, triggerType, tenant,
+                finalStatus, items.size(), done, failed, results));
+    }
+
+    /**
+     * 节点是否位于「已停算节点的下游子树」（沿 dependencies 的传递闭包）。
+     * dependencies 中 key 依赖 value（上游）；本方法检查 node 的上游链上是否有停算节点。
+     */
+    private String findStoppedUpstream(String node, Map<String, Set<String>> dependencies,
+                                       Set<String> stoppedThisRun, long tenantId) {
+        java.util.Deque<String> stack = new java.util.ArrayDeque<>();
+        Set<String> seen = new java.util.HashSet<>();
+        stack.push(node);
+        while (!stack.isEmpty()) {
+            String cur = stack.pop();
+            if (!seen.add(cur)) {
+                continue;
+            }
+            for (String up : dependencies.getOrDefault(cur, Set.of())) {
+                if (stateMachine.isStopped(tenantId, up) || stoppedThisRun.contains(up)) {
+                    return up;
+                }
+                stack.push(up);
+            }
+        }
+        return null;
+    }
