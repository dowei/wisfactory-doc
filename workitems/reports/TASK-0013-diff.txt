+    @Override

+        if (instance == null) {
+            throw paramError("实例不存在或已删除: " + instanceId);
+        }
+        // 组织节点（工厂/车间/产线未选模板）零生成：无清单可生成，非错误
+        if (instance.getTemplateId() == null) {

+        }
+        IotModelTemplate template = modelTemplateMapper.selectById(instance.getTemplateId());
+        if (template == null) {
+            throw paramError("来源模板不存在或已删除: " + instance.getTemplateId());
+        }
+        // 模板数据项清单（生成源，模板域只读——清单序按模板项 id 升序，生成序确定性）
+        List<IotDataItemTemplate> templateItems = dataItemTemplateMapper.selectList(
+                new LambdaQueryWrapper<IotDataItemTemplate>()
+                        .eq(IotDataItemTemplate::getTemplateId, template.getId())
+                        .orderByAsc(IotDataItemTemplate::getId));
+        if (templateItems.isEmpty()) {
+            return 0;
+        }
+
+        // 幂等跳过预检（双道闸第一道，含已删行旁路）：已占用 (instance_id, data_code) 跳过
+        // 不重建——活跃行已存在跳过（幂等），已删行占用亦跳过（两 uk 均不含 deleted 列，
+        // 已删键永久占用；补插必撞物理索引且被误报并发冲突，照 0011 修订轮 I-1 口径）
+        Map<String, IotDataItem> occupiedByCode = new HashMap<>();
+        for (IotDataItem row : baseMapper.selectOccupancyByInstance(TenantContext.getTenantId(), instance.getId())) {
+            occupiedByCode.put(row.getDataCode(), row);
+        }
+        List<IotDataItemTemplate> newItems = templateItems.stream()
+                .filter(t -> !occupiedByCode.containsKey(t.getDataCode()))
+                .collect(Collectors.toList());
+        if (newItems.isEmpty()) {
+            return 0;
+        }
+
+        // 父实例（PARENT 展开目标，lazy）：仅公式含 PARENT 引用时解析（parentId null/0=顶级
+        // 返回 null；其余按 id 查活跃行，多租户插件注入会话租户条件即同租户保证）——

+        boolean parentResolved = false;
+        // SELF 引用存在性集合 = 实例既有活跃项编码（占用行 deleted=0 部分）∪ 本批全部模板项编码
+        Set<String> selfScope = new LinkedHashSet<>();

+                selfScope.add(code);
+            }
+        });
+        templateItems.forEach(t -> selfScope.add(t.getDataCode()));
+
+        // 父实例活跃项 (full_key → id)（lazy：仅 PARENT 引用存在性核对/回填需要时首查触发）
+        Map<String, Long> parentItemIdByFullKey = null;
+
+        // 逐项映射 + 展开（展开失败/引用缺失即整批失败——本方法 @Transactional 整批回滚，
+        // 触发一的实例创建同事务一并回滚）
+        List<IotDataItem> toInsert = new ArrayList<>(newItems.size());

+        for (IotDataItemTemplate templateItem : newItems) {

+            String templateFormula = templateItem.getDefaultFormula();
+            if (templateFormula != null && !templateFormula.isBlank()) {
+                if (RelativeRefExpander.containsParentRef(templateFormula) && !parentResolved) {
+                    parent = resolveParent(instance);
+                    parentResolved = true;
+                }
+                String parentInstanceCode = parentResolved && parent != null ? parent.getInstanceCode() : null;
+                RelativeRefExpander.Outcome outcome = RelativeRefExpander.expand(
+                        templateFormula, instance.getInstanceCode(), parentInstanceCode);
+                for (String selfCode : outcome.selfCodes()) {
+                    if (!selfScope.contains(selfCode)) {
+                        throw paramError("公式引用的数据项不存在: SELF." + selfCode
+                                + "（实例 " + instance.getInstanceCode() + "，项 " + templateItem.getDataCode() + "）");
+                    }
+                }
+                for (String parentCode : outcome.parentCodes()) {

+                        throw paramError("公式引用的父实例数据项不存在: PARENT." + parentCode
+                                + "（父实例 " + parent.getInstanceCode() + "，项 " + templateItem.getDataCode() + "）");
+                    }
+                }
+                if (outcome.expandedFormula().length() > MAX_LEN_FORMULA) {
+                    throw paramError("展开后公式超长（上限 " + MAX_LEN_FORMULA + "）: 项 " + templateItem.getDataCode());

+        if (templateFormula == null || templateFormula.isBlank()) {
+            return false;
+        }
+        Matcher matcher = RELATIVE_REF.matcher(templateFormula);
+        while (matcher.find()) {

+     * @param templateFormula     模板默认公式（花括号占位形态；null/空白原样返回空引用集）
+     * @param instanceCode        目标实例全码（SELF 展开目标）

+     *                            {@link BusinessException}（400，文案含「顶级实例」）
+     * @return 展开结果（公式 + 各类引用集）
+     * @throws BusinessException 顶级实例的 PARENT 引用（整批失败语义由调用方承担）
+     */
+    public static Outcome expand(String templateFormula, String instanceCode, String parentInstanceCode) {
+        LinkedHashSet<String> selfCodes = new LinkedHashSet<>();
+        LinkedHashSet<String> parentCodes = new LinkedHashSet<>();
+        LinkedHashSet<String> resolvedRefs = new LinkedHashSet<>();
+        if (templateFormula == null || templateFormula.isBlank()) {
+            return new Outcome(templateFormula, selfCodes, parentCodes, resolvedRefs);
+        }
+

+        StringBuilder expanded = new StringBuilder();
+        while (matcher.find()) {
+            String kind = matcher.group(1).toUpperCase();
+            String code = matcher.group(2);
+            if ("SELF".equals(kind)) {
+                selfCodes.add(code);
+                String fullKey = instanceCode + "." + code;
+                matcher.appendReplacement(expanded, Matcher.quoteReplacement(fullKey));
+                resolvedRefs.add(fullKey);
+            } else {
+                if (parentInstanceCode == null) {
+                    throw new BusinessException(ResultCode.PARAM_ERROR.getCode(),
+                            "顶级实例不支持 PARENT 相对引用（无父实例可解析）: PARENT." + code
+                                    + "（实例 " + instanceCode + "）");
+                }
+                parentCodes.add(code);
+                String fullKey = parentInstanceCode + "." + code;
+                matcher.appendReplacement(expanded, Matcher.quoteReplacement(fullKey));
+                resolvedRefs.add(fullKey);
+            }
+        }
+        matcher.appendTail(expanded);
+
+        // 残留花括号 token = 绝对全键引用（{F01.WS01.EQ002.ELEC_CONS} 形态）：剥壳保留键、
+        // 逐引用入依赖图（不核存在性——被引实例可能后建，一期口径）

+        StringBuilder bare = new StringBuilder();
+        while (absolute.find()) {
+            String key = absolute.group(1);
+            absolute.appendReplacement(bare, Matcher.quoteReplacement(key));
+            resolvedRefs.add(key);
+        }
+        absolute.appendTail(bare);
+
+        return new Outcome(bare.toString(), selfCodes, parentCodes, resolvedRefs);
+    }
