+func loadConfig(getenv func(string) string) Config {
+	cfg := Config{

+		BackendBaseURL:   getenv("INGEST_BACKEND_BASE_URL"),
+		InternalToken:    getenv("INGEST_INTERNAL_TOKEN"),

+		MQTTClientID:     getenv("INGEST_MQTT_CLIENT_ID"),
+		RedisAddr:        getenv("INGEST_REDIS_ADDR"),
+		SnapshotTenant:   getenv("INGEST_SNAPSHOT_TENANT"),
+		SnapshotInterval: 30 * time.Second,
+		DedupTTL:         time.Hour,
+		GatewayMap:       map[string]gwtoken.ResolvedChannel{},
+	}
+	if cfg.HTTPAddr == "" {
+		cfg.HTTPAddr = ":8090"
+	}
+	if v := getenv("INGEST_SNAPSHOT_SCOPE"); v != "" {
+		if st, sid, err := parseScope(v); err == nil {
+			cfg.SnapshotScopeType, cfg.SnapshotScopeID = st, sid
+		}
+	}
+	if v := getenv("INGEST_SNAPSHOT_INTERVAL"); v != "" {
+		if d, err := time.ParseDuration(v); err == nil && d > 0 {
+			cfg.SnapshotInterval = d
+		}
+	}
+	if v := getenv("INGEST_DEDUP_TTL"); v != "" {
+		if d, err := time.ParseDuration(v); err == nil && d > 0 {
+			cfg.DedupTTL = d
+		}
+	}
+	for _, entry := range strings.Split(getenv("INGEST_GATEWAY_MAP"), ";") {
+		entry = strings.TrimSpace(entry)
+		if entry == "" {
+			continue
+		}
+		parts := strings.Split(entry, ":")
+		if len(parts) != 3 {
+			continue
+		}
+		chID, err := strconv.ParseInt(strings.TrimSpace(parts[1]), 10, 64)
+		if err != nil {
+			continue
+		}
+		cfg.GatewayMap[strings.TrimSpace(parts[0])] = gwtoken.ResolvedChannel{
+			ChannelID:  chID,
+			TenantCode: strings.TrimSpace(parts[2]),
+		}
+	}
+	return cfg
+}

+func parseScope(v string) (int, int64, error) {
+	parts := strings.SplitN(v, ":", 2)
+	if len(parts) != 2 {
+		return 0, 0, fmt.Errorf("scope 形态须为 scopeType:scopeId")
+	}
+	st, err := strconv.Atoi(strings.TrimSpace(parts[0]))
+	if err != nil {
+		return 0, 0, err
+	}
+	sid, err := strconv.ParseInt(strings.TrimSpace(parts[1]), 10, 64)
+	if err != nil {
+		return 0, 0, err
+	}
+	return st, sid, nil
+}

+	ConnectFailures int64 // broker 连接失败次数（fail-soft）
+	Reconnects      int64 // 断线重连次数
+}

+// Consumer MQTT 消费源：按 §3.2 主题矩阵路由，data 信封进流水线消费通道（实现
+// pipeline.Consumer 接口，零改动替换 0073 假消费源），status/config 信封进对应出口；
+// 监督循环承担连接（指数退避 fail-soft）与重连重订阅。
+type Consumer struct {
+	port Port
+	opts Options
+
+	msgs chan pipeline.Inbound // 有界队列（背压阻塞，不丢弃）
+
+	ctxMu sync.Mutex
+	ctx   context.Context // Run 期间的路由取消语境（满队阻塞的退出通道）
+
+	dataRouted      atomic.Int64
+	statusRouted    atomic.Int64
+	configRouted    atomic.Int64
+	malformedTopics atomic.Int64
+	connectFailures atomic.Int64
+	reconnects      atomic.Int64
+}
+
+// NewConsumer 构造消费源（port 必填；opts 零值字段取默认）。
+// 装配形态：pipeline.Deps.Consumer = 本消费源（Messages() 即 data 主题信封通道），
+// Run 与 pipeline.Run 并行（各自 goroutine）。
+func NewConsumer(port Port, opts Options) *Consumer {
+	if opts.QueueDepth <= 0 {
+		opts.QueueDepth = 1024
+	}
+	if opts.BackoffInitial <= 0 {
+		opts.BackoffInitial = 500 * time.Millisecond
+	}
+	if opts.BackoffMax <= 0 {
+		opts.BackoffMax = 30 * time.Second
+	}
+	if opts.Logger == nil {
+		opts.Logger = log.Default()
+	}
+	return &Consumer{
+		port: port,
+		opts: opts,
+		msgs: make(chan pipeline.Inbound, opts.QueueDepth),
+	}
+}
+
+// Messages 实现 pipeline.Consumer：data 主题信封通道（有界；满则路由侧背压阻塞）。
+func (c *Consumer) Messages() <-chan pipeline.Inbound { return c.msgs }
+
+// Stats 计数快照。
+func (c *Consumer) Stats() Stats {
+	return Stats{
+		DataRouted:      c.dataRouted.Load(),
+		StatusRouted:    c.statusRouted.Load(),
+		ConfigRouted:    c.configRouted.Load(),
+		MalformedTopics: c.malformedTopics.Load(),
+		ConnectFailures: c.connectFailures.Load(),
+		Reconnects:      c.reconnects.Load(),
+	}
+}
+
+func (c *Consumer) logger() *log.Logger { return c.opts.Logger }
+
+// Run 监督循环：连接（失败指数退避重试——broker 不可达 fail-soft，不阻断 HTTP 上报
+// 路径）→ 订阅矩阵 → 等待断线/停机；断线后重连并重订阅；ctx 取消时释放连接退出。
+func (c *Consumer) Run(ctx context.Context) {
+	c.ctxMu.Lock()
+	c.ctx = ctx
+	c.ctxMu.Unlock()
+
+	lost := make(chan struct{}, 1)
+	c.port.SetConnectionLostHandler(func(err error) {
+		c.logger().Printf("mqttsource: broker 连接丢失（fail-soft，重连重订阅）: %v", err)
+		select {
+		case lost <- struct{}{}:
+		default:
+		}
+	})
+
+	backoff := c.opts.BackoffInitial
+	for {
+		if ctx.Err() != nil {
+			c.port.Close()
+			return
+		}
+		if err := c.port.Connect(); err != nil {
+			c.connectFailures.Add(1)
+			c.logger().Printf("mqttsource: broker 不可达（fail-soft 不阻断 HTTP 上报路径，%v 后重试——指数退避）: %v", backoff, err)
+			if !sleepCtx(ctx, backoff) {
+				c.port.Close()
+				return
+			}
+			backoff = minDur(backoff*2, c.opts.BackoffMax)
+			continue
+		}
+		backoff = c.opts.BackoffInitial // 连接成功：退避重置
+		if err := c.subscribeAll(); err != nil {
+			c.logger().Printf("mqttsource: 订阅失败（fail-soft 重试）: %v", err)
+			c.port.Close()
+			if !sleepCtx(ctx, backoff) {
+				return
+			}
+			continue
+		}
+		select {
+		case <-ctx.Done():
+			c.port.Close()
+			return
+		case <-lost:
+			c.reconnects.Add(1)
+			if !sleepCtx(ctx, backoff) {
+				c.port.Close()
+				return
+			}
+		}
+	}
+}
+
+// subscribeAll 订阅 §3.2 矩阵（三类主题形态；LWT 与 status 同主题无需独立订阅）。
+func (c *Consumer) subscribeAll() error {
+	for _, topic := range Subscriptions {
+		if err := c.port.Subscribe(topic, c.route); err != nil {
+			return fmt.Errorf("mqttsource: 订阅 %q 失败: %w", topic, err)
+		}
+	}
+	return nil
+}
+
+// route 主题路由（§3.2 矩阵——四主题→信封类型）：broker 消息回调内执行。
+func (c *Consumer) route(topic string, payload []byte) {
+	tenant, gateway, kind, ok := SplitTopic(topic)
+	if !ok {
+		c.malformedTopics.Add(1)
+		return
+	}
+	switch kind {
+	case KindData:
+		// 上行·数据：统一消息信封 → 数据流水线消费源（租户取主题槽位——认证回调
+		// 授予的主题矩阵即令牌侧产物）。满队背压阻塞（不丢弃），ctx 取消退出。
+		c.ctxMu.Lock()
+		ctx := c.ctx
+		c.ctxMu.Unlock()
+		if ctx == nil {
+			ctx = context.Background()
+		}
+		select {
+		case c.msgs <- pipeline.Inbound{Tenant: tenant, Payload: payload}:
+			c.dataRouted.Add(1)
+		case <-ctx.Done():
+		}
+	case KindStatus:
+		c.statusRouted.Add(1)
+		env := parseStatus(topic, tenant, gateway, payload)
+		if c.opts.StatusSink != nil {
+			c.opts.StatusSink.OnStatus(env)
+		} else {
+			c.defaultStatusSink(env)
+		}
+	case KindConfig:
+		c.configRouted.Add(1)
+		n := parseConfigNotice(topic, tenant, gateway, payload)
+		if c.opts.ConfigSink != nil {
+			c.opts.ConfigSink.OnConfigNotice(n)
+		} else {
+			c.defaultConfigSink(n)
+		}
+	}
+}
+
+// defaultStatusSink 默认状态出口：计数+日志（§3.2 LWT 行「接入服务据此置 online_status=0
+// 并告警」的告警挂点——存储与告警通道归联调，TD-0057 登记）。
+func (c *Consumer) defaultStatusSink(env StatusEnvelope) {
+	if env.Graceful {
+		c.logger().Printf("mqttsource: 网关正常下线（graceful）tenant=%s gateway=%s", env.Tenant, env.Gateway)
+		return
+	}
+	c.logger().Printf("mqttsource: 网关状态上报（心跳/LWT——LWT 异常断开须告警，online_status 存储归联调）tenant=%s gateway=%s status=%q",
+		env.Tenant, env.Gateway, env.Status)
+}
+
+// defaultConfigSink 默认配置出口：本期不消费（决策 4——拉取对账兜底已覆盖），计数+日志。
+func (c *Consumer) defaultConfigSink(n ConfigNotice) {
+	c.logger().Printf("mqttsource: config.changed 通知（本期不消费——拉取对账兜底）tenant=%s gateway=%s version=%d hash=%s",
+		n.Tenant, n.Gateway, n.Notice.Version, n.Notice.Hash)
+}
+
+// parseStatus 宽松解析状态载荷（契约未冻结字段级 Schema——未知字段忽略）。
+func parseStatus(topic, tenant, gateway string, payload []byte) StatusEnvelope {
+	env := StatusEnvelope{Topic: topic, Tenant: tenant, Gateway: gateway, Payload: payload}
+	var m map[string]any
+	if err := json.Unmarshal(payload, &m); err != nil {
+		return env // 非法 JSON：保留现场，graceful=false（按异常断开语义告警）
+	}
+	if v, ok := m["graceful"]; ok {
+		env.Graceful = truthy(v)
+	}
+	if v, ok := m["status"].(string); ok {
+		env.Status = v
+	}
+	return env
+}
+
+// truthy 宽松真值判定（bool/非零数/非空真值串）。
+func truthy(v any) bool {
+	switch t := v.(type) {
+	case bool:
+		return t
+	case float64:
+		return t != 0
+	case string:
+		return t != "" && t != "false" && t != "0"
+	default:
+		return false
+	}
+}
+
+// parseConfigNotice 解析 config.changed 通知（§3.3 载荷五键；解析失败保留现场）。
+func parseConfigNotice(topic, tenant, gateway string, payload []byte) ConfigNotice {
+	n := ConfigNotice{Topic: topic, Tenant: tenant, Gateway: gateway, Payload: payload}
+	if cc, err := snapshot.ParseConfigChanged(payload); err == nil {
+		n.Notice = *cc
+	}
+	return n
+}
+
+func sleepCtx(ctx context.Context, d time.Duration) bool {
+	t := time.NewTimer(d)
+	defer t.Stop()
+	select {
+	case <-ctx.Done():
+		return false
+	case <-t.C:
+		return true
+	}
+}
+
+func minDur(a, b time.Duration) time.Duration {
+	if a < b {
+		return a
+	}
+	return b
+}

+// DefaultConnectTimeout 初连超时（监督循环快速失败——退避节奏不被悬挂拉长）。
+const DefaultConnectTimeout = 5 * time.Second
+
+// PahoPort paho.mqtt.golang 适配。
+type PahoPort struct {
+	// BrokerURL broker 地址（如 tcp://emqx-host:1883）。
+	BrokerURL string
+	// Username/Password broker 账号（空 = 匿名）。
+	Username string
+	Password string
+	// ClientID 客户端 ID；空 = ingestd-<pid>（多实例不得共享——会话互踢）。
+	ClientID string
+	// ConnectTimeout 初连超时；<=0 → DefaultConnectTimeout。
+	ConnectTimeout time.Duration
+
+	client pahomqtt.Client
+	lost   func(err error)
+}
+
+func (p *PahoPort) clientID() string {
+	if p.ClientID != "" {
+		return p.ClientID
+	}
+	return fmt.Sprintf("ingestd-%d", os.Getpid())
+}
+
+// Connect 建立连接（初连失败返回错误；会话内断线由 paho 自动重连）。
+func (p *PahoPort) Connect() error {
+	timeout := p.ConnectTimeout
+	if timeout <= 0 {
+		timeout = DefaultConnectTimeout
+	}
+	opts := pahomqtt.NewClientOptions().
+		AddBroker(p.BrokerURL).
+		SetClientID(p.clientID()).
+		SetUsername(p.Username).
+		SetPassword(p.Password).
+		SetCleanSession(false).
+		SetAutoReconnect(true).
+		SetResumeSubs(true).
+		SetConnectRetry(false). // 初连失败快速返回——重连节奏归监督循环（指数退避）
+		SetConnectTimeout(timeout).
+		SetConnectionLostHandler(func(_ pahomqtt.Client, err error) {
+			if p.lost != nil {
+				p.lost(err)
+			}
+		})
+	p.client = pahomqtt.NewClient(opts)
+	tok := p.client.Connect()
+	if !tok.WaitTimeout(timeout + 2*time.Second) {
+		return fmt.Errorf("paho: 连接超时（%v）", timeout)
+	}
+	if err := tok.Error(); err != nil {
+		return fmt.Errorf("paho: 连接失败: %w", err)
+	}
+	return nil
+}
+
+// Subscribe QoS 1 订阅（§3.2 QoS 列逐字）。
+func (p *PahoPort) Subscribe(topic string, handler func(topic string, payload []byte)) error {
+	tok := p.client.Subscribe(topic, 1, func(_ pahomqtt.Client, m pahomqtt.Message) {
+		handler(m.Topic(), m.Payload())
+	})
+	if !tok.WaitTimeout(10 * time.Second) {
+		return fmt.Errorf("paho: 订阅 %q 超时", topic)
+	}
+	if err := tok.Error(); err != nil {
+		return fmt.Errorf("paho: 订阅 %q 失败: %w", topic, err)
+	}
+	return nil
+}
+
+// SetConnectionLostHandler 注册连接丢失回调（监督循环重连重订阅）。
+func (p *PahoPort) SetConnectionLostHandler(fn func(err error)) {
+	p.lost = fn
+}
+
+// Close 断开连接（250ms 静默期让 in-flight 消息收尾；幂等）。
+func (p *PahoPort) Close() {
+	if p.client != nil && p.client.IsConnectionOpen() {
+		p.client.Disconnect(250)
+	}
+}

+// TASK-0027 决策 3：幂等真实现（单实例形态）——进程内有界 TTL 去重，替换内存假去重。
+//
+// 口径：
+//   - 原子性沿用 0076 修订轮范式（BUG-0002 红线）：MarkIfSeen 首见判定与标记在单锁内
+//     一次完成——check-then-act 两段式不可达，同键并发重报恰一处理；本实现不得降级为
+//     先查后写（与 BatchDeduper/PointDeduper 接口文档同纪律）。
+//   - TTL 即幂等窗口：窗口内重报=非首见；窗口过后同键视为新键（QoS1 重投超出窗口属
+//     新样本场景）。窗口默认 DefaultDedupTTL，必须 ≥ 网关 QoS1 重投窗口（broker 未确认
+//     消息在重连会话内重投，小时级）——配置过小的责任在装配方。
+//   - 有界性双保险：容量上限（live 条目数 ≤ cap，超限 FIFO 淘汰最早标记项，对齐
+//     ADR-0014 丢最旧口径）+ TTL 惰性过期回收（标记时顺带清理过期项，每项只出队一次，
+//     分摊 O(1)）——内存占用有界，不随运行时长无界增长。
+//   - 多实例演进位（登记）：单实例形态（README 并发模型起步口径）用进程内 TTL 去重；
+//     多实例水平扩展时幂等键必须共享存储 → Redis 化（SETNX+EXPIRE 等价单命令原子语义），
+//     经 BatchDeduper/PointDeduper 接口零改动替换（归 TASK-0028 存储面定版）。
+package pipeline
+
+import (
+	"fmt"
+	"sync"
+	"time"
+)
+
+// DefaultDedupTTL 幂等键保留窗口默认值（1h）：须 ≥ 常见 QoS1 重投窗口（broker 会话/
+// 重连窗口，小时级）；ingestd 经 INGEST_DEDUP_TTL 配置覆盖。
+const DefaultDedupTTL = time.Hour
+
+// defaultDedupCap 与内存假实现同默认容量。
+const defaultDedupCap = 65536
+
+// ttlEntry FIFO 队列节点：键 + 代次 + 标记时刻。同键 TTL 过期后重标记会追加新节点
+// （旧节点不出队、以代次隔离）——旧节点出队时代次不再匹配，不会误删新标记。
+type ttlEntry struct {
+	key string
+	gen uint64
+	at  time.Time
+}
+
+// ttlTable 有界 TTL 去重表内核（batchId 与 (pointCode,ts) 两适配器共用同一实现，
+// 消除第二份幂等语义逻辑）。
+type ttlTable struct {
+	mu    sync.Mutex
+	cap   int
+	ttl   time.Duration
+	now   func() time.Time // 时钟（真实时间；测试经短 TTL 驱动，不留注入口）
+	gen   uint64           // 代次发生器（单调递增）
+	live  int              // 活条目数（=map 大小；order 可含已过期/被覆盖的陈旧节点）
+	seen  map[string]ttlEntry
+	order []ttlEntry // 插入序 FIFO（淘汰与惰性过期都从头部出队）
+}
+
+func newTTLTable(capacity int, ttl time.Duration) *ttlTable {
+	if capacity <= 0 {
+		capacity = defaultDedupCap
+	}
+	if ttl <= 0 {
+		ttl = DefaultDedupTTL
+	}
+	return &ttlTable{
+		cap:  capacity,
+		ttl:  ttl,
+		now:  time.Now,
+		seen: make(map[string]ttlEntry, capacity),
+	}
+}
+
+// markIfSeen 原子标记并报告是否首见（单锁内完成：过期回收 → 窗口判定 → 容量淘汰 →
+// 标记，check-then-act 不可达）。
+func (t *ttlTable) markIfSeen(key string) bool {
+	t.mu.Lock()
+	defer t.mu.Unlock()
+	now := t.now()
+
+	// 惰性过期回收：仅出队头部过期项（插入序≈到期序，每项只出队一次，分摊 O(1)）。
+	// 队中部项到期由下方「标记时判过期」兜底，不依赖出队及时性。
+	for len(t.order) > 0 && now.Sub(t.order[0].at) >= t.ttl {
+		head := t.order[0]
+		t.order = t.order[1:]
+		if cur, ok := t.seen[head.key]; ok && cur.gen == head.gen {
+			delete(t.seen, head.key)
+			t.live--
+		}
+	}
+
+	if e, ok := t.seen[key]; ok {
+		if now.Sub(e.at) < t.ttl {
+			return false // 窗口内重报：非首见
+		}
+		// 已过期：视为首见（TTL=幂等窗口）。覆盖标记；旧节点仍在队中，出队时按代次
+		// 隔离（gen 不再匹配则跳过删除），live 计数不变（键仍在 map）。
+	}
+
+	// 容量淘汰：live ≥ cap 时出队至淘汰掉一个活条目为止（FIFO 最早标记项）。
+	if t.live >= t.cap {
+		for len(t.order) > 0 {
+			head := t.order[0]
+			t.order = t.order[1:]
+			if cur, ok := t.seen[head.key]; ok && cur.gen == head.gen {
+				delete(t.seen, head.key)
+				t.live--
+				break
+			}
+		}
+	}
+
+	t.gen++
+	t.seen[key] = ttlEntry{key: key, gen: t.gen, at: now}
+	t.order = append(t.order, ttlEntry{key: key, gen: t.gen, at: now})
+	t.live++
+	return true
+}
+
+// len 活条目数（观测：内存有界的直接证据）。
+func (t *ttlTable) len() int {
+	t.mu.Lock()
+	defer t.mu.Unlock()
+	return t.live
+}
+
+// ttlTTL 返回生效的幂等窗口（观测/装配方自检）。
+func (t *ttlTable) ttlTTL() time.Duration {
+	return t.ttl
+}
+
+// TTLBatchDeduper batchId 整批幂等真实现（实现 BatchDeduper 接口）。

+			return false
+		}
+	}

+// TD 登记随 TD-0057），周期拉取兜底覆盖 REQ-0033 的下游对账语义。
+//
+// 消费方式（快照 Schema v1.0.1 §3.3「接入消费方式」逐字对齐）：
+//	收到通知后按版本拉新快照、校验、原子替换缓存（新快照校验失败则继续用旧版并回执失败）

+// 哈希校验 + Schema 校验通过 → Cache.Replace 原子换入（0073 缓存）；失败保持旧版、
+// 计数并告警日志（fail-soft——拉取失败不阻断 HTTP 上报路径），下个窗口重试。
+// 版本与哈希均未变化时跳过替换（对账幂等，不无谓换入）。

+// DefaultReconcileInterval 拉取对账默认周期（30s）：通知缺位时的变更感知上限；
+// ingestd 经 INGEST_SNAPSHOT_INTERVAL 配置覆盖。
+const DefaultReconcileInterval = 30 * time.Second
+
+// Reconciler 拉取对账循环。
+type Reconciler struct {
+	// Fetcher 快照拉取客户端（真实现 InternalSnapshotClient；测试可注入）。
+	Fetcher SnapshotFetcher
+	// Cache 本地快照缓存（0073 原子替换缓存）。
+	Cache *Cache
+	// Scope 对账范围（Version 字段忽略——恒拉最新版）。
+	Scope Scope
+	// Interval 对账周期；<=0 → DefaultReconcileInterval。
+	Interval time.Duration
+	// Logger 告警日志（失败/替换）；nil → log.Default()。
+	Logger *log.Logger
+
+	pulls    atomic.Int64
+	failures atomic.Int64
+	replaced atomic.Int64
+	skipped  atomic.Int64
+
+	// mu/lastVersion/lastHash 本进程拉取锚（§3.3「校验哈希」对账口径）：同版本哈希
+	// 漂移 = 存储被篡改/端点异常探测（fail-soft 拒收保持旧版）。
+	mu          sync.Mutex
+	lastVersion int64
+	lastHash    string
+}
+
+func (r *Reconciler) logger() *log.Logger {
+	if r.Logger != nil {
+		return r.Logger
+	}
+	return log.Default()
+}
+
+// Pulls 累计拉取次数（观测）。
+func (r *Reconciler) Pulls() int64 { return r.pulls.Load() }
+
+// Failures 累计拉取失败次数（观测）。
+func (r *Reconciler) Failures() int64 { return r.failures.Load() }
+
+// Replaced 累计替换次数（观测）。
+func (r *Reconciler) Replaced() int64 { return r.replaced.Load() }
+
+// Skipped 同版本同哈希跳过次数（观测）。
+func (r *Reconciler) Skipped() int64 { return r.skipped.Load() }
+
+// Run 启动对账循环：启动即拉取一次（不等首个 tick——空缓存尽快装载），此后每
+// Interval 对账一次；阻塞直至 ctx 取消（优雅停机组成部分）。
+func (r *Reconciler) Run(ctx context.Context) {
+	r.pullOnce(ctx)
+	interval := r.Interval
+	if interval <= 0 {
+		interval = DefaultReconcileInterval
+	}
+	ticker := time.NewTicker(interval)
+	defer ticker.Stop()
+	for {
+		select {
+		case <-ctx.Done():
+			return
+		case <-ticker.C:
+			r.pullOnce(ctx)
+		}
+	}
+}
+
+// PullOnce 单次对账：拉最新 → 哈希/校验由 Fetcher 契约承担 → 同版本哈希漂移探测 →
+// 版本哈希未变跳过 → 原子替换；失败保持旧版、计数、告警日志（fail-soft）。
+func (r *Reconciler) PullOnce(ctx context.Context) error { return r.pullOnce(ctx) }
+
+func (r *Reconciler) pullOnce(ctx context.Context) error {
+	r.pulls.Add(1)
+	res, err := r.Fetcher.Fetch(ctx, r.Scope)
+	if err != nil {
+		r.failures.Add(1)
+		r.logger().Printf("snapshot: 对账拉取失败（fail-soft 保持旧版，下个窗口重试）: %v", err)
+		return err
+	}
+
+	r.mu.Lock()
+	lastVersion, lastHash := r.lastVersion, r.lastHash
+	r.mu.Unlock()
+	if lastVersion == res.Version && lastHash != "" && res.SnapshotHash != "" {
+		if res.SnapshotHash != lastHash {
+			// 同版本哈希漂移：存储被篡改/端点异常探测（§3.3「校验哈希」对账口径）——
+			// fail-soft 拒收保持旧版；恢复后下个窗口收敛。
+			r.failures.Add(1)
+			r.logger().Printf("snapshot: 对账哈希漂移（同 version=%d，存储被篡改/端点异常探测——保持旧版）last=%s got=%s",
+				res.Version, lastHash, res.SnapshotHash)
+			return fmt.Errorf("snapshot: 同版本哈希漂移 version=%d last=%s got=%s", res.Version, lastHash, res.SnapshotHash)
+		}
+		// 版本未变且哈希稳定：跳过替换（对账幂等），避免无谓换入。
+		r.skipped.Add(1)
+		return nil
+	}
+	if err := r.Cache.Replace(res.Snapshot); err != nil {
+		r.failures.Add(1)
+		r.logger().Printf("snapshot: 对账快照替换失败（fail-soft 保持旧版）: %v", err)
+		return err
+	}
+	r.mu.Lock()
+	r.lastVersion, r.lastHash = res.Version, res.SnapshotHash
+	r.mu.Unlock()
+	r.replaced.Add(1)
+	r.logger().Printf("snapshot: 对账快照已替换 version=%d hash=%s", res.Version, res.SnapshotHash)
+	return nil
+}

+// 契约依据：ctx 取消后对账循环退出（优雅停机组成部分）。
+func TestContractSnapshotReconcilerStopsOnContextCancel(t *testing.T) {
+	s := newReconcileServer(t, 17)
+	cache := snapshot.NewCache()
+	r := newReconciler(s.srv.URL, cache, 5*time.Millisecond)
+
+	ctx, cancel := context.WithCancel(context.Background())
+	done := make(chan struct{})
+	go func() { r.Run(ctx); close(done) }()
+
+	waitFor(t, "循环已拉取", func() bool { return r.Pulls() >= 1 })
+	cancel()
+	select {
+	case <-done:
+	case <-time.After(2 * time.Second):
+		t.Fatal("ctx 取消后 Reconciler 未退出")
+	}
+	n := r.Pulls()
+	time.Sleep(60 * time.Millisecond)
+	if r.Pulls() != n {
+		t.Errorf("取消后仍在拉取: %d → %d", n, r.Pulls())
+	}
+}

+func waitFor(t *testing.T, what string, cond func() bool) {
+	t.Helper()
+	deadline := time.Now().Add(3 * time.Second)
+	for time.Now().Before(deadline) {
+		if cond() {
+			return
+		}
+		time.Sleep(2 * time.Millisecond)
+	}
+	t.Fatalf("等待超时: %s", what)
+}
