+/** 默认公式限长口径（与 iot_data_item_template DDL default_formula varchar(2000)、0011 MAX_LEN_FORMULA 一致） */
+export const FORMULA_MAX_LENGTH = 2000;

+/** 相对引用关键字白名单（Expander RELATIVE_REF：(SELF|PARENT)，大小写不敏感） */
+export const FORMULA_RELATIVE_PREFIXES = ['SELF', 'PARENT'] as const;
+
+/** 花括号 token 字符集（Expander BRACE_REF：[A-Za-z0-9_.]+，含绝对全键多级点号） */
+export const FORMULA_BRACE_TOKEN_PATTERN = /^[A-Za-z0-9_.]+$/;

+/** 相对引用完整形态（Expander RELATIVE_REF 同款含 CASE_INSENSITIVE：关键字大小写不敏感，编码 [A-Za-z0-9_]+） */

+  | 'FORMULA_BRACE_UNBALANCED'
+  | 'FORMULA_BRACE_NESTED'
+  | 'FORMULA_TOKEN_EMPTY'
+  | 'FORMULA_TOKEN_CHARSET'
+  | 'FORMULA_RELATIVE_SHAPE'
+  | 'FORMULA_PAREN_UNBALANCED';

+/**
+ * 校验一条模板公式（客户端实时提示；与后端口径关系见文件头「从严登记」）。
+ *
+ * 规则顺序对齐后端：① 长度（0011）② 花括号配平/嵌套 ③ token 字符集与前缀形态
+ * （0013 口径）④ 圆括号配平（0011）。空/空白公式可空通过（0011 口径）。
+ * 返回全部命中问题（前端提示聚合展示；保存校验以后端为准）。
+ */
+export function validateFormula(formula: string | null | undefined): FormulaIssue[] {
+  if (formula == null || formula.trim() === '') {
+    return [];
+  }
+  const issues: FormulaIssue[] = [];
+  if (formula.length > FORMULA_MAX_LENGTH) {
+    issues.push({ code: 'FORMULA_LENGTH_EXCEED', message: FORMULA_ERROR_MESSAGES.FORMULA_LENGTH_EXCEED });
+  }
+  const braceIssues = checkBraces(formula);
+  issues.push(...braceIssues.issues);
+  // 花括号未配平/嵌套时 token 边界不可靠，跳过 token 级规则（配平类文案已给出）
+  if (braceIssues.extractable) {
+    for (const token of braceIssues.tokens) {
+      if (!token) {
+        issues.push({ code: 'FORMULA_TOKEN_EMPTY', message: FORMULA_ERROR_MESSAGES.FORMULA_TOKEN_EMPTY });
+        continue;
+      }
+      if (!FORMULA_BRACE_TOKEN_PATTERN.test(token)) {
+        issues.push({ code: 'FORMULA_TOKEN_CHARSET', message: FORMULA_ERROR_MESSAGES.FORMULA_TOKEN_CHARSET });
+        continue;
+      }
+      // 相对引用前缀白名单：token 以 SELF./PARENT. 开头（大小写不敏感）时必须满足
+      // Expander RELATIVE_REF 完整形态（编码 [A-Za-z0-9_]+）；其余 token 一律按
+      // 绝对全键引用放行（{OTHER.X} 合法，0013 口径）
+      const lower = token.toLowerCase();
+      const relativeShaped = FORMULA_RELATIVE_PREFIXES.some((p) => lower.startsWith(`${p.toLowerCase()}.`));
+      if (relativeShaped && !FORMULA_RELATIVE_REF_PATTERN.test(`{${token}}`)) {
+        issues.push({ code: 'FORMULA_RELATIVE_SHAPE', message: FORMULA_ERROR_MESSAGES.FORMULA_RELATIVE_SHAPE });
+      }
+    }
+  }
+  if (!checkParensBalanced(formula)) {
+    issues.push({ code: 'FORMULA_PAREN_UNBALANCED', message: FORMULA_ERROR_MESSAGES.FORMULA_PAREN_UNBALANCED });
+  }
+  return issues;
+}

+/** 花括号扫描结果：问题列表 + 是否可安全提取 token（配平且无嵌套） */
+interface BraceScanResult {
+  issues: FormulaIssue[];
+  extractable: boolean;
+  tokens: string[];
+}
+
+/**
+ * 花括号配平 + 嵌套检查（配对序扫描：计数不允许转负、结束必须归零——0011 计数
+ * 口径的从严版，}{ 形态前端拒绝）；配平且无嵌套时顺带提取 token 内容。
+ */
+function checkBraces(formula: string): BraceScanResult {
+  let depth = 0;
+  let nested = false;
+  const tokens: string[] = [];
+  let current = '';
+  for (const ch of formula) {
+    if (ch === '{') {
+      depth += 1;
+      if (depth > 1) {
+        nested = true;
+      }
+      current = '';
+      continue;
+    }
+    if (ch === '}') {
+      depth -= 1;
+      if (depth < 0) {
+        // 多余右括号：保持负值让归零判定失败，current 不再累积
+        depth = Number.MIN_SAFE_INTEGER;
+        continue;
+      }
+      if (depth === 0) {
+        tokens.push(current);
+        current = '';
+      }
+      continue;
+    }
+    if (depth === 1) {
+      current += ch;
+    }
+  }
+  const balanced = depth === 0;
+  const issues: FormulaIssue[] = [];
+  if (!balanced) {
+    issues.push({ code: 'FORMULA_BRACE_UNBALANCED', message: FORMULA_ERROR_MESSAGES.FORMULA_BRACE_UNBALANCED });
+  }
+  if (nested) {
+    issues.push({ code: 'FORMULA_BRACE_NESTED', message: FORMULA_ERROR_MESSAGES.FORMULA_BRACE_NESTED });
+  }
+  return { issues, extractable: balanced && !nested, tokens: balanced && !nested ? tokens : [] };
+}
+
+/** 圆括号配平（配对序扫描，0011 计数口径的从严版） */
+function checkParensBalanced(formula: string): boolean {
+  let depth = 0;
+  for (const ch of formula) {
+    if (ch === '(') {
+      depth += 1;
+    } else if (ch === ')') {
+      depth -= 1;
+      if (depth < 0) {
+        return false;
+      }
+    }
+  }
+  return depth === 0;
+}
+
+/**
+ * 提取花括号 token 内容（保序；仅在配平且无嵌套时返回，否则空数组——
+ * 与 Expander resolvedRefs 的收集前提供语法保证，结果供引用清单预演/回显用）。
+ */
+export function extractBraceTokens(formula: string): string[] {
+  if (formula == null || formula.trim() === '') {
+    return [];
+  }
+  const scan = checkBraces(formula);
+  return scan.extractable ? scan.tokens.filter((t) => t !== '') : [];
+}

+// ---------------- 引用选择器数据包装（编辑器与 0036/0016 复用） ----------------
+
+/** 引用选择器数据项（code=唯一键编码；name=展示名，可空） */
+export interface FormulaRefOption {
+  code: string;
+  name?: string;
+}
+
+/** 引用类型：SELF 本实例 / PARENT 父实例 / ABSOLUTE 跨设备绝对全键 */
+export type FormulaRefType = 'SELF' | 'PARENT' | 'ABSOLUTE';
+
+/** 引用选择器分组（两级：引用域分组 → 组内数据项；按 instance/数据项两级口径） */
+export interface FormulaRefGroup {
+  key: string;
+  label: string;
+  refType: FormulaRefType;
+  disabled?: boolean;
+  disabledReason?: string;
+  items: FormulaRefOption[];
+}
+
+/** 按引用类型包装 token：SELF/PARENT → {SELF.code}；ABSOLUTE → {code}（0013 口径） */
+export function buildRefToken(refType: FormulaRefType, code: string): string {
+  if (refType === 'ABSOLUTE') {
+    return `{${code}}`;
+  }
+  return `{${refType}.${code}}`;
+}
+
+/** buildDefaultRefGroups 参数（selfItems 必填；parentItems 缺位走置灰提示） */
+export interface BuildDefaultRefGroupsParams {
+  selfItems: FormulaRefOption[];
+  selfLabel?: string;
+  selfEmptyHint?: string;
+  parentItems?: FormulaRefOption[];
+  parentLabel?: string;
+  parentHint?: string;
+}
+
+/**
+ * 构建编辑器默认引用分组（SELF 本域 + PARENT 父域两级；本实例/模板数据项清单
+ * 由调用方按既有 API 取得后传入——编辑器组件自身不发请求，保持可复用）。
+ */
+export function buildDefaultRefGroups(params: BuildDefaultRefGroupsParams): FormulaRefGroup[] {
+  const selfEmpty = params.selfItems.length === 0;
+  const self: FormulaRefGroup = {
+    key: 'self',
+    label: params.selfLabel || '本实例数据项（SELF）',
+    refType: 'SELF',
+    disabled: selfEmpty && !!params.selfEmptyHint,
+    disabledReason: selfEmpty ? params.selfEmptyHint : undefined,
+    items: params.selfItems,
+  };
+  const parentEmpty = !params.parentItems || params.parentItems.length === 0;
+  const parent: FormulaRefGroup = {
+    key: 'parent',
+    label: params.parentLabel || '父实例数据项（PARENT）',
+    refType: 'PARENT',
+    disabled: parentEmpty && !!params.parentHint,
+    disabledReason: parentEmpty ? params.parentHint : undefined,
+    items: params.parentItems || [],
+  };
+  return [self, parent];
+}
