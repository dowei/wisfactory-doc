+        Map<Long, String[]> actualFiltered = new LinkedHashMap<>();
+        for (Map.Entry<Long, String[]> e : actual.entrySet()) {

+    @DisplayName("双向核对 - 控制器四码全部被 SQL 登记，4075-4078 四行全部挂端点（无死行）")
+    void controllerPermsAllRegisteredInSql() throws Exception {
+        Set<String> fromSql = new TreeSet<>();
+        parseMenuRows(readScript(SCRIPT_RELATIVE)).values().forEach(row -> fromSql.add(row[1]));

+        Set<String> fromCode = collectControllerPerms();
+        assertTrue(fromSql.containsAll(fromCode),
+                "管理面权限码必须全部在 sys_menu SQL 登记（BUG-0046 口径）——否则任何角色（含超管）都无法持有");

+    /** 收集调用方管理面控制器的全部 @SaCheckPermission 权限码（方法级 + 类级；SaMode.OR 多码全收） */
+    private Set<String> collectControllerPerms() {
+        Set<String> perms = new TreeSet<>();
+        Class<?> clazz = OpenApiClientController.class;
+        assertTrue(clazz.isAnnotationPresent(RestController.class),
+                clazz.getSimpleName() + " 必须是 @RestController（路由扫描前提）");
+        SaCheckPermission classLevel = clazz.getAnnotation(SaCheckPermission.class);
+        if (classLevel != null) {
+            perms.addAll(Arrays.asList(classLevel.value()));
+        }
+        for (Method method : clazz.getDeclaredMethods()) {
+            SaCheckPermission annotation = method.getAnnotation(SaCheckPermission.class);
+            if (annotation != null) {
+                perms.addAll(Arrays.asList(annotation.value()));
+            }
+        }
+        return perms;
+    }

+    /** 解析升级脚本 sys_menu INSERT 中的 (id → [parent_id, perms]) 映射（4000/4075 之下的行） */
+    private Map<Long, String[]> parseMenuRows(String sqlText) {
+        Map<Long, String[]> rows = new LinkedHashMap<>();
+        Matcher matcher = MENU_ROW_PATTERN.matcher(sqlText);
+        while (matcher.find()) {
+            rows.put(Long.parseLong(matcher.group(1)),
+                    new String[]{matcher.group(2), matcher.group(4)});
+        }
+        return rows;
+    }

+    private String readScript(String relative) throws Exception {
+        Path dir = Paths.get("").toAbsolutePath();
+        while (dir != null && !Files.exists(dir.resolve("resources/sql/01_base.sql"))) {
+            dir = dir.getParent();
+        }
+        assertTrue(dir != null, "未定位到仓库 resources/sql：请从仓库根目录或模块目录运行测试");
+        Path script = dir.resolve(relative);
+        assertTrue(Files.exists(script), "升级脚本不存在：" + script);
+        return Files.readString(script, StandardCharsets.UTF_8);
+    }
