+package net.otlab.calc.metric;
+
+import java.util.ArrayList;
+import java.util.LinkedHashSet;
+import java.util.List;
+import java.util.Set;
+
+/**
+ * 指标聚合计算（TASK-0045 brief 决策 2/3）：
+ *
+ * <ul>
+ * <li>读数走已物化结果 {@link MetricValueStore}——不现算原始点（0042/0049 ResultSink 数据 + Redis）；</li>
+ * <li>小时→日→月→年逐级聚合（级联 src_type 由 {@link MetricDef} 携带，来源取下级物化结果）；</li>
+ * <li>比值型（aggType=5）分子分母分别聚合再相除（不是逐条比值再平均），分母 0 结果为空并打标记；</li>
+ * <li>平均口径：period=DAY 且 aggType=1 时「当日总量÷24」（固定 24 小时分母），≠ 对到达值取均值；</li>
+ * <li>完整度：应到周期值到达率，非 GOOD 质量码与停用项计缺失；低于阈值带不完整标记；</li>
+ * <li>无数据出无值（value=null）+ 完整度 0，不写 0 冒充（CONV-02）。</li>
+ * </ul>
+ *
+ * @author dowei
+ * @date 2026-11-24
+ */
+public final class MetricAggregator {
+
+    /**
+     * 聚合一个窗口。
+     *
+     * @param def           指标定义
+     * @param source        已物化读数源
+     * @param ts            窗口起始（写 st_metric_value.ts）
+     * @param expectedCount 应到周期值数（完整度分母；平均口径 DAY 固定传 24）
+     * @param minCompleteness 完整度阈值（默认 0.8）
+     */
+    public MetricResult aggregate(MetricDef def, MetricValueStore source, long ts,
+                                  int expectedCount, double minCompleteness) {
+        if (def.isRatio()) {
+            return aggregateRatio(def, source, ts, expectedCount, minCompleteness);
+        }
+        // 平均口径：DAY + aggType=1 固定按「当日总量÷24」——非对到达值取均值（TC-0015）。
+        // 用 expectedCount（DAY 传 24）作固定分母，区别于普通 AVG 的到达数分母。
+        if (def.aggType() == 1 && "DAY".equalsIgnoreCase(def.period())) {
+            return aggregateAverageDiv24(def, source, ts, expectedCount, minCompleteness);
+        }
+        List<Double> values = readValues(def, source);
+        int arrived = values.size();
+        double completeness = expectedCount <= 0 ? 0.0
+                : Math.min(1.0, arrived / (double) expectedCount);
+        boolean noData = arrived == 0;
+        Double value = noData ? null : aggregateValue(def, values);
+        boolean incomplete = noData || completeness < minCompleteness;
+        return new MetricResult(value, completeness, incomplete, noData, ts);
+    }
+
+    /** 从物化源读值（sourceKeys 显式键 + 同前缀级联展开——不现算、去重） */
+    private List<Double> readValues(MetricDef def, MetricValueStore source) {
+        Set<String> keys = new LinkedHashSet<>(def.sourceKeys());
+        Set<String> seen = new LinkedHashSet<>();
+        List<Double> out = new ArrayList<>();
+        for (String k : keys) {
+            // 显式键（精确存在时直接读；否则按前缀展开物化子结果）
+            Double v = source.goodValue(k);
+            if (v != null) {
+                if (seen.add(k)) {
+                    out.add(v);
+                }
+            }
+            for (java.util.Map.Entry<String, Double> e : source.allGood().entrySet()) {
+                if (e.getKey().startsWith(k) && seen.add(e.getKey())) {
+                    out.add(e.getValue());
+                }
+            }
+        }
+        return out;
+    }
+
+    /** 普通型聚合值：1 平均（总量÷24）/ 2 累计 / 3 最大 / 4 最小 */
+    private Double aggregateValue(MetricDef def, List<Double> values) {
+        double sum = values.stream().mapToDouble(Double::doubleValue).sum();
+        return switch (def.aggType()) {
+            case 1 -> values.stream().mapToDouble(Double::doubleValue).average().orElse(sum);
+            case 2 -> sum;
+            case 3 -> values.stream().mapToDouble(Double::doubleValue).max().orElse(0.0);
+            case 4 -> values.stream().mapToDouble(Double::doubleValue).min().orElse(0.0);
+            default -> throw new IllegalArgumentException("不支持的聚合类型: " + def.aggType());
+        };
+    }
+
+    /** 比值型：分子分母分别聚合再相除 */
+    private MetricResult aggregateRatio(MetricDef def, MetricValueStore source, long ts,
+                                        int expectedCount, double minCompleteness) {
+        List<Double> nums = source.goodValuesWithPrefix(def.ratioNumKey());
+        List<Double> dens = source.goodValuesWithPrefix(def.ratioDenKey());
+        int arrived = nums.size() + dens.size();
+        double completeness = expectedCount <= 0 ? 0.0
+                : Math.min(1.0, arrived / (double) expectedCount);
+        boolean noData = nums.isEmpty() && dens.isEmpty();
+        Double value;
+        if (noData) {
+            value = null;
+        } else {
+            double num = nums.stream().mapToDouble(Double::doubleValue).sum();
+            double den = dens.stream().mapToDouble(Double::doubleValue).sum();
+            value = den == 0.0 ? null : num / den;
+        }
+        boolean incomplete = noData || value == null || completeness < minCompleteness;
+        return new MetricResult(value, completeness, incomplete, noData, ts);
+    }
+
+    /**
+     * 日平均专用：总量÷24（固定分母），与「对到达值取均值」严格区分（TC-0015/REQ-0025）。
+     *
+     * @param expectedCount 应到小时数（DAY=24）
+     */
+    public MetricResult aggregateAverageDiv24(MetricDef def, MetricValueStore source, long ts,
+                                              int expectedCount, double minCompleteness) {
+        List<Double> values = readValues(def, source);
+        int arrived = values.size();
+        double completeness = expectedCount <= 0 ? 0.0
+                : Math.min(1.0, arrived / (double) expectedCount);
+        boolean noData = arrived == 0;
+        double total = values.stream().mapToDouble(Double::doubleValue).sum();
+        Double value = noData ? null : total / expectedCount;
+        boolean incomplete = noData || completeness < minCompleteness;
+        return new MetricResult(value, completeness, incomplete, noData, ts);
+    }
+}

+ * 范式），不引入 XXL-Job 依赖、不改 backend sys_job。

+ * + 月末一次 + 宽限期到期」；自实现 daemon 线程、fail-soft 异常兜底，与 SnapshotReconciler/
+ * PeriodicScheduler 同款。窗口到期（end + graceDays）触发；同窗由 windowKey 幂等去重（后算覆盖）。

+ * @author dowei
+ * @date 2026-11-24

+public final class MetricScheduler {
+
+    private static final Logger log = LoggerFactory.getLogger(MetricScheduler.class);
+
+    /** 提交待算窗口的计划书（测试替身/Runtime 装配） */
+    public interface Plan {
+        void submit(MetricWindow window);
+    }
+
+    private final int graceDays;
+    private final ZoneId zone;
+    private final String tenant;
+    private final Set<String> fired = ConcurrentHashMap.newKeySet();
+    private volatile ScheduledExecutorService scheduler;
+    private long tickSeconds = 60;
+
+    public MetricScheduler(int graceDays) {
+        this(graceDays, ZoneOffset.UTC, "t45");
+    }
+
+    public MetricScheduler(int graceDays, ZoneId zone) {
+        this(graceDays, zone, "t45");
+    }
+
+    /** 生产装配用：显式租户编码参与窗口唯一键 */
+    public MetricScheduler(int graceDays, ZoneId zone, String tenant) {
+        this.graceDays = graceDays;
+        this.zone = zone;
+        this.tenant = tenant;
+    }
+
+    /** 自实现调度线程启动（tickSeconds 可配；daemon + fail-soft） */
+    public synchronized void start(long tickSeconds, Plan plan, java.util.function.Supplier<List<MetricDef>> defs) {
+        if (scheduler != null) {
+            return;
+        }
+        this.tickSeconds = Math.max(1, tickSeconds);
+        scheduler = Executors.newSingleThreadScheduledExecutor(r -> {
+            Thread t = new Thread(r, "calc-metric-scheduler");
+            t.setDaemon(true);
+            return t;
+        });
+        scheduler.scheduleAtFixedRate(() -> {
+            try {
+                tick(System.currentTimeMillis(), defs.get(), plan);
+            } catch (RuntimeException e) {
+                log.error("[CALC-METRIC-SCHED] tick 未预期异常（fail-soft，下个 tick 重试）", e);
+            }
+        }, this.tickSeconds, this.tickSeconds, TimeUnit.SECONDS);
+        log.info("[CALC-METRIC-SCHED] 指标聚合调度启动 tick={}s graceDays={}（自实现，非 XXL-Job）",
+                this.tickSeconds, graceDays);
+    }
+
+    public synchronized void stop() {
+        if (scheduler != null) {
+            scheduler.shutdownNow();
+            scheduler = null;
+            log.info("[CALC-METRIC-SCHED] 指标聚合调度已停止");
+        }
+    }
+
+    public boolean isRunning() {
+        return scheduler != null && !scheduler.isShutdown();
+    }
+
+    /**
+     * 单次调度扫描（可测试入口）：对每个指标定义，向前回溯找到最近一个「已结束且过了宽限期」
+     * 的窗口并提交；已提交窗由 windowKey 幂等去重（后算覆盖不追加）。
+     *
+     * <p>为什么要回溯：宽限期（默认 3 天）> 日窗长（1 天）时，紧邻的上一窗永远未到期，
+     * 必须向前找到 end+grace <= now 的窗。MONTH/YEAR 窗长大于宽限期，回溯一步即可。
+     */
+    public void tick(long nowMillis, List<MetricDef> defs, Plan plan) {
+        for (MetricDef def : defs) {
+            MetricWindow w = latestDueWindow(def, nowMillis);
+            if (w == null) {
+                continue;
+            }
+            String key = w.key();
+            if (!fired.add(key)) {
+                continue; // 同窗幂等：已提交过
+            }
+            log.info("[CALC-METRIC-SCHED] 窗口到期 metric={} period={} window=[{},{}) graceDays={}（自实现调度）",
+                    def.metricCode(), def.period(), w.startMillis(), w.endMillis(), graceDays);
+            plan.submit(w);
+        }
+    }
+
+    /** 从「上一窗」起向前回溯，找最近一个过宽限期的已结束窗 */
+    private MetricWindow latestDueWindow(MetricDef def, long nowMillis) {
+        MetricWindow w = MetricWindow.previous(tenant, def.metricCode(), def.period(),
+                1, nowMillis, zone);
+        // 最多回溯 400 窗（3 天宽限 / 1 小时窗 = 72 窗；留足余量，fail-closed 有界）
+        for (int i = 0; i < 400 && !w.due(nowMillis, graceDays); i++) {
+            w = shiftBack(w);
+        }
+        return w.due(nowMillis, graceDays) ? w : null;
+    }
+
+    /** 窗口前移一个自身周期（自然日历对齐） */
+    private MetricWindow shiftBack(MetricWindow w) {
+        java.time.ZonedDateTime start = Instant.ofEpochMilli(w.startMillis()).atZone(zone);
+        java.time.ZonedDateTime prevStart = switch (w.period().toUpperCase(java.util.Locale.ROOT)) {
+            case "HOUR" -> start.minusHours(1);
+            case "DAY" -> start.minusDays(1);
+            case "MONTH" -> start.minusMonths(1);
+            case "YEAR" -> start.minusYears(1);
+            default -> throw new IllegalArgumentException("不支持的指标周期: " + w.period());
+        };
+        return new MetricWindow(w.tenant(), w.metricCode(), w.period(), w.calendarType(),
+                prevStart.toInstant().toEpochMilli(), w.startMillis());
+    }
+
+    long tickSeconds() {
+        return tickSeconds;
+    }
+}

+ * 不存在「本月至今」中间窗；宽限期 3 天可配（env CALC_METRIC_GRACE_DAYS）。
+ *
+ * @author dowei
+ * @date 2026-11-24
+ */
+public record MetricWindow(String tenant, String metricCode, String period, int calendarType,
+                           long startMillis, long endMillis) {
+
+    /** 当前时刻所在窗口（对齐自然日历边界） */
+    public static MetricWindow forInstant(String tenant, String metricCode, String period,
+                                          int calendarType, long epochMillis, ZoneId zone) {
+        ZonedDateTime t = Instant.ofEpochMilli(epochMillis).atZone(zone);
+        ZonedDateTime start;
+        ZonedDateTime end;
+        switch (period.toUpperCase(java.util.Locale.ROOT)) {
+            case "HOUR" -> {
+                start = t.truncatedTo(ChronoUnit.HOURS);
+                end = start.plusHours(1);
+            }
+            case "DAY" -> {
+                start = t.truncatedTo(ChronoUnit.DAYS);
+                end = start.plusDays(1);
+            }
+            case "MONTH" -> {
+                start = t.withDayOfMonth(1).truncatedTo(ChronoUnit.DAYS);
+                end = start.plusMonths(1);
+            }
+            case "YEAR" -> {
+                start = t.withDayOfYear(1).truncatedTo(ChronoUnit.DAYS);
+                end = start.plusYears(1);
+            }
+            default -> throw new IllegalArgumentException("不支持的指标周期: " + period);
+        }
+        return new MetricWindow(tenant, metricCode, period, calendarType,
+                start.toInstant().toEpochMilli(), end.toInstant().toEpochMilli());
+    }
+
+    /** 上一个已结束窗口（聚合对象）；current 窗口未结束，不产出 */
+    public static MetricWindow previous(String tenant, String metricCode, String period,
+                                        int calendarType, long epochMillis, ZoneId zone) {
+        MetricWindow current = forInstant(tenant, metricCode, period, calendarType, epochMillis, zone);
+        ZonedDateTime start = Instant.ofEpochMilli(current.startMillis).atZone(zone);
+        ZonedDateTime previousStart = switch (period.toUpperCase(java.util.Locale.ROOT)) {
+            case "HOUR" -> start.minusHours(1);
+            case "DAY" -> start.minusDays(1);
+            case "MONTH" -> start.minusMonths(1);
+            case "YEAR" -> start.minusYears(1);
+            default -> throw new IllegalArgumentException("不支持的指标周期: " + period);
+        };
+        ZonedDateTime previousEnd = start;
+        return new MetricWindow(tenant, metricCode, period, calendarType,
+                previousStart.toInstant().toEpochMilli(), previousEnd.toInstant().toEpochMilli());
+    }
+
+    /** 窗口到期判据：now >= end + graceDays 天（宽限期内到达数据可纳入） */
+    public boolean due(long nowMillis, int graceDays) {
+        long dueAt = endMillis + (long) Math.max(0, graceDays) * 24L * 3600L * 1000L;
+        return nowMillis >= dueAt;
+    }
+
+    /** 月指标月末一次（ADR-0010） */
+    public boolean isMonthEndOnce() {
+        return "MONTH".equalsIgnoreCase(period);
+    }
+
+    /** 窗口唯一标识：(tenant, metricCode, periodStart, calendarType) */
+    public String key() {
+        return tenant + "|" + metricCode + "|" + startMillis + "|" + calendarType;
+    }
+}
