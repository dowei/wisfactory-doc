+-- ============================================================
+-- 存量库升级脚本：sys_operate_log 身份三列改可空（TASK-0075 / BUG-0003）
+--
+-- 用途（BUG-0003 修复方案①逐字口径的落实）：
+--   未认证态触发带 @OperateLog 的失败请求（如 prod /auth/login/test 拒绝、登录失败路径）时，
+--   OperateLog 切面取不到登录身份，MyBatis Plus 按 NOT_NULL 策略把身份列从 INSERT 列清单整体
+--   省略（2026-11-22 02:09:43 prod 实测复现报错 SQL 佐证），INSERT 撞 NOT NULL 无默认值约束：
+--   `Field 'user_id' doesn't have a default value` → 操作日志写入失败丢弃，审计留痕缺失。
+--
+--   实测报错 SQL 的列清单不含 user_id/user_type/user_name 三列（三者均 NOT NULL 无默认值，
+--   01_base.sql 439-443 行），故只放宽 user_id 一列 INSERT 仍会撞 user_type/user_name——
+--   三列一并改 NULL 可空，一次放宽到位：
+--     user_id   bigint      NOT NULL → NULL（未认证操作尝试无登录身份，user_id=NULL 落库）
+--     user_type int         NOT NULL → NULL（未认证操作尝试无用户类型）
+--     user_name varchar(500)NOT NULL → NULL（切面尽力记登录尝试用户名，取不到不造数）
+--   字符集/长度等其余列属性保持 01_base.sql 原口径不变（user_name 保持 utf8mb4_general_ci）。
+--
+-- 01_base.sql 不改（TASK-0075 风险核对第 2 行）：存量库与新库统一由本增量脚本放宽；
+--   全新库按部署口径执行 01_base.sql + 全部 upgrade 脚本后，与本脚本口径一致。
+--
+-- 兼容性核对（TASK-0075 风险核对第 2 行，全库 grep sys_operate_log 证据见任务报告）：
+--   写入点仅 OperateLogAspect / PermissionDeniedAuditListener 两处（均已在请求线程把审计字段
+--   填好，身份列未认证态留空）；查询页 sys_operate_log 列表按 user_name/部门/业务类型/状态
+--   筛选且均为可选项，OperateLogServiceImpl#selectMyList 的 user_id 等值筛选只在已认证态触发
+--   （未认证行 user_id=NULL 天然不入该结果），既有查询兼容。
+--   登录失败日志已有独立 LoginLogService（两日志不混存口径不变）。
+--
+-- 适用环境：
+--   已执行 01_base.sql（sys_operate_log 身份三列为 NOT NULL 原口径）的存量库及全新库。
+--
+-- 幂等口径：
+--   以 information_schema.COLUMNS.IS_NULLABLE 判存——已是 NULL 可空（IS_NULLABLE='YES'）
+--   则跳过（MySQL 8.0 无 MODIFY IF 语法，用条件化存储过程实现）。重复执行不报错、
+--   不重复 MODIFY。可安全重复执行。
+--
+-- 注意（兼容 backend/scripts/sql-exec.sh）：
+--   该执行器按「行尾分号」切分语句块——存储过程体内的分号必须写在一行内（不得行尾出现），
+--   故 CREATE PROCEDURE 全部压缩为单行；语句前不得挂 -- 注释行（执行器逐行剥除 -- 行，
+--   说明文字全部集中在本头部与文件尾）。
+-- ============================================================
+
+SET NAMES utf8mb4;
+
+DROP PROCEDURE IF EXISTS ot_ol_nullable;
+CREATE PROCEDURE ot_ol_nullable(IN cn VARCHAR(64), IN ddl VARCHAR(1000)) BEGIN DECLARE n INT DEFAULT 0; SELECT COUNT(*) INTO n FROM information_schema.COLUMNS WHERE TABLE_SCHEMA = DATABASE() AND TABLE_NAME = 'sys_operate_log' AND COLUMN_NAME = cn AND IS_NULLABLE = 'NO'; IF n > 0 THEN SET @s = ddl; PREPARE st FROM @s; EXECUTE st; DEALLOCATE PREPARE st; END IF; END;
